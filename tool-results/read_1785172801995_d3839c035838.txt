     1→     1→import type { LiquidGlassRenderer } from './index'
     2→     2→import type { GlassRenderState } from './methods-render-glass'
     3→     3→import { DP } from './spring'
     4→     4→import { generateInnerShadowMask, type InnerShadowMaskParams } from './inner-shadow-mask'
     5→     5→import { buildMaskKey, getOrCreateMaskEntry, uploadMaskTexture } from './inner-shadow-cache'
     6→     6→import { continuousCurvatureRoundedRectPath } from './continuous-curve'
     7→     7→
     8→     8→declare module './index' {
     9→     9→  interface LiquidGlassRenderer {
    10→    10→    renderGlassPostPasses(state: GlassRenderState): void
    11→    11→  }
    12→    12→}
    13→    13→
    14→    14→export const glassPostPassMethods = {
    15→    15→  /** Steps 2b–2f: Inner shadow, press glow, white overlay, foreground, rim highlight.
    16→    16→   *  These all composite on top of the glass body (already drawn to
    17→    17→   *  otherFbo by renderGlassElementPass). */
    18→    18→  renderGlassPostPasses(this: LiquidGlassRenderer, state: GlassRenderState) {
    19→    19→    const gl = this.gl
    20→    20→    const { el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress, elHighlightAlpha } = state
    21→    21→
    22→    22→    // Original-space SDF uniforms — shared by all post-pass shaders so their
    23→    23→    // capsule clip is computed in ORIGINAL space (faithful to graphicsLayer
    24→    24→    // { scaleX, scaleY } post-scaling). See element.ts / highlight.ts.
    25→    25→    const origSizeX = state.origW * this.dpr
    26→    26→    const origSizeY = state.origH * this.dpr
    27→    27→    const origRadius = state.origCornerRadius * this.dpr
    28→    28→    const layerScaleX = state.layerScaleX
    29→    29→    const layerScaleY = state.layerScaleY
    30→    30→
    31→    31→    // --- Step 2b: Inner shadow post-passes (Canvas2D ring mask approach) ---
    32→    32→    // Inner shadows are drawn on the element surface, underneath the press glow.
    33→    33→    // Each shadow uses a Canvas2D-generated blurred ring mask (fill shape →
    34→    34→    // destination-out offset shape → blur), composited via the
    35→    35→    // INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER with SrcOver blend.
    36→    36→    // SDF clip in the shader ensures the shadow stays inside the shape boundary.
    37→    37→
    38→    38→    /** Helper: draw one inner shadow post-pass (shadow1 or shadow2). */
    39→    39→    const drawInnerShadowPass = (
    40→    40→      shadowCfg: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] },
    41→    41→      shadowIndex: number // 0 = the only inner shadow (original has just ONE)
    42→    42→    ) => {
    43→    43→      // Progress modulation — faithful to the original inline shader:
    44→    44→      //   - Toggle knobs: radius, alpha, offset all modulated by togglePressProgress
    45→    45→      //   - Bottom-tab indicator: radius, alpha, offset modulated by togglePressProgress
    46→    46→      //   - Other elements (magnifier): static params, no modulation
    47→    47→      const progress = (el.isToggleKnob || el.isBottomTabIndicator) ? togglePressProgress : 1
    48→    48→      let shadowAlpha = shadowCfg.alpha * progress * state.enterAlpha
    49→    49→      let shadowRadius = shadowCfg.radius * progress
    50→    50→      let shadowOffsetX = shadowCfg.offsetX * progress
    51→    51→      let shadowOffsetY = shadowCfg.offsetY * progress
    52→    52→
    53→    53→      if (shadowAlpha <= 0.001 || shadowRadius <= 0.5) return
    54→    54→
    55→    55→      // Blur sigma = radius * dpr (BlurEffect semantics: sigma = radius directly).
    56→    56→      const blurSigma = shadowRadius * this.dpr  // device px — sigma = radius, not radius/3
    57→    57→      // Margin for blur spread (3σ) + AA
    58→    58→      const margin = Math.ceil(blurSigma * 3) + 2
    59→    59→      // Mask dimensions in device px (origSize + 2*margin)
    60→    60→      const maskW = Math.max(1, Math.ceil(origSizeX + 2 * margin))
    61→    61→      const maskH = Math.max(1, Math.ceil(origSizeY + 2 * margin))
    62→    62→      // Supersampling for sharper mask rasterization
    63→    63→      const deviceDpr = window.devicePixelRatio || 1
    64→    64→      const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
    65→    65→      const useG2 = !!el.useContinuousSdf
    66→    66→
    67→    67→      // Offset in device px (already × progress)
    68→    68→      const offsetXDp = shadowOffsetX * this.dpr
    69→    69→      const offsetYDp = shadowOffsetY * this.dpr
    70→    70→
    71→    71→      // Build mask params for the mask generator
    72→    72→      const maskParams: InnerShadowMaskParams = {
    73→    73→        w: origSizeX,
    74→    74→        h: origSizeY,
    75→    75→        radius: origRadius,
    76→    76→        offsetX: offsetXDp,
    77→    77→        offsetY: offsetYDp,
    78→    78→        blurSigma,
    79→    79→        margin,
    80→    80→        useG2,
    81→    81→        supersample: SS,
    82→    82→      }
    83→    83→
    84→    84→      // Build cache key and get/create cache entry
    85→    85→      const key = buildMaskKey(shadowIndex, maskParams)
    86→    86→      const entry = getOrCreateMaskEntry(this.innerShadowMaskCache, gl, key, maskW, maskH)
    87→    87→
    88→    88→      // Generate mask and upload texture if not ready
    89→    89→      if (!entry.ready) {
    90→    90→        const result = generateInnerShadowMask(maskParams)
    91→    91→        uploadMaskTexture(gl, entry, result)
    92→    92→      }
    93→    93→
    94→    94→      // --- Composite: inner shadow mask × shadowAlpha × shadowColor → scene ---
    95→    95→      // PREMULTIPLIED SrcOver: the shader outputs vec4(color*alpha, alpha) (premultiplied).
    96→    96→      // Using blendFunc(ONE, ONE_MINUS_SRC_ALPHA) avoids squaring the alpha (which
    97→    97→      // would make innerShadow at alpha=0.15 contribute only 0.15²=0.0225 — invisible).
    98→    98→      gl.enable(gl.BLEND)
    99→    99→      gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   100→   100→      gl.useProgram(this.innerShadowMaskCompositeProgram)
   101→   101→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   102→   102→      gl.enableVertexAttribArray(this.aPosLocIs)
   103→   103→      gl.vertexAttribPointer(this.aPosLocIs, 2, gl.FLOAT, false, 0, 0)
   104→   104→
   105→   105→      gl.uniform2f(this.uIs['uCanvasSize'], this.canvas.width, this.canvas.height)
   106→   106→      gl.uniform2f(this.uIs['uOffset'], sx * this.dpr, sy * this.dpr)
   107→   107→      gl.uniform2f(this.uIs['uSize'], sw * this.dpr, sh * this.dpr)
   108→   108→      gl.uniform4f(this.uIs['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   109→   109→
   110→   110→      gl.activeTexture(gl.TEXTURE0)
   111→   111→      gl.bindTexture(gl.TEXTURE_2D, entry.tex)
   112→   112→      gl.uniform1i(this.uIs['uInnerShadowMask'], 0)
   113→   113→      // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   114→   114→      // physical canvas is SS× larger but the shader uses 1x coords for UV.
   115→   115→      gl.uniform2f(this.uIs['uMaskOffset'], margin, margin)
   116→   116→      gl.uniform2f(this.uIs['uMaskSize'], entry.w, entry.h)
   117→   117→
   118→   118→      // Shadow color (defaults to black [0,0,0] if not specified)
   119→   119→      const color = shadowCfg.color ?? [0, 0, 0]
   120→   120→      gl.uniform3f(this.uIs['uInnerShadowColor'], color[0], color[1], color[2])
   121→   121→      gl.uniform1f(this.uIs['uInnerShadowAlpha'], shadowAlpha)
   122→   122→
   123→   123→      gl.uniform2f(this.uIs['uOriginalSize'], origSizeX, origSizeY)
   124→   124→      gl.uniform1f(this.uIs['uOriginalCornerRadius'], origRadius)
   125→   125→      gl.uniform2f(this.uIs['uLayerScale'], layerScaleX, layerScaleY)
   126→   126→      gl.uniform1f(this.uIs['uElementRotation'], state.elementRotation)
   127→   127→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   128→   128→    }
   129→   129→
   130→   130→    // Inner shadow (faithful to the original — only ONE black inner shadow,
   131→   131→    // no innerShadow2. LiquidToggle.kt: InnerShadow(radius=4dp*progress, alpha=progress))
   132→   132→    if (el.innerShadow) {
   133→   133→      drawInnerShadowPass(el.innerShadow, 0)
   134→   134→    }
   135→   135→
   136→   136→    // --- Step 2c: Press glow (button + bottom-tab container) ---
   137→   137→    // Faithful to InteractiveHighlight.kt: a flat white Plus-blend overlay
   138→   138→    // + a radial white glow. For buttons, position = finger, intensity = st.pressProgress.
   139→   139→    // For the bottom-tab container, position = indicator center (via the
   140→   140→    // original's position lambda), intensity = toggleState.pressProgress.
   141→   141→    const isContainer = !!el.isBottomTabContainer
   142→   142→    const glowP = isButton ? p : (isContainer ? togglePressProgress : 0)
   143→   143→    if ((isButton && el.isInteractive && st && p > 0.001) || (isContainer && togglePressProgress > 0.001)) {
   144→   144→      // a. Flat white overlay
   145→   145→      gl.useProgram(this.tintProgram)
   146→   146→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   147→   147→      gl.enableVertexAttribArray(this.aPosLocTn)
   148→   148→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   149→   149→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   150→   150→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   151→   151→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   152→   152→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   153→   153→      gl.uniform4f(
   154→   154→        this.uTn['uCornerRadii'],
   155→   155→        radii[0] * this.dpr,
   156→   156→        radii[1] * this.dpr,
   157→   157→        radii[2] * this.dpr,
   158→   158→        radii[3] * this.dpr
   159→   159→      )
   160→   160→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   161→   161→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   162→   162→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   163→   163→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   164→   164→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   165→   165→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.08 * glowP)
   166→   166→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   167→   167→
   168→   168→      // b. Radial highlight at finger position (button) or indicator center (container).
   169→   169→      gl.useProgram(this.highlightProgram)
   170→   170→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   171→   171→      gl.enableVertexAttribArray(this.aPosLocHl)
   172→   172→      gl.vertexAttribPointer(this.aPosLocHl, 2, gl.FLOAT, false, 0, 0)
   173→   173→      gl.blendFunc(gl.ONE, gl.ONE)
   174→   174→      gl.uniform2f(this.uHl['uCanvasSize'], this.canvas.width, this.canvas.height)
   175→   175→      gl.uniform2f(this.uHl['uOffset'], sx * this.dpr, sy * this.dpr)
   176→   176→      gl.uniform2f(this.uHl['uSize'], sw * this.dpr, sh * this.dpr)
   177→   177→      gl.uniform4f(
   178→   178→        this.uHl['uCornerRadii'],
   179→   179→        radii[0] * this.dpr,
   180→   180→        radii[1] * this.dpr,
   181→   181→        radii[2] * this.dpr,
   182→   182→        radii[3] * this.dpr
   183→   183→      )
   184→   184→      gl.uniform2f(this.uHl['uOriginalSize'], origSizeX, origSizeY)
   185→   185→      gl.uniform1f(this.uHl['uOriginalCornerRadius'], origRadius)
   186→   186→      gl.uniform2f(this.uHl['uLayerScale'], layerScaleX, layerScaleY)
   187→   187→      gl.uniform1f(this.uHl['uElementRotation'], state.elementRotation)
   188→   188→      gl.uniform1f(this.uHl['uCornerStyle'], this.cornerStyle)
   189→   189→      gl.uniform4f(this.uHl['uColor'], 1, 1, 1, 0.15 * glowP)
   190→   190→      const minDim = Math.min(sw, sh) * this.dpr
   191→   191→      gl.uniform1f(this.uHl['uRadius'], minDim * 1.5)
   192→   192→      // Position: for buttons, the finger position. For the container, the
   193→   193→      // indicator center — faithful to the original's position lambda which
   194→   194→      // returns (dampedDragAnimation.value + 0.5) * tabWidth (indicator center
   195→   195→      // relative to the container).
   196→   196→      let px: number, py: number
   197→   197→      if (isContainer) {
   198→   198→        const tg = this.toggleStates.get(el.isBottomTabContainer!.groupId)
   199→   199→        // tabW = container ORIGINAL width / tabsCount (not scaled — the position
   200→   200→        // lambda runs in pre-scale local coords). indicator center =
   201→   201→        // (fraction + 0.5) * tabW. Faithful to LiquidBottomTabs.kt:
   202→   202→        //   position = (dampedDragAnimation.value + 0.5) * tabWidth
   203→   203→        const tabsCount = el.isBottomTabContainer!.tabsCount ?? 4
   204→   204→        const tabW = el.rect.w / tabsCount
   205→   205→        const fraction = tg ? tg.fraction : 0
   206→   206→        const indCenterX = (fraction + 0.5) * tabW
   207→   207→        // Map original-local to scaled-local (the shader's uPosition is in
   208→   208→        // scaled-local coords, 0..sw). scale = sw / el.rect.w.
   209→   209→        const scaleToLocal = sw / el.rect.w
   210→   210→        px = Math.max(0, Math.min(sw, indCenterX * scaleToLocal)) * this.dpr
   211→   211→        py = (sh / 2) * this.dpr
   212→   212→      } else {
   213→   213→        px = Math.max(0, Math.min(sw, st!.dragX * state.layerScaleX)) * this.dpr
   214→   214→        py = Math.max(0, Math.min(sh, st!.dragY * state.layerScaleY)) * this.dpr
   215→   215→      }
   216→   216→      gl.uniform2f(this.uHl['uPosition'], px, py)
   217→   217→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   218→   218→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   219→   219→    }
   220→   220→
   221→   221→    // --- Step 2d: Toggle knob white overlay (faithful to LiquidToggle.kt
   222→   222→    // / LiquidSlider.kt onDrawSurface) ---
   223→   223→    if (el.isToggleKnob && togglePressProgress < 0.999) {
   224→   224→      const whiteAlpha = 1.0 * (1 - togglePressProgress)
   225→   225→      gl.useProgram(this.tintProgram)
   226→   226→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   227→   227→      gl.enableVertexAttribArray(this.aPosLocTn)
   228→   228→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   229→   229→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   230→   230→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   231→   231→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   232→   232→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   233→   233→      gl.uniform4f(
   234→   234→        this.uTn['uCornerRadii'],
   235→   235→        radii[0] * this.dpr,
   236→   236→        radii[1] * this.dpr,
   237→   237→        radii[2] * this.dpr,
   238→   238→        radii[3] * this.dpr
   239→   239→      )
   240→   240→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   241→   241→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   242→   242→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   243→   243→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   244→   244→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   245→   245→      // Faithful to LiquidToggle.kt onDrawSurface:
   246→   246→      //   drawRect(Color.White.copy(alpha = 1f - progress))
   247→   247→      // Solid white pebble at rest (alpha=1), fading to transparent when
   248→   248→      // pressed (alpha=0) to reveal the glass refraction beneath.
   249→   249→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, whiteAlpha)
   250→   250→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   251→   251→    }
   252→   252→
   253→   253→    // --- Step 2d2: Bottom tab indicator onDrawSurface (faithful to
   254→   254→    // LiquidBottomTabs.kt indicator):
   255→   255→    //   drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)
   256→   256→    //   drawRect(Color.Black.copy(alpha = 0.03f * progress))
   257→   257→    // First: theme-aware dim overlay fading OUT on press (SrcOver).
   258→   258→    // Second: subtle black tint fading IN on press (SrcOver).
   259→   259→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.dimColor) {
   260→   260→      const dc = el.isBottomTabIndicator.dimColor
   261→   261→      const p = togglePressProgress
   262→   262→      gl.useProgram(this.tintProgram)
   263→   263→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   264→   264→      gl.enableVertexAttribArray(this.aPosLocTn)
   265→   265→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   266→   266→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   267→   267→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   268→   268→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   269→   269→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   270→   270→      gl.uniform4f(this.uTn['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   271→   271→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   272→   272→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   273→   273→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   274→   274→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   275→   275→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   276→   276→      // First overlay: dim color at 0.1 * (1 - progress) — fades out on press.
   277→   277→      gl.uniform4f(this.uTn['uColor'], dc[0], dc[1], dc[2], 0.1 * (1 - p))
   278→   278→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   279→   279→      // Second overlay: black at 0.03 * progress — fades in on press.
   280→   280→      gl.uniform4f(this.uTn['uColor'], 0, 0, 0, 0.03 * p)
   281→   281→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   282→   282→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   283→   283→    }
   284→   284→
   285→   285→    // --- Step 2e: Foreground (label or icon) pass (button only) ---
   286→   286→    if (isButton && (el.label || el.icon)) {
   287→   287→      const fgTex = this.fgTextures.get(el.id)
   288→   288→      if (fgTex) {
   289→   289→        gl.useProgram(this.foregroundProgram)
   290→   290→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   291→   291→        gl.enableVertexAttribArray(this.aPosLocFg)
   292→   292→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   293→   293→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   294→   294→
   295→   295→        gl.activeTexture(gl.TEXTURE0)
   296→   296→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   297→   297→        gl.uniform1i(this.uFg['uTexture'], 0)
   298→   298→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   299→   299→        gl.uniform2f(this.uFg['uOffset'], sx * this.dpr, sy * this.dpr)
   300→   300→        gl.uniform2f(this.uFg['uSize'], sw * this.dpr, sh * this.dpr)
   301→   301→        gl.uniform4f(
   302→   302→          this.uFg['uCornerRadii'],
   303→   303→          radii[0] * this.dpr,
   304→   304→          radii[1] * this.dpr,
   305→   305→          radii[2] * this.dpr,
   306→   306→          radii[3] * this.dpr
   307→   307→        )
   308→   308→        gl.uniform2f(this.uFg['uOriginalSize'], origSizeX, origSizeY)
   309→   309→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], origRadius)
   310→   310→      gl.uniform2f(this.uFg['uLayerScale'], layerScaleX, layerScaleY)
   311→   311→      gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   312→   312→      // Continuous-curvature mask for capsule foregrounds.
   313→   313→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   314→   314→        gl.activeTexture(gl.TEXTURE2)
   315→   315→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   316→   316→        gl.uniform1i(this.uFg['uContinuousSdf'], 2)
   317→   317→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 1.0)
   318→   318→        gl.uniform2f(this.uFg['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   319→   319→        gl.uniform2f(this.uFg['uContinuousSdfElementSize'], state.origW * this.dpr, state.origH * this.dpr)
   320→   320→      } else {
   321→   321→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 0.0)
   322→   322→      }
   323→   323→        gl.uniform1f(this.uFg['uAlpha'], 1.0 - 0.15 * p)
   324→   324→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   325→   325→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   326→   326→      }
   327→   327→    }
   328→   328→
   329→   329→    // --- Step 2f: Rim highlight (Canvas2D stroke mask approach) ---
   330→   330→    // Uses Canvas2D ctx.stroke() (browser-native Skia) to rasterize the G2
   331→   331→    // Bezier path as a stroke mask. This is the SAME method the original
   332→   332→    // uses (Skia drawOutline + paint.Stroke) — the browser's Canvas2D
   333→   333→    // internally calls Skia's SkCanvas::drawPath, which tessellates the
   334→   334→    // Bezier into triangles and rasterizes with hardware coverage AA.
   335→   335→    //
   336→   336→    // Advantages over the SDF approach:
   337→   337→    //   - Exact G2 Bezier shape (not SDF approximation)
   338→   338→    //   - Hardware coverage AA (sub-pixel accurate, no smoothstep needed)
   339→   339→    //   - No SDF computation in shader (just one texture fetch)
   340→   340→    //   - Adaptive tessellation (more samples at high-curvature corners)
   341→   341→    //
   342→   342→    // The stroke mask is cached by exact geometry. Highlight angle/alpha/press
   343→   343→    // progress only affect the composite uniforms, not the mask, so caching is
   344→   344→    // safe and avoids both per-frame rasterization and UV mismatch between a
   345→   345→    // reused backing canvas and the logical mask size.
   346→   346→    if (el.highlight && el.highlight.alpha > 0.001) {
   347→   347→      const rimAlpha = (el.isToggleKnob || el.isBottomTabIndicator) ? elHighlightAlpha : el.highlight.alpha
   348→   348→      // Ambient mode: paint.color = White.copy(alpha=0.38). Although the AGSL shader
   349→   349→      // overrides the paint COLOR, Skia still applies paint.alpha on top of the shader
   350→   350→      // output. Without this 0.38 factor the dark side dims the scene by 100% (pure
   351→   351→      // black) instead of 38% (semi-transparent, faithful to the original).
   352→   352→      const paintAlpha = el.highlight.mode === 1 ? 0.38 : 1.0
   353→   353→      const finalAlpha = rimAlpha * state.enterAlpha * paintAlpha
   354→   354→      if (finalAlpha > 0.001) {
   355→   355→        // HighlightModifier.kt: ceil(width.toPx().coerceAtMost(minDimension / 2)) * 2
   356→   356→        const widthPx = Math.min(
   357→   357→          el.highlight.widthDp * this.dpr,
   358→   358→          Math.min(origSizeX, origSizeY) * 0.5
   359→   359→        )
   360→   360→        const strokeWidthDevice = Math.max(1, Math.ceil(widthPx) * 2)
   361→   361→        // Highlight data class: blurRadius defaults to width / 2. Honor it in the
   362→   362→        // Canvas2D mask as well (for the default 0.25dp this is sub-pixel, but it
   363→   363→        // keeps non-default highlights faithful too).
   364→   364→        const blurPx = Math.max(0, (el.highlight.blurRadiusDp ?? el.highlight.widthDp / 2) * this.dpr)
   365→   365→
   366→   366→        // --- Generate stroke mask via Canvas2D (browser-native Skia) ---
   367→   367→        // Element-local coordinates: (0,0) = element top-left in device px.
   368→   368→        // Margin for stroke width + AA + blur spread.
   369→   369→        const strokeMargin = Math.ceil(strokeWidthDevice) + 4  // logical margin
   370→   370→        // Logical mask size (1x device px) — used for shader UV mapping
   371→   371→        const maskW = Math.max(1, Math.ceil(origSizeX + 2 * strokeMargin))
   372→   372→        const maskH = Math.max(1, Math.ceil(origSizeY + 2 * strokeMargin))
   373→   373→        // Supersample for sharper stroke/blur rasterization.
   374→   374→        // Cap SS so total mask DPR (this.dpr × SS) ≤ device's native DPR —
   375→   375→        // no point rendering pixels beyond what the screen can display.
   376→   376→        const deviceDpr = window.devicePixelRatio || 1
   377→   377→        const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
   378→   378→        const canvasW = maskW * SS
   379→   379→        const canvasH = maskH * SS
   380→   380→        const useG2 = !!el.useContinuousSdf
   381→   381→        const maskKey = [
   382→   382→          useG2 ? 'g2' : 'rr',
   383→   383→          origSizeX.toFixed(3),
   384→   384→          origSizeY.toFixed(3),
   385→   385→          origRadius.toFixed(3),
   386→   386→          strokeWidthDevice,
   387→   387→          blurPx.toFixed(3),
   388→   388→          strokeMargin,
   389→   389→          maskW,
   390→   390→          maskH,
   391→   391→          `ss${SS}`,  // cache key includes supersample factor
   392→   392→        ].join(':')
   393→   393→
   394→   394→        let mask = this.strokeMaskCache.get(maskKey)
   395→   395→        if (!mask) {
   396→   396→          const canvas = document.createElement('canvas')
   397→   397→          canvas.width = canvasW  // 2x supersampled physical size
   398→   398→          canvas.height = canvasH
   399→   399→          const ctx = canvas.getContext('2d', { alpha: true })
   400→   400→          if (!ctx) throw new Error('2D canvas not supported')
   401→   401→          const tex = gl.createTexture()
   402→   402→          if (!tex) throw new Error('WebGL texture allocation failed')
   403→   403→          // w/h store the LOGICAL (1x) size for shader UV mapping;
   404→   404→          // the physical canvas is SS times larger.
   405→   405→          mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   406→   406→          this.strokeMaskCache.set(maskKey, mask)
   407→   407→
   408→   408→          // Keep the cache bounded. 32 entries is far above the catalog's
   409→   409→          // simultaneous highlight geometries and avoids unbounded growth on
   410→   410→          // highly dynamic pages.
   411→   411→          if (this.strokeMaskCache.size > 32) {
   412→   412→            const oldestKey = this.strokeMaskCache.keys().next().value as string | undefined
   413→   413→            if (oldestKey && oldestKey !== maskKey) {
   414→   414→              const oldest = this.strokeMaskCache.get(oldestKey)
   415→   415→              if (oldest) gl.deleteTexture(oldest.tex)
   416→   416→              this.strokeMaskCache.delete(oldestKey)
   417→   417→            }
   418→   418→          }
   419→   419→        }
   420→   420→
   421→   421→        if (!mask.ready) {
   422→   422→          const smCtx = mask.ctx
   423→   423→          smCtx.clearRect(0, 0, canvasW, canvasH)
   424→   424→          smCtx.save()
   425→   425→          // Scale up for 2x supersampling: draw in logical (1x) coordinates
   426→   426→          // while the physical canvas is 2x. This gives sharper stroke + blur.
   427→   427→          smCtx.scale(SS, SS)
   428→   428→          smCtx.translate(strokeMargin, strokeMargin)
   429→   429→
   430→   430→          // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   431→   431→          let path: Path2D
   432→   432→          if (useG2) {
   433→   433→            path = continuousCurvatureRoundedRectPath(smCtx, origSizeX, origSizeY, origRadius)
   434→   434→          } else {
   435→   435→            path = new Path2D()
   436→   436→            const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   437→   437→            path.moveTo(r, 0)
   438→   438→            path.lineTo(origSizeX - r, 0)
   439→   439→            path.arcTo(origSizeX, 0, origSizeX, r, r)
   440→   440→            path.lineTo(origSizeX, origSizeY - r)
   441→   441→            path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   442→   442→            path.lineTo(r, origSizeY)
   443→   443→            path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   444→   444→            path.lineTo(0, r)
   445→   445→            path.arcTo(0, 0, r, 0, r)
   446→   446→            path.closePath()
   447→   447→          }
   448→   448→
   449→   449→          // Stroke — browser-native Skia stroke rasterization.
   450→   450→          // Faithful to HighlightModifier.kt: clipOutline(outline) BEFORE drawOutline,
   451→   451→          // so only the INSIDE half of the centered stroke remains. Without this
   452→   452→          // clip, Canvas2D keeps the outer half too and the rim highlight leaks
   453→   453→          // outside the glass / looks twice as thick.
   454→   454→          smCtx.clip(path)
   455→   455→          smCtx.lineWidth = strokeWidthDevice
   456→   456→          smCtx.strokeStyle = 'rgba(255,255,255,1)'
   457→   457→          smCtx.lineJoin = 'round'
   458→   458→          smCtx.lineCap = 'round'
   459→   459→          // Approximate Skia BlurMaskFilter on the stroke paint. The clip above
   460→   460→          // keeps the result inside the outline; the blur only softens that
   461→   461→          // inside stroke band, like the original HighlightModifier layer.
   462→   462→          smCtx.filter = blurPx > 0.01 ? `blur(${blurPx}px)` : 'none'
   463→   463→          smCtx.stroke(path)
   464→   464→          smCtx.filter = 'none'
   465→   465→          smCtx.restore()
   466→   466→
   467→   467→          // Upload to GPU texture. Keep the same top-left UV convention as the
   468→   468→          // foreground pass: with UNPACK_FLIP_Y_WEBGL=false, local y=0 samples
   469→   469→          // the top row of the mask.
   470→   470→          gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   471→   471→          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   472→   472→          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   473→   473→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   474→   474→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   475→   475→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   476→   476→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   477→   477→          mask.ready = true
   478→   478→        }
   479→   479→
   480→   480→        // --- Composite: stroke mask × intensity × color → scene FBO ---
   481→   481→        gl.enable(gl.BLEND)
   482→   482→        if (el.highlight.mode === 1) {
   483→   483→          // Ambient — premultiplied SrcOver blend. The shader outputs
   484→   484→          // vec4(color.rgb*t*i, i) which is premultiplied (rgb already has alpha).
   485→   485→          // blendFunc(ONE, ONE_MINUS_SRC_ALPHA) = premultiplied SrcOver, which
   486→   486→          // avoids squaring the alpha (SRC_ALPHA would multiply rgb by alpha AGAIN).
   487→   487→          // This produces the correct "half black, half white" sphere effect:
   488→   488→          // dark side: result.rgb = 0 + dst*(1-i) → dims scene
   489→   489→          // bright side: result.rgb = color*i + dst*(1-i) → adds highlight
   490→   490→          gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   491→   491→        } else {
   492→   492→          // Default + Plain — Plus blend (additive). Premultiplied Plus = ONE, ONE.
   493→   493→          gl.blendFunc(gl.ONE, gl.ONE)
   494→   494→        }
   495→   495→        gl.useProgram(this.strokeMaskCompositeProgram)
   496→   496→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   497→   497→        gl.enableVertexAttribArray(this.aPosLocSm)
   498→   498→        gl.vertexAttribPointer(this.aPosLocSm, 2, gl.FLOAT, false, 0, 0)
   499→   499→        gl.uniform2f(this.uSm['uCanvasSize'], this.canvas.width, this.canvas.height)
   500→   500→        gl.uniform2f(this.uSm['uOffset'], sx * this.dpr, sy * this.dpr)
   501→   501→        gl.uniform2f(this.uSm['uSize'], sw * this.dpr, sh * this.dpr)
   502→   502→        gl.uniform4f(this.uSm['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   503→   503→        gl.activeTexture(gl.TEXTURE0)
   504→   504→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   505→   505→        gl.uniform1i(this.uSm['uStrokeMask'], 0)
   506→   506→        // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   507→   507→        // physical canvas is SS× larger but the shader uses 1x coords for UV:
   508→   508→        //   UV = (localCoord + margin) / logicalMaskSize
   509→   509→        // LINEAR filtering on the SS× texture gives supersampled quality.
   510→   510→        gl.uniform2f(this.uSm['uMaskOffset'], strokeMargin, strokeMargin)
   511→   511→        gl.uniform2f(this.uSm['uMaskSize'], mask.w, mask.h)
   512→   512→        gl.uniform4f(this.uSm['uHighlightColor'], el.highlight.color[0], el.highlight.color[1], el.highlight.color[2], 1.0)
   513→   513→        gl.uniform1f(this.uSm['uHighlightAngle'], el.useGravityAngle ? this.gravityAngle : el.highlight.angle)
   514→   514→        gl.uniform1f(this.uSm['uHighlightFalloff'], el.highlight.falloff)
   515→   515→        gl.uniform1f(this.uSm['uHighlightAlpha'], finalAlpha)
   516→   516→        gl.uniform1f(this.uSm['uHighlightMode'], el.highlight.mode)
   517→   517→        gl.uniform2f(this.uSm['uOriginalSize'], origSizeX, origSizeY)
   518→   518→        gl.uniform1f(this.uSm['uOriginalCornerRadius'], origRadius)
   519→   519→        gl.uniform2f(this.uSm['uLayerScale'], layerScaleX, layerScaleY)
   520→   520→        gl.uniform1f(this.uSm['uElementRotation'], state.elementRotation)
   521→   521→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   522→   522→
   523→   523→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   524→   524→      }
   525→   525→    }
   526→   526→
   527→   527→    // --- Step 2g: 内层背景板 rim highlight (指示器 only) ---
   528→   528→    // The 内层背景板 (hidden Row's 56dp glass inside the 指示器) has its own
   529→   529→    // Highlight.Default.copy(alpha=progress). This is drawn INSIDE the
   530→   530→    // 指示器's element shader (sampleIndicatorBackdrop) so it's clipped by
   531→   531→    // the 指示器's capsule SDF — no separate pass needed here.
   532→   532→  },
   533→   533→}

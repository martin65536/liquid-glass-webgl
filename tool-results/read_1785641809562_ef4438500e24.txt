    33→function useSystemTheme(): Theme {
    34→  const [theme, setTheme] = React.useState<Theme>('light')
    35→  React.useEffect(() => {
    36→    if (typeof window === 'undefined' || !window.matchMedia) return
    37→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    38→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    39→    update()
    40→    mq.addEventListener('change', update)
    41→    return () => mq.removeEventListener('change', update)
    42→  }, [])
    43→  return theme
    44→}

   300→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   301→    const MIN_FPS = 55
   302→    const MIN_DPR = 0.5 // slider minimum
   303→    if (fps >= MIN_FPS) {
   304→      // This DPR works — set lo to it (we know it's viable)
   305→      perfLoRef.current = candidateDpr
   306→    } else {
   307→      // Too slow — set hi to it (search lower)
   308→      perfHiRef.current = candidateDpr
   309→    }
   310→
   311→    // Special case: iteration 2 tested the minimum (0.5) and it failed.
   312→    // Performance is truly bad — force bestDpr = 0.5 and finish immediately.
   313→    if (iteration === 2 && candidateDpr === MIN_DPR && fps < MIN_FPS) {
   314→      perfPhaseRef.current = 'done'
   315→      const statusText = `检测完成 · 推荐 DPR：${MIN_DPR} · 性能有限，已自动降低画质`
   316→      try { window.localStorage.setItem(PERF_KEY, String(MIN_DPR)) } catch {}
   317→      setState({
   318→        customDpr: MIN_DPR,
   319→        perfDone: true,
   320→        perfResultDpr: MIN_DPR,
   321→        perfStatusText: statusText,
   322→        perfProgress: null,
   323→        perfProgressFrac: 1,
   324→      })
   325→      return
   326→    }
   327→
   328→    // Convergence: adjacent slider values (hi-lo ≤ 0.25) or max iterations reached.
   329→    // Both lo and hi are always on 0.25 steps after extremes are tested,
   330→    // so stalling cannot occur — the midpoint always lands strictly between them.
   331→    if (iteration >= maxIterations || (perfHiRef.current - perfLoRef.current) <= 0.25) {
   332→      // Converged — lo is the highest DPR that passed (already on 0.25 steps)
   333→      const bestDpr = Math.max(MIN_DPR, Math.min(deviceDpr, perfLoRef.current))
   334→      perfPhaseRef.current = 'done'
   335→      // isGood: recommended DPR is at least 75% of deviceDpr
   336→      const isGood = bestDpr >= deviceDpr * 0.75
   337→      const statusText = isGood
   338→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   339→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   340→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   341→      setState({
   342→        customDpr: bestDpr,
   343→        perfDone: true,
   344→        perfResultDpr: bestDpr,
   345→        perfStatusText: statusText,
   346→        perfProgress: null,
   347→        // Don't reset perfGlassAngle — settle animation will handle it
   348→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   349→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   350→        perfProgressFrac: 1,
   351→      })
   352→    } else {
   353→      // More iterations needed — increment trigger to re-fire the effect
   354→      perfPhaseRef.current = 'idle'
   355→      setState((prev) => ({
   356→        perfProgress: 'running',
   357→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   358→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   359→      }))
   360→    }
   361→  }

   386→  // Glass deformation animation — 16 glasses deform simultaneously.
   387→  // perfGlassAngle is stored in state; each glass in the 4×4 grid computes
   388→  // its own W/H from (angle + phaseOffset). Rotation speed: 1 circle/sec.
   389→  // Also smoothly animates perfProgressFracAnimated toward perfProgressFrac
   390→  // (replacing the CSS transition that the DOM overlay used).
   391→  React.useEffect(() => {
   392→    if (destination !== CatalogDestination.PerfBenchmark) return
   393→    if (state.perfProgress !== 'running' && state.perfProgress !== 'stop-requested') return
   394→    if (!rendererReady) return
   395→
   396→    const ORBIT_SPEED = 2 * Math.PI // radians per second — 1 full circle per second
   397→    const PROG_LERP_SPEED = 4 // progress bar lerp speed (fraction per second, mimics ~600ms CSS ease-out)
   398→    let angle = state.perfGlassAngle || 0
   399→    let progAnimated = state.perfProgressFracAnimated ?? 0
   400→    let lastT = 0
   401→    let raf = 0
   402→
   403→    const tick = (timestamp: number) => {
   404→      if (!lastT) lastT = timestamp
   405→      const dt = Math.min(0.05, (timestamp - lastT) / 1000)
   406→      lastT = timestamp
   407→      angle += ORBIT_SPEED * dt
   408→      // Smoothly animate progress toward current target (use ref for latest value)
   409→      const target = perfProgressFracRef.current
   410→      progAnimated += (target - progAnimated) * Math.min(1, PROG_LERP_SPEED * dt)
   411→      setState({ perfGlassAngle: angle, perfProgressFracAnimated: progAnimated })
   412→      raf = requestAnimationFrame(tick)
   413→    }
   414→    raf = requestAnimationFrame(tick)
   415→    return () => cancelAnimationFrame(raf)
   416→  }, [destination, state.perfProgress, rendererReady])

   418→  // Settle animation — after benchmark ends, smoothly transition glasses
   419→  // from deformed to square (perfDeformMul decays 1→0) and animate the
   420→  // exit button sliding in (perfExitProgress 0→1 with ease-out).
   421→  // Also animates perfProgressFracAnimated to 1.0.
   422→  React.useEffect(() => {
   423→    if (destination !== CatalogDestination.PerfBenchmark) return
   424→    if (!rendererReady) return
   425→    if (!state.perfDone) return
   426→    // Already settled — no need to animate
   427→    if (state.perfDeformMul <= 0.01 && state.perfExitProgress >= 0.99 && (state.perfProgressFracAnimated ?? 0) >= 0.99) return
   428→
   429→    const ORBIT_SPEED = 2 * Math.PI // keep rotation going during settle
   430→    const PROG_LERP_SPEED = 4 // progress bar lerp speed
   431→    let angle = state.perfGlassAngle || 0
   432→    let progAnimated = state.perfProgressFracAnimated ?? 0
   433→    let elapsed = 0
   434→    let lastT = 0
   435→    let raf = 0
   436→    // Exponential decay rate for deformMul: mul = e^(-3*t), reaches ~0.05 by 1s
   437→    const DECAY_RATE = 3
   438→    // Exit button: starts appearing after 0.3s, fully visible by 0.8s
   439→    const EXIT_DELAY = 0.3
   440→    const EXIT_DURATION = 0.5
   441→
   442→    const tick = (timestamp: number) => {
   443→      if (!lastT) lastT = timestamp
   444→      const dt = Math.min(0.05, (timestamp - lastT) / 1000)
   445→      lastT = timestamp
   446→      elapsed += dt
   447→
   448→      // Continue rotation (so glasses don't freeze at a weird angle)
   449→      angle += ORBIT_SPEED * dt
   450→
   451→      // Exponential decay: deformMul goes from 1 → ~0 over ~1 second
   452→      const mul = Math.exp(-DECAY_RATE * elapsed)
   453→
   454→      // Exit button: ease-out quadratic after delay
   455→      let exitProg = 0
   456→      if (elapsed > EXIT_DELAY) {
   457→        const t = Math.min(1, (elapsed - EXIT_DELAY) / EXIT_DURATION)
   458→        exitProg = 1 - (1 - t) * (1 - t) // ease-out
   459→      }
   460→
   461→      // Smoothly animate progress to 1.0 (use ref for latest value)
   462→      const progTarget = perfProgressFracRef.current
   463→      progAnimated += (progTarget - progAnimated) * Math.min(1, PROG_LERP_SPEED * dt)
   464→
   465→      setState({ perfGlassAngle: angle, perfDeformMul: mul, perfExitProgress: exitProg, perfProgressFracAnimated: progAnimated })
   466→

   645→  // Device motion → gravity angle for glass highlight direction.
   646→  // Faithful to UISensor.kt:
   647→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   648→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   649→  //
   650→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   651→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   652→  // rebuilding the catalog (which would lose drag state and cause jank).
   653→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   654→  // renderer.gravityAngle live each frame.
   655→  //
   656→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   657→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   658→  // device's acceleration INCLUDING gravity in device-space axes:
   659→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   660→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   661→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   662→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   663→  // vectors — atan2(beta, gamma) does NOT match the original.)
   664→  //
   665→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   666→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   667→  // Default 45° (matches UISensor.kt's initial value).
   668→  React.useEffect(() => {
   669→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   670→    // Only listen for gravity angle on Control Center — it's the only page
   671→    // that uses it (el.useGravityAngle=true). Listening on all pages causes
   672→    // ~60 setGravityAngle calls/sec → 60 full WebGL renders/sec → high GPU.
   673→    if (destination !== CatalogDestination.ControlCenter) return
   674→    let smoothed = 45
   675→    const alpha = 0.5
   676→    const handler = (e: DeviceMotionEvent) => {
   677→      const acc = e.accelerationIncludingGravity
   678→      if (!acc || acc.x == null || acc.y == null) return
   679→      const x = acc.x
   680→      const y = acc.y
   681→      let target = Math.atan2(y, x) * 180 / Math.PI
   682→      let delta = target - smoothed
   683→      while (delta > 180) delta -= 360
   684→      while (delta < -180) delta += 360
   685→      smoothed += delta * alpha
   686→      while (smoothed > 180) smoothed -= 360
   687→      while (smoothed < -180) smoothed += 360
   688→      const rad = smoothed * Math.PI / 180
   689→      rendererRef.current?.setGravityAngle(rad)
   690→    }
   691→    window.addEventListener('devicemotion', handler)
   692→    return () => window.removeEventListener('devicemotion', handler)
   693→  }, [rendererRef, destination])

   782→  // AdaptiveLuminanceGlass: compute the average luminance of the WALLPAPER
   783→  // behind the glass region and animate state.adaptiveLuminance toward it.
   784→  //
   785→  // Faithful to AdaptiveLuminanceGlassContent.kt:

   787→  //   averageLuminance → luminanceAnimation.animateTo(target, tween(1000))

   789→  // The original reads the glass's rendered output (the backdrop WITH effects

   791→  // after compositing, so gl.readPixels on the canvas returns 0 (the bug that
   792→  // caused luminance to always be 0). Reading from a scene FBO is fragile
   793→  // (ping-pong state). Instead, we sample the WALLPAPER on the CPU via a
   794→  // hidden 2D canvas — this is the backdrop luminance (stable, no feedback
   795→  // divergence) and matches the original's intent of "how bright is the

    33→function useSystemTheme(): Theme {
    34→  const [theme, setTheme] = React.useState<Theme>('light')
    35→  React.useEffect(() => {
    36→    if (typeof window === 'undefined' || !window.matchMedia) return
    37→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    38→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    39→    update()
    40→    mq.addEventListener('change', update)
    41→    return () => mq.removeEventListener('change', update)
    42→  }, [])
    43→  return theme
    44→}

   305→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   306→    const MIN_FPS = 55
   307→    const MIN_DPR = 0.5 // slider minimum
   308→    if (fps >= MIN_FPS) {
   309→      // This DPR works — set lo to it (we know it's viable)
   310→      perfLoRef.current = candidateDpr
   311→    } else {
   312→      // Too slow — set hi to it (search lower)
   313→      perfHiRef.current = candidateDpr
   314→    }
   315→
   316→    // Special case: iteration 2 tested the minimum (0.5) and it failed.
   317→    // Performance is truly bad — force bestDpr = 0.5 and finish immediately.
   318→    if (iteration === 2 && candidateDpr === MIN_DPR && fps < MIN_FPS) {
   319→      perfPhaseRef.current = 'done'
   320→      const statusText = `检测完成 · 推荐 DPR：${MIN_DPR} · 性能有限，已自动降低画质`
   321→      try { window.localStorage.setItem(PERF_KEY, String(MIN_DPR)) } catch {}
   322→      setState({
   323→        customDpr: MIN_DPR,
   324→        perfDone: true,
   325→        perfResultDpr: MIN_DPR,
   326→        perfStatusText: statusText,
   327→        perfProgress: null,
   328→        perfProgressFrac: 1,
   329→      })
   330→      return
   331→    }
   332→
   333→    // Convergence: adjacent slider values (hi-lo ≤ 0.25) or max iterations reached.
   334→    // Both lo and hi are always on 0.25 steps after extremes are tested,
   335→    // so stalling cannot occur — the midpoint always lands strictly between them.
   336→    if (iteration >= maxIterations || (perfHiRef.current - perfLoRef.current) <= 0.25) {
   337→      // Converged — lo is the highest DPR that passed (already on 0.25 steps)
   338→      const bestDpr = Math.max(MIN_DPR, Math.min(deviceDpr, perfLoRef.current))
   339→      perfPhaseRef.current = 'done'
   340→      // isGood: recommended DPR is at least 75% of deviceDpr
   341→      const isGood = bestDpr >= deviceDpr * 0.75
   342→      const statusText = isGood
   343→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   344→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   345→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   346→      setState({
   347→        customDpr: bestDpr,
   348→        perfDone: true,
   349→        perfResultDpr: bestDpr,
   350→        perfStatusText: statusText,
   351→        perfProgress: null,
   352→        // Don't reset perfGlassAngle — settle animation will handle it
   353→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   354→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   355→        perfProgressFrac: 1,
   356→      })
   357→    } else {
   358→      // More iterations needed — increment trigger to re-fire the effect
   359→      perfPhaseRef.current = 'idle'
   360→      setState((prev) => ({
   361→        perfProgress: 'running',
   362→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   363→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   364→      }))
   365→    }
   366→  }

   391→  // Glass deformation animation — 16 glasses deform simultaneously.
   392→  // perfGlassAngle is stored in state; each glass in the 4×4 grid computes
   393→  // its own W/H from (angle + phaseOffset). Rotation speed: 1 circle/sec.
   394→  // Also smoothly animates perfProgressFracAnimated toward perfProgressFrac

   396→  React.useEffect(() => {

   402→    const PROG_LERP_SPEED = 4 // progress bar lerp speed (fraction per second, mimics ~600ms CSS ease-out)
   403→    let angle = state.perfGlassAngle || 0
   404→    let progAnimated = state.perfProgressFracAnimated ?? 0
   405→    let lastT = 0
   406→    let raf = 0

   414→      const target = perfProgressFracRef.current

   416→      setState({ perfGlassAngle: angle, perfProgressFracAnimated: progAnimated })
   417→      raf = requestAnimationFrame(tick)
   418→    }
   419→    raf = requestAnimationFrame(tick)
   420→    return () => cancelAnimationFrame(raf)
   421→  }, [destination, state.perfProgress, rendererReady])

   423→  // Settle animation — after benchmark ends, smoothly transition glasses
   424→  // from deformed to square (perfDeformMul decays 1→0) and animate the
   425→  // exit button sliding in (perfExitProgress 0→1 with ease-out).
   426→  // Also animates perfProgressFracAnimated to 1.0.
   427→  React.useEffect(() => {
   428→    if (destination !== CatalogDestination.PerfBenchmark) return
   429→    if (!rendererReady) return
   430→    if (!state.perfDone) return
   431→    // Already settled — no need to animate
   432→    if (state.perfDeformMul <= 0.01 && state.perfExitProgress >= 0.99 && (state.perfProgressFracAnimated ?? 0) >= 0.99) return
   433→
   434→    const ORBIT_SPEED = 2 * Math.PI // keep rotation going during settle
   435→    const PROG_LERP_SPEED = 4 // progress bar lerp speed
   436→    let angle = state.perfGlassAngle || 0
   437→    let progAnimated = state.perfProgressFracAnimated ?? 0
   438→    let elapsed = 0
   439→    let lastT = 0
   440→    let raf = 0
   441→    // Exponential decay rate for deformMul: mul = e^(-3*t), reaches ~0.05 by 1s
   442→    const DECAY_RATE = 3
   443→    // Exit button: starts appearing after 0.3s, fully visible by 0.8s
   444→    const EXIT_DELAY = 0.3
   445→    const EXIT_DURATION = 0.5
   446→
   447→    const tick = (timestamp: number) => {
   448→      if (!lastT) lastT = timestamp
   449→      const dt = Math.min(0.05, (timestamp - lastT) / 1000)
   450→      lastT = timestamp
   451→      elapsed += dt
   452→
   453→      // Continue rotation (so glasses don't freeze at a weird angle)
   454→      angle += ORBIT_SPEED * dt
   455→
   456→      // Exponential decay: deformMul goes from 1 → ~0 over ~1 second
   457→      const mul = Math.exp(-DECAY_RATE * elapsed)
   458→
   459→      // Exit button: ease-out quadratic after delay
   460→      let exitProg = 0
   461→      if (elapsed > EXIT_DELAY) {
   462→        const t = Math.min(1, (elapsed - EXIT_DELAY) / EXIT_DURATION)
   463→        exitProg = 1 - (1 - t) * (1 - t) // ease-out
   464→      }
   465→
   466→      // Smoothly animate progress to 1.0 (use ref for latest value)
   467→      const progTarget = perfProgressFracRef.current
   468→      progAnimated += (progTarget - progAnimated) * Math.min(1, PROG_LERP_SPEED * dt)

   476→      }

   650→  // Device motion → gravity angle for glass highlight direction.
   651→  // Faithful to UISensor.kt:
   652→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   653→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   654→  //
   655→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   656→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   657→  // rebuilding the catalog (which would lose drag state and cause jank).
   658→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   659→  // renderer.gravityAngle live each frame.
   660→  //
   661→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   662→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   663→  // device's acceleration INCLUDING gravity in device-space axes:
   664→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   665→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   666→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   667→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   668→  // vectors — atan2(beta, gamma) does NOT match the original.)
   669→  //
   670→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   671→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   672→  // Default 45° (matches UISensor.kt's initial value).
   673→  React.useEffect(() => {
   674→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   675→    let smoothed = 45
   676→    const alpha = 0.5
   677→    const handler = (e: DeviceMotionEvent) => {
   678→      const acc = e.accelerationIncludingGravity
   679→      if (!acc || acc.x == null || acc.y == null) return
   680→      const x = acc.x
   681→      const y = acc.y
   682→      let target = Math.atan2(y, x) * 180 / Math.PI
   683→      // Shortest-path interpolation: wrap the delta to [-180, 180] so
   684→      // rotating through ±180° doesn't cause a 358° jump.
   685→      let delta = target - smoothed
   686→      while (delta > 180) delta -= 360
   687→      while (delta < -180) delta += 360
   688→      smoothed += delta * alpha
   689→      // Wrap smoothed to [-180, 180] to keep the value bounded.
   690→      while (smoothed > 180) smoothed -= 360
   691→      while (smoothed < -180) smoothed += 360
   692→      const rad = smoothed * Math.PI / 180
   693→      rendererRef.current?.setGravityAngle(rad)
   694→    }
   695→    window.addEventListener('devicemotion', handler)
   696→    return () => window.removeEventListener('devicemotion', handler)
   697→  }, [rendererRef])

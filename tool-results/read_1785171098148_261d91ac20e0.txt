    36→/* ------------------------------------------------------------------ *
    37→ * LiquidGlassRenderer
    38→ *
    39→ * One opaque WebGL canvas. Render pipeline per frame:
    40→ *   1. Wallpaper background pass (cover-fit).
    41→ *   2. For each button (in order):
    42→ *      a. Outer drop shadow pass (if configured).
    43→ *      b. Element pass (refraction + vibrancy + tint + highlight),
    44→ *         with InteractiveHighlight-driven scale/translation/stretch.
    45→ *      c. White overlay pass (Plus blend, 8% * progress) on press.
    46→ *      d. Radial highlight pass (Plus blend, at finger position) on press.
    47→ *      e. Foreground pass (button label, with press-driven alpha fade).
    48→ *
    49→ * No DOM children — the canvas owns the entire visual surface.
    50→ *
    51→ * NOTE: The class methods are split across multiple files in this
    52→ * directory (methods-*.ts). Each methods file declares its method
    53→ * signatures via `declare module './index'` and exports a record of
    54→ * named functions. Those records are merged onto the prototype at the
    55→ * bottom of this file via Object.assign.
    56→ * ------------------------------------------------------------------ */

   395→      'uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii',
   396→      'uStrokeMask', 'uMaskOffset', 'uMaskSize',
   397→      'uHighlightColor', 'uHighlightAngle', 'uHighlightFalloff', 'uHighlightAlpha', 'uHighlightMode',
   398→      'uOriginalSize', 'uOriginalCornerRadius', 'uLayerScale', 'uElementRotation',
   399→    ]
   400→    for (const n of smNames) this.uSm[n] = gl.getUniformLocation(this.strokeMaskCompositeProgram, n)
   401→    // Inner shadow mask composite (Canvas2D ring mask approach)
   402→    const isNames = [
   403→      'uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii',
   404→      'uInnerShadowMask', 'uMaskOffset', 'uMaskSize',
   405→      'uInnerShadowColor', 'uInnerShadowAlpha',
   406→      'uOriginalSize', 'uOriginalCornerRadius', 'uLayerScale', 'uElementRotation',
   407→    ]
   408→    for (const n of isNames) this.uIs[n] = gl.getUniformLocation(this.innerShadowMaskCompositeProgram, n)
   409→    const prNames = ['uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii', 'uColor', 'uCornerStyle',
   410→      'uUseContinuousSdf', 'uContinuousSdf', 'uContinuousSdfTexSize', 'uContinuousSdfElementSize']
   411→    for (const n of prNames) this.uPr[n] = gl.getUniformLocation(this.plainRectProgram, n)
   412→    const pbNames = [
   413→      'uBackdrop', 'uCanvasSize', 'uWallpaperSize', 'uOffset', 'uSize',
   414→      'uBlurRadius', 'uTintColor', 'uTintIntensity',
   415→    ]
   416→    for (const n of pbNames) this.uPb[n] = gl.getUniformLocation(this.progressiveBlurProgram, n)
   417→    const cpNames = ['uTexture', 'uCanvasSize']
   418→    for (const n of cpNames) this.uCp[n] = gl.getUniformLocation(this.copyProgram, n)
   419→    const sfNames = ['uColor']
   420→    for (const n of sfNames) this.uSf[n] = gl.getUniformLocation(this.solidFillProgram, n)
   421→    const ccNames = ['uTexture', 'uTexSize', 'uBrightness', 'uContrast', 'uSaturation']
   422→    for (const n of ccNames) this.uCc[n] = gl.getUniformLocation(this.colorControlsProgram, n)
   423→    const stNames = ['uTexture', 'uCanvasSize', 'uTintColor']
   424→    for (const n of stNames) this.uSt[n] = gl.getUniformLocation(this.sceneTintProgram, n)
   425→  }
   426→
   427→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   428→  ensureBlurPrograms(tapCount: number): void {
   429→    if (this.blurPrograms.has(tapCount)) return
   430→    const gl = this.gl
   431→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   432→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   433→    const mk = (fs: WebGLShader) => {
   434→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   435→      const p = gl.createProgram()!
   436→      gl.attachShader(p, vs)
   437→      gl.attachShader(p, fs)
   438→      gl.bindAttribLocation(p, 0, 'aPos')
   439→      gl.linkProgram(p)
   440→      gl.deleteShader(vs)
   441→      gl.deleteShader(fs)
   442→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   443→        const log = gl.getProgramInfoLog(p)
   444→        gl.deleteProgram(p)
   445→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   446→      }
   447→      return p
   448→    }
   449→    const hProg = mk(hFs)
   450→    const vProg = mk(vFs)
   451→    const uH: Record<string, WebGLUniformLocation | null> = {
   452→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   453→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   454→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   455→    }
   456→    const uV: Record<string, WebGLUniformLocation | null> = {
   457→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   458→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   459→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   460→    }
   461→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   462→  }
   463→
   464→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   465→   *  blurred result into blurFboB, returns blurFboBTex.
   466→   *  Saves/restores the currently-bound framebuffer.
   467→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   468→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   469→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   470→    const gl = this.gl
   471→    const w = this.fboW
   472→    const h = this.fboH
   473→    // Compute tap count, capped by blurTapCap (performance knob).
   474→    let taps = computeBlur1DTapCount(radius)
   475→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   476→    this.ensureBlurPrograms(taps)
   477→    const entry = this.blurPrograms.get(taps)!
   478→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   479→    gl.disable(gl.BLEND)
   480→
   481→    // Pass 1: horizontal — srcTex → blurFboA
   482→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   483→    gl.viewport(0, 0, w, h)
   484→    gl.useProgram(entry.hProg)
   485→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   486→    gl.enableVertexAttribArray(entry.aPosH)
   487→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   488→    gl.activeTexture(gl.TEXTURE0)
   489→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   490→    gl.uniform1i(entry.uH['uTexture'], 0)
   491→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   492→    gl.uniform1f(entry.uH['uRadius'], radius)
   493→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   494→
   495→    // Pass 2: vertical — blurFboATex → blurFboB
   496→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   497→    gl.viewport(0, 0, w, h)
   498→    gl.useProgram(entry.vProg)
   499→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   500→    gl.enableVertexAttribArray(entry.aPosV)
   501→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   502→    gl.activeTexture(gl.TEXTURE0)
   503→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   504→    gl.uniform1i(entry.uV['uTexture'], 0)
   505→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   506→    gl.uniform1f(entry.uV['uRadius'], radius)
   507→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   508→
   509→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   510→    gl.viewport(0, 0, w, h)
   511→    return this.blurFboBTex!
   512→  }
   513→
   514→  /** Lazy-compile highlight blur programs (alpha-blurring, sigma semantics).
   515→   *  Separate from ensureBlurPrograms because the shader is different
   516→   *  (blurs alpha, no early-return, integer-σ-spaced taps). */
   517→  ensureHighlightBlurPrograms(tapCount: number): void {
   518→    if (this.highlightBlurPrograms.has(tapCount)) return
   519→    const gl = this.gl
   520→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateHighlightBlurShader(tapCount, 'horizontal'))
   521→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateHighlightBlurShader(tapCount, 'vertical'))
   522→    const mk = (fs: WebGLShader) => {

   524→      const p = gl.createProgram()!

   538→    const hProg = mk(hFs)
   539→    const vProg = mk(vFs)
   540→    const uH: Record<string, WebGLUniformLocation | null> = {
   541→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),

   553→  /** 2-pass Gaussian blur on a highlight stroke MASK (alpha only).
   554→   *  Faithful to Android BlurMaskFilter(NORMAL, sigma):
   555→   *    - sigma = blurRadiusPx (the Android radius param IS sigma)
   556→   *    - convolves the mask's ALPHA with a Gaussian kernel
   557→   *    - sub-pixel sigma (0.25px) still blurs (no 0.5 early-return)
   558→   *  Reads srcTex (alpha mask), writes blurFboB, returns blurFboBTex.
   559→   *  Saves/restores the currently-bound framebuffer. */
   560→  blurHighlightMask(srcTex: WebGLTexture, sigmaPx: number): WebGLTexture {
   561→    const gl = this.gl
   562→    const w = this.fboW
   563→    const h = this.fboH
   564→    let taps = computeHighlightBlurTapCount(sigmaPx)
   565→    taps = Math.min(taps, Math.max(3, this.blurTapCap | 0))
   566→    this.ensureHighlightBlurPrograms(taps)
   567→    const entry = this.highlightBlurPrograms.get(taps)!
   568→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   569→    gl.disable(gl.BLEND)
   570→
   571→    // Pass 1: horizontal — srcTex → blurFboA
   572→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   573→    gl.viewport(0, 0, w, h)
   574→    gl.useProgram(entry.hProg)
   575→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   576→    gl.enableVertexAttribArray(entry.aPosH)
   577→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   578→    gl.activeTexture(gl.TEXTURE0)
   579→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   580→    gl.uniform1i(entry.uH['uTexture'], 0)
   581→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   582→    gl.uniform1f(entry.uH['uRadius'], sigmaPx)
   583→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   584→
   585→    // Pass 2: vertical — blurFboATex → blurFboB
   586→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   587→    gl.viewport(0, 0, w, h)
   588→    gl.useProgram(entry.vProg)
   589→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   590→    gl.enableVertexAttribArray(entry.aPosV)
   591→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   592→    gl.activeTexture(gl.TEXTURE0)
   593→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   594→    gl.uniform1i(entry.uV['uTexture'], 0)
   595→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   596→    gl.uniform1f(entry.uV['uRadius'], sigmaPx)
   597→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   598→
   599→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   600→    gl.viewport(0, 0, w, h)
   601→    return this.blurFboBTex!
   602→  }

   604→  dispose() {
   605→    if (this.rafId !== null) cancelAnimationFrame(this.rafId)
   606→    this.rafId = null
   607→    if (this.animRafId !== null) cancelAnimationFrame(this.animRafId)
   608→    this.animRafId = null
   609→    const gl = this.gl
   610→    if (this.wallpaperTexture) gl.deleteTexture(this.wallpaperTexture)
   611→    for (const tex of this.fgTextures.values()) gl.deleteTexture(tex)
   612→    this.fgTextures.clear()
   613→    for (const entry of this.strokeMaskCache.values()) gl.deleteTexture(entry.tex)
   614→    this.strokeMaskCache.clear()
   615→    destroyCache(gl, this.innerShadowMaskCache)
   616→    if (this.fboA) gl.deleteFramebuffer(this.fboA)
   617→    if (this.fboATex) gl.deleteTexture(this.fboATex)
   618→    if (this.fboB) gl.deleteFramebuffer(this.fboB)
   619→    if (this.fboBTex) gl.deleteTexture(this.fboBTex)
   620→    this.fboA = this.fboB = null
   621→    this.fboATex = this.fboBTex = null
   622→    if (this.tabsBackdropFbo) gl.deleteFramebuffer(this.tabsBackdropFbo)
   623→    if (this.tabsBackdropTex) gl.deleteTexture(this.tabsBackdropTex)
   624→    this.tabsBackdropFbo = null
   625→    this.tabsBackdropTex = null
   626→    // GP element FBO + blur FBOs + programs
   627→    if (this.gpElementFbo) gl.deleteFramebuffer(this.gpElementFbo)
   628→    if (this.gpElementTex) gl.deleteTexture(this.gpElementTex)
   629→    if (this.blurFboA) gl.deleteFramebuffer(this.blurFboA)
   630→    if (this.blurFboATex) gl.deleteTexture(this.blurFboATex)
   631→    if (this.blurFboB) gl.deleteFramebuffer(this.blurFboB)
   632→    if (this.blurFboBTex) gl.deleteTexture(this.blurFboBTex)
   633→    this.gpElementFbo = this.blurFboA = this.blurFboB = null
   634→    this.gpElementTex = this.blurFboATex = this.blurFboBTex = null
   635→    if (this.highlightMaskFbo) gl.deleteFramebuffer(this.highlightMaskFbo)
   636→    if (this.highlightMaskTex) gl.deleteTexture(this.highlightMaskTex)
   637→    this.highlightMaskFbo = null
   638→    this.highlightMaskTex = null
   639→    if (this.dialogBackdropFbo) gl.deleteFramebuffer(this.dialogBackdropFbo)
   640→    if (this.dialogBackdropTex) gl.deleteTexture(this.dialogBackdropTex)
   641→    this.dialogBackdropFbo = null
   642→    this.dialogBackdropTex = null
   643→    this.dialogBackdropKey = null
   644→    for (const { hProg, vProg } of this.blurPrograms.values()) {
   645→      gl.deleteProgram(hProg)
   646→      gl.deleteProgram(vProg)
   647→    }
   648→    this.blurPrograms.clear()
   649→    for (const { hProg, vProg } of this.highlightBlurPrograms.values()) {
   650→      gl.deleteProgram(hProg)
   651→      gl.deleteProgram(vProg)
   652→    }
   653→    this.highlightBlurPrograms.clear()
   654→    if (this.sdfTexture) gl.deleteTexture(this.sdfTexture)
   655→    this.sdfTexture = null
   656→    for (const { tex } of this.continuousSdfPool.values()) gl.deleteTexture(tex)
   657→    this.continuousSdfPool.clear()
   658→    this.continuousSdfTexture = null
   659→    this.continuousSdfKey = null
   660→    gl.deleteProgram(this.elementProgram)
   661→    gl.deleteProgram(this.shadowProgram)
   662→    gl.deleteProgram(this.wallpaperProgram)
   663→    gl.deleteProgram(this.foregroundProgram)
   664→    gl.deleteProgram(this.highlightProgram)
   665→    gl.deleteProgram(this.tintProgram)
   666→    gl.deleteProgram(this.rimHighlightProgram)
   667→    gl.deleteProgram(this.highlightStrokeProgram)
   668→    gl.deleteProgram(this.highlightCompositeProgram)
   669→    gl.deleteProgram(this.strokeMaskCompositeProgram)
   670→    gl.deleteProgram(this.innerShadowMaskCompositeProgram)
   671→    gl.deleteProgram(this.plainRectProgram)
   672→    gl.deleteProgram(this.progressiveBlurProgram)
   673→    gl.deleteProgram(this.copyProgram)
   674→    gl.deleteProgram(this.solidFillProgram)
   675→    gl.deleteProgram(this.colorControlsProgram)
   676→    gl.deleteProgram(this.sceneTintProgram)
   677→    gl.deleteBuffer(this.quadBuffer)
   678→  }

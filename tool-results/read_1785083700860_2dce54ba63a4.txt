     1→     1→import type { LiquidGlassRenderer } from './index'
     2→     2→import type { GlassRenderState } from './methods-render-glass'
     3→     3→import { DP } from './spring'
     4→     4→import { continuousCurvatureRoundedRectPath } from './continuous-curve'
     5→     5→
     6→     6→declare module './index' {
     7→     7→  interface LiquidGlassRenderer {
     8→     8→    renderGlassPostPasses(state: GlassRenderState): void
     9→     9→  }
    10→    10→}
    11→    11→
    12→    12→export const glassPostPassMethods = {
    13→    13→  /** Steps 2b–2f: Inner shadow, press glow, white overlay, foreground, rim highlight.
    14→    14→   *  These all composite on top of the glass body (already drawn to
    15→    15→   *  otherFbo by renderGlassElementPass). */
    16→    16→  renderGlassPostPasses(this: LiquidGlassRenderer, state: GlassRenderState) {
    17→    17→    const gl = this.gl
    18→    18→    const { el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress, elHighlightAlpha } = state
    19→    19→
    20→    20→    // Original-space SDF uniforms — shared by all post-pass shaders so their
    21→    21→    // capsule clip is computed in ORIGINAL space (faithful to graphicsLayer
    22→    22→    // { scaleX, scaleY } post-scaling). See element.ts / highlight.ts.
    23→    23→    const origSizeX = state.origW * this.dpr
    24→    24→    const origSizeY = state.origH * this.dpr
    25→    25→    const origRadius = state.origCornerRadius * this.dpr
    26→    26→    const layerScaleX = state.layerScaleX
    27→    27→    const layerScaleY = state.layerScaleY
    28→    28→
    29→    29→    // --- Step 2b: Inner shadow post-passes (Canvas2D ring mask approach) ---
    30→    30→    // Inner shadows are drawn on the element surface, underneath the press glow.
    31→    31→    // Each shadow uses a Canvas2D-generated blurred ring mask (fill shape →
    32→    32→    // destination-out offset shape → blur), composited via the
    33→    33→    // INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER with SrcOver blend.
    34→    34→    // SDF clip in the shader ensures the shadow stays inside the shape boundary.
    35→    35→
    36→    36→    /** Helper: draw one inner shadow post-pass (shadow1 or shadow2). */
    37→    37→    const drawInnerShadowPass = (
    38→    38→      shadowCfg: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] },
    39→    39→      shadowIndex: number // 0 = the only inner shadow (original has just ONE)
    40→    40→    ) => {
    41→    41→      // Progress modulation — faithful to the original inline shader:
    42→    42→      //   - Toggle knobs: radius, alpha, offset all modulated by togglePressProgress
    43→    43→      //   - Bottom-tab indicator: radius, alpha, offset modulated by togglePressProgress
    44→    44→      //   - Other elements (magnifier): static params, no modulation
    45→    45→      const progress = (el.isToggleKnob || el.isBottomTabIndicator) ? togglePressProgress : 1
    46→    46→      let shadowAlpha = shadowCfg.alpha * progress * state.enterAlpha
    47→    47→      let shadowRadius = shadowCfg.radius * progress
    48→    48→      let shadowOffsetX = shadowCfg.offsetX * progress
    49→    49→      let shadowOffsetY = shadowCfg.offsetY * progress
    50→    50→
    51→    51→      if (shadowAlpha <= 0.001 || shadowRadius <= 0.5) return
    52→    52→
    53→    53→      // Blur sigma = radius * dpr (BlurEffect semantics: sigma = radius directly).
    54→    54→      // The original InnerShadowModifier.kt uses BlurEffect(radius, radius, TileMode.Decal)
    55→    55→      // where the radius parameter IS the Gaussian sigma (NOT radius/3 like BlurMaskFilter).
    56→    56→      // BlurEffect wraps SkImageFilter::MakeBlur which takes sigma directly.
    57→    57→      const blurSigma = shadowRadius * this.dpr  // device px — sigma = radius, not radius/3
    58→    58→      // Margin for blur spread (3σ) + AA
    59→    59→      const margin = Math.ceil(blurSigma * 3) + 2
    60→    60→      // Mask dimensions in device px (origSize + 2*margin)
    61→    61→      const maskW = Math.max(1, Math.ceil(origSizeX + 2 * margin))
    62→    62→      const maskH = Math.max(1, Math.ceil(origSizeY + 2 * margin))
    63→    63→      // Supersampling for sharper mask rasterization
    64→    64→      const deviceDpr = window.devicePixelRatio || 1
    65→    65→      const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
    66→    66→      const canvasW = maskW * SS
    67→    67→      const canvasH = maskH * SS
    68→    68→      const useG2 = !!el.useContinuousSdf
    69→    69→
    70→    70→      // Offset in device px (already × progress)
    71→    71→      const offsetXDp = shadowOffsetX * this.dpr
    72→    72→      const offsetYDp = shadowOffsetY * this.dpr
    73→    73→
    74→    74→      const maskKey = [
    75→    75→        'is',
    76→    76→        shadowIndex,
    77→    77→        useG2 ? 'g2' : 'rr',
    78→    78→        origSizeX.toFixed(3),
    79→    79→        origSizeY.toFixed(3),
    80→    80→        origRadius.toFixed(3),
    81→    81→        offsetXDp.toFixed(3),
    82→    82→        offsetYDp.toFixed(3),
    83→    83→        blurSigma.toFixed(3),
    84→    84→        margin,
    85→    85→        maskW,
    86→    86→        maskH,
    87→    87→        `ss${SS}`,
    88→    88→      ].join(':')
    89→    89→
    90→    90→      let mask = this.innerShadowMaskCache.get(maskKey)
    91→    91→      if (!mask) {
    92→    92→        const canvas = document.createElement('canvas')
    93→    93→        canvas.width = canvasW
    94→    94→        canvas.height = canvasH
    95→    95→        const ctx = canvas.getContext('2d', { alpha: true })
    96→    96→        if (!ctx) throw new Error('2D canvas not supported')
    97→    97→        const tex = gl.createTexture()
    98→    98→        if (!tex) throw new Error('WebGL texture allocation failed')
    99→    99→        mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   100→   100→        this.innerShadowMaskCache.set(maskKey, mask)
   101→   101→
   102→   102→        // Keep the cache bounded (32 entries)
   103→   103→        if (this.innerShadowMaskCache.size > 32) {
   104→   104→          const oldestKey = this.innerShadowMaskCache.keys().next().value as string | undefined
   105→   105→          if (oldestKey && oldestKey !== maskKey) {
   106→   106→            const oldest = this.innerShadowMaskCache.get(oldestKey)
   107→   107→            if (oldest) gl.deleteTexture(oldest.tex)
   108→   108→            this.innerShadowMaskCache.delete(oldestKey)
   109→   109→          }
   110→   110→        }
   111→   111→      }
   112→   112→
   113→   113→      if (!mask.ready) {
   114→   114→        // --- Generate the ring mask via Canvas2D (two-canvas approach) ---
   115→   115→        // Step 1: draw hard-edge ring on a temp canvas (with SS× supersampling)
   116→   116→        // Step 2: blur the ring onto the output canvas at 1:1 physical pixels
   117→   117→        //         (no ctx.scale — drawImage and blur must be in physical-px space)
   118→   118→        //   - blur radius = blurSigma * SS physical px (= blurSigma logical px)
   119→   119→        //   - drawImage at 1:1 physical (canvasW × canvasH → canvasW × canvasH)
   120→   120→
   121→   121→        // --- Temp canvas for the hard-edge ring ---
   122→   122→        const tempCanvas = document.createElement('canvas')
   123→   123→        tempCanvas.width = canvasW
   124→   124→        tempCanvas.height = canvasH
   125→   125→        const tCtx = tempCanvas.getContext('2d', { alpha: true })
   126→   126→        if (!tCtx) throw new Error('2D canvas not supported')
   127→   127→
   128→   128→        // Draw ring on temp canvas: clip → fill shape → destination-out offset shape
   129→   129→        // Faithful to InnerShadowModifier.kt shadowLayer.record:
   130→   130→        //   canvas.clipOutline(outline, clipPath)    // clip to shape FIRST
   131→   131→        //   canvas.drawOutline(outline, paint)        // shadow-colored fill
   132→   132→        //   canvas.translate(offsetX, offsetY)
   133→   133→        //   canvas.drawOutline(outline, ShadowMaskPaint) // BlendMode.Clear
   134→   134→        // The clip ensures the ring is strictly inside the shape boundary,
   135→   135→        // and the destination-out only affects pixels within the clip.
   136→   136→        // Without this clip, subtle AA differences at the shape edge can
   137→   137→        // cause the inner shadow to not "严丝合缝" (not tight-fitting).
   138→   138→        tCtx.save()
   139→   139→        tCtx.scale(SS, SS)
   140→   140→        tCtx.translate(margin, margin)
   141→   141→
   142→   142→        // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   143→   143→        let path: Path2D
   144→   144→        if (useG2) {
   145→   145→          path = continuousCurvatureRoundedRectPath(tCtx, origSizeX, origSizeY, origRadius)
   146→   146→        } else {
   147→   147→          path = new Path2D()
   148→   148→          const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   149→   149→          path.moveTo(r, 0)
   150→   150→          path.lineTo(origSizeX - r, 0)
   151→   151→          path.arcTo(origSizeX, 0, origSizeX, r, r)
   152→   152→          path.lineTo(origSizeX, origSizeY - r)
   153→   153→          path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   154→   154→          path.lineTo(r, origSizeY)
   155→   155→          path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   156→   156→          path.lineTo(0, r)
   157→   157→          path.arcTo(0, 0, r, 0, r)
   158→   158→          path.closePath()
   159→   159→        }
   160→   160→
   161→   161→        // Clip to the shape FIRST (faithful to InnerShadowModifier.kt clipOutline)
   162→   162→        tCtx.clip(path)
   163→   163→
   164→   164→        // Fill the shape with white (creates full interior)
   165→   165→        tCtx.globalCompositeOperation = 'source-over'
   166→   166→        tCtx.fillStyle = 'white'
   167→   167→        tCtx.fill(path)
   168→   168→
   169→   169→        // Draw the OFFSET shape with destination-out (removes offset interior)
   170→   170→        tCtx.globalCompositeOperation = 'destination-out'
   171→   171→        tCtx.save()
   172→   172→        tCtx.translate(offsetXDp, offsetYDp)
   173→   173→        tCtx.fill(path) // same path, shifted by offset
   174→   174→        tCtx.restore()
   175→   175→
   176→   176→        tCtx.globalCompositeOperation = 'source-over'
   177→   177→        tCtx.restore()
   178→   178→
   179→   179→        // --- Blur the ring onto the output canvas (1:1 physical pixels) ---
   180→   180→        // NO ctx.scale here — drawImage and ctx.filter blur must operate in
   181→   181→        // physical-pixel space to avoid SS× size distortion.
   182→   182→        //   blurSigma is in device px (logical); the SS× canvas has SS physical px
   183→   183→        //   per logical px, so the physical blur radius = blurSigma * SS.
   184→   184→        const mCtx = mask.ctx
   185→   185→        mCtx.clearRect(0, 0, canvasW, canvasH)
   186→   186→        if (blurSigma > 0.01) {
   187→   187→          mCtx.filter = `blur(${blurSigma * SS}px)`
   188→   188→        } else {
   189→   189→          mCtx.filter = 'none'
   190→   190→        }
   191→   191→        // 1:1 physical-pixel drawImage: temp(canvasW×canvasH) → mask(canvasW×canvasH)
   192→   192→        mCtx.drawImage(tempCanvas, 0, 0)
   193→   193→        mCtx.filter = 'none'
   194→   194→
   195→   195→        // Upload to GPU texture
   196→   196→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   197→   197→        gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   198→   198→        gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   199→   199→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   200→   200→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   201→   201→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   202→   202→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   203→   203→        mask.ready = true
   204→   204→      }
   205→   205→
   206→   206→      // --- Composite: inner shadow mask × shadowAlpha × shadowColor → scene ---
   207→   207→      gl.enable(gl.BLEND)
   208→   208→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   209→   209→      gl.useProgram(this.innerShadowMaskCompositeProgram)
   210→   210→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   211→   211→      gl.enableVertexAttribArray(this.aPosLocIs)
   212→   212→      gl.vertexAttribPointer(this.aPosLocIs, 2, gl.FLOAT, false, 0, 0)
   213→   213→
   214→   214→      gl.uniform2f(this.uIs['uCanvasSize'], this.canvas.width, this.canvas.height)
   215→   215→      gl.uniform2f(this.uIs['uOffset'], sx * this.dpr, sy * this.dpr)
   216→   216→      gl.uniform2f(this.uIs['uSize'], sw * this.dpr, sh * this.dpr)
   217→   217→      gl.uniform4f(this.uIs['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   218→   218→
   219→   219→      gl.activeTexture(gl.TEXTURE0)
   220→   220→      gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   221→   221→      gl.uniform1i(this.uIs['uInnerShadowMask'], 0)
   222→   222→      // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   223→   223→      // physical canvas is SS× larger but the shader uses 1x coords for UV.
   224→   224→      gl.uniform2f(this.uIs['uMaskOffset'], margin, margin)
   225→   225→      gl.uniform2f(this.uIs['uMaskSize'], mask.w, mask.h)
   226→   226→
   227→   227→      // Shadow color (defaults to black [0,0,0] if not specified)
   228→   228→      const color = shadowCfg.color ?? [0, 0, 0]
   229→   229→      gl.uniform3f(this.uIs['uInnerShadowColor'], color[0], color[1], color[2])
   230→   230→      gl.uniform1f(this.uIs['uInnerShadowAlpha'], shadowAlpha)
   231→   231→
   232→   232→      gl.uniform2f(this.uIs['uOriginalSize'], origSizeX, origSizeY)
   233→   233→      gl.uniform1f(this.uIs['uOriginalCornerRadius'], origRadius)
   234→   234→      gl.uniform2f(this.uIs['uLayerScale'], layerScaleX, layerScaleY)
   235→   235→      gl.uniform1f(this.uIs['uElementRotation'], state.elementRotation)
   236→   236→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   237→   237→    }
   238→   238→
   239→   239→    // Inner shadow (faithful to the original — only ONE black inner shadow,
   240→   240→    // no innerShadow2. LiquidToggle.kt: InnerShadow(radius=4dp*progress, alpha=progress))
   241→   241→    if (el.innerShadow) {
   242→   242→      drawInnerShadowPass(el.innerShadow, 0)
   243→   243→    }
   244→   244→
   245→   245→    // --- Step 2c: Press glow (button + bottom-tab container) ---
   246→   246→    // Faithful to InteractiveHighlight.kt: a flat white Plus-blend overlay
   247→   247→    // + a radial white glow. For buttons, position = finger, intensity = st.pressProgress.
   248→   248→    // For the bottom-tab container, position = indicator center (via the
   249→   249→    // original's position lambda), intensity = toggleState.pressProgress.
   250→   250→    const isContainer = !!el.isBottomTabContainer
   251→   251→    const glowP = isButton ? p : (isContainer ? togglePressProgress : 0)
   252→   252→    if ((isButton && el.isInteractive && st && p > 0.001) || (isContainer && togglePressProgress > 0.001)) {
   253→   253→      // a. Flat white overlay
   254→   254→      gl.useProgram(this.tintProgram)
   255→   255→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   256→   256→      gl.enableVertexAttribArray(this.aPosLocTn)
   257→   257→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   258→   258→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   259→   259→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   260→   260→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   261→   261→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   262→   262→      gl.uniform4f(
   263→   263→        this.uTn['uCornerRadii'],
   264→   264→        radii[0] * this.dpr,
   265→   265→        radii[1] * this.dpr,
   266→   266→        radii[2] * this.dpr,
   267→   267→        radii[3] * this.dpr
   268→   268→      )
   269→   269→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   270→   270→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   271→   271→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   272→   272→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   273→   273→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   274→   274→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.08 * glowP)
   275→   275→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   276→   276→
   277→   277→      // b. Radial highlight at finger position (button) or indicator center (container).
   278→   278→      gl.useProgram(this.highlightProgram)
   279→   279→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   280→   280→      gl.enableVertexAttribArray(this.aPosLocHl)
   281→   281→      gl.vertexAttribPointer(this.aPosLocHl, 2, gl.FLOAT, false, 0, 0)
   282→   282→      gl.blendFunc(gl.ONE, gl.ONE)
   283→   283→      gl.uniform2f(this.uHl['uCanvasSize'], this.canvas.width, this.canvas.height)
   284→   284→      gl.uniform2f(this.uHl['uOffset'], sx * this.dpr, sy * this.dpr)
   285→   285→      gl.uniform2f(this.uHl['uSize'], sw * this.dpr, sh * this.dpr)
   286→   286→      gl.uniform4f(
   287→   287→        this.uHl['uCornerRadii'],
   288→   288→        radii[0] * this.dpr,
   289→   289→        radii[1] * this.dpr,
   290→   290→        radii[2] * this.dpr,
   291→   291→        radii[3] * this.dpr
   292→   292→      )
   293→   293→      gl.uniform2f(this.uHl['uOriginalSize'], origSizeX, origSizeY)
   294→   294→      gl.uniform1f(this.uHl['uOriginalCornerRadius'], origRadius)
   295→   295→      gl.uniform2f(this.uHl['uLayerScale'], layerScaleX, layerScaleY)
   296→   296→      gl.uniform1f(this.uHl['uElementRotation'], state.elementRotation)
   297→   297→      gl.uniform1f(this.uHl['uCornerStyle'], this.cornerStyle)
   298→   298→      gl.uniform4f(this.uHl['uColor'], 1, 1, 1, 0.15 * glowP)
   299→   299→      const minDim = Math.min(sw, sh) * this.dpr
   300→   300→      gl.uniform1f(this.uHl['uRadius'], minDim * 1.5)
   301→   301→      // Position: for buttons, the finger position. For the container, the
   302→   302→      // indicator center — faithful to the original's position lambda which
   303→   303→      // returns (dampedDragAnimation.value + 0.5) * tabWidth (indicator center
   304→   304→      // relative to the container).
   305→   305→      let px: number, py: number
   306→   306→      if (isContainer) {
   307→   307→        const tg = this.toggleStates.get(el.isBottomTabContainer!.groupId)
   308→   308→        // tabW = container ORIGINAL width / tabsCount (not scaled — the position
   309→   309→        // lambda runs in pre-scale local coords). indicator center =
   310→   310→        // (fraction + 0.5) * tabW. Faithful to LiquidBottomTabs.kt:
   311→   311→        //   position = (dampedDragAnimation.value + 0.5) * tabWidth
   312→   312→        const tabsCount = el.isBottomTabContainer!.tabsCount ?? 4
   313→   313→        const tabW = el.rect.w / tabsCount
   314→   314→        const fraction = tg ? tg.fraction : 0
   315→   315→        const indCenterX = (fraction + 0.5) * tabW
   316→   316→        // Map original-local to scaled-local (the shader's uPosition is in
   317→   317→        // scaled-local coords, 0..sw). scale = sw / el.rect.w.
   318→   318→        const scaleToLocal = sw / el.rect.w
   319→   319→        px = Math.max(0, Math.min(sw, indCenterX * scaleToLocal)) * this.dpr
   320→   320→        py = (sh / 2) * this.dpr
   321→   321→      } else {
   322→   322→        px = Math.max(0, Math.min(sw, st!.dragX * state.layerScaleX)) * this.dpr
   323→   323→        py = Math.max(0, Math.min(sh, st!.dragY * state.layerScaleY)) * this.dpr
   324→   324→      }
   325→   325→      gl.uniform2f(this.uHl['uPosition'], px, py)
   326→   326→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   327→   327→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   328→   328→    }
   329→   329→
   330→   330→    // --- Step 2d: Toggle knob white overlay (faithful to LiquidToggle.kt
   331→   331→    // / LiquidSlider.kt onDrawSurface) ---
   332→   332→    if (el.isToggleKnob && togglePressProgress < 0.999) {
   333→   333→      const whiteAlpha = 1.0 * (1 - togglePressProgress)
   334→   334→      gl.useProgram(this.tintProgram)
   335→   335→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   336→   336→      gl.enableVertexAttribArray(this.aPosLocTn)
   337→   337→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   338→   338→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   339→   339→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   340→   340→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   341→   341→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   342→   342→      gl.uniform4f(
   343→   343→        this.uTn['uCornerRadii'],
   344→   344→        radii[0] * this.dpr,
   345→   345→        radii[1] * this.dpr,
   346→   346→        radii[2] * this.dpr,
   347→   347→        radii[3] * this.dpr
   348→   348→      )
   349→   349→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   350→   350→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   351→   351→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   352→   352→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   353→   353→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   354→   354→      // Faithful to LiquidToggle.kt onDrawSurface:
   355→   355→      //   drawRect(Color.White.copy(alpha = 1f - progress))
   356→   356→      // Solid white pebble at rest (alpha=1), fading to transparent when
   357→   357→      // pressed (alpha=0) to reveal the glass refraction beneath.
   358→   358→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, whiteAlpha)
   359→   359→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   360→   360→    }
   361→   361→
   362→   362→    // --- Step 2d2: Bottom tab indicator onDrawSurface (faithful to
   363→   363→    // LiquidBottomTabs.kt indicator):
   364→   364→    //   drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)
   365→   365→    //   drawRect(Color.Black.copy(alpha = 0.03f * progress))
   366→   366→    // First: theme-aware dim overlay fading OUT on press (SrcOver).
   367→   367→    // Second: subtle black tint fading IN on press (SrcOver).
   368→   368→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.dimColor) {
   369→   369→      const dc = el.isBottomTabIndicator.dimColor
   370→   370→      const p = togglePressProgress
   371→   371→      gl.useProgram(this.tintProgram)
   372→   372→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   373→   373→      gl.enableVertexAttribArray(this.aPosLocTn)
   374→   374→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   375→   375→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   376→   376→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   377→   377→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   378→   378→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   379→   379→      gl.uniform4f(this.uTn['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   380→   380→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   381→   381→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   382→   382→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   383→   383→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   384→   384→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   385→   385→      // First overlay: dim color at 0.1 * (1 - progress) — fades out on press.
   386→   386→      gl.uniform4f(this.uTn['uColor'], dc[0], dc[1], dc[2], 0.1 * (1 - p))
   387→   387→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   388→   388→      // Second overlay: black at 0.03 * progress — fades in on press.
   389→   389→      gl.uniform4f(this.uTn['uColor'], 0, 0, 0, 0.03 * p)
   390→   390→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   391→   391→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   392→   392→    }
   393→   393→
   394→   394→    // --- Step 2e: Foreground (label or icon) pass (button only) ---
   395→   395→    if (isButton && (el.label || el.icon)) {
   396→   396→      const fgTex = this.fgTextures.get(el.id)
   397→   397→      if (fgTex) {
   398→   398→        gl.useProgram(this.foregroundProgram)
   399→   399→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   400→   400→        gl.enableVertexAttribArray(this.aPosLocFg)
   401→   401→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   402→   402→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   403→   403→
   404→   404→        gl.activeTexture(gl.TEXTURE0)
   405→   405→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   406→   406→        gl.uniform1i(this.uFg['uTexture'], 0)
   407→   407→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   408→   408→        gl.uniform2f(this.uFg['uOffset'], sx * this.dpr, sy * this.dpr)
   409→   409→        gl.uniform2f(this.uFg['uSize'], sw * this.dpr, sh * this.dpr)
   410→   410→        gl.uniform4f(
   411→   411→          this.uFg['uCornerRadii'],
   412→   412→          radii[0] * this.dpr,
   413→   413→          radii[1] * this.dpr,
   414→   414→          radii[2] * this.dpr,
   415→   415→          radii[3] * this.dpr
   416→   416→        )
   417→   417→        gl.uniform2f(this.uFg['uOriginalSize'], origSizeX, origSizeY)
   418→   418→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], origRadius)
   419→   419→      gl.uniform2f(this.uFg['uLayerScale'], layerScaleX, layerScaleY)
   420→   420→      gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   421→   421→      // Continuous-curvature mask for capsule foregrounds.
   422→   422→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   423→   423→        gl.activeTexture(gl.TEXTURE2)
   424→   424→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   425→   425→        gl.uniform1i(this.uFg['uContinuousSdf'], 2)
   426→   426→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 1.0)
   427→   427→        gl.uniform2f(this.uFg['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   428→   428→        gl.uniform2f(this.uFg['uContinuousSdfElementSize'], state.origW * this.dpr, state.origH * this.dpr)
   429→   429→      } else {
   430→   430→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 0.0)
   431→   431→      }
   432→   432→        gl.uniform1f(this.uFg['uAlpha'], 1.0 - 0.15 * p)
   433→   433→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   434→   434→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   435→   435→      }
   436→   436→    }
   437→   437→
   438→   438→    // --- Step 2f: Rim highlight (Canvas2D stroke mask approach) ---
   439→   439→    // Uses Canvas2D ctx.stroke() (browser-native Skia) to rasterize the G2
   440→   440→    // Bezier path as a stroke mask. This is the SAME method the original
   441→   441→    // uses (Skia drawOutline + paint.Stroke) — the browser's Canvas2D
   442→   442→    // internally calls Skia's SkCanvas::drawPath, which tessellates the
   443→   443→    // Bezier into triangles and rasterizes with hardware coverage AA.
   444→   444→    //
   445→   445→    // Advantages over the SDF approach:
   446→   446→    //   - Exact G2 Bezier shape (not SDF approximation)
   447→   447→    //   - Hardware coverage AA (sub-pixel accurate, no smoothstep needed)
   448→   448→    //   - No SDF computation in shader (just one texture fetch)
   449→   449→    //   - Adaptive tessellation (more samples at high-curvature corners)
   450→   450→    //
   451→   451→    // The stroke mask is cached by exact geometry. Highlight angle/alpha/press
   452→   452→    // progress only affect the composite uniforms, not the mask, so caching is
   453→   453→    // safe and avoids both per-frame rasterization and UV mismatch between a
   454→   454→    // reused backing canvas and the logical mask size.
   455→   455→    if (el.highlight && el.highlight.alpha > 0.001) {
   456→   456→      const rimAlpha = (el.isToggleKnob || el.isBottomTabIndicator) ? elHighlightAlpha : el.highlight.alpha
   457→   457→      const finalAlpha = rimAlpha * state.enterAlpha
   458→   458→      if (finalAlpha > 0.001) {
   459→   459→        // HighlightModifier.kt: ceil(width.toPx().coerceAtMost(minDimension / 2)) * 2
   460→   460→        const widthPx = Math.min(
   461→   461→          el.highlight.widthDp * this.dpr,
   462→   462→          Math.min(origSizeX, origSizeY) * 0.5
   463→   463→        )
   464→   464→        const strokeWidthDevice = Math.max(1, Math.ceil(widthPx) * 2)
   465→   465→        // Highlight data class: blurRadius defaults to width / 2. Honor it in the
   466→   466→        // Canvas2D mask as well (for the default 0.25dp this is sub-pixel, but it
   467→   467→        // keeps non-default highlights faithful too).
   468→   468→        const blurPx = Math.max(0, (el.highlight.blurRadiusDp ?? el.highlight.widthDp / 2) * this.dpr)
   469→   469→
   470→   470→        // --- Generate stroke mask via Canvas2D (browser-native Skia) ---
   471→   471→        // Element-local coordinates: (0,0) = element top-left in device px.
   472→   472→        // Margin for stroke width + AA + blur spread.
   473→   473→        const strokeMargin = Math.ceil(strokeWidthDevice) + 4  // logical margin
   474→   474→        // Logical mask size (1x device px) — used for shader UV mapping
   475→   475→        const maskW = Math.max(1, Math.ceil(origSizeX + 2 * strokeMargin))
   476→   476→        const maskH = Math.max(1, Math.ceil(origSizeY + 2 * strokeMargin))
   477→   477→        // Supersample for sharper stroke/blur rasterization.
   478→   478→        // Cap SS so total mask DPR (this.dpr × SS) ≤ device's native DPR —
   479→   479→        // no point rendering pixels beyond what the screen can display.
   480→   480→        const deviceDpr = window.devicePixelRatio || 1
   481→   481→        const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
   482→   482→        const canvasW = maskW * SS
   483→   483→        const canvasH = maskH * SS
   484→   484→        const useG2 = !!el.useContinuousSdf
   485→   485→        const maskKey = [
   486→   486→          useG2 ? 'g2' : 'rr',
   487→   487→          origSizeX.toFixed(3),
   488→   488→          origSizeY.toFixed(3),
   489→   489→          origRadius.toFixed(3),
   490→   490→          strokeWidthDevice,
   491→   491→          blurPx.toFixed(3),
   492→   492→          strokeMargin,
   493→   493→          maskW,
   494→   494→          maskH,
   495→   495→          `ss${SS}`,  // cache key includes supersample factor
   496→   496→        ].join(':')
   497→   497→
   498→   498→        let mask = this.strokeMaskCache.get(maskKey)
   499→   499→        if (!mask) {
   500→   500→          const canvas = document.createElement('canvas')
   501→   501→          canvas.width = canvasW  // 2x supersampled physical size
   502→   502→          canvas.height = canvasH
   503→   503→          const ctx = canvas.getContext('2d', { alpha: true })
   504→   504→          if (!ctx) throw new Error('2D canvas not supported')
   505→   505→          const tex = gl.createTexture()
   506→   506→          if (!tex) throw new Error('WebGL texture allocation failed')
   507→   507→          // w/h store the LOGICAL (1x) size for shader UV mapping;
   508→   508→          // the physical canvas is SS times larger.
   509→   509→          mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   510→   510→          this.strokeMaskCache.set(maskKey, mask)
   511→   511→
   512→   512→          // Keep the cache bounded. 32 entries is far above the catalog's
   513→   513→          // simultaneous highlight geometries and avoids unbounded growth on
   514→   514→          // highly dynamic pages.
   515→   515→          if (this.strokeMaskCache.size > 32) {
   516→   516→            const oldestKey = this.strokeMaskCache.keys().next().value as string | undefined
   517→   517→            if (oldestKey && oldestKey !== maskKey) {
   518→   518→              const oldest = this.strokeMaskCache.get(oldestKey)
   519→   519→              if (oldest) gl.deleteTexture(oldest.tex)
   520→   520→              this.strokeMaskCache.delete(oldestKey)
   521→   521→            }
   522→   522→          }
   523→   523→        }
   524→   524→
   525→   525→        if (!mask.ready) {
   526→   526→          const smCtx = mask.ctx
   527→   527→          smCtx.clearRect(0, 0, canvasW, canvasH)
   528→   528→          smCtx.save()
   529→   529→          // Scale up for 2x supersampling: draw in logical (1x) coordinates
   530→   530→          // while the physical canvas is 2x. This gives sharper stroke + blur.
   531→   531→          smCtx.scale(SS, SS)
   532→   532→          smCtx.translate(strokeMargin, strokeMargin)
   533→   533→
   534→   534→          // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   535→   535→          let path: Path2D
   536→   536→          if (useG2) {
   537→   537→            path = continuousCurvatureRoundedRectPath(smCtx, origSizeX, origSizeY, origRadius)
   538→   538→          } else {
   539→   539→            path = new Path2D()
   540→   540→            const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   541→   541→            path.moveTo(r, 0)
   542→   542→            path.lineTo(origSizeX - r, 0)
   543→   543→            path.arcTo(origSizeX, 0, origSizeX, r, r)
   544→   544→            path.lineTo(origSizeX, origSizeY - r)
   545→   545→            path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   546→   546→            path.lineTo(r, origSizeY)
   547→   547→            path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   548→   548→            path.lineTo(0, r)
   549→   549→            path.arcTo(0, 0, r, 0, r)
   550→   550→            path.closePath()
   551→   551→          }
   552→   552→
   553→   553→          // Stroke — browser-native Skia stroke rasterization.
   554→   554→          // Faithful to HighlightModifier.kt: clipOutline(outline) BEFORE drawOutline,
   555→   555→          // so only the INSIDE half of the centered stroke remains. Without this
   556→   556→          // clip, Canvas2D keeps the outer half too and the rim highlight leaks
   557→   557→          // outside the glass / looks twice as thick.
   558→   558→          smCtx.clip(path)
   559→   559→          smCtx.lineWidth = strokeWidthDevice
   560→   560→          smCtx.strokeStyle = 'rgba(255,255,255,1)'
   561→   561→          smCtx.lineJoin = 'round'
   562→   562→          smCtx.lineCap = 'round'
   563→   563→          // Approximate Skia BlurMaskFilter on the stroke paint. The clip above
   564→   564→          // keeps the result inside the outline; the blur only softens that
   565→   565→          // inside stroke band, like the original HighlightModifier layer.
   566→   566→          smCtx.filter = blurPx > 0.01 ? `blur(${blurPx}px)` : 'none'
   567→   567→          smCtx.stroke(path)
   568→   568→          smCtx.filter = 'none'
   569→   569→          smCtx.restore()
   570→   570→
   571→   571→          // Upload to GPU texture. Keep the same top-left UV convention as the
   572→   572→          // foreground pass: with UNPACK_FLIP_Y_WEBGL=false, local y=0 samples
   573→   573→          // the top row of the mask.
   574→   574→          gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   575→   575→          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   576→   576→          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   577→   577→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   578→   578→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   579→   579→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   580→   580→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   581→   581→          mask.ready = true
   582→   582→        }
   583→   583→
   584→   584→        // --- Composite: stroke mask × intensity × color → scene FBO ---
   585→   585→        gl.enable(gl.BLEND)
   586→   586→        if (el.highlight.mode === 1) {
   587→   587→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   588→   588→        } else {
   589→   589→          gl.blendFunc(gl.ONE, gl.ONE)
   590→   590→        }
   591→   591→        gl.useProgram(this.strokeMaskCompositeProgram)
   592→   592→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   593→   593→        gl.enableVertexAttribArray(this.aPosLocSm)
   594→   594→        gl.vertexAttribPointer(this.aPosLocSm, 2, gl.FLOAT, false, 0, 0)
   595→   595→        gl.uniform2f(this.uSm['uCanvasSize'], this.canvas.width, this.canvas.height)
   596→   596→        gl.uniform2f(this.uSm['uOffset'], sx * this.dpr, sy * this.dpr)
   597→   597→        gl.uniform2f(this.uSm['uSize'], sw * this.dpr, sh * this.dpr)
   598→   598→        gl.uniform4f(this.uSm['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   599→   599→        gl.activeTexture(gl.TEXTURE0)
   600→   600→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   601→   601→        gl.uniform1i(this.uSm['uStrokeMask'], 0)
   602→   602→        // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   603→   603→        // physical canvas is SS× larger but the shader uses 1x coords for UV:
   604→   604→        //   UV = (localCoord + margin) / logicalMaskSize
   605→   605→        // LINEAR filtering on the SS× texture gives supersampled quality.
   606→   606→        gl.uniform2f(this.uSm['uMaskOffset'], strokeMargin, strokeMargin)
   607→   607→        gl.uniform2f(this.uSm['uMaskSize'], mask.w, mask.h)
   608→   608→        gl.uniform4f(this.uSm['uHighlightColor'], el.highlight.color[0], el.highlight.color[1], el.highlight.color[2], 1.0)
   609→   609→        gl.uniform1f(this.uSm['uHighlightAngle'], el.useGravityAngle ? this.gravityAngle : el.highlight.angle)
   610→   610→        gl.uniform1f(this.uSm['uHighlightFalloff'], el.highlight.falloff)
   611→   611→        gl.uniform1f(this.uSm['uHighlightAlpha'], finalAlpha)
   612→   612→        gl.uniform1f(this.uSm['uHighlightMode'], el.highlight.mode)
   613→   613→        gl.uniform2f(this.uSm['uOriginalSize'], origSizeX, origSizeY)
   614→   614→        gl.uniform1f(this.uSm['uOriginalCornerRadius'], origRadius)
   615→   615→        gl.uniform2f(this.uSm['uLayerScale'], layerScaleX, layerScaleY)
   616→   616→        gl.uniform1f(this.uSm['uElementRotation'], state.elementRotation)
   617→   617→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   618→   618→
   619→   619→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   620→   620→      }
   621→   621→    }
   622→   622→
   623→   623→    // --- Step 2g: 内层背景板 rim highlight (指示器 only) ---
   624→   624→    // The 内层背景板 (hidden Row's 56dp glass inside the 指示器) has its own
   625→   625→    // Highlight.Default.copy(alpha=progress). This is drawn INSIDE the
   626→   626→    // 指示器's element shader (sampleIndicatorBackdrop) so it's clipped by
   627→   627→    // the 指示器's capsule SDF — no separate pass needed here.
   628→   628→  },
   629→   629→}

     6→/** Shared state between renderGlassElement and its sub-passes.
     7→ *  Rect/radius values are in CSS px (same units as the original code —
     8→ *  each pass multiplies by `dpr` when setting GL uniforms). */
     9→export interface GlassRenderState {
    10→  el: GlassElementConfig
    11→  st: ElementState | undefined
    12→  isButton: boolean
    13→  p: number // press progress
    14→  sx: number // screen x (CSS px) — SCALED rect top-left
    15→  sy: number // screen y (CSS px) — SCALED rect top-left
    16→  sw: number // screen width (CSS px) — SCALED (includes graphicsLayer scaleX)
    17→  sh: number // screen height (CSS px) — SCALED (includes graphicsLayer scaleY)
    18→  radii: [number, number, number, number] // CSS px — SCALED corner radii (for shadow pass)
    19→  togglePressProgress: number
    20→  elHighlightAlpha: number
    21→  // Global element alpha (from enterProgress / ControlCenter). Multiplies the
    22→  // final fragment alpha so the whole glass element fades in/out.
    23→  enterAlpha: number
    24→  // Layer transform scale factors (from the layerBlock). Used to scale
    25→  // shader params (refraction, blur, shadow) so they stretch WITH the
    26→  // layer — faithful to the original which applies graphicsLayer AFTER
    27→  // the shader, causing the entire rendered layer to scale as a unit.
    28→  layerScaleX: number
    29→  layerScaleY: number
    30→  layerScale: number // min(scaleX, scaleY) — for isotropic params
    31→  // ORIGINAL (unscaled) element geometry — for the element-pass shader which
    32→  // computes SDF/refraction in original space then maps to screen (faithful
    33→  // to graphicsLayer post-scaling). See element.ts.
    34→  origW: number
    35→  origH: number
    36→  origCornerRadius: number
    37→  // Element rotation in radians (graphicsLayer rotationZ). 0 for most.
    38→  elementRotation: number
    39→  // Whether this element is using the independent backdrop path (skip ping-pong,
    40→  // sample wallpaper directly). Passed to the element pass so it can set
    41→  // uSampleWallpaper correctly.
    42→  independent: boolean
    43→  // Per-element FBO: when true, the element is being rendered into a small
    44→  // bbox-sized FBO. The element pass sets uUsePerElementFbo=1 + uSceneRectOffset
    45→  // + uElFboSize so the shader reconstructs screenCoord correctly.
    46→  usePerElementFbo: boolean
    47→  // Element bbox top-left in canvas px (top-left origin, DEVICE px) — the
    48→  // scene-space offset of the per-element FBO's origin.
    49→  sceneRectOffsetX: number
    50→  sceneRectOffsetY: number
    51→  // Per-element FBO size in device px.
    52→  elFboW: number
    53→  elFboH: number
    54→}

    96→    // renderGlassElementPass and renderGlassPostPasses are declared in

   101→
   102→export const glassRenderMethods = {
   103→  /** Render a glass element (button / glass-shape) via FBO ping-pong.
   104→   *  Returns the swapped curFbo/curTex/otherFbo/otherTex so the caller
   105→   *  can continue iteration with the new "current scene". */
   106→  renderGlassElement(
   107→    this: LiquidGlassRenderer,
   108→    el: GlassElementConfig,
   109→    st: ElementState | undefined,
   110→    curFbo: WebGLFramebuffer,
   111→    curTex: WebGLTexture,
   112→    otherFbo: WebGLFramebuffer,
   113→    otherTex: WebGLTexture,
   114→    r: { x: number; y: number; w: number; h: number }
   115→  ): {
   116→    curFbo: WebGLFramebuffer
   117→    curTex: WebGLTexture
   118→    otherFbo: WebGLFramebuffer
   119→    otherTex: WebGLTexture
   120→  } {
   121→    const gl = this.gl
   122→    const isButton = el.kind === 'button'
   123→    const p = st?.pressProgress ?? 0
   124→
   125→    // --- Compute press transform (button only) ---
   126→    // Faithful to LiquidButton.kt layerBlock — ALWAYS runs when isInteractive,
   127→    // even if pressProgress≈0 (at rest, scale=1, translation=0 naturally).
   128→    // The original Compose layerBlock is applied unconditionally; we must
   129→    // not short-circuit on pressProgress threshold or the translation will
   130→    // snap to 0 prematurely during release-overshoot (pressProgress may be
   131→    // slightly negative due to underdamped spring while offset is still
   132→    // animating back to start).
   133→    const PRESS_SCALE_RATIO = 4 / 48
   134→    let scale = 1
   135→    let translationX = 0
   136→    let translationY = 0
   137→    let scaleX = 1
   138→    let scaleY = 1
   139→    // Control-center enter progress (faithful to ControlCenterContent.kt glassLayer)
   140→    // The original applies a DERIVED progress (via ProgressConverter) for
   141→    // translation/scale, and a SEPARATE safe progress (clamped 0..1) for alpha.
   142→    if (el.enterProgress != null) {
   143→      // ProgressConverter: dampens overscroll exponentially.
   144→      //   p < 0  → (1 - e^-|p|) * -1   (approaches -1)
   145→      //   0..1   → p                    (linear)
   146→      //   p > 1  → 1 + (1 - e^-(p-1))  (approaches 2)
   147→      const raw = el.enterProgress
   148→      const derived = raw < 0
   149→        ? (1 - Math.exp(-Math.abs(raw))) * -1
   150→        : raw <= 1 ? raw
   151→        : 1 + (1 - Math.exp(-(raw - 1)))
   152→      // translationY = -48dp * (1 - derived) — slides up 48dp when collapsed
   153→      translationY += -48 * DP * (1 - derived)
   154→      // Overscroll row-stretch: when derived > 1, grow inter-row spacing
   155→      // by 32dp per unit of DERIVED overshoot (faithful to spacerLayoutModifier
   156→      // which uses the derived progress, not raw).
   157→      if (el.enterStretchFactor != null && derived > 1) {
   158→        translationY += el.enterStretchFactor * (derived - 1) * 32 * DP
   159→      }
   160→      // scale: scaleX /= 1 + 0.1*max(0, derived-1), scaleY *= 1 + 0.1*max(0, derived-1)
   161→      const sFactor = 1 + 0.1 * Math.max(0, derived - 1)
   162→      scaleX /= sFactor
   163→      scaleY *= sFactor
   164→    }
   165→    if (isButton && el.isInteractive && st) {
   166→      const width = el.rect.w
   167→      const height = el.rect.h
   168→      const maxDim = Math.max(width, height)
   169→      const minDim = Math.min(width, height)
   170→      const maxOffset = minDim
   171→      const initialDerivative = 0.05
   172→      const maxDragScale = PRESS_SCALE_RATIO
   173→
   174→      scale = 1 + PRESS_SCALE_RATIO * p
   175→      const dx = st.dragX - st.startDragX
   176→      const dy = st.dragY - st.startDragY
   177→      translationX = maxOffset * Math.tanh(initialDerivative * dx / maxOffset)
   178→      translationY = maxOffset * Math.tanh(initialDerivative * dy / maxOffset)
   179→
   180→      const offsetAngle = Math.atan2(dy, dx)
   181→      const whCap = Math.min(width / height, 1)
   182→      const hwCap = Math.min(height / width, 1)
   183→      scaleX = scale + maxDragScale * Math.abs(Math.cos(offsetAngle) * dx / maxDim) * whCap
   184→      scaleY = scale + maxDragScale * Math.abs(Math.sin(offsetAngle) * dy / maxDim) * hwCap
   185→    } else if (el.enterProgress == null) {
   186→      // Only override scaleX/Y if enterProgress didn't set them.
   187→      scaleX = scale
   188→      scaleY = scale
   189→    }
   190→
   191→    // --- Toggle knob transform (faithful to LiquidToggle.kt + DampedDragAnimation.kt) ---
   192→    // The knob's layerBlock applies:
   193→    //   scaleX = dampedDragAnimation.scaleX
   194→    //   scaleY = dampedDragAnimation.scaleY
   195→    //   velocity = dampedDragAnimation.velocity / 50
   196→    //   scaleX /= 1 - clamp(velocity * 0.75, -0.2, 0.2)
   197→    //   scaleY *= 1 - clamp(velocity * 0.25, -0.2, 0.2)
   198→    // The X and Y scales use SEPARATE underdamped springs (ζ=0.6 / ζ=0.7),
   199→    // giving X a tiny bit more bounce than Y on release.
   200→    let toggleXOffset = 0
   201→    let toggleScaleX = 1
   202→    let toggleScaleY = 1
   203→    let togglePressProgress = 0
   204→    if (el.isToggleKnob) {
   205→      const tg = this.toggleStates.get(el.isToggleKnob.groupId)
   206→      if (tg) {
   207→        toggleXOffset = tg.fraction * el.isToggleKnob.dragWidth
   208→        toggleScaleX = tg.scaleX
   209→        toggleScaleY = tg.scaleY
   210→        togglePressProgress = tg.pressProgress
   211→        // Velocity-driven squash-and-stretch (faithful to LiquidToggle.kt / LiquidSlider.kt layerBlock).
   212→        //   velocity = dampedDragAnimation.velocity / divisor
   213→        //   scaleX /= 1 - clamp(velocity * 0.75, -0.2, 0.2)
   214→        //   scaleY *= 1 - clamp(velocity * 0.25, -0.2, 0.2)
   215→        // Divisor: 50 for toggle knob, 10 for slider knob (faithful to original).
   216→        const divisor = el.isToggleKnob.velocityDivisor ?? 50
   217→        const vel = tg.velocity / divisor
   218→        const velX = Math.max(-0.2, Math.min(0.2, vel * 0.75))
   219→        const velY = Math.max(-0.2, Math.min(0.2, vel * 0.25))
   220→        toggleScaleX = toggleScaleX / (1 - velX)
   221→        toggleScaleY = toggleScaleY * (1 - velY)
   222→      }
   223→    }
   224→    scaleX *= toggleScaleX
   225→    scaleY *= toggleScaleY
   226→
   227→    // --- Bottom tabs container transform (faithful to LiquidBottomTabs.kt container layerBlock) ---
   228→    //   val scale = lerp(1f, 1f + 16f.dp.toPx() / size.width, progress)
   229→    //   scaleX = scaleY = scale
   230→    //   translationX = panelOffset (whole bar shifts during drag)
   231→    if (el.isBottomTabContainer) {
   232→      const tg = this.toggleStates.get(el.isBottomTabContainer.groupId)
   233→      if (tg) {
   234→        const containerScale = 1 + (16 * DP) / el.rect.w * tg.pressProgress
   235→        scaleX *= containerScale
   236→        scaleY *= containerScale
   237→        translationX += tg.panelOffset
   238→        // Drive press glow (InteractiveHighlight) via togglePressProgress.
   239→        togglePressProgress = tg.pressProgress
   240→      }
   241→    }
   242→
   243→    // --- Bottom tabs content transform (faithful to LiquidBottomTabs.kt) ---
   244→    // Tab content sits INSIDE the container Row. In the original:
   245→    //   - Container's layerBlock: scale = lerp(1, 1+16dp/containerWidth, pressProgress)
   246→    //     applied to the ENTIRE Row (including tab content).
   247→    //   - Each tab's own graphicsLayer: scale = lerp(1, 1.2, pressProgress)
   248→    //     applied on top of the container scale.
   249→    //   - The Row has translationX = panelOffset.
   250→    //
   251→    // In our single-pass renderer, we apply both scales to the tab content's
   252→    // scaleX/Y (multiplied), and panelOffset to translationX. We do NOT
   253→    // scale the tab content's POSITION around the container center — that
   254→    // would shift tabs away from their hit-test rects. The container scale
   255→    // only affects the tab content SIZE, not position.
   256→    if (el.isBottomTabContent) {
   257→      const tg = this.toggleStates.get(el.isBottomTabContent.groupId)
   258→      if (tg) {
   259→        // Use the actual container width from the element config (not a guess).
   260→        const containerW = el.isBottomTabContent.containerWidth ?? el.rect.w
   261→        const containerScale = 1 + (16 * DP) / containerW * tg.pressProgress
   262→        scaleX *= containerScale
   263→        // Content's own scale (lerp(1, 1.2, pressProgress)).
   264→        const contentScale = 1 + 0.2 * tg.pressProgress
   265→        scaleX *= contentScale
   266→        scaleY *= containerScale * contentScale
   267→        translationX += tg.panelOffset
   268→      }
   269→    }
   270→
   271→    // --- Bottom tabs indicator transform (faithful to LiquidBottomTabs.kt indicator layerBlock) ---
   272→    //   translationX = dampedDragAnimation.value * tabWidth + panelOffset
   273→    //   scaleX = dampedDragAnimation.scaleX  (spring 0.6, 250, 1→78/56)
   274→    //   scaleY = dampedDragAnimation.scaleY  (spring 0.7, 250, 1→78/56)
   275→    //   velocity = dampedDragAnimation.velocity / 10  (NOT 50 like toggle!)
   276→    //   scaleX /= 1 - clamp(velocity * 0.75, -0.2, 0.2)
   277→    //   scaleY *= 1 - clamp(velocity * 0.25, -0.2, 0.2)
   278→    //
   279→    // NOTE: the indicator is a CHILD of the container Row. In the original,
   280→    // the container's layerBlock (scale around center) applies to ALL children
   281→    // including the indicator. The indicator's own graphicsLayer (translationX,
   282→    // scaleX/Y, velocity stretch) is applied INSIDE the container's scaled
   283→    // space. So the indicator's translationX is ALSO scaled by the container.
   284→    //
   285→    // In our single-pass renderer, we can't nest graphicsLayer. Instead, we
   286→    // must NOT apply the container scale to the indicator — the indicator's
   287→    // position (toggleXOffset) is already in screen space, and applying
   288→    // container scale on top shifts it away from the finger.
   289→    // The container scale only affects the indicator's SIZE (scaleX/Y), not
   290→    // its position. We apply the container scale to the indicator's scale
   291→    // but NOT to its center position.
   292→    if (el.isBottomTabIndicator) {
   293→      const tg = this.toggleStates.get(el.isBottomTabIndicator.groupId)
   294→      if (tg) {
   295→        // Position: indicator slides between tabs + panelOffset.
   296→        // NO container scale applied to position — the indicator follows
   297→        // the finger directly. Container scale only affects size.
   298→        toggleXOffset += tg.fraction * el.isBottomTabIndicator.dragWidth
   299→        toggleXOffset += tg.panelOffset
   300→        // Scale from DampedDragAnimation (1 → 78/56 on press).
   301→        const indScaleX = tg.scaleX
   302→        const indScaleY = tg.scaleY
   303→        // Velocity squash — divisor is 10 (not 50 like toggle knob).
   304→        const vel = tg.velocity / 10
   305→        const velX = Math.max(-0.2, Math.min(0.2, vel * 0.75))
   306→        const velY = Math.max(-0.2, Math.min(0.2, vel * 0.25))
   307→        const finalIndScaleX = indScaleX / (1 - velX)
   308→        const finalIndScaleY = indScaleY * (1 - velY)
   309→        scaleX *= finalIndScaleX
   310→        scaleY *= finalIndScaleY
   311→        // Drive white overlay alpha + surface color by pressProgress
   312→        // (faithful to indicator onDrawSurface).
   313→        togglePressProgress = Math.max(togglePressProgress, tg.pressProgress)
   314→        // NOTE: the indicator does NOT get the container's layerBlock scale.
   315→        // In LiquidBottomTabs.kt, the indicator Box is a SIBLING of the
   316→        // container Row (both children of BoxWithConstraints), NOT a child
   317→        // of the container. The container's layerBlock (1 + 16dp/width)
   318→        // only applies to the container + its tab-content children. The
   319→        // indicator only gets its own DampedDragAnimation scale (78/56) +
   320→        // velocity stretch.
   321→      }
   322→    }
   323→
   324→    // --- Arbitrary element scale (elementScaleX/Y from GlassElementConfig) ---
   325→    // Multiplied into the final scaleX/Y — used by perf benchmark's outer
   326→    // glasses for scale-only animation while keeping rect.w/h fixed.
   327→    if (el.elementScaleX != null) scaleX *= el.elementScaleX
   328→    if (el.elementScaleY != null) scaleY *= el.elementScaleY
   329→
   330→    // Compute final on-screen rect (in CSS px, matching the original code).
   331→    let cx: number, cy: number
   332→    cx = r.x + el.rect.w / 2 + translationX + toggleXOffset
   333→    cy = r.y + el.rect.h / 2 + translationY
   334→    const sw = el.rect.w * scaleX
   335→    const sh = el.rect.h * scaleY
   336→    const sx = cx - sw / 2
   337→    const sy = cy - sh / 2
   338→    const cornerRadius = el.cornerRadius * Math.min(scaleX, scaleY)
   339→    const radii: [number, number, number, number] = [
   340→      cornerRadius, cornerRadius, cornerRadius, cornerRadius,
   341→    ]
   342→
   343→    // --- tabsBackdrop FBO pass removed ---
   344→    // The faithful sampleIndicatorBackdrop (element-utils.ts) computes the
   345→    // tinted layer inline (wallpaper + accentColor at containerColor alpha
   346→    // inside the 容器 capsule SDF), approximating LiquidBottomTabs.kt's
   347→    // ColorFilter.tint(SrcIn) on the 内层背景板 (hidden Row)'s content. No separate FBO
   348→    // capture is needed.
   349→
   350→    // --- Independent backdrop optimization ---
   351→    // When independentBackdrop=true AND the page has a wallpaper (not a solid
   352→    // backgroundColor), the element's glass refraction samples the wallpaper
   353→    // directly (via uSampleWallpaper in the shader). This allows SKIPPING the
   354→    // FBO ping-pong blit — the most expensive per-element operation (~850K px
   355→    // fullscreen copy at DPR 1.5). The element renders directly to curFbo with
   356→    // alpha blending, matching the original Android app where most elements use
   357→    // LayerBackdrop (wallpaper) via RenderEffect rather than compositing the scene.
   358→    //
   359→    // TODO: The skip-ping-pong path is currently DISABLED because it causes
   360→    // visual issues (upside-down, stretching, missing effects). The
   361→    // independent flag is still computed and passed to the element pass so
   362→    // that uSampleWallpaper can be set correctly for the shader path that
   363→    // samples the wallpaper via coverUv (which is already working for toggle
   364→    // knobs and indicators). The ping-pong blit is always performed for now.
   365→    const independent = !!(el.independentBackdrop && !this.backgroundColor && this.wallpaperTexture)
   366→    const skipPingPong = false // TODO: re-enable after fixing visual issues
   367→
   368→    // --- Per-element FBO optimization (Settings toggle, default ON) ---
   369→    // Instead of the fullscreen ping-pong blit (Step 1 below), render the
   370→    // element into a small bbox-sized FBO. The backdrop is scissor-cropped
   371→    // into a small texture, the element pass samples that, then composites
   372→    // back onto curFbo at the element's bbox. curFbo is NEVER swapped — it
   373→    // stays the fixed accumulation target, so all previously-rendered
   374→    // elements remain available for subsequent elements to sample.
   375→    //
   376→    // Excluded elements (kept on the legacy ping-pong path for correctness):
   377→    //  - backdropFbo elements (dialog card): they sample the pre-rendered
   378→    //    dialogBackdropTex, not the scene FBO — the crop would be wrong.
   379→    //  - SDF-texture elements (LockScreen): use a separate glass path.
   380→    // These fall through to the legacy path below.
   381→    if (this.usePerElementFbo && !el.backdropFbo && !el.useSdfTexture && !skipPingPong) {
   382→      return this.renderGlassElementPerFbo(el, st, curFbo, curTex, otherFbo, otherTex, {
   383→        sx, sy, sw, sh, radii, scaleX, scaleY, isButton, p, togglePressProgress,
   384→        independent, translationX, translationY,
   385→      })
   386→    }
   387→
   388→    // --- Step 1: Blit curFbo → otherFbo (FULLSCREEN — must copy the entire
   389→    // scene so ping-pong preserves all previously-rendered elements. Scissor
   390→    // cannot be used here because otherFbo's regions outside the current
   391→    // element still need the correct scene content for subsequent elements
   392→    // to sample from.) ---
   393→    if (skipPingPong) {
   394→      this.bindFBO(curFbo)
   395→      // Blending is already enabled from the element loop setup.
   396→    } else {
   397→      this.bindFBO(otherFbo)
   398→      this.drawCopy(curTex)
   399→      // Re-enable blending after the copy (drawCopy disables it).
   400→      gl.enable(gl.BLEND)
   401→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   402→    }
   403→
   404→    // --- Scissor: limit drawing passes (shadow + element + highlight) to the
   405→    // element's bounding box + margin. The blit above is fullscreen (needed
   406→    // for ping-pong correctness), but the actual element rendering only
   407→    // affects a small region. Scissor skips fragment shader execution for
   408→    // pixels far outside the element — the single biggest perf win.
   409→    // Margin covers: outer shadow (~24dp), highlight blur (~2px), press scale
   410→    // (up to 1.5x), toggle drag offset. 60 CSS px is safe.
   411→    const MARGIN_CSS = 60
   412→    const scissorX = Math.max(0, Math.round((sx - MARGIN_CSS) * this.dpr))
   413→    const scissorY = Math.max(0, Math.round((this.cssHeight - (sy + sh + MARGIN_CSS)) * this.dpr))
   414→    const scissorW = Math.min(this.fboW - scissorX, Math.round((sw + 2 * MARGIN_CSS) * this.dpr))
   415→    const scissorH = Math.min(this.fboH - scissorY, Math.round((sh + 2 * MARGIN_CSS) * this.dpr))
   416→    gl.enable(gl.SCISSOR_TEST)
   417→    gl.scissor(scissorX, scissorY, scissorW, scissorH)
   418→
   419→    const state: GlassRenderState = {
   420→      el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress,
   421→      // For toggle knobs + bottom-tab indicators, the highlight alpha is
   422→      // modulated by pressProgress (faithful to Highlight.Default.copy(alpha=progress)).
   423→      // At rest (progress=0) the alpha should be 0 — so we initialize to 0
   424→      // here, and renderGlassElementPass overrides it to alpha*progress when
   425→      // progress > 0. For non-toggle elements, use the static highlight alpha.
   426→      elHighlightAlpha: (el.isToggleKnob || el.isBottomTabIndicator) ? 0 : (el.highlight ? el.highlight.alpha : 0),
   427→      enterAlpha: el.enterProgress != null ? (() => {
   428→        // Faithful to ControlCenterContent.kt: alpha = EaseIn.transform(safeProgress)
   429→        // where safeProgress = safeEnterProgressAnimation.value (clamped 0..1).
   430→        // EaseIn = CubicBezierEasing(0.42, 0, 1, 1). Use enterSafeProgress
   431→        // if available, else fall back to clamped enterProgress.
   432→        const sp = el.enterSafeProgress != null
   433→          ? Math.max(0, Math.min(1, el.enterSafeProgress))
   434→          : Math.max(0, Math.min(1, el.enterProgress!))
   435→        return easeIn(sp)
   436→      })() : 1,
   437→      layerScaleX: scaleX,
   438→      layerScaleY: scaleY,
   439→      layerScale: Math.min(scaleX, scaleY),
   440→      // ORIGINAL geometry (unscaled) for the element-pass SDF. The shader
   441→      // computes SDF/refraction in original space, then maps the refraction
   442→      // offset to screen space via uLayerScale — faithful to the original
   443→      // which shades at original size then scales via graphicsLayer.
   444→      origW: el.rect.w,
   445→      origH: el.rect.h,
   446→      origCornerRadius: el.cornerRadius,
   447→      elementRotation: el.elementRotation ?? 0,
   448→      independent,
   449→      // Per-element FBO fields — populated below if the per-element path is
   450→      // taken; left at defaults (usePerElementFbo=false) for the legacy path.
   451→      usePerElementFbo: false,
   452→      sceneRectOffsetX: 0,
   453→      sceneRectOffsetY: 0,
   454→      elFboW: 0,
   455→      elFboH: 0,
   456→    }
   457→
   458→    // --- Step 2a: Shadow pass (to otherFbo, on top of copied scene) ---
   459→    this.renderGlassShadowPass(state)
   460→
   461→    // --- Step 2b: Element pass (refraction + vibrancy + tint) ---
   462→    if (el.useSeparableBlur && el.blurRadius >= 0.5) {
   463→      const blurRadiusPx = el.blurRadius * state.layerScale * this.dpr
   464→      // For backdropFbo elements (dialog card), blur the dialogBackdropTex
   465→      // (wallpaper+scrim+colorControls opaque layer) instead of the scene FBO.
   466→      const backdropSrc = (el.backdropFbo && this.dialogBackdropTex) ? this.dialogBackdropTex : curTex
   467→      const blurredBackdrop = this.blurTexture(backdropSrc, blurRadiusPx)
   468→      // blurTexture disables BLEND — re-enable it so renderGlassElementPass
   469→      // composites the glass onto otherFbo with alpha blending.
   470→      this.gl.enable(this.gl.BLEND)
   471→      this.gl.blendFunc(this.gl.SRC_ALPHA, this.gl.ONE_MINUS_SRC_ALPHA)
   472→      this.bindFBO(otherFbo)
   473→      this.gl.viewport(0, 0, this.fboW, this.fboH)
   474→      // Pass the pre-blurred texture as curTex. For backdropFbo elements,
   475→      // temporarily disable backdropFbo so the element pass binds curTex
   476→      // (the blurred backdrop) instead of the raw dialogBackdropTex.
   477→      const passState = el.backdropFbo ? { ...state, el: { ...el, backdropFbo: false } } : state
   478→      this.renderGlassElementPass(passState, blurredBackdrop)
   479→    } else {
   480→      this.renderGlassElementPass(state, curTex)
   481→    }
   482→
   483→    // --- Steps 2c–2f: Press glow, white overlay, foreground, rim highlight ---
   484→    this.renderGlassPostPasses(state)
   485→
   486→    // --- Disable scissor (restore full-screen rasterization for subsequent
   487→    // elements + the final blit to the default framebuffer) ---
   488→    gl.disable(gl.SCISSOR_TEST)
   489→
   490→    // --- Step 3: Swap curFbo ↔ otherFbo ---
   491→    // otherFbo now contains: previous scene + shadow + glass body +
   492→    // press glow + white overlay + foreground + rim highlight. It
   493→    // becomes the new "current scene" for subsequent elements to sample.
   494→    return {
   495→      curFbo: otherFbo,
   496→      curTex: otherTex,
   497→      otherFbo: curFbo,
   498→      otherTex: curTex,
   499→    }
   500→  },
   501→
   502→  /** Per-element FBO render path. Renders the glass element into a small
   503→   *  bbox-sized FBO (capped at MAX_ELEMENT_FBO_SIZE) instead of the fullscreen
   504→   *  ping-pong blit. Steps:
   505→   *    1. Shadow pass → curFbo (scissor to bbox, same as legacy).
   506→   *    2. Crop the element's bbox region from curTex into backdropCropFbo
   507→   *       (optionally 2-pass blur for useSeparableBlur elements).
   508→   *    3. Render the element pass into elFbo (sampling the cropped backdrop).
   509→   *    4. Composite elFbo back onto curFbo at the bbox (scissor + SrcOver).
   510→   *    5. Post passes (press glow, foreground, highlight) → curFbo (scissor).
   511→   *  curFbo is never swapped — it stays the fixed accumulation target. */
   512→  renderGlassElementPerFbo(
   513→    this: LiquidGlassRenderer,
   514→    el: GlassElementConfig,
   515→    st: ElementState | undefined,
   516→    curFbo: WebGLFramebuffer,
   517→    curTex: WebGLTexture,
   518→    otherFbo: WebGLFramebuffer,
   519→    otherTex: WebGLTexture,
   520→    computed: {
   521→      sx: number; sy: number; sw: number; sh: number
   522→      radii: [number, number, number, number]
   523→      scaleX: number; scaleY: number
   524→      isButton: boolean; p: number
   525→      togglePressProgress: number
   526→      independent: boolean
   527→      translationX: number; translationY: number
   528→    }
   529→  ): {
   530→    curFbo: WebGLFramebuffer
   531→    curTex: WebGLTexture
   532→    otherFbo: WebGLFramebuffer
   533→    otherTex: WebGLTexture
   534→  } {
   535→    const gl = this.gl
   536→    const { sx, sy, sw, sh, radii, scaleX, scaleY, isButton, p, togglePressProgress, independent } = computed
   537→
   538→    // --- Bbox in device px (top-left origin), clamped to the canvas ---
   539→    // Same MARGIN as the legacy path (shadow + highlight + press scale room).
   540→    const MARGIN_CSS = 60
   541→    const bx0 = Math.max(0, Math.round((sx - MARGIN_CSS) * this.dpr))
   542→    const by0Top = Math.max(0, Math.round((sy - MARGIN_CSS) * this.dpr))
   543→    const bx1 = Math.min(this.fboW, Math.round((sx + sw + MARGIN_CSS) * this.dpr))
   544→    const by1Top = Math.min(this.fboH, Math.round((sy + sh + MARGIN_CSS) * this.dpr))
   545→    const bboxW = Math.max(1, bx1 - bx0)
   546→    const bboxH = Math.max(1, by1Top - by0Top)
   547→    // Bottom-left origin Y for scissor (WebGL scissor uses BL origin).
   548→    const bboxScissorY = Math.max(0, this.fboH - by1Top)
   549→
   550→    // --- Ensure the per-element FBOs exist at bboxW×bboxH (capped) ---
   551→    const { w: elFboW, h: elFboH } = this.ensureElementFBO(bboxW, bboxH)
   552→
   553→    // --- Build the GlassRenderState (same as the legacy path) ---
   554→    const state: GlassRenderState = {
   555→      el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress,
   556→      elHighlightAlpha: (el.isToggleKnob || el.isBottomTabIndicator) ? 0 : (el.highlight ? el.highlight.alpha : 0),
   557→      enterAlpha: el.enterProgress != null ? (() => {
   558→        const sp = el.enterSafeProgress != null
   559→          ? Math.max(0, Math.min(1, el.enterSafeProgress))
   560→          : Math.max(0, Math.min(1, el.enterProgress!))
   561→        return easeIn(sp)
   562→      })() : 1,
   563→      layerScaleX: scaleX,
   564→      layerScaleY: scaleY,
   565→      layerScale: Math.min(scaleX, scaleY),
   566→      origW: el.rect.w,
   567→      origH: el.rect.h,
   568→      origCornerRadius: el.cornerRadius,
   569→      elementRotation: el.elementRotation ?? 0,
   570→      independent,
   571→      // Per-element FBO: the element pass renders into elFbo. screenCoord is
   572→      // reconstructed as uSceneRectOffset + localCoord. The offset is the
   573→      // element's bbox top-left in scene device px (top-left origin).
   574→      usePerElementFbo: true,
   575→      sceneRectOffsetX: bx0,
   576→      sceneRectOffsetY: by0Top,
   577→      elFboW,
   578→      elFboH,
   579→    }
   580→
   581→    // --- Step 1: Shadow pass → curFbo (scissor to bbox) ---
   582→    this.bindFBO(curFbo)
   583→    gl.enable(gl.SCISSOR_TEST)
   584→    gl.scissor(bx0, bboxScissorY, bboxW, bboxH)
   585→    gl.enable(gl.BLEND)
   586→    gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   587→    this.renderGlassShadowPass(state)
   588→
   589→    // --- Step 2: Crop + (optional) blur backdrop ---
   590→    // Crop the bbox region of curTex into backdropCropFbo. If useSeparableBlur,
   591→    // also run the 2-pass Gaussian on the cropped texture (operates on the
   592→    // small FBO — much cheaper than blurring the fullscreen scene).
   593→    // Set uBackdropRect FIRST so the element pass's sceneUv maps canvasPx into
   594→    // the cropped texture's [0..1] range (not the full canvas).
   595→    this.gl.useProgram(this.elementProgram)
   596→    this.gl.uniform4f(this.uEl['uBackdropRect'], bx0, by0Top, bboxW, bboxH)
   597→    let backdropTex: WebGLTexture
   598→    if (el.useSeparableBlur && el.blurRadius >= 0.5) {
   599→      const blurRadiusPx = el.blurRadius * state.layerScale * this.dpr
   600→      backdropTex = this.cropAndBlurBackdrop(curTex, bx0, by0Top, bboxW, bboxH, blurRadiusPx)
   601→    } else {
   602→      backdropTex = this.cropAndBlurBackdrop(curTex, bx0, by0Top, bboxW, bboxH, 0)
   603→    }
   604→
   605→    // --- Step 3: Render element pass → elFbo (transparent, then glass body) ---
   606→    // Clear elFbo to transparent first (the element shader discards outside
   607→    // the glass shape, leaving only the glass body's RGBA).
   608→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.elFbo)
   609→    gl.viewport(0, 0, elFboW, elFboH)
   610→    gl.disable(gl.SCISSOR_TEST)
   611→    gl.clearColor(0, 0, 0, 0)
   612→    gl.clear(gl.COLOR_BUFFER_BIT)
   613→    gl.enable(gl.BLEND)
   614→    gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   615→    // Render the element pass sampling the cropped backdrop. For
   616→    // useSeparableBlur elements, the backdrop is already blurred → the
   617→    // element pass should use inlineBlurRadius=0. renderGlassElementPass
   618→    // handles this when uSkipColorControls / useSeparableBlur is set on el.
   619→    // We pass the (possibly blurred) backdropTex as curTex.
   620→    this.renderGlassElementPass(state, backdropTex)
   621→
   622→    // --- Step 4: Composite elFbo → curFbo at the bbox (SrcOver) ---
   623→    this.bindFBO(curFbo)
   624→    gl.enable(gl.SCISSOR_TEST)
   625→    gl.scissor(bx0, bboxScissorY, bboxW, bboxH)
   626→    this.drawElFboComposite(this.elFboTex!, elFboW, elFboH, bx0, by0Top, bboxW, bboxH)
   627→
   628→    // --- Step 5: Post passes (press glow, white overlay, foreground, rim
   629→    // highlight) → curFbo (scissor to bbox, same as legacy) ---
   630→    this.renderGlassPostPasses(state)
   631→
   632→    gl.disable(gl.SCISSOR_TEST)
   633→
   634→    // --- No swap: curFbo remains the accumulation target ---
   635→    return { curFbo, curTex, otherFbo, otherTex }
   636→  },
   637→
   638→  renderGlassShadowPass(this: LiquidGlassRenderer, state: GlassRenderState) {
   639→    const gl = this.gl
   640→    const { el, sx, sy, sw, sh, radii } = state
   641→    if (!el.outerShadow || el.outerShadow.radius <= 0.5) return
   642→    // Shadow alpha: for bottom tab indicator, modulate by pressProgress
   643→    // (faithful to Kotlin: Shadow(alpha = progress)). At rest (progress=0),
   644→    // shadow is invisible; when pressed, Shadow.Default becomes visible.
   645→    let shadowAlpha = el.outerShadow.alpha
   646→    if (el.isBottomTabIndicator) {
   647→      shadowAlpha *= state.togglePressProgress
   648→    }
   649→    if (shadowAlpha <= 0.001) return
   650→    gl.useProgram(this.shadowProgram)
   651→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   652→    gl.enableVertexAttribArray(this.aPosLocSh)
   653→    gl.vertexAttribPointer(this.aPosLocSh, 2, gl.FLOAT, false, 0, 0)
   654→    gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   655→
   656→    gl.uniform2f(this.uSh['uCanvasSize'], this.canvas.width, this.canvas.height)
   657→    gl.uniform2f(this.uSh['uElementOffset'], sx * this.dpr, sy * this.dpr)
   658→    gl.uniform2f(this.uSh['uElementSize'], sw * this.dpr, sh * this.dpr)
   659→    gl.uniform4f(
   660→      this.uSh['uCornerRadii'],
   661→      radii[0] * this.dpr,
   662→      radii[1] * this.dpr,
   663→      radii[2] * this.dpr,
   664→      radii[3] * this.dpr
   665→    )
   666→    // ORIGINAL-space SDF uniforms — the shadow shader now computes its SDF
   667→    // in original space (faithful to graphicsLayer { scaleX, scaleY } post-
   668→    // scaling), so the shadow capsule shape stays correct when the element
   669→    // is stretched. See shadow.ts for the full rationale.
   670→    gl.uniform2f(this.uSh['uOriginalSize'], state.origW * this.dpr, state.origH * this.dpr)
   671→    gl.uniform1f(this.uSh['uOriginalCornerRadius'], state.origCornerRadius * this.dpr)
   672→    gl.uniform2f(this.uSh['uLayerScale'], state.layerScaleX, state.layerScaleY)
   673→    gl.uniform1f(this.uSh['uElementRotation'], state.elementRotation)
   674→    gl.uniform1f(this.uSh['uCornerStyle'], this.cornerStyle)
   675→    // Shadow radius + offset in ORIGINAL px (NOT scaled by layerScale).
   676→    // Faithful to original: BlurMaskFilter blurs the shadow at original size,
   677→    // then graphicsLayer scales the entire shadow layer — so the blur sigma
   678→    // and offset are defined at original resolution and stretched with the
   679→    // layer. The shader's original-space SDF already models this, so we pass
   680→    // the unscaled values here.
   681→    gl.uniform1f(this.uSh['uShadowRadius'], el.outerShadow.radius * this.dpr)
   682→    gl.uniform2f(
   683→      this.uSh['uShadowOffset'],
   684→      el.outerShadow.offsetX * this.dpr,
   685→      el.outerShadow.offsetY * this.dpr
   686→    )
   687→    gl.uniform4f(
   688→      this.uSh['uShadowColor'],
   689→      el.outerShadow.color[0],
   690→      el.outerShadow.color[1],
   691→      el.outerShadow.color[2],
   692→      shadowAlpha
   693→    )
   694→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   695→  },

     1→'use client'
     2→
     3→import * as React from 'react'
     4→import { LiquidGlassRenderer, type GlassElementConfig } from './renderer'
     5→import { draggingGroups } from './catalog'
     6→
     7→/* ------------------------------------------------------------------ *
     8→ * LiquidGlassCanvas
     9→ *
    10→ * A self-contained WebGL canvas that renders a wallpaper + a list of
    11→ * liquid-glass elements. No DOM children — the canvas owns the entire
    12→ * visual surface (wallpaper, glass, labels, chevrons, press glow).
    13→ *
    14→ * Elements may be of several kinds (button / glass-shape / plain-rect /
    15→ * progressive-blur / text). Pointer events are hit-tested against each
    16→ * element rect:
    17→ *   - 'button' kind → triggers InteractiveHighlight press + drag
    18→ *   - any kind with onTap → fires the callback on pointerup if the
    19→ *     pointer is still inside the element
    20→ *   - any kind with onDrag → fires live during pointermove
    21→ *   - empty space OR vertical-drag-takeover → starts a scroll drag
    22→ *
    23→ * Scroll behavior (matches user feedback):
    24→ *   - During drag, scrollY follows the finger directly (no spring).
    25→ *   - On release, the drag velocity becomes inertia, which exponentially
    26→ *     decays. No rebound at edges — scrolling just stops at the boundary.
    27→ *   - If the press starts on an element but the user drags mostly
    28→ *     vertically, the gesture is converted to a scroll (so list items
    29→ *     and buttons don't trap the scroll).
    30→ *
    31→ * Wheel events scroll the canvas directly (no inertia).
    32→ * ------------------------------------------------------------------ */
    33→
    34→export interface LiquidGlassCanvasProps {
    35→  wallpaperSrc: string
    36→  elements: GlassElementConfig[]
    37→  /** Total scrollable content height in CSS px. */
    38→  contentHeight?: number
    39→  /** Called once after the renderer finishes its first frame (after wallpaper + SDF texture loads). */
    40→  onReady?: () => void
    41→  /** Optional callbacks map: id → { onTap, onDragStart, onDrag, onDragEnd }. */
    42→  interactions?: Record<string, ElementInteraction>
    43→  /** When this number changes, the canvas resets scrollY to 0 (used for
    44→   *  destination switches in the catalog). */
    45→  scrollResetToken?: number
    46→  /** If set, the renderer fills the canvas with this RGB color instead
    47→   *  of drawing the wallpaper. Used for the Home page (black background). */
    48→  backgroundColor?: [number, number, number] | null
    49→  /** Map of toggle groupId → target fraction (0 or 1). The canvas syncs
    50→   *  these to the renderer whenever the map changes (programmatic toggle). */
    51→  toggleTargets?: Record<string, number>
    52→  /** Map of tab groupId → { tabIndex, tabsCount }. The canvas syncs these
    53→   *  to the renderer via setTabSelected (which uses pressedScale=78/56). */
    54→  tabTargets?: Record<string, { tabIndex: number; tabsCount: number }>
    55→  /** Ref that will be populated with the renderer instance once created.
    56→   *  Allows the parent (e.g. catalog builders) to call renderer methods
    57→   *  like setToggleTarget / beginToggleDrag / dragToggle / endToggleDrag. */
    58→  rendererRef?: React.MutableRefObject<LiquidGlassRenderer | null>
    59→  className?: string
    60→  /** Device pixel ratio override (0 = use device DPR, capped at 1.5 by
    61→   *  the renderer's resize). Applied on renderer init + when it changes. */
    62→  dpr?: number
    63→  /** Max 1D taps per separable blur pass (1..33). Performance knob for
    64→   *  useSeparableBlur elements. Applied on renderer init + when it changes.
    65→   *  Small blur radii automatically use fewer taps (computeBlur1DTapCount);
    66→   *  this caps the MAXIMUM. */
    67→  blurTapCap?: number
    68→  /** Corner style: 0 = circular, 1 = continuous (squircle). */
    69→  cornerStyle?: number
    70→  /** Per-element FBO optimization toggle. When true (default), each glass
    71→   *  element renders into a small bbox-sized FBO instead of a fullscreen
    72→   *  ping-pong blit — the biggest per-element cost saver. Pure optimization. */
    73→  usePerElementFbo?: boolean
    74→}
    75→
    76→export interface ElementInteraction {
    77→  onTap?: (pos: { x: number; y: number }) => void
    78→  /** Fires on first pointermove after press. */
    79→  onDragStart?: (pos: { x: number; y: number }) => void
    80→  /** Fires on each pointermove while pressed. */
    81→  onDrag?: (pos: { x: number; y: number }, delta: { x: number; y: number }) => void
    82→  /** Fires on pointerup. `velocity` is the release velocity in px/s
    83→   *  (positive y = downward), computed from recent pointer samples.
    84→   *  Faithful to Compose's `draggable.onDragStopped(velocity)`. */
    85→  onDragEnd?: (pos: { x: number; y: number }, velocity: { x: number; y: number }) => void
    86→  /** Fires during a multi-pointer transform gesture (pinch zoom + rotate).
    87→   *  `gestureZoom` is the multiplicative zoom factor (1.0 = no change),
    88→   *  `gestureRotate` is the additive rotation delta in radians,
    89→   *  `pan` is the centroid movement delta (already rotation-aware).
    90→   *  Faithful to Compose's detectTransformGestures. Only fires when 2+ pointers
    91→   *  are active on the element. */
    92→  onTransform?: (pan: { x: number; y: number }, gestureZoom: number, gestureRotate: number) => void
    93→}
    94→
    95→/** Internal gesture mode — set on pointerdown, may transition during move. */
    96→type GestureMode =
    97→  | 'pending' // pointer down, no movement yet — could become tap, drag, or scroll
    98→  | 'drag' // committed to an element drag (horizontal or onDrag element)
    99→  | 'scroll' // committed to a scroll drag
   100→  | 'transform' // 2-pointer pinch zoom + rotate (onTransform element)
   101→  | 'none' // no active gesture
   102→
   103→/** Per-pointer gesture state. Stored in a Map<pointerId, GestureState> so
   104→ *  multiple pointers can interact with different elements simultaneously
   105→ *  (multi-touch). When 2 pointers land on the same element with onTransform,
   106→ *  they form a transform pair (pinch zoom + rotate) — both entries have
   107→ *  mode='transform' and point at each other via transformPartner. */
   108→interface GestureState {
   109→  /** Hit element id (or null for scroll/empty space). */
   110→  pressedId: string | null
   111→  /** Canvas-local CSS px at press. */
   112→  startX: number
   113→  startY: number
   114→  /** Client-Y at press (for scroll delta computation). */
   115→  startClientY: number
   116→  /** ScrollY at press (for scroll delta computation). */
   117→  startScrollY: number
   118→  /** Whether onDragStart has fired for this gesture. */
   119→  dragStarted: boolean
   120→  /** Current gesture mode. */
   121→  mode: GestureMode
   122→  /** Whether the hit element had an onDrag handler (so we know
   123→   *  whether to commit to drag or scroll on horizontal/vertical move). */
   124→  hasDrag: boolean
   125→  /** Recent (timestamp, clientX, clientY) samples for inertia + release
   126→   *  velocity. Both axes are tracked — faithful to Compose's VelocityTracker
   127→   *  which returns an Offset(x, y). */
   128→  velocitySamples: { t: number; x: number; y: number }[]
   129→  /** Current canvas-local CSS px. */
   130→  x: number
   131→  y: number
   132→  /** pointerId of the other pointer in a transform pair (else null). */
   133→  transformPartner: number | null
   134→}
   135→
   136→export function LiquidGlassCanvas({
   137→  wallpaperSrc,
   138→  elements,
   139→  contentHeight,
   140→  onReady,
   141→  interactions,
   142→  scrollResetToken,
   143→  backgroundColor = null,
   144→  toggleTargets,
   145→  tabTargets,
   146→  rendererRef,
   147→  className,
   148→  dpr,
   149→  blurTapCap,
   150→  cornerStyle,
   151→  usePerElementFbo,
   152→}: LiquidGlassCanvasProps) {
   153→  const canvasRef = React.useRef<HTMLCanvasElement>(null)
   154→  const containerRef = React.useRef<HTMLDivElement>(null)
   155→  const rendererRefInternal = React.useRef<LiquidGlassRenderer | null>(null)
   156→  // Keep refs to the latest state so pointer handlers can read them
   157→  // without being re-created on every change.
   158→  const elementsRef = React.useRef(elements)
   159→  React.useEffect(() => { elementsRef.current = elements })
   160→  const interactionsRef = React.useRef(interactions)
   161→  React.useEffect(() => { interactionsRef.current = interactions })
   162→  // Module-level draggingGroups from catalog.tsx — tracks which toggle
   163→  // groups are being dragged (setToggleTarget skipped to avoid drift).
   164→
   165→  // --- Gesture state (per-pointer, in a Map so handlers don't need re-creation) ---
   166→  // Each active pointer has its own GestureState. Multiple pointers can
   167→  // interact with different elements simultaneously (multi-touch). When 2
   168→  // pointers land on the same element with onTransform, they form a
   169→  // transform pair (pinch zoom + rotate) — both entries' mode is 'transform'
   170→  // and they reference each other via transformPartner.
   171→  const gesturesRef = React.useRef<Map<number, GestureState>>(new Map())
   172→
   173→  // --- Multi-pointer transform tracking ---
   174→  // Previous 2-pointer state for delta computation: { dist, angle, centroid }.
   175→  // Only set while 2 pointers are in transform mode on the same element.
   176→  const prevPinchRef = React.useRef<{ dist: number; angle: number; cx: number; cy: number } | null>(null)
   177→
   178→  // --- Init renderer + wallpaper + resize observer ---
   179→  React.useEffect(() => {
   180→    if (!canvasRef.current || !containerRef.current) return
   181→    const renderer = new LiquidGlassRenderer(canvasRef.current)
   182→    rendererRefInternal.current = renderer
   183→    if (rendererRef) rendererRef.current = renderer
   184→    renderer.setBackgroundColor(backgroundColor)
   185→    renderer.loadWallpaper(wallpaperSrc).then(() => {
   186→      // Fire onReady after wallpaper loads + one frame renders
   187→      requestAnimationFrame(() => { onReady?.() })
   188→    }).catch((e) => {
   189→      console.error(e)
   190→      // Still fire onReady even on wallpaper failure so the loader goes away
   191→      requestAnimationFrame(() => { onReady?.() })
   192→    })
   193→    renderer.loadSdfTexture('/clock_sdf.webp').catch((e) => console.error(e))
   194→
   195→    const resize = () => {
   196→      const r = containerRef.current?.getBoundingClientRect()
   197→      if (!r) return
   198→      canvasRef.current!.style.width = r.width + 'px'
   199→      canvasRef.current!.style.height = r.height + 'px'
   200→      renderer.resize(r.width, r.height)
   201→    }
   202→    // Apply the DPR override BEFORE the first resize so the renderer uses
   203→    // the correct DPR from the start (otherwise resize caps at 1.5).
   204→    if (dpr != null) {
   205→      const deviceDpr = typeof window !== 'undefined' ? window.devicePixelRatio || 1 : 1
   206→      renderer.dpr = dpr > 0 ? Math.max(0.5, Math.min(deviceDpr, dpr)) : deviceDpr
   207→    }
   208→    // Apply blur tap cap (Settings slider) so 2-pass separable blur uses it.
   209→    if (blurTapCap != null) renderer.blurTapCap = Math.max(1, Math.min(33, blurTapCap | 0))
   210→    if (cornerStyle != null) renderer.cornerStyle = cornerStyle
   211→    if (usePerElementFbo != null) renderer.usePerElementFbo = usePerElementFbo
   212→    resize()
   213→    const ro = new ResizeObserver(resize)
   214→    ro.observe(containerRef.current)
   215→
   216→    // --- Wheel scroll (direct, no inertia) ---
   217→    // Trackpad two-finger swipes come through as wheel events with
   218→    // deltaY. We apply them directly to scrollY.
   219→    const onWheel = (e: WheelEvent) => {
   220→      e.preventDefault()
   221→      const delta = e.deltaY !== 0 ? e.deltaY : e.deltaX
   222→      const cur = renderer.getScrollY()
   223→      renderer.setScrollY(cur + delta)
   224→    }
   225→    const canvas = canvasRef.current
   226→    canvas.addEventListener('wheel', onWheel, { passive: false })
   227→
   228→    return () => {
   229→      ro.disconnect()
   230→      canvas.removeEventListener('wheel', onWheel)
   231→      renderer.dispose()
   232→      rendererRefInternal.current = null
   233→      if (rendererRef) rendererRef.current = null
   234→    }
   235→  }, [wallpaperSrc])
   236→
   237→  // Push backgroundColor changes (e.g. destination switch Home → other).
   238→  React.useEffect(() => {
   239→    rendererRefInternal.current?.setBackgroundColor(backgroundColor)
   240→  }, [backgroundColor])
   241→
   242→  // Apply DPR override when it changes (Settings page slider).
   243→  React.useEffect(() => {
   244→    const renderer = rendererRefInternal.current
   245→    if (!renderer || dpr == null) return
   246→    const deviceDpr = typeof window !== 'undefined' ? window.devicePixelRatio || 1 : 1
   247→    renderer.dpr = dpr > 0 ? Math.max(0.5, Math.min(deviceDpr, dpr)) : deviceDpr
   248→    const r = containerRef.current?.getBoundingClientRect()
   249→    if (r) renderer.resize(r.width, r.height)
   250→  }, [dpr])
   251→
   252→  // Apply blur tap cap when it changes (Settings page slider).
   253→  React.useEffect(() => {
   254→    const renderer = rendererRefInternal.current
   255→    if (!renderer || blurTapCap == null) return
   256→    renderer.blurTapCap = Math.max(1, Math.min(33, blurTapCap | 0))
   257→  }, [blurTapCap])
   258→
   259→  // Apply corner style when it changes (Settings page toggle).
   260→  React.useEffect(() => {
   261→    const renderer = rendererRefInternal.current
   262→    if (!renderer || cornerStyle == null) return
   263→    renderer.cornerStyle = cornerStyle
   264→    renderer.requestRender()
   265→  }, [cornerStyle])
   266→
   267→  // Apply per-element FBO optimization toggle when it changes (Settings page).
   268→  React.useEffect(() => {
   269→    const renderer = rendererRefInternal.current
   270→    if (!renderer || usePerElementFbo == null) return
   271→    renderer.usePerElementFbo = usePerElementFbo
   272→    renderer.requestRender()
   273→  }, [usePerElementFbo])
   274→
   275→  // Push the latest element list to the renderer.
   276→  React.useEffect(() => {
   277→    rendererRefInternal.current?.setElements(elements)
   278→  }, [elements])
   279→
   280→  // Push content height (for scroll clamping).
   281→  React.useEffect(() => {
   282→    if (contentHeight !== undefined) {
   283→      rendererRefInternal.current?.setContentHeight(contentHeight)
   284→    }
   285→  }, [contentHeight])
   286→
   287→  // Reset scroll when scrollResetToken changes (destination switch).
   288→  React.useEffect(() => {
   289→    rendererRefInternal.current?.setScrollY(0)
   290→  }, [scrollResetToken])
   291→
   292→  // Sync toggleTargets → renderer (programmatic toggle, e.g. via tap).
   293→  React.useEffect(() => {
   294→    if (!toggleTargets) return
   295→    const r = rendererRefInternal.current
   296→    if (!r) return
   297→    for (const [groupId, target] of Object.entries(toggleTargets)) {
   298→      // Skip groups currently being dragged — dragToggle controls their
   299→      // fraction, setToggleTarget would conflict and cause drift.
   300→      if (draggingGroups.has(groupId)) continue
   301→      r.setToggleTarget(groupId, target)
   302→    }
   303→  }, [toggleTargets])
   304→
   305→  // Sync tabTargets → renderer (programmatic tab selection, e.g. via tap).
   306→  // Uses setTabSelected which sets pressedScale=78/56 (vs toggle's 1.5).
   307→  React.useEffect(() => {
   308→    if (!tabTargets) return
   309→    const r = rendererRefInternal.current
   310→    if (!r) return
   311→    for (const [groupId, { tabIndex, tabsCount }] of Object.entries(tabTargets)) {
   312→      // Skip groups currently being dragged or recently released (spring
   313→      // still animating). setTabSelected would zero velocity and fight
   314→      // the spring, causing the indicator to jump/snapping wrong.
   315→      if (draggingGroups.has(groupId)) continue
   316→      r.setTabSelected(groupId, tabIndex, tabsCount)
   317→    }
   318→  }, [tabTargets])
   319→
   320→  // --- Pointer handlers ---------------------------------------------
   321→  const localPos = (e: React.PointerEvent<HTMLCanvasElement>) => {
   322→    const canvas = canvasRef.current!
   323→    const rect = canvas.getBoundingClientRect()
   324→    return { x: e.clientX - rect.left, y: e.clientY - rect.top }
   325→  }
   326→
   327→  /** Compute scroll velocity (px/s) from recent samples. Returns the
   328→   *  vertical scroll velocity (negative = finger moved down = scroll up).
   329→   *  Takes the per-pointer sample buffer so each pointer computes its own. */
   330→  const computeReleaseVelocity = (
   331→    samples: { t: number; x: number; y: number }[]
   332→  ): number => {
   333→    if (samples.length < 2) return 0
   334→    // Use the last ~100ms of samples for a stable estimate.
   335→    const now = samples[samples.length - 1].t
   336→    const cutoff = now - 100
   337→    let oldest = samples[samples.length - 1]
   338→    for (let i = samples.length - 1; i >= 0; i--) {
   339→      if (samples[i].t < cutoff) break
   340→      oldest = samples[i]
   341→    }
   342→    const dt = (now - oldest.t) / 1000
   343→    if (dt < 0.001) return 0
   344→    const dy = samples[samples.length - 1].y - oldest.y
   345→    // Positive dy (finger moved down) → negative scroll velocity (scroll up).
   346→    return -dy / dt
   347→  }
   348→
   349→  /** Compute release velocity (px/s) on both axes from recent samples.
   350→   *  Faithful to Compose's VelocityTracker which returns an Offset(x, y).
   351→   *  Takes the per-pointer sample buffer so each pointer computes its own. */
   352→  const computeReleaseVelocity2D = (
   353→    samples: { t: number; x: number; y: number }[]
   354→  ): { x: number; y: number } => {
   355→    if (samples.length < 2) return { x: 0, y: 0 }
   356→    const last = samples[samples.length - 1]
   357→    const now = last.t
   358→    const cutoff = now - 100
   359→    let oldest = last
   360→    for (let i = samples.length - 1; i >= 0; i--) {
   361→      if (samples[i].t < cutoff) break
   362→      oldest = samples[i]
   363→    }
   364→    const dt = (now - oldest.t) / 1000
   365→    if (dt < 0.001) return { x: 0, y: 0 }
   366→    return {
   367→      x: (last.x - oldest.x) / dt,
   368→      y: (last.y - oldest.y) / dt,
   369→    }
   370→  }
   371→
   372→  const handlePointerDown = React.useCallback(
   373→    (e: React.PointerEvent<HTMLCanvasElement>) => {
   374→      const els = elementsRef.current
   375→      const canvas = canvasRef.current
   376→      const renderer = rendererRefInternal.current
   377→      if (!canvas || !renderer) return
   378→
   379→      const { x, y } = localPos(e)
   380→      const scrollY = renderer.getScrollY()
   381→      const interactions = interactionsRef.current
   382→
   383→      // Hit-test topmost first (last in array = topmost in z-order).
   384→      // Skip decorative elements (no interactions AND not isInteractive)
   385→      // so they don't block hit-test on interactive elements below them.
   386→      // E.g. the slider fill (plain-rect, no interactions) sits on top of
   387→      // the slider track (plain-rect, has onTap/onDrag) — without this
   388→      // skip, pressing on the colored fill would miss the track.
   389→      let hit: GlassElementConfig | null = null
   390→      for (let i = els.length - 1; i >= 0; i--) {
   391→        const el = els[i]
   392→        // Use hitRect (expanded touch target) if set, else fall back to rect.
   393→        // This lets slider tracks (visually 6dp tall) have a ~48dp touch target.
   394→        const hr = el.hitRect ?? el.rect
   395→        const visibleHY = el.scroll ? hr.y - scrollY : hr.y
   396→        // If the element has a rotation (e.g. Glass Playground square), the
   397→        // visual shape is rotated but the hit-test rect is the un-rotated
   398→        // AABB. To match the visual shape, un-rotate the pointer point
   399→        // around the rect center, then test against the un-rotated rect.
   400→        // Faithful to graphicsLayer { rotationZ } which rotates the visual
   401→        // but not the touch target (Compose's pointerInput works in the
   402→        // un-rotated local space).
   403→        let testX = x, testY = y
   404→        const elRot = (el as GlassElementConfig & { elementRotation?: number }).elementRotation
   405→        if (elRot && Math.abs(elRot) > 0.001) {
   406→          const cx = hr.x + hr.w * 0.5
   407→          const cy = (el.scroll ? hr.y - scrollY : hr.y) + hr.h * 0.5
   408→          const dx = x - cx
   409→          const dy = y - cy
   410→          const cos = Math.cos(-elRot)
   411→          const sin = Math.sin(-elRot)
   412→          testX = cx + dx * cos - dy * sin
   413→          testY = cy + dx * sin + dy * cos
   414→        }
   415→        if (
   416→          testX >= hr.x &&
   417→          testX <= hr.x + hr.w &&
   418→          testY >= visibleHY &&
   419→          testY <= visibleHY + hr.h
   420→        ) {
   421→          const hasInteraction = !!interactions?.[el.id]
   422→          if (!hasInteraction && !el.isInteractive) {
   423→            // Decorative element — fall through to elements below.
   424→            continue
   425→          }
   426→          hit = el
   427→          break
   428→        }
   429→      }
   430→
   431→      // If a second pointer lands on the SAME element as an existing pointer
   432→      // AND that element has onTransform, enter transform mode (pinch zoom +
   433→      // rotate). Both pointers transition to 'transform' mode and reference
   434→      // each other via transformPartner. We skip this if the existing pointer
   435→      // is already in transform mode (a third finger on the same element just
   436→      // starts its own pending gesture — it can't join an existing pair).
   437→      if (hit) {
   438→        const hitId = hit.id
   439→        const existingEntry = Array.from(gesturesRef.current.entries()).find(
   440→          ([, g]) => g.pressedId === hitId && g.mode !== 'transform'
   441→        )
   442→        if (existingEntry && interactions?.[hitId]?.onTransform) {
   443→          const [partnerPid, partnerGs] = existingEntry
   444→          // Cancel any pending press highlight on the shared element.
   445→          if (hit.isInteractive && (hit.kind === 'button' || hit.kind === 'text')) {
   446→            renderer.setPressed(hitId, false)
   447→          }
   448→          // Initialize prevPinch from the 2-pointer state.
   449→          const p1 = { x: partnerGs.x, y: partnerGs.y }
   450→          const p2 = { x, y }
   451→          const dx = p2.x - p1.x
   452→          const dy = p2.y - p1.y
   453→          prevPinchRef.current = {
   454→            dist: Math.hypot(dx, dy),
   455→            angle: Math.atan2(dy, dx),
   456→            cx: (p1.x + p2.x) / 2,
   457→            cy: (p1.y + p2.y) / 2,
   458→          }
   459→          // Promote the existing pointer to transform mode.
   460→          partnerGs.mode = 'transform'
   461→          partnerGs.transformPartner = e.pointerId
   462→          // Initialize the new pointer's gesture state directly in transform mode.
   463→          gesturesRef.current.set(e.pointerId, {
   464→            pressedId: hitId,
   465→            startX: x,
   466→            startY: y,
   467→            startClientY: e.clientY,
   468→            startScrollY: renderer.getScrollY(),
   469→            dragStarted: false,
   470→            mode: 'transform',
   471→            hasDrag: !!interactions?.[hitId]?.onDrag,
   472→            velocitySamples: [{ t: performance.now(), x: e.clientX, y: e.clientY }],
   473→            x,
   474→            y,
   475→            transformPartner: partnerPid,
   476→          })
   477→          try {
   478→            canvas.setPointerCapture(e.pointerId)
   479→          } catch {
   480→            // ignore
   481→          }
   482→          return
   483→        }
   484→      }
   485→
   486→      // Otherwise: this pointer starts its own independent gesture. Multiple
   487→      // pointers can be down simultaneously, each with its own GestureState —
   488→      // e.g. drag a slider with one finger while pressing a button with
   489→      // another, or scroll the canvas while dragging a toggle.
   490→      const hasDrag = !!(hit && interactions?.[hit.id]?.onDrag)
   491→      gesturesRef.current.set(e.pointerId, {
   492→        pressedId: hit ? hit.id : null,
   493→        startX: x,
   494→        startY: y,
   495→        startClientY: e.clientY,
   496→        startScrollY: renderer.getScrollY(),
   497→        dragStarted: false,
   498→        mode: 'pending',
   499→        hasDrag,
   500→        velocitySamples: [{ t: performance.now(), x: e.clientX, y: e.clientY }],
   501→        x,
   502→        y,
   503→        transformPartner: null,
   504→      })
   505→
   506→      // For 'button' kind with isInteractive, trigger press highlight
   507→      // immediately. If the gesture later becomes a scroll, we'll cancel.
   508→      // Also support 'text' kind with isInteractive — used by the home page
   509→      // list items, which get a subtle white tint on press.
   510→      // Also support 'glass-shape' with isInteractive + onTap + NO onDrag
   511→      // (e.g. dialog Cancel/Okay buttons) — they get the InteractiveHighlight
   512→      // press glow. Elements WITH onDrag (toggle/slider knobs, CC tiles) are
   513→      // NOT included here — they don't use the button press highlight.
   514→      if (hit && hit.isInteractive) {
   515→        const hasDrag0 = !!interactions?.[hit.id]?.onDrag
   516→        if (hit.kind === 'button' || hit.kind === 'text' || (hit.kind === 'glass-shape' && !hasDrag0 && !!interactions?.[hit.id]?.onTap)) {
   517→          renderer.setPressed(hit.id, true, { x, y })
   518→        }
   519→      }
   520→
   521→      try {
   522→        canvas.setPointerCapture(e.pointerId)
   523→      } catch {
   524→        // ignore
   525→      }
   526→    },
   527→    []
   528→  )
   529→
   530→  const handlePointerMove = React.useCallback(
   531→    (e: React.PointerEvent<HTMLCanvasElement>) => {
   532→      const renderer = rendererRefInternal.current
   533→      const canvas = canvasRef.current
   534→      if (!canvas || !renderer) return
   535→      const { x, y } = localPos(e)
   536→
   537→      // Look up this pointer's gesture state. If there's no entry, the
   538→      // pointer isn't part of any gesture (shouldn't happen — every
   539→      // pointerdown creates an entry — but be defensive).
   540→      const gs = gesturesRef.current.get(e.pointerId)
   541→      if (!gs) return
   542→
   543→      // Update current position (used by transform delta computation and
   544→      // by the partner pointer if it's in transform mode).
   545→      gs.x = x
   546→      gs.y = y
   547→
   548→      // --- Transform mode (2-pointer pinch zoom + rotate) ---
   549→      // Both pointers in a transform pair handle the move independently;
   550→      // each fires onTransform with the deltas computed from the current
   551→      // 2-pointer geometry vs the previous snapshot. This matches Compose's
   552→      // detectTransformGestures which recomputes on every pointer move.
   553→      if (gs.mode === 'transform') {
   554→        const partnerPid = gs.transformPartner
   555→        if (partnerPid == null) return
   556→        const partner = gesturesRef.current.get(partnerPid)
   557→        if (!partner) return
   558→        const id = gs.pressedId
   559→        if (!id) return
   560→        const dx = partner.x - gs.x
   561→        const dy = partner.y - gs.y
   562→        const dist = Math.hypot(dx, dy)
   563→        const angle = Math.atan2(dy, dx)
   564→        const cx = (gs.x + partner.x) / 2
   565→        const cy = (gs.y + partner.y) / 2
   566→        const prev = prevPinchRef.current
   567→        if (prev && prev.dist > 0.001) {
   568→          const gestureZoom = dist / prev.dist
   569→          let gestureRotate = angle - prev.angle
   570→          // Wrap to [-PI, PI].
   571→          if (gestureRotate > Math.PI) gestureRotate -= 2 * Math.PI
   572→          if (gestureRotate < -Math.PI) gestureRotate += 2 * Math.PI
   573→          const pan = { x: cx - prev.cx, y: cy - prev.cy }
   574→          interactionsRef.current?.[id]?.onTransform?.(pan, gestureZoom, gestureRotate)
   575→        }
   576→        prevPinchRef.current = { dist, angle, cx, cy }
   577→        return
   578→      }
   579→
   580→      // Track velocity samples for inertia (always, while pressed).
   581→      gs.velocitySamples.push({ t: performance.now(), x: e.clientX, y: e.clientY })
   582→      // Cap the buffer at ~20 samples.
   583→      if (gs.velocitySamples.length > 20) {
   584→        gs.velocitySamples.shift()
   585→      }
   586→
   587→      const dx = x - gs.startX
   588→      const dy = y - gs.startY
   589→      const absDx = Math.abs(dx)
   590→      const absDy = Math.abs(dy)
   591→
   592→      // --- Pending → commit to drag or scroll ---
   593→      if (gs.mode === 'pending') {
   594→        // Small wiggle threshold — keep press highlight alive for tiny
   595→        // movements (finger jitter on tap). Press highlight position
   596→        // follows the finger during this phase.
   597→        const MOVE_THRESHOLD = 4
   598→
   599→        // While pending, update press highlight position so the glow
   600→        // tracks the finger even before we commit to drag or scroll.
   601→        const id0 = gs.pressedId
   602→        if (id0) {
   603→          const els0 = elementsRef.current
   604→          const el0 = els0.find((b) => b.id === id0)
   605→          if (el0?.kind === 'button' && el0.isInteractive) {
   606→            renderer.setDragPosition(id0, { x, y })
   607→          }
   608→        }
   609→
   610→        if (absDx < MOVE_THRESHOLD && absDy < MOVE_THRESHOLD) return
   611→
   612→        // Decide gesture ownership:
   613→        //   - Buttons (interactive 'button' kind): KEEP the press — do not
   614→        //     scroll-takeover. The press highlight follows the finger
   615→        //     everywhere, matching the original InteractiveHighlight
   616→        //     behavior (press only releases on pointerup). This is the
   617→        //     "和之前一样" the user asked for.
   618→        //   - Elements with onDrag (e.g. lock-screen glass, slider knobs):
   619→        //     the drag owns the gesture — commit immediately on any
   620→        //     directional movement. This prevents the scroll-takeover from
   621→        //     hijacking the lock-screen glass drag (which previously made
   622→        //     the page feel frozen because the glass never moved).
   623→        //   - Text list items (interactive 'text' kind, no onDrag): allow
   624→        //     vertical-dominant scroll-takeover so the home page scrolls.
   625→        //   - Empty / non-interactive: scroll.
   626→        const id = gs.pressedId
   627→        const els = elementsRef.current
   628→        const hitEl = id ? els.find((b) => b.id === id) : null
   629→        const isButton = hitEl?.kind === 'button' && hitEl?.isInteractive
   630→        const hasDrag = !!hitEl && !!interactionsRef.current?.[id!]?.onDrag
   631→        // glass-shape with onTap + isInteractive + NO onDrag = button-like
   632→        // (e.g. dialog Cancel/Okay). Treat like a button: keep press, no
   633→        // scroll-takeover. Elements WITH onDrag (toggle knobs, slider knobs,
   634→        // CC tiles, lock-screen glass) are handled by the hasDrag branch.
   635→        const isShapeButton = !hasDrag && hitEl?.kind === 'glass-shape' && hitEl?.isInteractive && !!interactionsRef.current?.[id!]?.onTap
   636→
   637→        if (hasDrag) {
   638→          // Element owns the gesture — commit to drag immediately.
   639→          gs.mode = 'drag'
   640→          gs.dragStarted = true
   641→          interactionsRef.current?.[id!]?.onDragStart?.({ x, y })
   642→          // Fall through to the committed 'drag' branch below.
   643→        } else if (isButton || isShapeButton) {
   644→          // Button keeps its press — press highlight follows the finger.
   645→          // Update drag position so the glow tracks even large movements.
   646→          renderer.setDragPosition(id!, { x, y })
   647→          // Fall through: mode stays 'pending' so the press never commits
   648→          // to a scroll. On pointerup it will be treated as a tap.
   649→        } else {
   650→          // Text items / empty space → allow scroll-takeover.
   651→          const SCROLL_TAKEOVER_THRESHOLD = 14
   652→          const verticalDominant =
   653→            absDy > absDx + 2 && absDy >= SCROLL_TAKEOVER_THRESHOLD
   654→
   655→          if (verticalDominant) {
   656→            // SCROLL LOCK: only one pointer drives scroll at a time. If
   657→            // another pointer is already scrolling, this pointer stays in
   658→            // 'pending' (no scroll takeover) — prevents two fingers from
   659→            // fighting over scroll.
   660→            const otherScrolling = Array.from(gesturesRef.current.entries()).some(
   661→              ([pid, g]) => pid !== e.pointerId && g.mode === 'scroll'
   662→            )
   663→            if (otherScrolling) {
   664→              return
   665→            }
   666→            // Convert to scroll. Cancel any pending text press.
   667→            if (id) {
   668→              const el = els.find((b) => b.id === id)
   669→              if (el?.isInteractive && el.kind === 'text') {
   670→                renderer.setPressed(id, false)
   671→              }
   672→            }
   673→            gs.mode = 'scroll'
   674→            const scrollDelta = e.clientY - gs.startClientY
   675→            renderer.setScrollY(gs.startScrollY - scrollDelta)
   676→            return
   677→          }
   678→        }
   679→      }
   680→
   681→      // --- Committed modes ---
   682→      if (gs.mode === 'scroll') {
   683→        const scrollDelta = e.clientY - gs.startClientY
   684→        renderer.setScrollY(gs.startScrollY - scrollDelta)
   685→        return
   686→      }
   687→
   688→      if (gs.mode === 'drag') {
   689→        const id = gs.pressedId
   690→        if (!id) return
   691→        const els = elementsRef.current
   692→        const el = els.find((b) => b.id === id)
   693→        if (!el) return
   694→
   695→        // For 'button' kind, forward to renderer for InteractiveHighlight.
   696→        if (el.kind === 'button' && el.isInteractive) {
   697→          renderer.setDragPosition(id, { x, y })
   698→        }
   699→
   700→        // Fire onDrag.
   701→        interactionsRef.current?.[id]?.onDrag?.({ x, y }, { x: dx, y: dy })
   702→      }
   703→    },
   704→    []
   705→  )
   706→
   707→  const handlePointerUp = React.useCallback(
   708→    (e: React.PointerEvent<HTMLCanvasElement>) => {
   709→      const renderer = rendererRefInternal.current
   710→      const canvas = canvasRef.current
   711→
   712→      const gs = gesturesRef.current.get(e.pointerId)
   713→
   714→      // No gesture for this pointer — just release capture if any.
   715→      if (!gs) {
   716→        if (canvas && canvas.hasPointerCapture(e.pointerId)) {
   717→          try { canvas.releasePointerCapture(e.pointerId) } catch { /* ignore */ }
   718→        }
   719→        return
   720→      }
   721→
   722→      const mode = gs.mode
   723→      const id = gs.pressedId
   724→
   725→      // --- Transform mode exit ---
   726→      // When one of the 2 transform pointers lifts, the remaining pointer
   727→      // switches to drag mode (faithful to Compose: a pinch that loses a
   728→      // finger becomes a pan). The shared element id is preserved.
   729→      if (mode === 'transform') {
   730→        const partnerPid = gs.transformPartner
   731→        // Remove this pointer's gesture state.
   732→        gesturesRef.current.delete(e.pointerId)
   733→        prevPinchRef.current = null
   734→        if (partnerPid != null) {
   735→          const partner = gesturesRef.current.get(partnerPid)
   736→          if (partner) {
   737→            partner.transformPartner = null
   738→            partner.mode = 'drag'
   739→            partner.dragStarted = true
   740→            // Re-anchor the drag at the partner's current position so the
   741→            // delta computation is continuous from here.
   742→            partner.startX = partner.x
   743→            partner.startY = partner.y
   744→            if (partner.pressedId) {
   745→              interactionsRef.current?.[partner.pressedId]?.onDragStart?.({ x: partner.x, y: partner.y })
   746→            }
   747→          }
   748→        }
   749→        if (canvas && canvas.hasPointerCapture(e.pointerId)) {
   750→          try { canvas.releasePointerCapture(e.pointerId) } catch { /* ignore */ }
   751→        }
   752→        return
   753→      }
   754→
   755→      // --- Non-transform: release press, fire tap/dragEnd, scroll inertia ---
   756→      if (renderer) {
   757→        // Release button/text/shape-button press.
   758→        if (id) {
   759→          const els = elementsRef.current
   760→          const el = els.find((b) => b.id === id)
   761→          if (el?.isInteractive) {
   762→            const hasDrag1 = !!interactionsRef.current?.[id]?.onDrag
   763→            if (el.kind === 'button' || el.kind === 'text' || (el.kind === 'glass-shape' && !hasDrag1 && !!interactionsRef.current?.[id]?.onTap)) {
   764→              renderer.setPressed(id, false)
   765→            }
   766→          }
   767→        }
   768→
   769→        // Apply scroll inertia on release.
   770→        if (mode === 'scroll') {
   771→          const v = computeReleaseVelocity(gs.velocitySamples)
   772→          if (Math.abs(v) > 50) {
   773→            renderer.setScrollVelocity(v)
   774→          }
   775→        }
   776→
   777→        // Fire onDragEnd / onTap.
   778→        if (id) {
   779→          const { x, y } = localPos(e)
   780→          if (gs.dragStarted) {
   781→            // Compute release velocity (px/s, positive y = downward) from
   782→            // recent pointer samples on BOTH axes — faithful to Compose's
   783→            // VelocityTracker which returns an Offset(x, y). Previously vx
   784→            // was always 0 (only y was tracked), which was wrong for any
   785→            // horizontal-drag consumer.
   786→            const { x: vx, y: vy } = computeReleaseVelocity2D(gs.velocitySamples)
   787→            interactionsRef.current?.[id]?.onDragEnd?.({ x, y }, { x: vx, y: vy })
   788→          } else if (mode === 'pending' || mode === 'drag') {
   789→            // Treat as a tap (no scroll takeover happened and no drag started).
   790→            interactionsRef.current?.[id]?.onTap?.({ x, y })
   791→          }
   792→        }
   793→      }
   794→
   795→      // Remove this pointer's gesture state.
   796→      gesturesRef.current.delete(e.pointerId)
   797→      if (canvas && canvas.hasPointerCapture(e.pointerId)) {
   798→        try { canvas.releasePointerCapture(e.pointerId) } catch { /* ignore */ }
   799→      }
   800→    },
   801→    []
   802→  )
   803→
   804→  return (
   805→    <div ref={containerRef} className={className} style={{ position: 'relative' }}>
   806→      <canvas
   807→        ref={canvasRef}
   808→        onPointerDown={handlePointerDown}
   809→        onPointerMove={handlePointerMove}
   810→        onPointerUp={handlePointerUp}
   811→        onPointerLeave={handlePointerUp}
   812→        onPointerCancel={handlePointerUp}
   813→        style={{
   814→          display: 'block',
   815→          width: '100%',
   816→          height: '100%',
   817→          cursor: 'pointer',
   818→          touchAction: 'none',
   819→        }}
   820→      />
   821→    </div>
   822→  )
   823→}

    34→/* ------------------------------------------------------------------ *
    35→ * LiquidGlassRenderer
    36→ *
    37→ * One opaque WebGL canvas. Render pipeline per frame:
    38→ *   1. Wallpaper background pass (cover-fit).
    39→ *   2. For each button (in order):
    40→ *      a. Outer drop shadow pass (if configured).
    41→ *      b. Element pass (refraction + vibrancy + tint + highlight),
    42→ *         with InteractiveHighlight-driven scale/translation/stretch.
    43→ *      c. White overlay pass (Plus blend, 8% * progress) on press.
    44→ *      d. Radial highlight pass (Plus blend, at finger position) on press.
    45→ *      e. Foreground pass (button label, with press-driven alpha fade).
    46→ *
    47→ * No DOM children — the canvas owns the entire visual surface.
    48→ *
    49→ * NOTE: The class methods are split across multiple files in this
    50→ * directory (methods-*.ts). Each methods file declares its method
    51→ * signatures via `declare module './index'` and exports a record of
    52→ * named functions. Those records are merged onto the prototype at the
    53→ * bottom of this file via Object.assign.
    54→ * ------------------------------------------------------------------ */

   398→    // Stroke mask composite (Canvas2D stroke mask approach)
   399→    const smNames = [
   400→      'uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii',
   401→      'uStrokeMask', 'uMaskOffset', 'uMaskSize',
   402→      'uHighlightColor', 'uHighlightAngle', 'uHighlightFalloff', 'uHighlightAlpha', 'uHighlightMode',
   403→      'uOriginalSize', 'uOriginalCornerRadius', 'uLayerScale', 'uElementRotation',
   404→    ]
   405→    for (const n of smNames) this.uSm[n] = gl.getUniformLocation(this.strokeMaskCompositeProgram, n)
   406→    // Inner shadow mask composite (Canvas2D ring mask approach)
   407→    const isNames = [
   408→      'uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii',
   409→      'uInnerShadowMask', 'uMaskOffset', 'uMaskSize',
   410→      'uInnerShadowColor', 'uInnerShadowAlpha',
   411→      'uOriginalSize', 'uOriginalCornerRadius', 'uLayerScale', 'uElementRotation',
   412→    ]
   413→    for (const n of isNames) this.uIs[n] = gl.getUniformLocation(this.innerShadowMaskCompositeProgram, n)
   414→    const prNames = ['uCanvasSize', 'uOffset', 'uSize', 'uCornerRadii', 'uColor', 'uCornerStyle',
   415→      'uUseContinuousSdf', 'uContinuousSdf', 'uContinuousSdfTexSize', 'uContinuousSdfElementSize']
   416→    for (const n of prNames) this.uPr[n] = gl.getUniformLocation(this.plainRectProgram, n)
   417→    const pbNames = [
   418→      'uBackdrop', 'uCanvasSize', 'uWallpaperSize', 'uOffset', 'uSize',
   419→      'uBlurRadius', 'uTintColor', 'uTintIntensity',
   420→    ]
   421→    for (const n of pbNames) this.uPb[n] = gl.getUniformLocation(this.progressiveBlurProgram, n)
   422→    const cpNames = ['uTexture', 'uCanvasSize']
   423→    for (const n of cpNames) this.uCp[n] = gl.getUniformLocation(this.copyProgram, n)
   424→    const sfNames = ['uColor']
   425→    for (const n of sfNames) this.uSf[n] = gl.getUniformLocation(this.solidFillProgram, n)
   426→    const ccNames = ['uTexture', 'uTexSize', 'uBrightness', 'uContrast', 'uSaturation']
   427→    for (const n of ccNames) this.uCc[n] = gl.getUniformLocation(this.colorControlsProgram, n)
   428→    const stNames = ['uTexture', 'uCanvasSize', 'uTintColor']
   429→    for (const n of stNames) this.uSt[n] = gl.getUniformLocation(this.sceneTintProgram, n)
   430→  }

   432→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   433→  ensureBlurPrograms(tapCount: number): void {
   434→    if (this.blurPrograms.has(tapCount)) return
   435→    const gl = this.gl
   436→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   437→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   438→    const mk = (fs: WebGLShader) => {
   439→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   440→      const p = gl.createProgram()!
   441→      gl.attachShader(p, vs)
   442→      gl.attachShader(p, fs)
   443→      gl.bindAttribLocation(p, 0, 'aPos')
   444→      gl.linkProgram(p)
   445→      gl.deleteShader(vs)
   446→      gl.deleteShader(fs)
   447→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   448→        const log = gl.getProgramInfoLog(p)
   449→        gl.deleteProgram(p)
   450→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   451→      }
   452→      return p
   453→    }
   454→    const hProg = mk(hFs)
   455→    const vProg = mk(vFs)
   456→    const uH: Record<string, WebGLUniformLocation | null> = {
   457→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   458→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   459→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   460→    }
   461→    const uV: Record<string, WebGLUniformLocation | null> = {
   462→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   463→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   464→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   465→    }
   466→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   467→  }
   468→
   469→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   470→   *  blurred result into blurFboB, returns blurFboBTex.
   471→   *  Saves/restores the currently-bound framebuffer.
   472→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   473→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   474→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   475→    const gl = this.gl
   476→    const w = this.fboW
   477→    const h = this.fboH
   478→    // Compute tap count, capped by blurTapCap (performance knob).
   479→    let taps = computeBlur1DTapCount(radius)
   480→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   481→    this.ensureBlurPrograms(taps)
   482→    const entry = this.blurPrograms.get(taps)!
   483→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   484→    gl.disable(gl.BLEND)
   485→
   486→    // Pass 1: horizontal — srcTex → blurFboA
   487→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   488→    gl.viewport(0, 0, w, h)
   489→    gl.useProgram(entry.hProg)
   490→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   491→    gl.enableVertexAttribArray(entry.aPosH)
   492→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   493→    gl.activeTexture(gl.TEXTURE0)
   494→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   495→    gl.uniform1i(entry.uH['uTexture'], 0)
   496→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   497→    gl.uniform1f(entry.uH['uRadius'], radius)
   498→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   499→
   500→    // Pass 2: vertical — blurFboATex → blurFboB
   501→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   502→    gl.viewport(0, 0, w, h)
   503→    gl.useProgram(entry.vProg)
   504→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   505→    gl.enableVertexAttribArray(entry.aPosV)
   506→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   507→    gl.activeTexture(gl.TEXTURE0)
   508→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   509→    gl.uniform1i(entry.uV['uTexture'], 0)
   510→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   511→    gl.uniform1f(entry.uV['uRadius'], radius)
   512→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   513→
   514→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   515→    gl.viewport(0, 0, w, h)
   516→    return this.blurFboBTex!
   517→  }
   518→
   519→  /** Lazy-compile highlight blur programs (alpha-blurring, sigma semantics).
   520→   *  Separate from ensureBlurPrograms because the shader is different
   521→   *  (blurs alpha, no early-return, integer-σ-spaced taps). */
   522→  ensureHighlightBlurPrograms(tapCount: number): void {
   523→    if (this.highlightBlurPrograms.has(tapCount)) return
   524→    const gl = this.gl
   525→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateHighlightBlurShader(tapCount, 'horizontal'))
   526→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateHighlightBlurShader(tapCount, 'vertical'))
   527→    const mk = (fs: WebGLShader) => {

   529→      const p = gl.createProgram()!

   543→    const hProg = mk(hFs)
   544→    const vProg = mk(vFs)
   545→    const uH: Record<string, WebGLUniformLocation | null> = {

   556→  }

   558→  /** 2-pass Gaussian blur on a highlight stroke MASK (alpha only).
   559→   *  Faithful to Android BlurMaskFilter(NORMAL, sigma):
   560→   *    - sigma = blurRadiusPx (the Android radius param IS sigma)
   561→   *    - convolves the mask's ALPHA with a Gaussian kernel
   562→   *    - sub-pixel sigma (0.25px) still blurs (no 0.5 early-return)
   563→   *  Reads srcTex (alpha mask), writes blurFboB, returns blurFboBTex.
   564→   *  Saves/restores the currently-bound framebuffer. */
   565→  blurHighlightMask(srcTex: WebGLTexture, sigmaPx: number): WebGLTexture {
   566→    const gl = this.gl
   567→    const w = this.fboW
   568→    const h = this.fboH
   569→    let taps = computeHighlightBlurTapCount(sigmaPx)
   570→    taps = Math.min(taps, Math.max(3, this.blurTapCap | 0))
   571→    this.ensureHighlightBlurPrograms(taps)
   572→    const entry = this.highlightBlurPrograms.get(taps)!
   573→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   574→    gl.disable(gl.BLEND)
   575→
   576→    // Pass 1: horizontal — srcTex → blurFboA
   577→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   578→    gl.viewport(0, 0, w, h)
   579→    gl.useProgram(entry.hProg)
   580→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   581→    gl.enableVertexAttribArray(entry.aPosH)
   582→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   583→    gl.activeTexture(gl.TEXTURE0)
   584→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   585→    gl.uniform1i(entry.uH['uTexture'], 0)
   586→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   587→    gl.uniform1f(entry.uH['uRadius'], sigmaPx)
   588→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   589→
   590→    // Pass 2: vertical — blurFboATex → blurFboB
   591→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   592→    gl.viewport(0, 0, w, h)
   593→    gl.useProgram(entry.vProg)
   594→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   595→    gl.enableVertexAttribArray(entry.aPosV)
   596→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   597→    gl.activeTexture(gl.TEXTURE0)
   598→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   599→    gl.uniform1i(entry.uV['uTexture'], 0)
   600→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   601→    gl.uniform1f(entry.uV['uRadius'], sigmaPx)
   602→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   603→
   604→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   605→    gl.viewport(0, 0, w, h)
   606→    return this.blurFboBTex!
   607→  }

   609→  dispose() {
   610→    if (this.rafId !== null) cancelAnimationFrame(this.rafId)
   611→    this.rafId = null
   612→    if (this.animRafId !== null) cancelAnimationFrame(this.animRafId)
   613→    this.animRafId = null
   614→    const gl = this.gl
   615→    if (this.wallpaperTexture) gl.deleteTexture(this.wallpaperTexture)
   616→    for (const tex of this.fgTextures.values()) gl.deleteTexture(tex)
   617→    this.fgTextures.clear()
   618→    for (const entry of this.strokeMaskCache.values()) gl.deleteTexture(entry.tex)
   619→    this.strokeMaskCache.clear()
   620→    for (const entry of this.innerShadowMaskCache.values()) gl.deleteTexture(entry.tex)
   621→    this.innerShadowMaskCache.clear()
   622→    if (this.fboA) gl.deleteFramebuffer(this.fboA)
   623→    if (this.fboATex) gl.deleteTexture(this.fboATex)
   624→    if (this.fboB) gl.deleteFramebuffer(this.fboB)
   625→    if (this.fboBTex) gl.deleteTexture(this.fboBTex)
   626→    this.fboA = this.fboB = null
   627→    this.fboATex = this.fboBTex = null
   628→    if (this.tabsBackdropFbo) gl.deleteFramebuffer(this.tabsBackdropFbo)
   629→    if (this.tabsBackdropTex) gl.deleteTexture(this.tabsBackdropTex)
   630→    this.tabsBackdropFbo = null
   631→    this.tabsBackdropTex = null
   632→    // GP element FBO + blur FBOs + programs
   633→    if (this.gpElementFbo) gl.deleteFramebuffer(this.gpElementFbo)
   634→    if (this.gpElementTex) gl.deleteTexture(this.gpElementTex)
   635→    if (this.blurFboA) gl.deleteFramebuffer(this.blurFboA)
   636→    if (this.blurFboATex) gl.deleteTexture(this.blurFboATex)
   637→    if (this.blurFboB) gl.deleteFramebuffer(this.blurFboB)
   638→    if (this.blurFboBTex) gl.deleteTexture(this.blurFboBTex)
   639→    this.gpElementFbo = this.blurFboA = this.blurFboB = null
   640→    this.gpElementTex = this.blurFboATex = this.blurFboBTex = null
   641→    if (this.highlightMaskFbo) gl.deleteFramebuffer(this.highlightMaskFbo)
   642→    if (this.highlightMaskTex) gl.deleteTexture(this.highlightMaskTex)
   643→    this.highlightMaskFbo = null
   644→    this.highlightMaskTex = null
   645→    if (this.dialogBackdropFbo) gl.deleteFramebuffer(this.dialogBackdropFbo)
   646→    if (this.dialogBackdropTex) gl.deleteTexture(this.dialogBackdropTex)
   647→    this.dialogBackdropFbo = null
   648→    this.dialogBackdropTex = null
   649→    this.dialogBackdropKey = null
   650→    for (const { hProg, vProg } of this.blurPrograms.values()) {
   651→      gl.deleteProgram(hProg)
   652→      gl.deleteProgram(vProg)
   653→    }
   654→    this.blurPrograms.clear()
   655→    for (const { hProg, vProg } of this.highlightBlurPrograms.values()) {
   656→      gl.deleteProgram(hProg)
   657→      gl.deleteProgram(vProg)
   658→    }
   659→    this.highlightBlurPrograms.clear()
   660→    if (this.sdfTexture) gl.deleteTexture(this.sdfTexture)
   661→    this.sdfTexture = null
   662→    for (const { tex } of this.continuousSdfPool.values()) gl.deleteTexture(tex)
   663→    this.continuousSdfPool.clear()
   664→    this.continuousSdfTexture = null
   665→    this.continuousSdfKey = null
   666→    gl.deleteProgram(this.elementProgram)
   667→    gl.deleteProgram(this.shadowProgram)
   668→    gl.deleteProgram(this.wallpaperProgram)
   669→    gl.deleteProgram(this.foregroundProgram)
   670→    gl.deleteProgram(this.highlightProgram)
   671→    gl.deleteProgram(this.tintProgram)
   672→    gl.deleteProgram(this.rimHighlightProgram)
   673→    gl.deleteProgram(this.highlightStrokeProgram)
   674→    gl.deleteProgram(this.highlightCompositeProgram)
   675→    gl.deleteProgram(this.strokeMaskCompositeProgram)
   676→    gl.deleteProgram(this.innerShadowMaskCompositeProgram)
   677→    gl.deleteProgram(this.plainRectProgram)
   678→    gl.deleteProgram(this.progressiveBlurProgram)
   679→    gl.deleteProgram(this.copyProgram)
   680→    gl.deleteProgram(this.solidFillProgram)
   681→    gl.deleteProgram(this.colorControlsProgram)
   682→    gl.deleteProgram(this.sceneTintProgram)
   683→    gl.deleteBuffer(this.quadBuffer)
   684→  }

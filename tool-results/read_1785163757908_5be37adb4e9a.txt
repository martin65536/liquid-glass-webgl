    86→    86→/* ------------------------------------------------------------------ *
    87→    87→ * Element kinds — extends the glass-button model to cover the catalog.
    88→    88→ *
    89→    89→ *   - 'button'          : existing glass button (default)
    90→    90→ *   - 'glass-shape'     : glass rect with NO label and NO press effect

   417→   417→   * The renderer applies ProgressConverter to derive the visual progress:
   418→   418→   *   derived = p<0 ? convert(p) : p<=1 ? p : 1+convert(p-1)
   419→   419→   *   where convert(x) = (1 - exp(-|x|)) * sign(x)
   420→   420→   * Then:
   421→   421→   *   translationY = -48dp * (1 - derived)
   422→   422→   *   scaleX /= 1 + 0.1 * max(0, derived - 1)
   423→   423→   *   scaleY *= 1 + 0.1 * max(0, derived - 1)
   424→   424→   */
   425→   425→  enterProgress?: number
   426→   426→  /**
   427→   427→   * Control-center SAFE enter progress (clamped 0..1). Faithful to
   428→   428→   * ControlCenterContent.kt safeEnterProgressAnimation. Drives alpha:
   429→   429→   *   alpha = EaseIn.transform(safeProgress)
   430→   430→   *   where EaseIn = CubicBezierEasing(0.42, 0, 1, 1)
   431→   431→   * The safe progress animates independently (critical spring, no velocity)
   432→   432→   * so alpha/dim/blur never overshoot even when the raw progress bounces.
   433→   433→   */
   434→   434→  enterSafeProgress?: number
   435→   435→  /**
   436→   436→   * ControlCenter overscroll row-stretch. When set AND derived progress > 1,
   437→   437→   * the element's y is offset by `enterStretchFactor * max(0, derived-1) * 32dp`.
   438→   438→   * Faithful to ControlCenterContent.kt's spacerLayoutModifier which grows
   439→   439→   * inter-row spacing by 32dp (large spacer) or 16dp (small spacer) per unit
   440→   440→   * of DERIVED overshoot (not raw — the ProgressConverter dampens it).
   441→   441→   */
   442→   442→  enterStretchFactor?: number
   443→   443→  /**
   444→   444→   * Render ON TOP of all other elements (after the main render loop).
   445→   445→   * Used by the control-center dim overlay, which must composite above
   446→   446→   * the glass tiles (faithful to ControlCenterContent.kt's drawWithContent
   447→   447→   * which draws drawRect(dim) AFTER drawContent()). Elements with this
   448→   448→   * flag are collected and rendered in a second pass. Hit-test order is
   449→   449→   * unaffected (still array order).
   450→   450→   */
   451→   451→  renderOnTop?: boolean
   452→   452→  /** Sample the WALLPAPER (not the scene FBO) as the glass backdrop for
   453→   453→   *  refraction. Used by the back button / theme toggle on pages with a
   454→   454→   *  dark scrim/dim overlay (Dialog, ControlCenter) so the glass refracts
   455→   455→   *  the clean wallpaper instead of the darkened scene. Only affects the
   456→   456→   *  uBackdrop texture binding in renderGlassElementPass — the element
   457→   457→   *  still composites onto the scene FBO normally. */
   458→   458→  sampleWallpaper?: boolean
   459→   459→  /** Scrim color applied to the wallpaper backdrop BEFORE colorControls/blur/lens.
   460→   460→   *  Only used when sampleWallpaper=true. Faithful to DialogContent.kt /
   461→   461→   *  ControlCenterContent.kt where the scrim is painted onto the wallpaper Image
   462→   462→   *  (via BackdropDemoScaffold's drawWithContent modifier), so the LayerBackdrop
   463→   463→   *  captures wallpaper+scrim as one opaque layer. [r,g,b,a] 0..1; a=0 = no scrim. */
   464→   464→  scrimColor?: [number, number, number, number]
   465→   465→  /** When true, this element samples a dedicated backdrop FBO (rendered by
   466→   466→   *  renderDialogBackdrop) holding wallpaper+scrim+colorControls as one opaque
   467→   467→   *  layer, instead of the scene FBO. Faithful to the original's LayerBackdrop
   468→   468→   *  which captures wallpaper+scrim. Used by the dialog card with 2-pass blur
   469→   469→   *  to get the correct colorControls→blur→lens order. */
   470→   470→  backdropFbo?: boolean
   471→   471→  /** GLOBAL backdrop blur radius (CSS px), applied to the wallpaper RIGHT
   472→   472→   *  AFTER renderBackground — before any element (dim, tiles) composites on
   473→   473→   *  top. Faithful to ControlCenterContent.kt's backdrop Image modifier:
   474→   474→   *    .drawWithContent { drawContent(); drawRect(dim) }
   475→   475→   *    .graphicsLayer { BlurEffect(4dp * progress) }
   476→   476→   *  The graphicsLayer wraps the wallpaper painter, so only the WALLPAPER is
   477→   477→   *  blurred; the dim (drawn by drawWithContent AFTER drawContent) is crisp.
   478→   478→   *  Set this on the cc-dim element (first element). The renderer scans for
   479→   479→   *  it and blurs fboA (wallpaper) → blurFboB → copies back to fboA before
   480→   480→   *  the element loop. blurRadius=0 → no blur. */
   481→   481→  sceneBlurRadius?: number
   482→   482→  /** When true, the rim highlight angle is driven by renderer.gravityAngle
   483→   483→   *  (updated live via setGravityAngle) instead of the static highlight.angle
   484→   484→   *  baked at build time. Used by CC tiles so the highlight rotates smoothly
   485→   485→   *  with device orientation WITHOUT rebuilding the catalog (gravityAngle is
   486→   486→   *  pushed to the renderer via setGravityAngle, not via React state). */
   487→   487→  useGravityAngle?: boolean
   488→   488→  /** When true, the element pass samples a precomputed continuous-curvature
   489→   489→   *  SDF texture (generated from the G2-continuous Bezier path in
   490→   490→   *  continuous-curve.ts) for its shape, instead of the analytic
   491→   491→   *  sdRoundedRect / sdContinuousRoundedRect SDF. Only applied to the dialog
   492→   492→   *  card for now. The renderer's loadContinuousSdf() must have been called
   493→   493→   *  for the element's (w, h, radius) before rendering — see methods-render.ts. */
   494→   494→  useContinuousSdf?: boolean
   495→   495→}

   522→   522→/* ------------------------------------------------------------------ *
   523→   523→ * ToggleGroupState — faithful port of DampedDragAnimation.kt.
   524→   524→ *
   525→   525→ * The renderer maintains one ToggleGroupState per groupId. The fraction
   526→   526→ * (0..1) is animated with a critically damped spring (k=1000, ζ=1) so
   527→   527→ * it tracks the target with no overshoot — matching the smooth knob
   528→   528→ * glide of the original. The pressProgress (also critically damped)
   529→   529→ * drives the knob scale (1→1.5) and the white overlay alpha (1→0).
   530→   530→ *
   531→   531→ * Faithful to DampedDragAnimation.kt:
   532→   532→ *   - valueAnimation:        spring(1f, 1000f)  — critically damped
   533→   533→ *   - velocityAnimation:     spring(0.5f, 300f) — underdamped (drag velocity)
   534→   534→ *   - pressProgressAnimation: spring(1f, 1000f) — critically damped
   535→   535→ *   - scaleXAnimation:       spring(0.6f, 250f) — underdamped, more bounce
   536→   536→ *   - scaleYAnimation:       spring(0.7f, 250f) — underdamped, less bounce
   537→   537→ *
   538→   538→ * The velocity (tracked via a VelocityTracker on the value animation)
   539→   539→ * drives the squash-and-stretch in the layerBlock:
   540→   540→ *   scaleX /= 1 - clamp(vel/50 * 0.75, -0.2, 0.2)
   541→   541→ *   scaleY *= 1 - clamp(vel/50 * 0.25, -0.2, 0.2)
   542→   542→ * ------------------------------------------------------------------ */
   543→   543→export interface ToggleGroupState {
   544→   544→  // Knob position fraction (0..1). Animated with critically damped spring.
   545→   545→  fraction: number
   546→   546→  fractionVelocity: number
   547→   547→  targetFraction: number
   548→   548→  // Press progress (0 = released, 1 = pressed). Drives scale + white overlay.
   549→   549→  pressProgress: number
   550→   550→  pressVelocity: number
   551→   551→  targetPress: number
   552→   552→  // Knob scale X (1..1.5). Underdamped spring (ζ=0.6, k=250) — more bounce.
   553→   553→  scaleX: number
   554→   554→  scaleXVelocity: number
   555→   555→  targetScaleX: number
   556→   556→  // Knob scale Y (1..1.5). Underdamped spring (ζ=0.7, k=250) — less bounce.
   557→   557→  scaleY: number
   558→   558→  scaleYVelocity: number
   559→   559→  targetScaleY: number
   560→   560→  // Drag velocity (normalized 0..1 per second). Underdamped spring (ζ=0.5, k=300).
   561→   561→  // Drives the squash-and-stretch in the layerBlock. Tracked via a simple
   562→   562→  // velocity tracker on the fraction animation.
   563→   563→  velocity: number
   564→   564→  velocityVelocity: number
   565→   565→  targetVelocity: number
   566→   566→  // True while the user is dragging the knob. While true, drag deltas
   567→   567→  // update targetFraction directly (no spring lag, matches onDrag in
   568→   568→  // DampedDragAnimation which updates `fraction` instantly).
   569→   569→  isDragging: boolean
   570→   570→  // True after a drag release (endToggleDrag/endSliderDrag) — enables
   571→   571→  // velocity tracking from fraction rate-of-change. Set false by
   572→   572→  // setToggleTarget/setTabSelected (tap = no velocity animation).
   573→   573→  // Faithful to DampedDragAnimation: velocity is only non-zero after a
   574→   574→  // drag (VelocityTracker accumulates during drag); animateToValue (tap)
   575→   575→  // checks `if (velocity != 0f)` — for taps velocity is 0 → no stretch.
   576→   576→  trackVelocityAfterRelease: boolean
   577→   577→  // VelocityTracker fed (time, fraction) every animation frame — faithful
   578→   578→  // port of DampedDragAnimation.updateVelocity() which calls
   579→   579→  // velocityTracker.addPosition(now, Offset(value, 0)) inside the
   580→   580→  // valueAnimation.animateTo block. Replaces the old naive Δfraction/Δt
   581→   581→  // difference (which was spike-prone on small dt) with a least-squares
   582→   582→  // fit, matching Compose's VelocityTracker.calculateVelocity().
   583→   583→  velocityTracker: VelocityTracker1D

   585→   585→  lastFractionForVelocity: number
   586→   586→  lastFractionTime: number
   587→   587→

   589→   589→  // 78/56 ≈ 1.393 for bottom tabs indicator.
   590→   590→  // Set when the toggle group is first created (via ensureToggleState).
   591→   591→  pressedScale: number
   592→   592→  // valueRangeSpan = (valueRange.endInclusive - valueRange.start).
   593→   593→  // Faithful to DampedDragAnimation.kt's updateVelocity():
   594→   594→  //   targetVelocity = velocityTracker.calculateVelocity().x / valueRangeSpan
   595→   595→  // Toggle/slider: valueRange = 0..1 → span 1 (no division).

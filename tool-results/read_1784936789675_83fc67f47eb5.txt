    32→    32→function useSystemTheme(): Theme {
    33→    33→  const [theme, setTheme] = React.useState<Theme>('light')
    34→    34→  React.useEffect(() => {
    35→    35→    if (typeof window === 'undefined' || !window.matchMedia) return
    36→    36→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    37→    37→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    38→    38→    update()
    39→    39→    mq.addEventListener('change', update)
    40→    40→    return () => mq.removeEventListener('change', update)
    41→    41→  }, [])
    42→    42→  return theme
    43→    43→}

   251→   251→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   252→   252→    const MIN_FPS = 55
   253→   253→    if (fps >= MIN_FPS) {
   254→   254→      // This DPR works — try higher
   255→   255→      perfLoRef.current = candidateDpr
   256→   256→    } else {
   257→   257→      // Too slow — try lower
   258→   258→      perfHiRef.current = candidateDpr
   259→   259→    }
   260→   260→
   261→   261→    if (iteration >= maxIterations || perfHiRef.current - perfLoRef.current <= 0.25) {
   262→   262→      // Converged — use the highest DPR that worked (lo)
   263→   263→      const bestDpr = Math.max(0.5, Math.min(deviceDpr, Math.round(perfLoRef.current * 4) / 4))
   264→   264→      perfPhaseRef.current = 'done'
   265→   265→      const isGood = bestDpr >= deviceDpr * 0.5
   266→   266→      const statusText = isGood
   267→   267→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   268→   268→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   269→   269→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   270→   270→      setState({
   271→   271→        customDpr: bestDpr,
   272→   272→        perfDone: true,
   273→   273→        perfResultDpr: bestDpr,
   274→   274→        perfStatusText: statusText,
   275→   275→        perfProgress: null,
   276→   276→        // Don't reset perfGlassAngle — settle animation will handle it
   277→   277→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   278→   278→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   279→   279→        perfProgressFrac: 1,
   280→   280→      })
   281→   281→    } else {
   282→   282→      // More iterations needed — increment trigger to re-fire the effect
   283→   283→      perfPhaseRef.current = 'idle'
   284→   284→      setState((prev) => ({
   285→   285→        perfProgress: 'running',
   286→   286→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   287→   287→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   288→   288→      }))
   289→   289→    }

   331→   331→    let raf = 0
   332→   332→    // Exponential decay rate for deformMul: mul = e^(-3*t), reaches ~0.05 by 1s
   333→   333→    const DECAY_RATE = 3
   334→   334→    // Exit button: starts appearing after 0.3s, fully visible by 0.8s

   362→   362→        return
   363→   363→      }
   364→   364→      raf = requestAnimationFrame(tick)
   365→   365→    }
   366→   366→    raf = requestAnimationFrame(tick)
   367→   367→    return () => cancelAnimationFrame(raf)
   368→   368→  }, [destination, state.perfDone, rendererReady])

   537→   537→  // Device motion → gravity angle for glass highlight direction.
   538→   538→  // Faithful to UISensor.kt:
   539→   539→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   540→   540→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   541→   541→  //
   542→   542→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   543→   543→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   544→   544→  // rebuilding the catalog (which would lose drag state and cause jank).
   545→   545→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   546→   546→  // renderer.gravityAngle live each frame.
   547→   547→  //
   548→   548→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   549→   549→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   550→   550→  // device's acceleration INCLUDING gravity in device-space axes:
   551→   551→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   552→   552→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   553→   553→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   554→   554→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   555→   555→  // vectors — atan2(beta, gamma) does NOT match the original.)
   556→   556→  //
   557→   557→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   558→   558→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   559→   559→  // Default 45° (matches UISensor.kt's initial value).
   560→   560→  React.useEffect(() => {
   561→   561→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   562→   562→    let smoothed = 45
   563→   563→    const alpha = 0.5
   564→   564→    const handler = (e: DeviceMotionEvent) => {
   565→   565→      const acc = e.accelerationIncludingGravity
   566→   566→      if (!acc || acc.x == null || acc.y == null) return
   567→   567→      const x = acc.x
   568→   568→      const y = acc.y
   569→   569→      let target = Math.atan2(y, x) * 180 / Math.PI
   570→   570→      // Shortest-path interpolation: wrap the delta to [-180, 180] so
   571→   571→      // rotating through ±180° doesn't cause a 358° jump.
   572→   572→      let delta = target - smoothed
   573→   573→      while (delta > 180) delta -= 360
   574→   574→      while (delta < -180) delta += 360
   575→   575→      smoothed += delta * alpha
   576→   576→      // Wrap smoothed to [-180, 180] to keep the value bounded.
   577→   577→      while (smoothed > 180) smoothed -= 360
   578→   578→      while (smoothed < -180) smoothed += 360
   579→   579→      const rad = smoothed * Math.PI / 180
   580→   580→      rendererRef.current?.setGravityAngle(rad)
   581→   581→    }
   582→   582→    window.addEventListener('devicemotion', handler)
   583→   583→    return () => window.removeEventListener('devicemotion', handler)
   584→   584→  }, [rendererRef])

   668→   668→  //   averageLuminance → luminanceAnimation.animateTo(target, tween(1000))

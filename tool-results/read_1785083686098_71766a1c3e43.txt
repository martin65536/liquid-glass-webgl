     1→import { SDF_GLSL } from './sdf'
     2→
     3→/* ------------------------------------------------------------------ *
     4→ * Interactive highlight pass — faithful port of the AGSL shader from
     5→ * InteractiveHighlight.kt:
     6→ *
     7→ *   uniform float2 size;
     8→ *   layout(color) uniform half4 color;
     9→ *   uniform float radius;
    10→ *   uniform float2 position;
    11→ *   half4 main(float2 coord) {
    12→ *       float dist = distance(coord, position);
    13→ *       float intensity = smoothstep(radius, radius * 0.5, dist);
    14→ *       return color * intensity;
    15→ *   }
    16→ *
    17→ * Combined with the white-overlay pass (drawRect white 8% Plus blend)
    18→ * this reproduces the catalog's drag-follow press glow. Drawn ABOVE the
    19→ * element pass and BELOW the foreground (label) pass, all clipped to the
    20→ * element rect.
    21→ *
    22→ * BLEND CORRECTNESS: Skia's BlendMode.Plus is `result = S + D` where S
    23→ * is the source color in PREMULTIPLIED form. Compose stores
    24→ * `White(1.0).copy(alpha = a)` premultiplied as `(a, a, a, a)`, so the
    25→ * actual contribution to dst.rgb is `color.rgb * color.a * intensity`.
    26→ *
    27→ * To replicate in WebGL we output the premultiplied contribution as RGB
    28→ * and use blendFunc(ONE, ONE) (pure additive). Using (SRC_ALPHA, ONE)
    29→ * here would re-multiply by src.a, squaring the alpha and making the
    30→ * glow ~50x dimmer (e.g. 0.15² = 0.0225 — essentially invisible).
    31→ * ------------------------------------------------------------------ */
    32→export const HIGHLIGHT_FRAGMENT_SHADER = /* glsl */ `
    33→precision highp float;
    34→
    35→uniform vec2  uCanvasSize;
    36→uniform vec2  uOffset;       // element top-left in canvas px (top-left origin) — SCALED rect
    37→uniform vec2  uSize;         // element size in canvas px — SCALED
    38→uniform vec4  uCornerRadii;  // capsule radii (topLeft, topRight, bottomRight, bottomLeft) in px — SCALED
    39→uniform vec4  uColor;        // rgba; usually white * (alpha = 0.15 * progress)
    40→uniform float uRadius;       // glow radius in canvas px (= minDim * 1.5, SCALED space)
    41→uniform vec2  uPosition;     // finger position in element-local px (top-left origin, SCALED space)
    42→// --- ORIGINAL-SPACE SDF clip (faithful to graphicsLayer { scaleX, scaleY }) ---
    43→// The press glow (InteractiveHighlight) is drawn INSIDE the graphicsLayer, so
    44→// it is clipped to the ORIGINAL capsule shape, then scaled with the layer.
    45→// The glow position + radius are in SCALED space (they track the finger in
    46→// screen px), but the clip SDF is in original space so the capsule clip stays
    47→// correct when the button is stretched.
    48→uniform vec2  uOriginalSize;
    49→uniform float uOriginalCornerRadius;
    50→uniform vec2  uLayerScale;
    51→uniform float uElementRotation;
    52→
    53→${SDF_GLSL}
    54→
    55→void main() {
    56→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
    57→    vec2 localCoord = screenCoord - uOffset;
    58→
    59→    // --- Capsule clip in ORIGINAL space (faithful to graphicsLayer clip) ---
    60→    vec2 elementCenter = uOffset + uSize * 0.5;
    61→    vec2 centeredScreen = screenCoord - elementCenter;
    62→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
    63→    vec2 centeredOrig = centeredScreen / layerScale;
    64→    vec2 origHalfSize = uOriginalSize * 0.5;
    65→    float sd = sdShape(rotateBy(centeredOrig, -uElementRotation), origHalfSize, uOriginalCornerRadius);
    66→    if (sd > 0.5) discard;
    67→    float clipAlpha = 1.0 - smoothstep(-0.5, 0.5, sd);
    68→
    69→    // Faithful AGSL port: smoothstep(radius, radius*0.5, dist) means
    70→    // intensity = 1 at dist <= radius*0.5, fading to 0 at dist >= radius.
    71→    // dist + uPosition are in SCALED local space (finger tracks screen px).
    72→    float dist = distance(localCoord, uPosition);
    73→    float intensity = smoothstep(uRadius, uRadius * 0.5, dist);
    74→
    75→    // Premultiplied Plus-blend contribution. Renderer uses blendFunc(ONE, ONE)
    76→    // so result.rgb = contribution + dst.rgb (clamped to 1).
    77→    vec3 contribution = uColor.rgb * uColor.a * intensity * clipAlpha;
    78→    gl_FragColor = vec4(contribution, 1.0);
    79→}
    80→`
    81→
    82→/* ------------------------------------------------------------------ *
    83→ * White-overlay pass — a flat white fill at low alpha, used for the
    84→ * first half of the InteractiveHighlight effect (drawRect white 8%
    85→ * Plus blend in the Kotlin source). Plus blend is approximated by
    86→ * additive blending in the renderer.
    87→ *
    88→ * IMPORTANT: the original Compose chain wraps everything (this drawRect
    89→ * included) in a graphicsLayer with clip=true and the capsule shape.
    90→ * Without SDF clipping here, the white fill would flood the AABB
    91→ * corners OUTSIDE the capsule. We discard sd > 0.5 with 1px AA.
    92→ * ------------------------------------------------------------------ */
    93→export const TINT_FRAGMENT_SHADER = /* glsl */ `
    94→precision highp float;
    95→
    96→uniform vec2  uCanvasSize;
    97→uniform vec2  uOffset;
    98→uniform vec2  uSize;
    99→uniform vec4  uCornerRadii;
   100→uniform vec4  uColor;
   101→// --- ORIGINAL-SPACE SDF clip (faithful to graphicsLayer { scaleX, scaleY }) ---
   102→// The white overlay (onDrawSurface drawRect) is drawn INSIDE the graphicsLayer,
   103→// so it is clipped to the ORIGINAL capsule shape, then scaled with the layer.
   104→// Computing the clip SDF in original space keeps the capsule clip correct when
   105→// the button is stretched (no corner bleed, no stretched-clip artifacts).
   106→uniform vec2  uOriginalSize;
   107→uniform float uOriginalCornerRadius;
   108→uniform vec2  uLayerScale;
   109→uniform float uElementRotation;
   110→
   111→${SDF_GLSL}
   112→
   113→void main() {
   114→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   115→    vec2 elementCenter = uOffset + uSize * 0.5;
   116→    vec2 centeredScreen = screenCoord - elementCenter;
   117→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   118→    vec2 centeredOrig = centeredScreen / layerScale;
   119→    vec2 origHalfSize = uOriginalSize * 0.5;
   120→    float sd = sdShape(rotateBy(centeredOrig, -uElementRotation), origHalfSize, uOriginalCornerRadius);
   121→    if (sd > 0.5) discard;
   122→    float clipAlpha = 1.0 - smoothstep(-0.5, 0.5, sd);
   123→

   125→}
   126→`

   128→/* ------------------------------------------------------------------ *
   129→ * Rim highlight pass — faithful port of HighlightModifier.kt +
   130→ * DefaultHighlightShaderString / AmbientHighlightShaderString.
   131→ *
   132→ * The original draws a STROKE of width strokeWidth centered on the shape
   133→ * edge, blurred by blurRadius, clipped to inside the shape. The stroke is
   134→ * colored by the AGSL shader (Default or Ambient), then composited with
   135→ * the layer's blendMode (Plus for Default, SrcOver for Ambient).
   136→ *
   137→ * We model the visible (post-clip) stroke band as a smooth mask: 1 from
   138→ * sd = -strokeWidth/2 up to sd = 0 (edge), fading to 0 over blur pixels
   139→ * further inward. The outward half is removed by the shape clip (discard
   140→ * for sd > 0).
   141→ *
   142→ * Default shader: returns color * intensity, color = White(1.0).
   143→ *   Plus blend (renderer uses gl.ONE/gl.ONE): output rgb = color * intensity * mask.
   144→ * Ambient shader: returns half4(t,t,t,1.0) * intensity, t = step(0,d).
   145→ *   SrcOver blend: output = (t*i, t*i, t*i, i) where i = intensity * mask.
   146→ * ------------------------------------------------------------------ */
   147→export const RIM_HIGHLIGHT_FRAGMENT_SHADER = /* glsl */ `
   148→precision highp float;
   149→
   150→uniform vec2  uCanvasSize;
   151→uniform vec2  uOffset;          // element top-left in canvas px (top-left origin) — SCALED rect
   152→uniform vec2  uSize;            // element size in canvas px — SCALED (includes graphicsLayer scale)
   153→uniform vec4  uCornerRadii;     // (topLeft, topRight, bottomRight, bottomLeft) in px — SCALED
   154→uniform vec4  uHighlightColor;  // rgb + 1.0
   155→uniform float uHighlightAngle;  // radians
   156→uniform float uHighlightFalloff;
   157→uniform float uHighlightAlpha;
   158→uniform float uHighlightMode;     // 0=Default, 1=Ambient, 2=Plain
   159→uniform float uHighlightStrokeWidth;
   160→uniform float uHighlightBlur;
   161→// --- ORIGINAL-SPACE SDF (faithful to graphicsLayer { scaleX, scaleY }) ---
   162→// Same approach as the element shader: compute SDF/stroke in ORIGINAL space
   163→// (shape is correct, not stretched), so the highlight clip + stroke remain a
   164→// correct capsule shape that is then scaled by graphicsLayer. Without this,
   165→// a horizontally-stretched button would stretch the highlight clip too,
   166→// making the stroke band uneven. See element.ts for the full rationale.
   167→uniform vec2  uOriginalSize;        // element size in px (ORIGINAL, unscaled)
   168→uniform float uOriginalCornerRadius; // corner radius in px (ORIGINAL, unscaled)
   169→uniform vec2  uLayerScale;          // (scaleX, scaleY) from graphicsLayer
   170→uniform float uElementRotation;     // rotation in radians (graphicsLayer rotationZ)
   171→
   172→${SDF_GLSL}
   173→
   174→void main() {
   175→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   176→    // elementCenter is the SAME for scaled and original rects (scaling is
   177→    // around the center), so uOffset + uSize*0.5 gives the correct center.
   178→    vec2 elementCenter = uOffset + uSize * 0.5;
   179→    vec2 centeredScreen = screenCoord - elementCenter;
   180→    // Map to ORIGINAL space (guard against divide-by-zero).
   181→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   182→    vec2 centeredOrig = centeredScreen / layerScale;
   183→    // Un-rotate into the element's local space so the SDF shape rotates.
   184→    vec2 centeredOrigRot = rotateBy(centeredOrig, -uElementRotation);
   185→
   186→    vec2 origHalfSize = uOriginalSize * 0.5;
   187→    float origRadius = uOriginalCornerRadius;
   188→
   189→    // SDF for stroke — analytic sdRoundedRect (matches the pre-capsule
   190→    // highlight implementation). When capsule is OFF, this is the exact
   191→    // shape. When capsule is ON, this is a close approximation (circular
   192→    // arc vs G2 Bezier — the difference is sub-pixel within the 2px stroke
   193→    // band, invisible in the highlight).
   194→    float sd = sdRoundedRect(centeredOrigRot, origHalfSize, origRadius);
   195→
   196→    // Outside the shape — clip (hard discard, matching pre-capsule behavior).
   197→    if (sd > 0.0) discard;
   198→
   199→    // Stroke mask — faithful to HighlightModifier.kt:
   200→    //   paint.style = Stroke
   201→    //   paint.strokeWidth = ceil(width.toPx()) * 2     // full stroke, centered on edge
   202→    //   paint.blur(blurRadius.toPx())                   // BlurMaskFilter, Blur.NORMAL
   203→    //   canvas.clipOutline(outline)                     // clip to inside the shape
   204→    //   canvas.drawOutline(outline, paint)              // stroke centered on edge
   205→    //
   206→    // Implementation: first compute a HARD-EDGE stroke mask (1.0 inside the
   207→    // stroke band, 0.0 outside), then convolve it with a Gaussian kernel by
   208→    // sampling the SDF at multiple offsets along the gradient direction.
   209→    // This mirrors the original's two-step process (draw stroke → blur),
   210→    // rather than using an analytic erf approximation.
   211→    //
   212→    // The hard stroke band: sd in [-strokeHalf, +strokeHalf].
   213→    // After clip (sd > 0 discarded by the outer if), only [-strokeHalf, 0] shows.
   214→    //
   215→    // Faithful to the original BlurMaskFilter:
   216→    //   paint.blur(blurRadius.toPx())  →  BlurMaskFilter(NORMAL, sigma=blurRadius_px)
   217→    // In Skia/Android, BlurMaskFilter's radius param IS the Gaussian sigma
   218→    // (not radius/3). blurRadius = width/2 = 0.25dp, so sigma = 0.25*dpr px.
   219→    // uHighlightBlur is already in device px (set by the renderer as widthDp*dpr*0.5).
   220→    float strokeHalf = uHighlightStrokeWidth * 0.5;
   221→    float sigma = max(uHighlightBlur, 0.1);
   222→
   223→    // Gaussian convolution of the hard stroke mask — 3-tap (σ-spaced).
   224→    // The original's BlurMaskFilter has σ = blurRadius = 0.25dp → 0.25px at
   225→    // dpr=1. At this sub-pixel sigma, only 3 taps (at -σ, 0, +σ) are needed
   226→    // — the Gaussian weight at ±2σ is exp(-2) ≈ 0.14, negligible. This
   227→    // replaces the old 65-tap loop (which computed 65 exp() calls per pixel,
   228→    // ~650 cycles — the single biggest shader cost). 3 taps = 3 exp() = ~30
   229→    // cycles, a 20× reduction with identical visual result at σ=0.25.
   230→    //   hardMask(sd) = 1.0 if |sd| < strokeHalf, else 0.0
   231→    //   blurred(sd) = Σ hardMask(sd - offset_k) * gauss(offset_k, σ)
   232→    // CLIP HALVING: the stroke is centered on sd=0; clip removes sd>0 (outer
   233→    // half), so peak ≈ 0.5. We halve to match.
   234→    float strokeMask = 0.0;
   235→    float wSum = 0.0;
   236→    for (int i = -1; i <= 1; i++) {
   237→        float offset = float(i) * sigma;  // taps at -σ, 0, +σ
   238→        float sampleSd = sd - offset;
   239→        float hard = (abs(sampleSd) < strokeHalf) ? 1.0 : 0.0;
   240→        float w = exp(-0.5 * (offset * offset) / (sigma * sigma));
   241→        strokeMask += hard * w;
   242→        wSum += w;
   243→    }
   244→    strokeMask /= wSum;
   245→    strokeMask *= 0.5;  // clip halves the symmetric stroke at the edge
   246→
   247→    if (uHighlightMode < 0.5) {
   248→        // Default — shader returns color * intensity, Plus blend.
   249→        float gradRadius = min(origRadius * 1.5, min(origHalfSize.x, origHalfSize.y));
   250→        vec2 grad = gradSdRoundedRect(centeredOrigRot, origHalfSize, gradRadius);
   251→        vec2 normal = vec2(cos(uHighlightAngle), sin(uHighlightAngle));
   252→        float d = dot(grad, normal);
   253→        float intensity = pow(abs(d), uHighlightFalloff);
   254→        vec3 c = uHighlightColor.rgb * intensity * strokeMask * uHighlightAlpha;
   255→        gl_FragColor = vec4(c, 1.0);
   256→    } else if (uHighlightMode < 1.5) {
   257→        // Ambient — faithful to AmbientHighlightShaderString:
   258→        //   float d = dot(grad, normal);
   259→        //   float intensity = pow(abs(d), falloff);
   260→        //   float t = step(0.0, d);  ← half-black-half-white split
   261→        //   return half4(t, t, t, 1.0) * intensity;
   262→        // SrcOver blend: bright side adds white light, dark side reduces
   263→        // scene brightness → 3D sphere effect (half black, half white).
   264→        float gradRadius = min(origRadius * 1.5, min(origHalfSize.x, origHalfSize.y));
   265→        vec2 grad = gradSdRoundedRect(centeredOrigRot, origHalfSize, gradRadius);
   266→        vec2 normal = vec2(cos(uHighlightAngle), sin(uHighlightAngle));
   267→        float d = dot(grad, normal);
   268→        float intensity = pow(abs(d), uHighlightFalloff);
   269→        float t = step(0.0, d);  // 0 on dark side (d<0), 1 on bright side (d>=0)
   270→        float i = intensity * strokeMask * uHighlightAlpha;
   271→        gl_FragColor = vec4(uHighlightColor.rgb * t * i, i);
   272→    } else {
   273→        // Plain — even stroke, paint.color, Plus blend.
   274→        vec3 c = uHighlightColor.rgb * strokeMask * uHighlightAlpha;
   275→        gl_FragColor = vec4(c, 1.0);
   276→    }
   277→}
   278→`
   279→
   280→/* ------------------------------------------------------------------ *
   281→ * HIGHLIGHT_STROKE_FRAGMENT_SHADER — pass 1 of the 3-pass faithful
   282→ * highlight (stroke mask → 2-pass blur → composite).
   283→ *
   284→ * Faithful to HighlightModifier.kt's GraphicsLayer record:
   285→ *   canvas.clipOutline(outline)            // clip to INSIDE the shape
   286→ *   canvas.drawOutline(outline, paint)     // paint.Stroke + BlurMaskFilter
   287→ *
   288→ * This shader ONLY renders the stroke alpha mask (clipped to inside):
   289→ *   - paint.style = Stroke, strokeWidth = ceil(width*dpr)*2, centered on edge
   290→ *   - clipOutline → discard sd > 0 (outside), keep inside half
   291→ *   - NO blur here (blur is done as a separate 2-pass Gaussian on this mask)
   292→ *   - NO intensity/color here (composite pass multiplies them in)
   293→ *
   294→ * Output: gl_FragColor.a = stroke mask (1.0 inside the stroke band, 0 outside),
   295→ *         rgb = 0 (unused; only alpha matters for the blur pass).
   296→ * The FBO is cleared to transparent (0,0,0,0) before this pass, so the
   297→ * surround is transparent and the 2-pass blur will naturally expand the
   298→ * stroke's alpha fringe outward (matching BlurMaskFilter NORMAL behavior).
   299→ * ------------------------------------------------------------------ */
   300→export const HIGHLIGHT_STROKE_FRAGMENT_SHADER = /* glsl */ `
   301→precision highp float;
   302→
   303→uniform vec2  uCanvasSize;
   304→uniform vec2  uOffset;          // element top-left (top-left origin) — SCALED
   305→uniform vec2  uSize;            // element size — SCALED
   306→uniform vec4  uCornerRadii;     // SCALED
   307→uniform float uHighlightStrokeWidth;  // ceil(width*dpr)*2, device px
   308→uniform vec2  uOriginalSize;
   309→uniform float uOriginalCornerRadius;
   310→uniform vec2  uLayerScale;
   311→uniform float uElementRotation;
   312→// uCornerStyle, uUseContinuousSdf, uContinuousSdf, uContinuousSdfTexSize,
   313→// uContinuousSdfElementSize are declared in SDF_GLSL (do NOT redeclare here).
   314→
   315→${SDF_GLSL}
   316→
   317→void main() {
   318→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   319→    vec2 elementCenter = uOffset + uSize * 0.5;
   320→    vec2 centeredScreen = screenCoord - elementCenter;
   321→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   322→    vec2 centeredOrig = centeredScreen / layerScale;
   323→    vec2 centeredOrigRot = rotateBy(centeredOrig, -uElementRotation);
   324→
   325→    vec2 origHalfSize = uOriginalSize * 0.5;
   326→    float origRadius = uOriginalCornerRadius;
   327→
   328→    float sd = sdShape(centeredOrigRot, origHalfSize, origRadius);
   329→
   330→    // clipOutline — clip to INSIDE the shape. Outside (sd > 0) is discarded.
   331→    float edgeAA;
   332→    if (uUseContinuousSdf > 0.5) {
   333→        float mask = sampleClipMask(centeredOrigRot, origHalfSize, origRadius);
   334→        if (mask < 0.01) discard;
   335→        edgeAA = mask;
   336→    } else {
   337→        if (sd > 0.0) discard;
   338→        edgeAA = 1.0 - smoothstep(-0.5, 0.5, sd);
   339→    }
   340→
   341→    // Stroke band centered on the edge (sd = 0), with 0.5px coverage AA on
   342→    // the inner boundary. The outer boundary (sd = +strokeHalf) is clipped
   343→    // away by edgeAA above. Faithful to Skia Paint.Stroke's coverage AA.
   344→    // The BlurMaskFilter pass (when sigma >= 0.5px) softens this further;
   345→    // at sub-pixel sigma (0.25px) the blur is skipped and this 0.5px AA
   346→    // is what matches the original's look (Skia's 0.25px blur is negligibly
   347→    // soft — essentially just AA).
   348→    float strokeHalf = uHighlightStrokeWidth * 0.5;
   349→    float strokeAA = 1.0 - smoothstep(strokeHalf - 0.5, strokeHalf, abs(sd));
   350→
   351→    gl_FragColor = vec4(0.0, 0.0, 0.0, strokeAA * edgeAA);
   352→}
   353→`
   354→
   355→/* ------------------------------------------------------------------ *
   356→ * HIGHLIGHT_COMPOSITE_FRAGMENT_SHADER — pass 3 of the 3-pass faithful
   357→ * highlight. Samples the blurred stroke mask (from the 2-pass blur) and
   358→ * multiplies by the AGSL RuntimeShader's intensity + color.
   359→ *
   360→ * Faithful to the original two-part pipeline:
   361→ *   1. Skia stroke + BlurMaskFilter produces the blurred alpha mask
   362→ *      (this is pass 1 + the 2-pass blur).
   363→ *   2. AGSL RuntimeShader (DefaultHighlightShaderString) computes
   364→ *      intensity = pow(abs(dot(grad, normal)), falloff) and returns
   365→ *      color * intensity. The RuntimeShader runs PER PIXEL of the stroke,
   366→ *      so the final visible color = strokeMask * intensity * color.
   367→ *
   368→ * This composite shader does step 2: read blurredMask.a, compute intensity
   369→ * from the SDF gradient, output color * intensity * mask (premultiplied).
   370→ *
   371→ * Blend modes (set by the renderer):
   372→ *   - Default (mode 0): Plus blend (gl.ONE, gl.ONE) — output rgb is added.
   373→ *     Premultiplied: rgb = color * intensity * mask * alpha.
   374→ *   - Ambient (mode 1): SrcOver blend (gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA).
   375→ *     Output = (color*i*mask, i*mask) premultiplied; the renderer's
   376→ *     gl.blendFunc(SRC_ALPHA, ONE_MINUS_SRC_ALPHA) un-premultiplies correctly.
   377→ *   - Plain (mode 2): Plus blend, no intensity (even stroke).
   378→ * ------------------------------------------------------------------ */
   379→export const HIGHLIGHT_COMPOSITE_FRAGMENT_SHADER = /* glsl */ `
   380→precision highp float;
   381→
   382→uniform vec2  uCanvasSize;
   383→uniform vec2  uOffset;
   384→uniform vec2  uSize;
   385→uniform vec4  uCornerRadii;
   386→uniform sampler2D uBlurredMask;   // the 2-pass-blurred stroke mask FBO
   387→uniform vec2  uMaskTexSize;       // size of the mask FBO (= canvas size)
   388→uniform vec4  uHighlightColor;    // rgb + 1.0
   389→uniform float uHighlightAngle;
   390→uniform float uHighlightFalloff;
   391→uniform float uHighlightAlpha;
   392→uniform float uHighlightMode;     // 0=Default, 1=Ambient, 2=Plain
   393→uniform vec2  uOriginalSize;
   394→uniform float uOriginalCornerRadius;
   395→uniform vec2  uLayerScale;
   396→uniform float uElementRotation;
   397→// uCornerStyle, uUseContinuousSdf, uContinuousSdf, uContinuousSdfTexSize,
   398→// uContinuousSdfElementSize are declared in SDF_GLSL (do NOT redeclare here).
   399→
   400→${SDF_GLSL}
   401→
   402→void main() {
   403→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   404→
   405→    // Sample the blurred stroke mask at this pixel. The mask FBO covers the
   406→    // full canvas (same size), so UV = gl_FragCoord / maskTexSize.
   407→    // Mask FBO is Y-down (top-left origin, like our scene FBOs), so flip Y
   408→    // to match the screenCoord convention.
   409→    vec2 maskUv = vec2(gl_FragCoord.x / uMaskTexSize.x, gl_FragCoord.y / uMaskTexSize.y);
   410→    float mask = texture2D(uBlurredMask, maskUv).a;
   411→    if (mask < 0.001) discard;
   412→
   413→    // Compute intensity from the SDF gradient (AGSL DefaultHighlightShaderString).
   414→    vec2 elementCenter = uOffset + uSize * 0.5;
   415→    vec2 centeredScreen = screenCoord - elementCenter;
   416→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   417→    vec2 centeredOrig = centeredScreen / layerScale;
   418→    vec2 centeredOrigRot = rotateBy(centeredOrig, -uElementRotation);
   419→    vec2 origHalfSize = uOriginalSize * 0.5;
   420→    float origRadius = uOriginalCornerRadius;
   421→
   422→    // Faithful clip-after-blur: the original does clipOutline → stroke(blur),
   423→    // but Skia applies clip at the canvas level AFTER the BlurMaskFilter
   424→    // spreads alpha. So alpha that blurred OUTSIDE the shape is clipped away.
   425→    // Our stroke shader clips before blur (discard sd>0), then blur spreads
   426→    // alpha back outside — we must clip AGAIN here to match. Without this,
   427→    // the highlight "leaks" outside the shape, making it brighter than the
   428→    // original (which has zero contribution outside the clip region).
   429→    float sd = sdShape(centeredOrigRot, origHalfSize, origRadius);
   430→    float clipAA;
   431→    if (uUseContinuousSdf > 0.5) {
   432→        clipAA = sampleClipMask(centeredOrigRot, origHalfSize, origRadius);
   433→    } else {
   434→        clipAA = 1.0 - smoothstep(-0.5, 0.5, sd);
   435→    }
   436→    mask *= clipAA;
   437→    if (mask < 0.001) discard;
   438→
   439→    // Compute d (with sign) for Default + Ambient modes — needed for
   440→    // Ambient's step(0,d) half-black-half-white split.
   441→    float d = 0.0;  // signed dot(grad, normal) — 0 for Plain mode
   442→    float intensity;
   443→    if (uHighlightMode < 1.5) {
   444→        // Default + Ambient use the SDF gradient · normal.
   445→        float gradRadius = min(origRadius * 1.5, min(origHalfSize.x, origHalfSize.y));
   446→        vec2 grad = gradSdRoundedRect(centeredOrigRot, origHalfSize, gradRadius);
   447→        vec2 normal = vec2(cos(uHighlightAngle), sin(uHighlightAngle));
   448→        d = dot(grad, normal);
   449→        intensity = pow(abs(d), uHighlightFalloff);
   450→    } else {
   451→        // Plain — no directional intensity (even stroke).
   452→        intensity = 1.0;
   453→    }
   454→
   455→    float a = mask * uHighlightAlpha;
   456→
   457→    if (uHighlightMode < 0.5) {
   458→        // Default — Plus blend. Output premultiplied rgb (alpha=1 so blendFunc
   459→        // (ONE, ONE) adds rgb directly).
   460→        vec3 c = uHighlightColor.rgb * intensity * a;
   461→        gl_FragColor = vec4(c, 1.0);
   462→    } else if (uHighlightMode < 1.5) {
   463→        // Ambient — SrcOver blend. Faithful to AmbientHighlightShaderString:
   464→        //   float t = step(0.0, d);  ← half-black-half-white split
   465→        // Bright side (d>=0): t=1 → white highlight. Dark side (d<0): t=0 →
   466→        // black overlay that reduces scene brightness via SrcOver → 3D sphere.
   467→        float t = step(0.0, d);
   468→        float i = intensity * a;
   469→        gl_FragColor = vec4(uHighlightColor.rgb * t * i, i);
   470→    } else {
   471→        // Plain — Plus blend, no intensity.
   472→        vec3 c = uHighlightColor.rgb * a;
   473→        gl_FragColor = vec4(c, 1.0);
   474→    }
   475→}
   476→`
   477→
   478→/* ------------------------------------------------------------------ *
   479→ * STROKE_MASK_COMPOSITE_FRAGMENT_SHADER — Canvas2D stroke mask approach.
   480→ *
   481→ * Samples a stroke mask texture generated by Canvas2D ctx.stroke()
   482→ * (browser-native Skia: exact Bezier tessellation + hardware coverage AA).
   483→ * The mask is element-local (small, element size + margin).
   484→ *
   485→ * Composite: strokeMask × intensity × color, Plus/SrcOver blend.
   486→ * ------------------------------------------------------------------ */
   487→export const STROKE_MASK_COMPOSITE_FRAGMENT_SHADER = /* glsl */ `
   488→precision highp float;
   489→
   490→uniform vec2  uCanvasSize;
   491→uniform vec2  uOffset;
   492→uniform vec2  uSize;
   493→uniform vec4  uCornerRadii;
   494→uniform sampler2D uStrokeMask;
   495→uniform vec2  uMaskOffset;
   496→uniform vec2  uMaskSize;
   497→uniform vec4  uHighlightColor;
   498→uniform float uHighlightAngle;
   499→uniform float uHighlightFalloff;
   500→uniform float uHighlightAlpha;
   501→uniform float uHighlightMode;
   502→uniform vec2  uOriginalSize;
   503→uniform float uOriginalCornerRadius;
   504→uniform vec2  uLayerScale;
   505→uniform float uElementRotation;
   506→
   507→${SDF_GLSL}
   508→
   509→void main() {
   510→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   511→
   512→    // Map screen coord → element-local ORIGINAL space (un-scale, un-rotate).
   513→    // The stroke mask is drawn in original space (origSizeX × origSizeY + margin).
   514→    // elementCenter is the same in scaled and original space (scaling is around center).
   515→    vec2 elementCenter = uOffset + uSize * 0.5;
   516→    vec2 centeredScreen = screenCoord - elementCenter;
   517→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   518→    vec2 centeredOrig = centeredScreen / layerScale;
   519→    vec2 centeredOrigRot = rotateBy(centeredOrig, -uElementRotation);
   520→
   521→    // Mask UV: map original-space coord → mask texture UV.
   522→    // The mask was drawn with translate(margin, margin), so mask (0,0) =
   523→    // element-local (-margin). Element-local coord 0..origSize maps to
   524→    // mask UV (0+margin)/maskSize .. (origSize+margin)/maskSize.
   525→    // uMaskOffset = margin (scalar, passed as vec2 for convenience).
   526→    // uMaskSize = (origSize + 2*margin).
   527→    vec2 origHalfSize = uOriginalSize * 0.5;
   528→    vec2 maskTexCoord = centeredOrigRot + origHalfSize;  // 0..origSize (element-local)
   529→    vec2 maskUv = (maskTexCoord + uMaskOffset) / uMaskSize;
   530→    if (maskUv.x < 0.0 || maskUv.x > 1.0 || maskUv.y < 0.0 || maskUv.y > 1.0) discard;
   531→    float mask = texture2D(uStrokeMask, maskUv).a;
   532→    if (mask < 0.001) discard;
   533→
   534→    float origRadius = uOriginalCornerRadius;
   535→
   536→    // Compute d (with sign) for Default + Ambient modes — needed for
   537→    // Ambient's step(0,d) half-black-half-white split.
   538→    float d = 0.0;  // signed dot(grad, normal) — 0 for Plain mode
   539→    float intensity;
   540→    if (uHighlightMode < 1.5) {
   541→        float gradRadius = min(origRadius * 1.5, min(origHalfSize.x, origHalfSize.y));
   542→        vec2 grad = gradSdRoundedRect(centeredOrigRot, origHalfSize, gradRadius);
   543→        vec2 normal = vec2(cos(uHighlightAngle), sin(uHighlightAngle));
   544→        d = dot(grad, normal);
   545→        intensity = pow(abs(d), uHighlightFalloff);
   546→    } else {
   547→        intensity = 1.0;
   548→    }
   549→
   550→    float a = mask * uHighlightAlpha;
   551→    if (uHighlightMode < 0.5) {
   552→        gl_FragColor = vec4(uHighlightColor.rgb * intensity * a, 1.0);
   553→    } else if (uHighlightMode < 1.5) {
   554→        // Ambient — faithful to AmbientHighlightShaderString:
   555→        //   float t = step(0.0, d);  ← bright/dark split
   556→        // Bright side: t=1 → white highlight. Dark side: t=0 → black overlay
   557→        // that reduces scene brightness via SrcOver → 3D sphere effect.
   558→        float t = step(0.0, d);
   559→        float i = intensity * a;
   560→        gl_FragColor = vec4(uHighlightColor.rgb * t * i, i);
   561→    } else {
   562→        gl_FragColor = vec4(uHighlightColor.rgb * a, 1.0);
   563→    }
   564→}
   565→`

   567→/* ------------------------------------------------------------------ *
   568→ * INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER — Canvas2D inner shadow
   569→ * mask composite. Samples a blurred ring mask texture generated by
   570→ * Canvas2D (fill → destination-out → blur) and composites onto the
   571→ * scene with SrcOver blend.
   572→ *
   573→ * The mask represents the inner shadow ring: the area of the shape
   574→ * that is NOT covered by the offset shape, blurred. This creates a
   575→ * soft shadow band inside the shape, thicker on the side opposite
   576→ * the offset direction — faithful to InnerShadowModifier.kt.
   577→ *
   578→ * Composite: shadowColor * maskAlpha * shadowAlpha, SrcOver blend.
   579→ * SDF clip ensures the shadow stays inside the shape boundary
   580→ * (blur that leaked outside is discarded).
   581→ * ------------------------------------------------------------------ */
   582→export const INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER = /* glsl */ `
   583→precision highp float;
   584→
   585→uniform vec2  uCanvasSize;
   586→uniform vec2  uOffset;           // element top-left in canvas px (top-left origin) — SCALED rect
   587→uniform vec2  uSize;             // element size in canvas px — SCALED
   588→uniform vec4  uCornerRadii;      // (topLeft, topRight, bottomRight, bottomLeft) — SCALED
   589→uniform sampler2D uInnerShadowMask; // Canvas2D-generated blurred ring mask
   590→uniform vec2  uMaskOffset;       // margin in device px (for UV mapping: element-local → mask UV)
   591→uniform vec2  uMaskSize;         // total mask size in device px (w+2*margin, h+2*margin)
   592→uniform vec3  uInnerShadowColor; // shadow color RGB
   593→uniform float uInnerShadowAlpha; // shadow alpha
   594→// --- ORIGINAL-SPACE SDF clip (faithful to graphicsLayer { scaleX, scaleY }) ---
   595→uniform vec2  uOriginalSize;        // element size in px (ORIGINAL, unscaled)
   596→uniform float uOriginalCornerRadius; // corner radius in px (ORIGINAL, unscaled)
   597→uniform vec2  uLayerScale;          // (scaleX, scaleY) from graphicsLayer
   598→uniform float uElementRotation;     // rotation in radians (graphicsLayer rotationZ)
   599→
   600→${SDF_GLSL}
   601→
   602→void main() {
   603→    vec2 screenCoord = vec2(gl_FragCoord.x, uCanvasSize.y - gl_FragCoord.y);
   604→
   605→    // Map screen coord → element-local ORIGINAL space (un-scale, un-rotate).
   606→    // The inner shadow mask is drawn in original space (origSize + margin).
   607→    // elementCenter is the same in scaled and original space (scaling is
   608→    // around center).
   609→    vec2 elementCenter = uOffset + uSize * 0.5;
   610→    vec2 centeredScreen = screenCoord - elementCenter;
   611→    vec2 layerScale = max(uLayerScale, vec2(1e-4));
   612→    vec2 centeredOrig = centeredScreen / layerScale;
   613→    vec2 centeredOrigRot = rotateBy(centeredOrig, -uElementRotation);
   614→
   615→    // Clip to element shape — hard discard for pixels outside.
   616→    // Faithful to InnerShadowModifier.kt's graphicsLayer clip which applies
   617→    // AFTER BlurMaskFilter: the blurred ring's outward spread is clipped away,
   618→    // but pixels AT the edge (sd ≈ 0) remain at full mask intensity.
   619→    // We use a HARD discard (no smoothstep clipAlpha) because multiplying by
   620→    // clipAlpha at the edge would reduce the inner shadow's peak intensity,
   621→    // creating a visible gap. The mask's own blur softness provides the
   622→    // edge transition, matching the original's clip-after-blur behavior.
   623→    vec2 origHalfSize = uOriginalSize * 0.5;
   624→    float sd = sdShape(centeredOrigRot, origHalfSize, uOriginalCornerRadius);
   625→    if (sd > 0.5) discard;
   626→
   627→    // Map to mask UV: original-space coord → mask texture UV.
   628→    // The mask was drawn with translate(margin, margin), so mask (0,0) =
   629→    // element-local (-margin). Element-local coord 0..origSize maps to
   630→    // mask UV (0+margin)/maskSize .. (origSize+margin)/maskSize.
   631→    vec2 maskTexCoord = centeredOrigRot + origHalfSize;  // 0..origSize (element-local)
   632→    vec2 maskUv = (maskTexCoord + uMaskOffset) / uMaskSize;
   633→    if (maskUv.x < 0.0 || maskUv.x > 1.0 || maskUv.y < 0.0 || maskUv.y > 1.0) discard;
   634→    float mask = texture2D(uInnerShadowMask, maskUv).a;
   635→    if (mask < 0.001) discard;
   636→
   637→    // SrcOver composite: shadow color × mask × alpha.
   638→    // No clipAlpha multiplier — hard discard at sd>0.5 is sufficient.
   639→    // Premultiplied alpha output with gl.blendFunc(SRC_ALPHA, ONE_MINUS_SRC_ALPHA).
   640→    float a = mask * uInnerShadowAlpha;
   641→    gl_FragColor = vec4(uInnerShadowColor * a, a);
   642→}
   643→`

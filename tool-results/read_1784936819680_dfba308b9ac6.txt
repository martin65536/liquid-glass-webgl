    32→function useSystemTheme(): Theme {
    33→  const [theme, setTheme] = React.useState<Theme>('light')
    34→  React.useEffect(() => {
    35→    if (typeof window === 'undefined' || !window.matchMedia) return
    36→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    37→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    38→    update()
    39→    mq.addEventListener('change', update)
    40→    return () => mq.removeEventListener('change', update)
    41→  }, [])
    42→  return theme
    43→}

   251→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   252→    const MIN_FPS = 55
   253→    if (fps >= MIN_FPS) {
   254→      // This DPR works — try higher
   255→      perfLoRef.current = candidateDpr
   256→    } else {
   257→      // Too slow — try lower
   258→      perfHiRef.current = candidateDpr
   259→    }
   260→
   261→    if (iteration >= maxIterations || perfHiRef.current - perfLoRef.current <= 0.25) {
   262→      // Converged — use the highest DPR that worked (lo)
   263→      const bestDpr = Math.max(0.5, Math.min(deviceDpr, Math.round(perfLoRef.current * 4) / 4))
   264→      perfPhaseRef.current = 'done'
   265→      const isGood = bestDpr >= deviceDpr * 0.5
   266→      const statusText = isGood
   267→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   268→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   269→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   270→      setState({
   271→        customDpr: bestDpr,
   272→        perfDone: true,
   273→        perfResultDpr: bestDpr,
   274→        perfStatusText: statusText,
   275→        perfProgress: null,
   276→        // Don't reset perfGlassAngle — settle animation will handle it
   277→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   278→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   279→        perfProgressFrac: 1,
   280→      })
   281→    } else {
   282→      // More iterations needed — increment trigger to re-fire the effect
   283→      perfPhaseRef.current = 'idle'
   284→      setState((prev) => ({
   285→        perfProgress: 'running',
   286→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   287→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   288→      }))
   289→    }
   290→  }

   537→  // Device motion → gravity angle for glass highlight direction.
   538→  // Faithful to UISensor.kt:
   539→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   540→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   541→  //
   542→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   543→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   544→  // rebuilding the catalog (which would lose drag state and cause jank).
   545→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   546→  // renderer.gravityAngle live each frame.
   547→  //
   548→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   549→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   550→  // device's acceleration INCLUDING gravity in device-space axes:
   551→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   552→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   553→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   554→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   555→  // vectors — atan2(beta, gamma) does NOT match the original.)
   556→  //
   557→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   558→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   559→  // Default 45° (matches UISensor.kt's initial value).
   560→  React.useEffect(() => {
   561→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   562→    let smoothed = 45
   563→    const alpha = 0.5
   564→    const handler = (e: DeviceMotionEvent) => {
   565→      const acc = e.accelerationIncludingGravity
   566→      if (!acc || acc.x == null || acc.y == null) return
   567→      const x = acc.x
   568→      const y = acc.y
   569→      let target = Math.atan2(y, x) * 180 / Math.PI
   570→      // Shortest-path interpolation: wrap the delta to [-180, 180] so
   571→      // rotating through ±180° doesn't cause a 358° jump.
   572→      let delta = target - smoothed
   573→      while (delta > 180) delta -= 360
   574→      while (delta < -180) delta += 360
   575→      smoothed += delta * alpha
   576→      // Wrap smoothed to [-180, 180] to keep the value bounded.
   577→      while (smoothed > 180) smoothed -= 360
   578→      while (smoothed < -180) smoothed += 360
   579→      const rad = smoothed * Math.PI / 180
   580→      rendererRef.current?.setGravityAngle(rad)
   581→    }
   582→    window.addEventListener('devicemotion', handler)
   583→    return () => window.removeEventListener('devicemotion', handler)
   584→  }, [rendererRef])

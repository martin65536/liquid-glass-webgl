    45→
    46→export const renderMethods = {
    47→  render(this: LiquidGlassRenderer) {
    48→    // PERFORMANCE: Skip render if nothing changed since last frame.
    49→    // This prevents redundant full-scene re-renders when rAF fires
    50→    // (e.g. from browser repaints) but no state actually changed.
    51→    if (!this.needsRedraw) return
    52→    this.needsRedraw = false
    53→
    54→    // --- PerfMonitor: start frame timing + reset per-frame counters ---
    55→    // Push canvas info first so the snapshot includes it.
    56→    this.perfMonitor.canvasCssW = this.cssWidth
    57→    this.perfMonitor.canvasCssH = this.cssHeight
    58→    this.perfMonitor.canvasDevW = this.canvas.width
    59→    this.perfMonitor.canvasDevH = this.canvas.height
    60→    this.perfMonitor.dpr = this.dpr
    61→    this.perfMonitor.deviceDpr = typeof window !== 'undefined' ? window.devicePixelRatio || 1 : 1
    62→    this.perfMonitor.frameStart()
    63→
    64→    if (!this.wallpaperReady && !this.backgroundColor) {
    65→      this.perfMonitor.frameEnd()
    66→      return
    67→    }
    68→    const gl = this.gl
    69→    // Ensure FBOs exist (created lazily on first render after resize).
    70→    this.resizeFBOs(this.canvas.width, this.canvas.height)
    71→
    72→    // Re-rasterize any dirty foregrounds.
    73→    for (const cfg of this.buttonConfigs) {
    74→      if (this.fgDirtyIds.has(cfg.id)) {
    75→        this.rasterizeForeground(cfg)
    76→      }
    77→    }
    78→
    79→    // --- 1. Render background (wallpaper or solid color) into fboA ----
    80→    // fboA is the "current scene" — everything rendered so far. Glass
    81→    // elements will sample from fboA.texture to compute refraction of
    82→    // the actual colors behind them (track color, card background, etc).
    83→    this.renderBackground()
    84→    this.perfMonitor.incDrawCall() // wallpaper pass = 1 draw call
    85→
    86→    if (this.buttonConfigs.length === 0) {
    87→      // No elements — blit fboA to the default framebuffer and done.
    88→      this.bindFBO(null)
    89→      this.drawCopy(this.fboATex!)
    90→      this.perfMonitor.incDrawCall() // final blit
    91→      this.perfMonitor.frameEnd()
    92→      return
    93→    }
    94→
    95→    // --- Global backdrop blur (ControlCenter) ---
    96→    // Faithful to ControlCenterContent.kt: the backdrop Image has
    97→    //   .graphicsLayer { BlurEffect(4dp * progress) }
    98→    // which blurs the WALLPAPER (not the dim, not the tiles). We replicate
    99→    // by blurring fboA (wallpaper) right after renderBackground, BEFORE any
   100→    // element composites on top. The cc-dim element (drawn next) renders a
   101→    // crisp dim on top of the blurred wallpaper — matching the original's
   102→    // drawWithContent { drawContent(); drawRect(dim) } where drawContent()
   103→    // draws the blurred wallpaper and drawRect(dim) is crisp.
   104→    //
   105→    // sceneBlurRadius is set on the cc-dim element (CSS px). We scan for it
   106→    // here (once per frame) and blur fboA in-place (blurTexture → blurFboB,
   107→    // then drawCopy back to fboA).
   108→    const sceneBlurEl = this.buttonConfigs.find((e) => (e.sceneBlurRadius ?? 0) >= 0.5)
   109→    if (sceneBlurEl) {
   110→      const r = sceneBlurEl.sceneBlurRadius! * this.dpr
   111→      const blurred = this.blurTexture(this.fboATex!, r)
   112→      // blurTexture restored the FBO binding to fboA (what renderBackground
   113→      // bound). Rebind explicitly + copy blurred result back into fboA.
   114→      this.bindFBO(this.fboA!)
   115→      this.drawCopy(blurred)
   116→      this.perfMonitor.incBlurPass()
   117→      this.perfMonitor.incDrawCall(2) // blur = 2 passes + 1 copy
   118→    }
   119→
   120→    // Enable blending for the remaining passes.
   121→    gl.enable(gl.BLEND)
   122→    gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   123→
   124→    // Cull + iterate. We render elements in DECLARED ORDER (no Wave 1 /
   125→    // Wave 2 split) because the FBO ping-pong makes z-ordering faithful:
   126→    // each element composites on top of everything declared before it.
   127→    //
   128→    // CULL MARGIN: 120px accounts for outer shadows (~24dp), press/toggle
   129→    // scale (up to 1.5x), and foreground halo blur.
   130→    //
   131→    // CULL MARGIN UNITS: All comparisons are in VIEWPORT coords (y=0 is
   132→    // the top of the visible canvas, y=cssHeight is the bottom). Mixing
   133→    // viewport y with content y (which is offset by scrollY) was the
   134→    // cause of the long-standing "elements disappear before sliding off
   135→    // screen" bug.
   136→    const scrollY = this.scrollY
   137→    const CULL_MARGIN = 120
   138→    const viewportTop = -CULL_MARGIN
   139→    const viewportBottom = this.cssHeight + CULL_MARGIN
   140→
   141→    // Helper to compute the element's effective rect (with scroll offset).
   142→    const effRect = (el: GlassElementConfig) => {
   143→      const y = el.scroll ? el.rect.y - scrollY : el.rect.y
   144→      return { x: el.rect.x, y, w: el.rect.w, h: el.rect.h }
   145→    }
   146→
   147→    // Iterate elements in declared order. Track which FBO is "current"
   148→    // (i.e. contains the scene built up so far). Glass elements trigger
   149→    // a ping-pong; non-glass elements render directly to the current FBO.
   150→    let curFbo: WebGLFramebuffer = this.fboA!
   151→    let curTex: WebGLTexture = this.fboATex!
   152→    let otherFbo: WebGLFramebuffer = this.fboB!
   153→    let otherTex: WebGLTexture = this.fboBTex!
   154→
   155→    for (const el of this.buttonConfigs) {
   156→      // Skip renderOnTop elements — they are rendered in a second pass
   157→      // after all other elements (faithful to ControlCenterContent.kt's
   158→      // drawWithContent which draws the dim AFTER drawContent).
   159→      if (el.renderOnTop) continue
   160→
   161→      // Compute the element's effective y in VIEWPORT coords (after scroll).
   162→      const y = el.scroll ? el.rect.y - scrollY : el.rect.y
   163→      if (y + el.rect.h < viewportTop || y > viewportBottom) continue
   164→
   165→      const r = effRect(el)
   166→      const st = this.buttonStates.get(el.id)
   167→
   168→      // --- Non-glass elements: render directly to current FBO ---
   169→      if (this.renderNonGlassElement(el, r, st, curFbo)) continue
   170→
   171→      // --- Backdrop FBO: render wallpaper+scrim+colorControls into
   172→      // dialogBackdropFbo (cached) for backdropFbo elements. ---
   173→      if (el.backdropFbo && el.scrimColor) {
   174→        this.renderDialogBackdrop(el.scrimColor, el.brightness, el.contrast, el.saturation)
   175→      }
   176→
   177→      // --- Continuous-curvature SDF texture (capsule shape) ---
   178→      // For elements with useContinuousSdf=true (dialog card), ensure the
   179→      // SDF texture for the element's (w, h, radius) is generated + uploaded
   180→      // BEFORE rendering. loadContinuousSdf() is cached — no-op if already
   181→      // loaded for this geometry. Generation is synchronous (Canvas2D raster
   182→      // + chamfer distance transform on a 256×256 grid) so it only happens
   183→      // once per (w, h, radius) tuple, on the first frame after a resize.
   184→      if (el.useContinuousSdf) {
   185→        this.loadContinuousSdf(el.rect.w, el.rect.h, el.cornerRadius)
   186→      }
   187→
   188→      // --- Glass elements (button / glass-shape): ping-pong ---
   189→      const result = this.renderGlassElement(el, st, curFbo, curTex, otherFbo, otherTex, r)
   190→      curFbo = result.curFbo
   191→      curTex = result.curTex
   192→      otherFbo = result.otherFbo
   193→      otherTex = result.otherTex
   194→
   195→      // After the container glass is rendered (before tab-content), snapshot
   196→      // the scene (wallpaper + glass, no text) into tabsBackdropFbo. The
   197→      // indicator samples this to avoid the white/black tab text bleeding through.
   198→      if (el.isBottomTabContainer && this.tabsBackdropFbo && this.tabsBackdropTex) {
   199→        this.bindFBO(this.tabsBackdropFbo)
   200→        // Clear to transparent first (avoid stale black from previous frames).
   201→        this.gl.clearColor(0, 0, 0, 0)
   202→        this.gl.clear(this.gl.COLOR_BUFFER_BIT)
   203→        this.drawCopy(curTex)
   204→        // Re-bind curFbo for continued rendering (tab-content draws on top).
   205→        this.bindFBO(curFbo)
   206→        // drawCopy disables blend; re-enable for subsequent tab-content rendering.
   207→        this.gl.enable(this.gl.BLEND)
   208→        this.gl.blendFunc(this.gl.SRC_ALPHA, this.gl.ONE_MINUS_SRC_ALPHA)
   209→      }
   210→    }
   211→
   212→    // --- Second pass: render renderOnTop elements ---
   213→    // Faithful to ControlCenterContent.kt / DialogContent.kt's drawWithContent:
   214→    //   drawContent()  ← first pass (card/tiles)
   215→    //   drawRect(dim)  ← second pass (dim/scrim on top, like the original)
   216→    // Also renders glass renderOnTop elements (back button / theme toggle)
   217→    // via normal ping-pong — they composite on top of the scrim. If they
   218→    // have sampleWallpaper=true, the refraction samples the clean wallpaper
   219→    // (handled in renderGlassElementPass), so the scrim doesn't darken them.
   220→    for (const el of this.buttonConfigs) {
   221→      if (!el.renderOnTop) continue
   222→      const y = el.scroll ? el.rect.y - scrollY : el.rect.y
   223→      if (y + el.rect.h < viewportTop || y > viewportBottom) continue
   224→      const r = effRect(el)
   225→      const st = this.buttonStates.get(el.id)
   226→
   227→      // Non-glass renderOnTop elements (scrim/dim) render directly on curFbo.
   228→      if (this.renderNonGlassElement(el, r, st, curFbo)) continue
   229→
   230→      // Glass renderOnTop elements (back button / theme toggle): normal
   231→      // ping-pong. The blit copies curTex (which now contains the scrim) to
   232→      // otherFbo, then the glass element renders on top. sampleWallpaper
   233→      // (if set) only changes the refraction sample, not the blit — so the
   234→      // scene is preserved and the button composites correctly on top.
   235→      const result = this.renderGlassElement(el, st, curFbo, curTex, otherFbo, otherTex, r)
   236→      curFbo = result.curFbo
   237→      curTex = result.curTex
   238→      otherFbo = result.otherFbo
   239→      otherTex = result.otherTex
   240→    }
   241→
   242→    // --- Final: blit curFbo → default framebuffer (visible canvas) ---
   243→    this.bindFBO(null)
   244→    this.drawCopy(curTex)
   245→    this.perfMonitor.incDrawCall() // final blit
   246→
   247→    // --- PerfMonitor: end frame timing + capture counters ---
   248→    this.perfMonitor.frameEnd()
   249→  },
   250→
   251→  /** Helper to set SDF uniforms (canvasSize + offset + size + cornerRadii)
   252→   *  for any of the SDF-using programs. */
   253→  setSdfUniforms(
   254→    this: LiquidGlassRenderer,
   255→    u: Record<string, WebGLUniformLocation | null>,
   256→    aPosLoc: number,
   257→    r: { x: number; y: number; w: number; h: number },
   258→    cornerRadius: number
   259→  ) {
   260→    const gl = this.gl
   261→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   262→    gl.enableVertexAttribArray(aPosLoc)
   263→    gl.vertexAttribPointer(aPosLoc, 2, gl.FLOAT, false, 0, 0)
   264→    gl.uniform2f(u['uCanvasSize'], this.canvas.width, this.canvas.height)
   265→    gl.uniform2f(u['uOffset'], r.x * this.dpr, r.y * this.dpr)
   266→    gl.uniform2f(u['uSize'], r.w * this.dpr, r.h * this.dpr)
   267→    gl.uniform4f(
   268→      u['uCornerRadii'],
   269→      cornerRadius * this.dpr,
   270→      cornerRadius * this.dpr,
   271→      cornerRadius * this.dpr,
   272→      cornerRadius * this.dpr
   273→    )
   274→  },
   275→
   276→  /** Render wallpaper or solid background color into fboA. */
   277→  renderBackground(this: LiquidGlassRenderer) {
   278→    const gl = this.gl
   279→    this.bindFBO(this.fboA)
   280→    gl.disable(gl.BLEND)
   281→    if (this.backgroundColor) {
   282→      const [r, g, b] = this.backgroundColor
   283→      this.drawSolidFill(r, g, b, 1)
   284→    } else {
   285→      gl.useProgram(this.wallpaperProgram)
   286→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   287→      gl.enableVertexAttribArray(this.aPosLocWp)
   288→      gl.vertexAttribPointer(this.aPosLocWp, 2, gl.FLOAT, false, 0, 0)
   289→      gl.activeTexture(gl.TEXTURE0)
   290→      gl.bindTexture(gl.TEXTURE_2D, this.wallpaperTexture!)
   291→      gl.uniform1i(this.uWp['uBackdrop'], 0)
   292→      gl.uniform2f(this.uWp['uCanvasSize'], this.canvas.width, this.canvas.height)
   293→      gl.uniform2f(this.uWp['uWallpaperSize'], this.wallpaperSize[0], this.wallpaperSize[1])
   294→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   295→    }
   296→  },
   297→
   298→  /** Render wallpaper+scrim+colorControls into dialogBackdropFbo as ONE OPAQUE
   299→   *  layer (alpha=1), replicating the original's LayerBackdrop (wallpaper+scrim)
   300→   *  with colorControls applied — matching the original's colorControls→blur→lens
   301→   *  effects order. The dialog card (backdropFbo + useSeparableBlur) 2-pass blurs
   302→   *  this FBO then does lens refraction.
   303→   *
   304→   *  Order: wallpaper (opaque) → scrim (glBlendFuncSeparate, correct alpha) →
   305→   *  colorControls (fullscreen pass). Cached by scrim+cc params. */
   306→  renderDialogBackdrop(
   307→    this: LiquidGlassRenderer,
   308→    scrim: [number, number, number, number],
   309→    brightness: number,
   310→    contrast: number,
   311→    saturation: number
   312→  ) {
   313→    const key = `${scrim.join(',')}|${brightness},${contrast},${saturation}`
   314→    if (this.dialogBackdropKey === key) return  // cached
   315→    this.dialogBackdropKey = key
   316→    const gl = this.gl
   317→    // Step 1: paint wallpaper (opaque) into dialogBackdropFbo.
   318→    this.bindFBO(this.dialogBackdropFbo!)
   319→    gl.disable(gl.BLEND)
   320→    if (this.backgroundColor) {
   321→      const [r, g, b] = this.backgroundColor
   322→      this.drawSolidFill(r, g, b, 1)
   323→    } else {
   324→      gl.useProgram(this.wallpaperProgram)
   325→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   326→      gl.enableVertexAttribArray(this.aPosLocWp)
   327→      gl.vertexAttribPointer(this.aPosLocWp, 2, gl.FLOAT, false, 0, 0)
   328→      gl.activeTexture(gl.TEXTURE0)
   329→      gl.bindTexture(gl.TEXTURE_2D, this.wallpaperTexture!)
   330→      gl.uniform1i(this.uWp['uBackdrop'], 0)
   331→      gl.uniform2f(this.uWp['uCanvasSize'], this.canvas.width, this.canvas.height)
   332→      gl.uniform2f(this.uWp['uWallpaperSize'], this.wallpaperSize[0], this.wallpaperSize[1])
   333→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   334→    }
   335→    // Step 2: composite scrim via glBlendFuncSeparate (correct SrcOver alpha,
   336→    // no src.a² squaring) so the FBO alpha stays 1.
   337→    if (scrim[3] > 0.001) {
   338→      gl.enable(gl.BLEND)
   339→      gl.blendFuncSeparate(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA, gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   340→      this.drawSolidFill(scrim[0], scrim[1], scrim[2], scrim[3])
   341→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   342→    }
   343→    // Step 3: colorControls (fullscreen pass) — applied to the opaque
   344→    // wallpaper+scrim, BEFORE blur. Faithful to colorControls→blur→lens order.
   345→    // Ping-pong through blurFboA to avoid reading/writing dialogBackdropFbo.
   346→    this.bindFBO(this.blurFboA!)
   347→    this.drawColorControls(this.dialogBackdropTex!, brightness, contrast, saturation)
   348→    // Copy blurred-cc result back to dialogBackdropFbo so the 2-pass blur in
   349→    // the useSeparableBlur path can blur it.
   350→    this.bindFBO(this.dialogBackdropFbo!)
   351→    this.drawCopy(this.blurFboATex!)
   352→  },
   353→
   354→  /** Render a non-glass element (plain-rect / progressive-blur / text).
   355→   *  Returns true if the element was handled (caller should `continue`).
   356→   *  Returns false for glass elements (caller should run the ping-pong path). */
   357→  renderNonGlassElement(
   358→    this: LiquidGlassRenderer,
   359→    el: GlassElementConfig,
   360→    r: { x: number; y: number; w: number; h: number },
   361→    st: ElementState | undefined,
   362→    curFbo: WebGLFramebuffer
   363→  ): boolean {
   364→    const gl = this.gl
   365→
   366→    // Apply enterProgress translationY (ControlCenter) to the rect.
   367→    // Uses DERIVED progress (with ProgressConverter) — faithful to
   368→    // ControlCenterContent.kt glassLayer which reads the derivedStateOf progress.
   369→    let r2 = r
   370→    if (el.enterProgress != null) {
   371→      const raw = el.enterProgress
   372→      const derived = raw < 0
   373→        ? (1 - Math.exp(-Math.abs(raw))) * -1
   374→        : raw <= 1 ? raw
   375→        : 1 + (1 - Math.exp(-(raw - 1)))
   376→      const ty = -48 * DP * (1 - derived)
   377→      // Overscroll row-stretch: when derived > 1, grow inter-row spacing
   378→      // by 32dp per unit of DERIVED overshoot.
   379→      const stretch = el.enterStretchFactor != null && derived > 1
   380→        ? el.enterStretchFactor * (derived - 1) * 32 * DP
   381→        : 0
   382→      r2 = { x: r.x, y: r.y + ty + stretch, w: r.w, h: r.h }
   383→    }
   384→
   385→    // --- plain-rect ---
   386→    if (el.kind === 'plain-rect' && el.plainRect) {
   387→      // Skip rendering fully-transparent plain-rects (e.g. invisible drag
   388→      // catchers). They have no visual effect but would otherwise waste a
   389→      // draw call and (with SRC_ALPHA blending) could interfere with the
   390→      // scene. Alpha=0 → no contribution → skip.
   391→      const baseC = el.isToggleTrack ? null : el.plainRect.color
   392→      if (baseC && baseC[3] <= 0) return true
   393→      this.bindFBO(curFbo)
   394→      // Toggle tracks: lerp between offColor and onColor based on the
   395→      // group's animated fraction. Faithful to LiquidToggle.kt's
   396→      // `drawRect(lerp(trackColor, accentColor, fraction))`.
   397→      let c: [number, number, number, number]
   398→      if (el.isToggleTrack) {
   399→        const tg = this.toggleStates.get(el.isToggleTrack.groupId)
   400→        const f = tg ? tg.fraction : 0
   401→        const off = el.isToggleTrack.offColor
   402→        const on = el.isToggleTrack.onColor
   403→        c = [
   404→          off[0] + (on[0] - off[0]) * f,
   405→          off[1] + (on[1] - off[1]) * f,
   406→          off[2] + (on[2] - off[2]) * f,
   407→          off[3] + (on[3] - off[3]) * f,
   408→        ]
   409→      } else {
   410→        c = el.plainRect.color
   411→      }
   412→      // Slider fill: dynamically adjust width from the toggle group's animated
   413→      // fraction so the fill tracks the knob's spring motion (no React-state
   414→      // lag). Faithful to LiquidSlider.kt:
   415→      //   width = constraints.maxWidth * dampedDragAnimation.progress
   416→      // i.e. fillW = trackW * fraction (NOT knob-center-aligned — the original
   417→      // lets the fill span the full track width, with the knob clamped at the
   418→      // ends so it sits w/4 inside the fill at progress=0 and w/4 past the
   419→      // fill end at progress=1).
   420→      let fillRect = r2
   421→      if (el.isSliderFill) {
   422→        const sf = this.toggleStates.get(el.isSliderFill.groupId)
   423→        const fraction = sf ? sf.fraction : 0
   424→        const fillW = Math.max(el.isSliderFill.minW, el.isSliderFill.trackW * fraction)
   425→        fillRect = { x: r.x, y: r.y, w: fillW, h: r.h }
   426→      }
   427→      gl.useProgram(this.plainRectProgram)
   428→      this.setSdfUniforms(this.uPr, this.aPosLocPr, fillRect, el.cornerRadius)
   429→      // glBlendFuncSeparate: correct SrcOver on the alpha channel (ONE instead
   430→      // of SRC_ALPHA) so the scene FBO's alpha stays 1.0 when translucent
   431→      // plain-rects (scrims, tracks, fills) composite onto it. Without this,
   432→      // glBlendFunc's alpha-squaring (out.a = src.a² + dst.a*(1-src.a)) decays
   433→      // the FBO alpha below 1, making glass that samples it erroneously
   434→      // translucent. RGB is unchanged (same SRC_ALPHA, ONE_MINUS_SRC_ALPHA).
   435→      gl.blendFuncSeparate(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA, gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   436→      // Apply enterProgress alpha (ControlCenter fade) to plainRects.
   437→      // Uses SAFE progress (clamped 0..1) — faithful to ControlCenterContent.kt
   438→      // which uses safeEnterProgressAnimation.value for alpha/dim/blur.
   439→      const enterA = el.enterProgress != null ? (() => {
   440→        const sp = el.enterSafeProgress != null
   441→          ? Math.max(0, Math.min(1, el.enterSafeProgress))
   442→          : Math.max(0, Math.min(1, el.enterProgress!))
   443→        return easeIn(sp)
   444→      })() : 1
   445→      gl.uniform4f(this.uPr['uColor'], c[0], c[1], c[2], c[3] * enterA)
   446→      gl.uniform1f(this.uPr['uCornerStyle'], this.cornerStyle)
   447→      // Continuous-curvature mask for capsule plain-rects (toggle tracks, etc.)
   448→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   449→        gl.activeTexture(gl.TEXTURE2)
   450→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   451→        gl.uniform1i(this.uPr['uContinuousSdf'], 2)
   452→        gl.uniform1f(this.uPr['uUseContinuousSdf'], 1.0)
   453→        gl.uniform2f(this.uPr['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   454→        gl.uniform2f(this.uPr['uContinuousSdfElementSize'], r2.w * this.dpr, r2.h * this.dpr)
   455→      } else {
   456→        gl.uniform1f(this.uPr['uUseContinuousSdf'], 0.0)
   457→      }
   458→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   459→      this.perfMonitor.incNonGlass()
   460→      this.perfMonitor.incDrawCall()
   461→      return true
   462→    }
   463→
   464→    // --- progressive-blur ---
   465→    // Progressive-blur samples the wallpaper directly (not the scene
   466→    // texture) — this matches the original catalog which uses
   467→    // AlphaMask over the canvas backdrop. If we wanted it to blur the
   468→    // scene (including plain-rects drawn before it), we'd sample
   469→    // curTex here instead. For now, keep the original behavior.
   470→    if (el.kind === 'progressive-blur' && el.progressiveBlur) {
   471→      this.bindFBO(curFbo)
   472→      gl.useProgram(this.progressiveBlurProgram)
   473→      this.setSdfUniforms(this.uPb, this.aPosLocPb, r2, el.cornerRadius)
   474→      // Premultiplied alpha blending — the shader outputs premultiplied rgb
   475→      // (rgb * alpha) faithful to the original AGSL AlphaMask shader.
   476→      // Using SRC_ALPHA would double-apply alpha (rgb*a*a) → black band at bottom.
   477→      gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   478→      gl.activeTexture(gl.TEXTURE0)
   479→      gl.bindTexture(gl.TEXTURE_2D, this.wallpaperTexture!)
   480→      gl.uniform1i(this.uPb['uBackdrop'], 0)
   481→      gl.uniform2f(this.uPb['uWallpaperSize'], this.wallpaperSize[0], this.wallpaperSize[1])
   482→      gl.uniform1f(this.uPb['uBlurRadius'], el.progressiveBlur.blurRadius * this.dpr)
   483→      const tc = el.progressiveBlur.tintColor
   484→      gl.uniform4f(this.uPb['uTintColor'], tc[0], tc[1], tc[2], tc[3])
   485→      gl.uniform1f(this.uPb['uTintIntensity'], el.progressiveBlur.tintIntensity)
   486→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   487→      this.perfMonitor.incNonGlass()
   488→      this.perfMonitor.incDrawCall()
   489→      return true
   490→    }
   491→
   492→    // --- text ---
   493→    if (el.kind === 'text') {
   494→      this.bindFBO(curFbo)
   495→      // Compute the effective draw rect for bottom-tab content.
   496→      // Faithful to LiquidBottomTabs.kt: the container is the parent of all
   497→      // tab-content, so the container's scale applies to the WHOLE Row as a
   498→      // unit — each tab scales around the CONTAINER's center, not its own.
   499→      // This means tabs spread apart as the bar grows:
   500→      //   scaledTabCenter = containerCenter + (tabCenter - containerCenter) * scale
   501→      let drawRect = r2
   502→      let fgScaleX = 1
   503→      let fgScaleY = 1
   504→      if (el.isBottomTabContent) {
   505→        const tg = this.toggleStates.get(el.isBottomTabContent.groupId)
   506→        if (tg) {
   507→          // Container scale = lerp(1, 1+16dp/width, pressProgress).
   508→          const containerW = el.isBottomTabContent.containerWidth ?? el.rect.w * 4
   509→          const containerScale = 1 + (16 * DP) / containerW * tg.pressProgress
   510→          fgScaleX = containerScale
   511→          fgScaleY = containerScale
   512→          // Scale around the CONTAINER center (not the tab's own center).
   513→          const pivotX = el.isBottomTabContent.containerCenterX ?? (el.rect.x + el.rect.w / 2)
   514→          const pivotY = el.isBottomTabContent.containerCenterY ?? (el.rect.y + el.rect.h / 2)
   515→          const tabCenterX = el.rect.x + el.rect.w / 2
   516→          const tabCenterY = el.rect.y + el.rect.h / 2
   517→          // scaledCenter = pivot + (center - pivot) * scale + panelOffset
   518→          const cx = pivotX + (tabCenterX - pivotX) * containerScale + tg.panelOffset
   519→          const cy = pivotY + (tabCenterY - pivotY) * containerScale
   520→          const sw = el.rect.w * fgScaleX
   521→          const sh = el.rect.h * fgScaleY
   522→          drawRect = { x: cx - sw / 2, y: cy - sh / 2, w: sw, h: sh }
   523→        }
   524→      }
   525→      // Press tint overlay for interactive text items (e.g. home list
   526→      // items). Faithful to MainContent.kt's
   527→      //   ripple(color = if (isLightTheme) Color.Black else Color.White)
   528→      //   RippleDefaults.pressedAlpha = 0.1f
   529→      // When el.pressTintColor is set, use SrcOver blend with that color
   530→      // (black in light theme, white in dark). When unset, fall back to the
   531→      // legacy white Plus-blend overlay for backward compat.
   532→      const pText = st?.pressProgress ?? 0
   533→      if (el.isInteractive && pText > 0.001) {
   534→        const pressTint = el.pressTintColor
   535→        gl.useProgram(this.tintProgram)
   536→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   537→        gl.enableVertexAttribArray(this.aPosLocTn)
   538→        gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   539→        if (pressTint) {
   540→          // Ripple (SrcOver): color over content at pressedAlpha.
   541→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   542→        } else {
   543→          // Legacy white Plus-blend overlay.
   544→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   545→        }
   546→        gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   547→        gl.uniform2f(this.uTn['uOffset'], drawRect.x * this.dpr, drawRect.y * this.dpr)
   548→        gl.uniform2f(this.uTn['uSize'], drawRect.w * this.dpr, drawRect.h * this.dpr)
   549→        gl.uniform4f(this.uTn['uCornerRadii'], 0, 0, 0, 0)
   550→        gl.uniform2f(this.uTn['uOriginalSize'], drawRect.w * this.dpr, drawRect.h * this.dpr)
   551→        gl.uniform1f(this.uTn['uOriginalCornerRadius'], 0)
   552→        gl.uniform2f(this.uTn['uLayerScale'], 1, 1)
   553→        if (pressTint) {
   554→          gl.uniform4f(this.uTn['uColor'], pressTint[0], pressTint[1], pressTint[2], 0.10 * pText)
   555→        } else {
   556→          gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.10 * pText)
   557→        }
   558→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   559→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   560→      }
   561→      const fgTex = this.fgTextures.get(el.id)
   562→      if (fgTex) {
   563→        gl.useProgram(this.foregroundProgram)
   564→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   565→        gl.enableVertexAttribArray(this.aPosLocFg)
   566→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   567→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   568→        gl.activeTexture(gl.TEXTURE0)
   569→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   570→        gl.uniform1i(this.uFg['uTexture'], 0)
   571→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   572→        gl.uniform2f(this.uFg['uOffset'], drawRect.x * this.dpr, drawRect.y * this.dpr)
   573→        gl.uniform2f(this.uFg['uSize'], drawRect.w * this.dpr, drawRect.h * this.dpr)
   574→        gl.uniform4f(
   575→          this.uFg['uCornerRadii'],
   576→          el.cornerRadius * this.dpr,
   577→          el.cornerRadius * this.dpr,
   578→          el.cornerRadius * this.dpr,
   579→          el.cornerRadius * this.dpr
   580→        )
   581→        // Pass the content scale so the foreground shader's SDF clip scales
   582→        // correctly (matching the glass-element pattern). For non-tab text,
   583→        // layerScale = 1 (origSize = scaled size).
   584→        gl.uniform2f(this.uFg['uOriginalSize'], el.rect.w * this.dpr, el.rect.h * this.dpr)
   585→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], el.cornerRadius * this.dpr)
   586→        gl.uniform2f(this.uFg['uLayerScale'], fgScaleX, fgScaleY)
   587→        gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   588→        gl.uniform1f(this.uFg['uAlpha'], el.enterProgress != null ? (() => {
   589→          const sp = el.enterSafeProgress != null
   590→            ? Math.max(0, Math.min(1, el.enterSafeProgress))
   591→            : Math.max(0, Math.min(1, el.enterProgress!))
   592→          return easeIn(sp)
   593→        })() : 1.0)
   594→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   595→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   596→      }
   597→      this.perfMonitor.incNonGlass()
   598→      this.perfMonitor.incDrawCall()
   599→      return true
   600→    }
   601→
   602→    return false
   603→  },
   604→}

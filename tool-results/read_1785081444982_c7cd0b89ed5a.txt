     1→     1→import type { LiquidGlassRenderer } from './index'
     2→     2→import type { GlassRenderState } from './methods-render-glass'
     3→     3→import { DP } from './spring'
     4→     4→import { continuousCurvatureRoundedRectPath } from './continuous-curve'
     5→     5→
     6→     6→declare module './index' {
     7→     7→  interface LiquidGlassRenderer {
     8→     8→    renderGlassPostPasses(state: GlassRenderState): void
     9→     9→  }
    10→    10→}
    11→    11→
    12→    12→export const glassPostPassMethods = {
    13→    13→  /** Steps 2b–2f: Inner shadow, press glow, white overlay, foreground, rim highlight.
    14→    14→   *  These all composite on top of the glass body (already drawn to
    15→    15→   *  otherFbo by renderGlassElementPass). */
    16→    16→  renderGlassPostPasses(this: LiquidGlassRenderer, state: GlassRenderState) {
    17→    17→    const gl = this.gl
    18→    18→    const { el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress, elHighlightAlpha } = state
    19→    19→
    20→    20→    // Original-space SDF uniforms — shared by all post-pass shaders so their
    21→    21→    // capsule clip is computed in ORIGINAL space (faithful to graphicsLayer
    22→    22→    // { scaleX, scaleY } post-scaling). See element.ts / highlight.ts.
    23→    23→    const origSizeX = state.origW * this.dpr
    24→    24→    const origSizeY = state.origH * this.dpr
    25→    25→    const origRadius = state.origCornerRadius * this.dpr
    26→    26→    const layerScaleX = state.layerScaleX
    27→    27→    const layerScaleY = state.layerScaleY
    28→    28→
    29→    29→    // --- Step 2b: Inner shadow post-passes (Canvas2D ring mask approach) ---
    30→    30→    // Inner shadows are drawn on the element surface, underneath the press glow.
    31→    31→    // Each shadow uses a Canvas2D-generated blurred ring mask (fill shape →
    32→    32→    // destination-out offset shape → blur), composited via the
    33→    33→    // INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER with SrcOver blend.
    34→    34→    // SDF clip in the shader ensures the shadow stays inside the shape boundary.
    35→    35→
    36→    36→    /** Helper: draw one inner shadow post-pass (shadow1 or shadow2). */
    37→    37→    const drawInnerShadowPass = (
    38→    38→      shadowCfg: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] },
    39→    39→      shadowIndex: number // 0 = innerShadow, 1 = innerShadow2
    40→    40→    ) => {
    41→    41→      // Progress modulation — faithful to the original inline shader:
    42→    42→      //   - Toggle knobs: radius, alpha, offset all modulated by togglePressProgress
    43→    43→      //   - Bottom-tab indicator: radius, alpha, offset modulated by togglePressProgress
    44→    44→      //   - Other elements (magnifier): static params, no modulation
    45→    45→      const progress = (el.isToggleKnob || el.isBottomTabIndicator) ? togglePressProgress : 1
    46→    46→      let shadowAlpha = shadowCfg.alpha * progress * state.enterAlpha
    47→    47→      let shadowRadius = shadowCfg.radius * progress
    48→    48→      let shadowOffsetX = shadowCfg.offsetX * progress
    49→    49→      let shadowOffsetY = shadowCfg.offsetY * progress
    50→    50→
    51→    51→      if (shadowAlpha <= 0.001 || shadowRadius <= 0.5) return
    52→    52→
    53→    53→      // Blur sigma = radius * dpr (BlurEffect semantics: sigma = radius directly).
    54→    54→      // The original InnerShadowModifier.kt uses BlurEffect(radius, radius, TileMode.Decal)
    55→    55→      // where the radius parameter IS the Gaussian sigma (NOT radius/3 like BlurMaskFilter).
    56→    56→      // BlurEffect wraps SkImageFilter::MakeBlur which takes sigma directly.
    57→    57→      const blurSigma = shadowRadius * this.dpr  // device px — sigma = radius, not radius/3
    58→    58→      // Margin for blur spread (3σ) + AA
    59→    59→      const margin = Math.ceil(blurSigma * 3) + 2
    60→    60→      // Mask dimensions in device px (origSize + 2*margin)
    61→    61→      const maskW = Math.max(1, Math.ceil(origSizeX + 2 * margin))
    62→    62→      const maskH = Math.max(1, Math.ceil(origSizeY + 2 * margin))
    63→    63→      // Supersampling for sharper mask rasterization
    64→    64→      const deviceDpr = window.devicePixelRatio || 1
    65→    65→      const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
    66→    66→      const canvasW = maskW * SS
    67→    67→      const canvasH = maskH * SS
    68→    68→      const useG2 = !!el.useContinuousSdf
    69→    69→
    70→    70→      // Offset in device px (already × progress)
    71→    71→      const offsetXDp = shadowOffsetX * this.dpr
    72→    72→      const offsetYDp = shadowOffsetY * this.dpr
    73→    73→
    74→    74→      const maskKey = [
    75→    75→        'is',
    76→    76→        shadowIndex,
    77→    77→        useG2 ? 'g2' : 'rr',
    78→    78→        origSizeX.toFixed(3),
    79→    79→        origSizeY.toFixed(3),
    80→    80→        origRadius.toFixed(3),
    81→    81→        offsetXDp.toFixed(3),
    82→    82→        offsetYDp.toFixed(3),
    83→    83→        blurSigma.toFixed(3),
    84→    84→        margin,
    85→    85→        maskW,
    86→    86→        maskH,
    87→    87→        `ss${SS}`,
    88→    88→      ].join(':')
    89→    89→
    90→    90→      let mask = this.innerShadowMaskCache.get(maskKey)
    91→    91→      if (!mask) {
    92→    92→        const canvas = document.createElement('canvas')
    93→    93→        canvas.width = canvasW
    94→    94→        canvas.height = canvasH
    95→    95→        const ctx = canvas.getContext('2d', { alpha: true })
    96→    96→        if (!ctx) throw new Error('2D canvas not supported')
    97→    97→        const tex = gl.createTexture()
    98→    98→        if (!tex) throw new Error('WebGL texture allocation failed')
    99→    99→        mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   100→   100→        this.innerShadowMaskCache.set(maskKey, mask)
   101→   101→
   102→   102→        // Keep the cache bounded (32 entries)
   103→   103→        if (this.innerShadowMaskCache.size > 32) {
   104→   104→          const oldestKey = this.innerShadowMaskCache.keys().next().value as string | undefined
   105→   105→          if (oldestKey && oldestKey !== maskKey) {
   106→   106→            const oldest = this.innerShadowMaskCache.get(oldestKey)
   107→   107→            if (oldest) gl.deleteTexture(oldest.tex)
   108→   108→            this.innerShadowMaskCache.delete(oldestKey)
   109→   109→          }
   110→   110→        }
   111→   111→      }
   112→   112→
   113→   113→      if (!mask.ready) {
   114→   114→        // --- Generate the ring mask via Canvas2D (two-canvas approach) ---
   115→   115→        // Step 1: draw hard-edge ring on a temp canvas (with SS× supersampling)
   116→   116→        // Step 2: blur the ring onto the output canvas at 1:1 physical pixels
   117→   117→        //         (no ctx.scale — drawImage and blur must be in physical-px space)
   118→   118→        //   - blur radius = blurSigma * SS physical px (= blurSigma logical px)
   119→   119→        //   - drawImage at 1:1 physical (canvasW × canvasH → canvasW × canvasH)
   120→   120→
   121→   121→        // --- Temp canvas for the hard-edge ring ---
   122→   122→        const tempCanvas = document.createElement('canvas')
   123→   123→        tempCanvas.width = canvasW
   124→   124→        tempCanvas.height = canvasH
   125→   125→        const tCtx = tempCanvas.getContext('2d', { alpha: true })
   126→   126→        if (!tCtx) throw new Error('2D canvas not supported')
   127→   127→
   128→   128→        // Draw ring on temp canvas: fill shape → destination-out offset shape
   129→   129→        // Scale(SS, SS) for supersampling: draw in logical (device-px) coordinates
   130→   130→        // while the physical canvas is SS× larger.
   131→   131→        tCtx.save()
   132→   132→        tCtx.scale(SS, SS)
   133→   133→        tCtx.translate(margin, margin)
   134→   134→
   135→   135→        // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   136→   136→        let path: Path2D
   137→   137→        if (useG2) {
   138→   138→          path = continuousCurvatureRoundedRectPath(tCtx, origSizeX, origSizeY, origRadius)
   139→   139→        } else {
   140→   140→          path = new Path2D()
   141→   141→          const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   142→   142→          path.moveTo(r, 0)
   143→   143→          path.lineTo(origSizeX - r, 0)
   144→   144→          path.arcTo(origSizeX, 0, origSizeX, r, r)
   145→   145→          path.lineTo(origSizeX, origSizeY - r)
   146→   146→          path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   147→   147→          path.lineTo(r, origSizeY)
   148→   148→          path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   149→   149→          path.lineTo(0, r)
   150→   150→          path.arcTo(0, 0, r, 0, r)
   151→   151→          path.closePath()
   152→   152→        }
   153→   153→
   154→   154→        // Fill the shape with white (creates full interior)
   155→   155→        tCtx.globalCompositeOperation = 'source-over'
   156→   156→        tCtx.fillStyle = 'white'
   157→   157→        tCtx.fill(path)
   158→   158→
   159→   159→        // Draw the OFFSET shape with destination-out (removes offset interior)
   160→   160→        tCtx.globalCompositeOperation = 'destination-out'
   161→   161→        tCtx.save()
   162→   162→        tCtx.translate(offsetXDp, offsetYDp)
   163→   163→        tCtx.fill(path) // same path, shifted by offset
   164→   164→        tCtx.restore()
   165→   165→
   166→   166→        tCtx.globalCompositeOperation = 'source-over'
   167→   167→        tCtx.restore()
   168→   168→
   169→   169→        // --- Blur the ring onto the output canvas (1:1 physical pixels) ---
   170→   170→        // NO ctx.scale here — drawImage and ctx.filter blur must operate in
   171→   171→        // physical-pixel space to avoid SS× size distortion.
   172→   172→        //   blurSigma is in device px (logical); the SS× canvas has SS physical px
   173→   173→        //   per logical px, so the physical blur radius = blurSigma * SS.
   174→   174→        const mCtx = mask.ctx
   175→   175→        mCtx.clearRect(0, 0, canvasW, canvasH)
   176→   176→        if (blurSigma > 0.01) {
   177→   177→          mCtx.filter = `blur(${blurSigma * SS}px)`
   178→   178→        } else {
   179→   179→          mCtx.filter = 'none'
   180→   180→        }
   181→   181→        // 1:1 physical-pixel drawImage: temp(canvasW×canvasH) → mask(canvasW×canvasH)
   182→   182→        mCtx.drawImage(tempCanvas, 0, 0)
   183→   183→        mCtx.filter = 'none'
   184→   184→
   185→   185→        // Upload to GPU texture
   186→   186→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   187→   187→        gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   188→   188→        gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   189→   189→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   190→   190→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   191→   191→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   192→   192→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   193→   193→        mask.ready = true
   194→   194→      }
   195→   195→
   196→   196→      // --- Composite: inner shadow mask × shadowAlpha × shadowColor → scene ---
   197→   197→      gl.enable(gl.BLEND)
   198→   198→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   199→   199→      gl.useProgram(this.innerShadowMaskCompositeProgram)
   200→   200→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   201→   201→      gl.enableVertexAttribArray(this.aPosLocIs)
   202→   202→      gl.vertexAttribPointer(this.aPosLocIs, 2, gl.FLOAT, false, 0, 0)
   203→   203→
   204→   204→      gl.uniform2f(this.uIs['uCanvasSize'], this.canvas.width, this.canvas.height)
   205→   205→      gl.uniform2f(this.uIs['uOffset'], sx * this.dpr, sy * this.dpr)
   206→   206→      gl.uniform2f(this.uIs['uSize'], sw * this.dpr, sh * this.dpr)
   207→   207→      gl.uniform4f(this.uIs['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   208→   208→
   209→   209→      gl.activeTexture(gl.TEXTURE0)
   210→   210→      gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   211→   211→      gl.uniform1i(this.uIs['uInnerShadowMask'], 0)
   212→   212→      // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   213→   213→      // physical canvas is SS× larger but the shader uses 1x coords for UV.
   214→   214→      gl.uniform2f(this.uIs['uMaskOffset'], margin, margin)
   215→   215→      gl.uniform2f(this.uIs['uMaskSize'], mask.w, mask.h)
   216→   216→
   217→   217→      // Shadow color (defaults to black [0,0,0] if not specified)
   218→   218→      const color = shadowCfg.color ?? [0, 0, 0]
   219→   219→      gl.uniform3f(this.uIs['uInnerShadowColor'], color[0], color[1], color[2])
   220→   220→      gl.uniform1f(this.uIs['uInnerShadowAlpha'], shadowAlpha)
   221→   221→
   222→   222→      gl.uniform2f(this.uIs['uOriginalSize'], origSizeX, origSizeY)
   223→   223→      gl.uniform1f(this.uIs['uOriginalCornerRadius'], origRadius)
   224→   224→      gl.uniform2f(this.uIs['uLayerScale'], layerScaleX, layerScaleY)
   225→   225→      gl.uniform1f(this.uIs['uElementRotation'], state.elementRotation)
   226→   226→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   227→   227→    }
   228→   228→
   229→   229→    // Inner shadow 1 (typically dark/black, offset downward)
   230→   230→    if (el.innerShadow) {
   231→   231→      drawInnerShadowPass(el.innerShadow, 0)
   232→   232→    }
   233→   233→    // Inner shadow 2 (typically white/bright, offset upward — 3D bevel)
   234→   234→    if (el.innerShadow2) {
   235→   235→      drawInnerShadowPass(el.innerShadow2, 1)
   236→   236→    }
   237→   237→
   238→   238→    // --- Step 2c: Press glow (button + bottom-tab container) ---
   239→   239→    // Faithful to InteractiveHighlight.kt: a flat white Plus-blend overlay
   240→   240→    // + a radial white glow. For buttons, position = finger, intensity = st.pressProgress.
   241→   241→    // For the bottom-tab container, position = indicator center (via the
   242→   242→    // original's position lambda), intensity = toggleState.pressProgress.
   243→   243→    const isContainer = !!el.isBottomTabContainer
   244→   244→    const glowP = isButton ? p : (isContainer ? togglePressProgress : 0)
   245→   245→    if ((isButton && el.isInteractive && st && p > 0.001) || (isContainer && togglePressProgress > 0.001)) {
   246→   246→      // a. Flat white overlay
   247→   247→      gl.useProgram(this.tintProgram)
   248→   248→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   249→   249→      gl.enableVertexAttribArray(this.aPosLocTn)
   250→   250→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   251→   251→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   252→   252→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   253→   253→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   254→   254→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   255→   255→      gl.uniform4f(
   256→   256→        this.uTn['uCornerRadii'],
   257→   257→        radii[0] * this.dpr,
   258→   258→        radii[1] * this.dpr,
   259→   259→        radii[2] * this.dpr,
   260→   260→        radii[3] * this.dpr
   261→   261→      )
   262→   262→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   263→   263→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   264→   264→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   265→   265→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   266→   266→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   267→   267→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.08 * glowP)
   268→   268→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   269→   269→
   270→   270→      // b. Radial highlight at finger position (button) or indicator center (container).
   271→   271→      gl.useProgram(this.highlightProgram)
   272→   272→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   273→   273→      gl.enableVertexAttribArray(this.aPosLocHl)
   274→   274→      gl.vertexAttribPointer(this.aPosLocHl, 2, gl.FLOAT, false, 0, 0)
   275→   275→      gl.blendFunc(gl.ONE, gl.ONE)
   276→   276→      gl.uniform2f(this.uHl['uCanvasSize'], this.canvas.width, this.canvas.height)
   277→   277→      gl.uniform2f(this.uHl['uOffset'], sx * this.dpr, sy * this.dpr)
   278→   278→      gl.uniform2f(this.uHl['uSize'], sw * this.dpr, sh * this.dpr)
   279→   279→      gl.uniform4f(
   280→   280→        this.uHl['uCornerRadii'],
   281→   281→        radii[0] * this.dpr,
   282→   282→        radii[1] * this.dpr,
   283→   283→        radii[2] * this.dpr,
   284→   284→        radii[3] * this.dpr
   285→   285→      )
   286→   286→      gl.uniform2f(this.uHl['uOriginalSize'], origSizeX, origSizeY)
   287→   287→      gl.uniform1f(this.uHl['uOriginalCornerRadius'], origRadius)
   288→   288→      gl.uniform2f(this.uHl['uLayerScale'], layerScaleX, layerScaleY)
   289→   289→      gl.uniform1f(this.uHl['uElementRotation'], state.elementRotation)
   290→   290→      gl.uniform1f(this.uHl['uCornerStyle'], this.cornerStyle)
   291→   291→      gl.uniform4f(this.uHl['uColor'], 1, 1, 1, 0.15 * glowP)
   292→   292→      const minDim = Math.min(sw, sh) * this.dpr
   293→   293→      gl.uniform1f(this.uHl['uRadius'], minDim * 1.5)
   294→   294→      // Position: for buttons, the finger position. For the container, the
   295→   295→      // indicator center — faithful to the original's position lambda which
   296→   296→      // returns (dampedDragAnimation.value + 0.5) * tabWidth (indicator center
   297→   297→      // relative to the container).
   298→   298→      let px: number, py: number
   299→   299→      if (isContainer) {
   300→   300→        const tg = this.toggleStates.get(el.isBottomTabContainer!.groupId)
   301→   301→        // tabW = container ORIGINAL width / tabsCount (not scaled — the position
   302→   302→        // lambda runs in pre-scale local coords). indicator center =
   303→   303→        // (fraction + 0.5) * tabW. Faithful to LiquidBottomTabs.kt:
   304→   304→        //   position = (dampedDragAnimation.value + 0.5) * tabWidth
   305→   305→        const tabsCount = el.isBottomTabContainer!.tabsCount ?? 4
   306→   306→        const tabW = el.rect.w / tabsCount
   307→   307→        const fraction = tg ? tg.fraction : 0
   308→   308→        const indCenterX = (fraction + 0.5) * tabW
   309→   309→        // Map original-local to scaled-local (the shader's uPosition is in
   310→   310→        // scaled-local coords, 0..sw). scale = sw / el.rect.w.
   311→   311→        const scaleToLocal = sw / el.rect.w
   312→   312→        px = Math.max(0, Math.min(sw, indCenterX * scaleToLocal)) * this.dpr
   313→   313→        py = (sh / 2) * this.dpr
   314→   314→      } else {
   315→   315→        px = Math.max(0, Math.min(sw, st!.dragX * state.layerScaleX)) * this.dpr
   316→   316→        py = Math.max(0, Math.min(sh, st!.dragY * state.layerScaleY)) * this.dpr
   317→   317→      }
   318→   318→      gl.uniform2f(this.uHl['uPosition'], px, py)
   319→   319→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   320→   320→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   321→   321→    }
   322→   322→
   323→   323→    // --- Step 2d: Toggle knob white overlay (faithful to LiquidToggle.kt
   324→   324→    // / LiquidSlider.kt onDrawSurface) ---
   325→   325→    if (el.isToggleKnob && togglePressProgress < 0.999) {
   326→   326→      const whiteAlpha = 1.0 * (1 - togglePressProgress)
   327→   327→      gl.useProgram(this.tintProgram)
   328→   328→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   329→   329→      gl.enableVertexAttribArray(this.aPosLocTn)
   330→   330→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   331→   331→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   332→   332→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   333→   333→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   334→   334→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   335→   335→      gl.uniform4f(
   336→   336→        this.uTn['uCornerRadii'],
   337→   337→        radii[0] * this.dpr,
   338→   338→        radii[1] * this.dpr,
   339→   339→        radii[2] * this.dpr,
   340→   340→        radii[3] * this.dpr
   341→   341→      )
   342→   342→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   343→   343→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   344→   344→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   345→   345→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   346→   346→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   347→   347→      // Faithful to LiquidToggle.kt onDrawSurface:
   348→   348→      //   drawRect(Color.White.copy(alpha = 1f - progress))
   349→   349→      // Solid white pebble at rest (alpha=1), fading to transparent when
   350→   350→      // pressed (alpha=0) to reveal the glass refraction beneath.
   351→   351→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, whiteAlpha)
   352→   352→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   353→   353→    }
   354→   354→
   355→   355→    // --- Step 2d2: Bottom tab indicator onDrawSurface (faithful to
   356→   356→    // LiquidBottomTabs.kt indicator):
   357→   357→    //   drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)
   358→   358→    //   drawRect(Color.Black.copy(alpha = 0.03f * progress))
   359→   359→    // First: theme-aware dim overlay fading OUT on press (SrcOver).
   360→   360→    // Second: subtle black tint fading IN on press (SrcOver).
   361→   361→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.dimColor) {
   362→   362→      const dc = el.isBottomTabIndicator.dimColor
   363→   363→      const p = togglePressProgress
   364→   364→      gl.useProgram(this.tintProgram)
   365→   365→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   366→   366→      gl.enableVertexAttribArray(this.aPosLocTn)
   367→   367→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   368→   368→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   369→   369→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   370→   370→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   371→   371→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   372→   372→      gl.uniform4f(this.uTn['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   373→   373→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   374→   374→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   375→   375→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   376→   376→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   377→   377→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   378→   378→      // First overlay: dim color at 0.1 * (1 - progress) — fades out on press.
   379→   379→      gl.uniform4f(this.uTn['uColor'], dc[0], dc[1], dc[2], 0.1 * (1 - p))
   380→   380→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   381→   381→      // Second overlay: black at 0.03 * progress — fades in on press.
   382→   382→      gl.uniform4f(this.uTn['uColor'], 0, 0, 0, 0.03 * p)
   383→   383→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   384→   384→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   385→   385→    }
   386→   386→
   387→   387→    // --- Step 2e: Foreground (label or icon) pass (button only) ---
   388→   388→    if (isButton && (el.label || el.icon)) {
   389→   389→      const fgTex = this.fgTextures.get(el.id)
   390→   390→      if (fgTex) {
   391→   391→        gl.useProgram(this.foregroundProgram)
   392→   392→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   393→   393→        gl.enableVertexAttribArray(this.aPosLocFg)
   394→   394→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   395→   395→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   396→   396→
   397→   397→        gl.activeTexture(gl.TEXTURE0)
   398→   398→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   399→   399→        gl.uniform1i(this.uFg['uTexture'], 0)
   400→   400→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   401→   401→        gl.uniform2f(this.uFg['uOffset'], sx * this.dpr, sy * this.dpr)
   402→   402→        gl.uniform2f(this.uFg['uSize'], sw * this.dpr, sh * this.dpr)
   403→   403→        gl.uniform4f(
   404→   404→          this.uFg['uCornerRadii'],
   405→   405→          radii[0] * this.dpr,
   406→   406→          radii[1] * this.dpr,
   407→   407→          radii[2] * this.dpr,
   408→   408→          radii[3] * this.dpr
   409→   409→        )
   410→   410→        gl.uniform2f(this.uFg['uOriginalSize'], origSizeX, origSizeY)
   411→   411→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], origRadius)
   412→   412→      gl.uniform2f(this.uFg['uLayerScale'], layerScaleX, layerScaleY)
   413→   413→      gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   414→   414→      // Continuous-curvature mask for capsule foregrounds.
   415→   415→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   416→   416→        gl.activeTexture(gl.TEXTURE2)
   417→   417→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   418→   418→        gl.uniform1i(this.uFg['uContinuousSdf'], 2)
   419→   419→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 1.0)
   420→   420→        gl.uniform2f(this.uFg['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   421→   421→        gl.uniform2f(this.uFg['uContinuousSdfElementSize'], state.origW * this.dpr, state.origH * this.dpr)
   422→   422→      } else {
   423→   423→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 0.0)
   424→   424→      }
   425→   425→        gl.uniform1f(this.uFg['uAlpha'], 1.0 - 0.15 * p)
   426→   426→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   427→   427→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   428→   428→      }
   429→   429→    }
   430→   430→
   431→   431→    // --- Step 2f: Rim highlight (Canvas2D stroke mask approach) ---
   432→   432→    // Uses Canvas2D ctx.stroke() (browser-native Skia) to rasterize the G2
   433→   433→    // Bezier path as a stroke mask. This is the SAME method the original
   434→   434→    // uses (Skia drawOutline + paint.Stroke) — the browser's Canvas2D
   435→   435→    // internally calls Skia's SkCanvas::drawPath, which tessellates the
   436→   436→    // Bezier into triangles and rasterizes with hardware coverage AA.
   437→   437→    //
   438→   438→    // Advantages over the SDF approach:
   439→   439→    //   - Exact G2 Bezier shape (not SDF approximation)
   440→   440→    //   - Hardware coverage AA (sub-pixel accurate, no smoothstep needed)
   441→   441→    //   - No SDF computation in shader (just one texture fetch)
   442→   442→    //   - Adaptive tessellation (more samples at high-curvature corners)
   443→   443→    //
   444→   444→    // The stroke mask is cached by exact geometry. Highlight angle/alpha/press
   445→   445→    // progress only affect the composite uniforms, not the mask, so caching is
   446→   446→    // safe and avoids both per-frame rasterization and UV mismatch between a
   447→   447→    // reused backing canvas and the logical mask size.
   448→   448→    if (el.highlight && el.highlight.alpha > 0.001) {
   449→   449→      const rimAlpha = (el.isToggleKnob || el.isBottomTabIndicator) ? elHighlightAlpha : el.highlight.alpha
   450→   450→      const finalAlpha = rimAlpha * state.enterAlpha
   451→   451→      if (finalAlpha > 0.001) {
   452→   452→        // HighlightModifier.kt: ceil(width.toPx().coerceAtMost(minDimension / 2)) * 2
   453→   453→        const widthPx = Math.min(
   454→   454→          el.highlight.widthDp * this.dpr,
   455→   455→          Math.min(origSizeX, origSizeY) * 0.5
   456→   456→        )
   457→   457→        const strokeWidthDevice = Math.max(1, Math.ceil(widthPx) * 2)
   458→   458→        // Highlight data class: blurRadius defaults to width / 2. Honor it in the
   459→   459→        // Canvas2D mask as well (for the default 0.25dp this is sub-pixel, but it
   460→   460→        // keeps non-default highlights faithful too).
   461→   461→        const blurPx = Math.max(0, (el.highlight.blurRadiusDp ?? el.highlight.widthDp / 2) * this.dpr)
   462→   462→
   463→   463→        // --- Generate stroke mask via Canvas2D (browser-native Skia) ---
   464→   464→        // Element-local coordinates: (0,0) = element top-left in device px.
   465→   465→        // Margin for stroke width + AA + blur spread.
   466→   466→        const strokeMargin = Math.ceil(strokeWidthDevice) + 4  // logical margin
   467→   467→        // Logical mask size (1x device px) — used for shader UV mapping
   468→   468→        const maskW = Math.max(1, Math.ceil(origSizeX + 2 * strokeMargin))
   469→   469→        const maskH = Math.max(1, Math.ceil(origSizeY + 2 * strokeMargin))
   470→   470→        // Supersample for sharper stroke/blur rasterization.
   471→   471→        // Cap SS so total mask DPR (this.dpr × SS) ≤ device's native DPR —
   472→   472→        // no point rendering pixels beyond what the screen can display.
   473→   473→        const deviceDpr = window.devicePixelRatio || 1
   474→   474→        const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
   475→   475→        const canvasW = maskW * SS
   476→   476→        const canvasH = maskH * SS
   477→   477→        const useG2 = !!el.useContinuousSdf
   478→   478→        const maskKey = [
   479→   479→          useG2 ? 'g2' : 'rr',
   480→   480→          origSizeX.toFixed(3),
   481→   481→          origSizeY.toFixed(3),
   482→   482→          origRadius.toFixed(3),
   483→   483→          strokeWidthDevice,
   484→   484→          blurPx.toFixed(3),
   485→   485→          strokeMargin,
   486→   486→          maskW,
   487→   487→          maskH,
   488→   488→          `ss${SS}`,  // cache key includes supersample factor
   489→   489→        ].join(':')
   490→   490→
   491→   491→        let mask = this.strokeMaskCache.get(maskKey)
   492→   492→        if (!mask) {
   493→   493→          const canvas = document.createElement('canvas')
   494→   494→          canvas.width = canvasW  // 2x supersampled physical size
   495→   495→          canvas.height = canvasH
   496→   496→          const ctx = canvas.getContext('2d', { alpha: true })
   497→   497→          if (!ctx) throw new Error('2D canvas not supported')
   498→   498→          const tex = gl.createTexture()
   499→   499→          if (!tex) throw new Error('WebGL texture allocation failed')
   500→   500→          // w/h store the LOGICAL (1x) size for shader UV mapping;
   501→   501→          // the physical canvas is SS times larger.
   502→   502→          mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   503→   503→          this.strokeMaskCache.set(maskKey, mask)
   504→   504→
   505→   505→          // Keep the cache bounded. 32 entries is far above the catalog's
   506→   506→          // simultaneous highlight geometries and avoids unbounded growth on
   507→   507→          // highly dynamic pages.
   508→   508→          if (this.strokeMaskCache.size > 32) {
   509→   509→            const oldestKey = this.strokeMaskCache.keys().next().value as string | undefined
   510→   510→            if (oldestKey && oldestKey !== maskKey) {
   511→   511→              const oldest = this.strokeMaskCache.get(oldestKey)
   512→   512→              if (oldest) gl.deleteTexture(oldest.tex)
   513→   513→              this.strokeMaskCache.delete(oldestKey)
   514→   514→            }
   515→   515→          }
   516→   516→        }
   517→   517→
   518→   518→        if (!mask.ready) {
   519→   519→          const smCtx = mask.ctx
   520→   520→          smCtx.clearRect(0, 0, canvasW, canvasH)
   521→   521→          smCtx.save()
   522→   522→          // Scale up for 2x supersampling: draw in logical (1x) coordinates
   523→   523→          // while the physical canvas is 2x. This gives sharper stroke + blur.
   524→   524→          smCtx.scale(SS, SS)
   525→   525→          smCtx.translate(strokeMargin, strokeMargin)
   526→   526→
   527→   527→          // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   528→   528→          let path: Path2D
   529→   529→          if (useG2) {
   530→   530→            path = continuousCurvatureRoundedRectPath(smCtx, origSizeX, origSizeY, origRadius)
   531→   531→          } else {
   532→   532→            path = new Path2D()
   533→   533→            const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   534→   534→            path.moveTo(r, 0)
   535→   535→            path.lineTo(origSizeX - r, 0)
   536→   536→            path.arcTo(origSizeX, 0, origSizeX, r, r)
   537→   537→            path.lineTo(origSizeX, origSizeY - r)
   538→   538→            path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   539→   539→            path.lineTo(r, origSizeY)
   540→   540→            path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   541→   541→            path.lineTo(0, r)
   542→   542→            path.arcTo(0, 0, r, 0, r)
   543→   543→            path.closePath()
   544→   544→          }
   545→   545→
   546→   546→          // Stroke — browser-native Skia stroke rasterization.
   547→   547→          // Faithful to HighlightModifier.kt: clipOutline(outline) BEFORE drawOutline,
   548→   548→          // so only the INSIDE half of the centered stroke remains. Without this
   549→   549→          // clip, Canvas2D keeps the outer half too and the rim highlight leaks
   550→   550→          // outside the glass / looks twice as thick.
   551→   551→          smCtx.clip(path)
   552→   552→          smCtx.lineWidth = strokeWidthDevice
   553→   553→          smCtx.strokeStyle = 'rgba(255,255,255,1)'
   554→   554→          smCtx.lineJoin = 'round'
   555→   555→          smCtx.lineCap = 'round'
   556→   556→          // Approximate Skia BlurMaskFilter on the stroke paint. The clip above
   557→   557→          // keeps the result inside the outline; the blur only softens that
   558→   558→          // inside stroke band, like the original HighlightModifier layer.
   559→   559→          smCtx.filter = blurPx > 0.01 ? `blur(${blurPx}px)` : 'none'
   560→   560→          smCtx.stroke(path)
   561→   561→          smCtx.filter = 'none'
   562→   562→          smCtx.restore()
   563→   563→
   564→   564→          // Upload to GPU texture. Keep the same top-left UV convention as the
   565→   565→          // foreground pass: with UNPACK_FLIP_Y_WEBGL=false, local y=0 samples
   566→   566→          // the top row of the mask.
   567→   567→          gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   568→   568→          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   569→   569→          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   570→   570→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   571→   571→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   572→   572→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   573→   573→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   574→   574→          mask.ready = true
   575→   575→        }
   576→   576→
   577→   577→        // --- Composite: stroke mask × intensity × color → scene FBO ---
   578→   578→        gl.enable(gl.BLEND)
   579→   579→        if (el.highlight.mode === 1) {
   580→   580→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   581→   581→        } else {
   582→   582→          gl.blendFunc(gl.ONE, gl.ONE)
   583→   583→        }
   584→   584→        gl.useProgram(this.strokeMaskCompositeProgram)
   585→   585→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   586→   586→        gl.enableVertexAttribArray(this.aPosLocSm)
   587→   587→        gl.vertexAttribPointer(this.aPosLocSm, 2, gl.FLOAT, false, 0, 0)
   588→   588→        gl.uniform2f(this.uSm['uCanvasSize'], this.canvas.width, this.canvas.height)
   589→   589→        gl.uniform2f(this.uSm['uOffset'], sx * this.dpr, sy * this.dpr)
   590→   590→        gl.uniform2f(this.uSm['uSize'], sw * this.dpr, sh * this.dpr)
   591→   591→        gl.uniform4f(this.uSm['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   592→   592→        gl.activeTexture(gl.TEXTURE0)
   593→   593→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   594→   594→        gl.uniform1i(this.uSm['uStrokeMask'], 0)
   595→   595→        // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   596→   596→        // physical canvas is SS× larger but the shader uses 1x coords for UV:
   597→   597→        //   UV = (localCoord + margin) / logicalMaskSize
   598→   598→        // LINEAR filtering on the SS× texture gives supersampled quality.
   599→   599→        gl.uniform2f(this.uSm['uMaskOffset'], strokeMargin, strokeMargin)
   600→   600→        gl.uniform2f(this.uSm['uMaskSize'], mask.w, mask.h)
   601→   601→        gl.uniform4f(this.uSm['uHighlightColor'], el.highlight.color[0], el.highlight.color[1], el.highlight.color[2], 1.0)
   602→   602→        gl.uniform1f(this.uSm['uHighlightAngle'], el.useGravityAngle ? this.gravityAngle : el.highlight.angle)
   603→   603→        gl.uniform1f(this.uSm['uHighlightFalloff'], el.highlight.falloff)
   604→   604→        gl.uniform1f(this.uSm['uHighlightAlpha'], finalAlpha)
   605→   605→        gl.uniform1f(this.uSm['uHighlightMode'], el.highlight.mode)
   606→   606→        gl.uniform2f(this.uSm['uOriginalSize'], origSizeX, origSizeY)
   607→   607→        gl.uniform1f(this.uSm['uOriginalCornerRadius'], origRadius)
   608→   608→        gl.uniform2f(this.uSm['uLayerScale'], layerScaleX, layerScaleY)
   609→   609→        gl.uniform1f(this.uSm['uElementRotation'], state.elementRotation)
   610→   610→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   611→   611→
   612→   612→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   613→   613→      }
   614→   614→    }
   615→   615→
   616→   616→    // --- Step 2g: 内层背景板 rim highlight (指示器 only) ---
   617→   617→    // The 内层背景板 (hidden Row's 56dp glass inside the 指示器) has its own
   618→   618→    // Highlight.Default.copy(alpha=progress). This is drawn INSIDE the
   619→   619→    // 指示器's element shader (sampleIndicatorBackdrop) so it's clipped by
   620→   620→    // the 指示器's capsule SDF — no separate pass needed here.
   621→   621→  },
   622→   622→}

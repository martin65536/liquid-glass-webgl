     1→import type { LiquidGlassRenderer } from './index'
     2→import type { GlassRenderState } from './methods-render-glass'
     3→import { DP } from './spring'
     4→import { continuousCurvatureRoundedRectPath } from './continuous-curve'
     5→
     6→declare module './index' {
     7→  interface LiquidGlassRenderer {
     8→    renderGlassPostPasses(state: GlassRenderState): void
     9→  }
    10→}
    11→
    12→export const glassPostPassMethods = {
    13→  /** Steps 2b–2f: Inner shadow, press glow, white overlay, foreground, rim highlight.
    14→   *  These all composite on top of the glass body (already drawn to
    15→   *  otherFbo by renderGlassElementPass). */
    16→  renderGlassPostPasses(this: LiquidGlassRenderer, state: GlassRenderState) {
    17→    const gl = this.gl
    18→    const { el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress, elHighlightAlpha } = state
    19→
    20→    // Original-space SDF uniforms — shared by all post-pass shaders so their
    21→    // capsule clip is computed in ORIGINAL space (faithful to graphicsLayer
    22→    // { scaleX, scaleY } post-scaling). See element.ts / highlight.ts.
    23→    const origSizeX = state.origW * this.dpr
    24→    const origSizeY = state.origH * this.dpr
    25→    const origRadius = state.origCornerRadius * this.dpr
    26→    const layerScaleX = state.layerScaleX
    27→    const layerScaleY = state.layerScaleY
    28→
    29→    // --- Step 2b: Inner shadow post-passes (Canvas2D ring mask approach) ---
    30→    // Inner shadows are drawn on the element surface, underneath the press glow.
    31→    // Each shadow uses a Canvas2D-generated blurred ring mask (fill shape →
    32→    // destination-out offset shape → blur), composited via the
    33→    // INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER with SrcOver blend.
    34→    // SDF clip in the shader ensures the shadow stays inside the shape boundary.
    35→
    36→    /** Helper: draw one inner shadow post-pass (shadow1 or shadow2). */
    37→    const drawInnerShadowPass = (
    38→      shadowCfg: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] },
    39→      shadowIndex: number // 0 = the only inner shadow (original has just ONE)
    40→    ) => {
    41→      // Progress modulation — faithful to the original inline shader:
    42→      //   - Toggle knobs: radius, alpha, offset all modulated by togglePressProgress
    43→      //   - Bottom-tab indicator: radius, alpha, offset modulated by togglePressProgress
    44→      //   - Other elements (magnifier): static params, no modulation
    45→      const progress = (el.isToggleKnob || el.isBottomTabIndicator) ? togglePressProgress : 1
    46→      let shadowAlpha = shadowCfg.alpha * progress * state.enterAlpha
    47→      let shadowRadius = shadowCfg.radius * progress
    48→      let shadowOffsetX = shadowCfg.offsetX * progress
    49→      let shadowOffsetY = shadowCfg.offsetY * progress
    50→
    51→      if (shadowAlpha <= 0.001 || shadowRadius <= 0.5) return
    52→
    53→      // Blur sigma = radius * dpr (BlurEffect semantics: sigma = radius directly).
    54→      // The original InnerShadowModifier.kt uses BlurEffect(radius, radius, TileMode.Decal)
    55→      // where the radius parameter IS the Gaussian sigma (NOT radius/3 like BlurMaskFilter).
    56→      // BlurEffect wraps SkImageFilter::MakeBlur which takes sigma directly.
    57→      const blurSigma = shadowRadius * this.dpr  // device px — sigma = radius, not radius/3
    58→      // Margin for blur spread (3σ) + AA
    59→      const margin = Math.ceil(blurSigma * 3) + 2
    60→      // Mask dimensions in device px (origSize + 2*margin)
    61→      const maskW = Math.max(1, Math.ceil(origSizeX + 2 * margin))
    62→      const maskH = Math.max(1, Math.ceil(origSizeY + 2 * margin))
    63→      // Supersampling for sharper mask rasterization
    64→      const deviceDpr = window.devicePixelRatio || 1
    65→      const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
    66→      const canvasW = maskW * SS
    67→      const canvasH = maskH * SS
    68→      const useG2 = !!el.useContinuousSdf
    69→
    70→      // Offset in device px (already × progress)
    71→      const offsetXDp = shadowOffsetX * this.dpr
    72→      const offsetYDp = shadowOffsetY * this.dpr
    73→
    74→      const maskKey = [
    75→        'is',
    76→        shadowIndex,
    77→        useG2 ? 'g2' : 'rr',
    78→        origSizeX.toFixed(3),
    79→        origSizeY.toFixed(3),
    80→        origRadius.toFixed(3),
    81→        offsetXDp.toFixed(3),
    82→        offsetYDp.toFixed(3),
    83→        blurSigma.toFixed(3),
    84→        margin,
    85→        maskW,
    86→        maskH,
    87→        `ss${SS}`,
    88→      ].join(':')
    89→
    90→      let mask = this.innerShadowMaskCache.get(maskKey)
    91→      if (!mask) {
    92→        const canvas = document.createElement('canvas')
    93→        canvas.width = canvasW
    94→        canvas.height = canvasH
    95→        const ctx = canvas.getContext('2d', { alpha: true })
    96→        if (!ctx) throw new Error('2D canvas not supported')
    97→        const tex = gl.createTexture()
    98→        if (!tex) throw new Error('WebGL texture allocation failed')
    99→        mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   100→        this.innerShadowMaskCache.set(maskKey, mask)
   101→
   102→        // Keep the cache bounded (32 entries)
   103→        if (this.innerShadowMaskCache.size > 32) {
   104→          const oldestKey = this.innerShadowMaskCache.keys().next().value as string | undefined
   105→          if (oldestKey && oldestKey !== maskKey) {
   106→            const oldest = this.innerShadowMaskCache.get(oldestKey)
   107→            if (oldest) gl.deleteTexture(oldest.tex)
   108→            this.innerShadowMaskCache.delete(oldestKey)
   109→          }
   110→        }
   111→      }
   112→
   113→      if (!mask.ready) {
   114→        // --- Generate the ring mask via Canvas2D (two-canvas approach) ---
   115→        // Step 1: draw hard-edge ring on a temp canvas (with SS× supersampling)
   116→        // Step 2: blur the ring onto the output canvas at 1:1 physical pixels
   117→        //         (no ctx.scale — drawImage and blur must be in physical-px space)
   118→        //   - blur radius = blurSigma * SS physical px (= blurSigma logical px)
   119→        //   - drawImage at 1:1 physical (canvasW × canvasH → canvasW × canvasH)
   120→
   121→        // --- Temp canvas for the hard-edge ring ---
   122→        const tempCanvas = document.createElement('canvas')
   123→        tempCanvas.width = canvasW
   124→        tempCanvas.height = canvasH
   125→        const tCtx = tempCanvas.getContext('2d', { alpha: true })
   126→        if (!tCtx) throw new Error('2D canvas not supported')
   127→
   128→        // Draw ring on temp canvas: clip → fill shape → destination-out offset shape
   129→        // Faithful to InnerShadowModifier.kt shadowLayer.record:
   130→        //   canvas.clipOutline(outline, clipPath)    // clip to shape FIRST
   131→        //   canvas.drawOutline(outline, paint)        // shadow-colored fill
   132→        //   canvas.translate(offsetX, offsetY)
   133→        //   canvas.drawOutline(outline, ShadowMaskPaint) // BlendMode.Clear
   134→        // The clip ensures the ring is strictly inside the shape boundary,
   135→        // and the destination-out only affects pixels within the clip.
   136→        // Without this clip, subtle AA differences at the shape edge can
   137→        // cause the inner shadow to not "严丝合缝" (not tight-fitting).
   138→        tCtx.save()
   139→        tCtx.scale(SS, SS)
   140→        tCtx.translate(margin, margin)
   141→
   142→        // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   143→        let path: Path2D
   144→        if (useG2) {
   145→          path = continuousCurvatureRoundedRectPath(tCtx, origSizeX, origSizeY, origRadius)
   146→        } else {
   147→          path = new Path2D()
   148→          const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   149→          path.moveTo(r, 0)
   150→          path.lineTo(origSizeX - r, 0)
   151→          path.arcTo(origSizeX, 0, origSizeX, r, r)
   152→          path.lineTo(origSizeX, origSizeY - r)
   153→          path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   154→          path.lineTo(r, origSizeY)
   155→          path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   156→          path.lineTo(0, r)
   157→          path.arcTo(0, 0, r, 0, r)
   158→          path.closePath()
   159→        }
   160→
   161→        // Clip to the shape FIRST (faithful to InnerShadowModifier.kt clipOutline)
   162→        tCtx.clip(path)
   163→
   164→        // Fill the shape with white (creates full interior)
   165→        tCtx.globalCompositeOperation = 'source-over'
   166→        tCtx.fillStyle = 'white'
   167→        tCtx.fill(path)
   168→
   169→        // Draw the OFFSET shape with destination-out (removes offset interior)
   170→        tCtx.globalCompositeOperation = 'destination-out'
   171→        tCtx.save()
   172→        tCtx.translate(offsetXDp, offsetYDp)
   173→        tCtx.fill(path) // same path, shifted by offset
   174→        tCtx.restore()
   175→
   176→        tCtx.globalCompositeOperation = 'source-over'
   177→        tCtx.restore()
   178→
   179→        // --- Blur the ring onto the output canvas (1:1 physical pixels) ---
   180→        // NO ctx.scale here — drawImage and ctx.filter blur must operate in
   181→        // physical-pixel space to avoid SS× size distortion.
   182→        //   blurSigma is in device px (logical); the SS× canvas has SS physical px
   183→        //   per logical px, so the physical blur radius = blurSigma * SS.
   184→        const mCtx = mask.ctx
   185→        mCtx.clearRect(0, 0, canvasW, canvasH)
   186→        if (blurSigma > 0.01) {
   187→          mCtx.filter = `blur(${blurSigma * SS}px)`
   188→        } else {
   189→          mCtx.filter = 'none'
   190→        }
   191→        // 1:1 physical-pixel drawImage: temp(canvasW×canvasH) → mask(canvasW×canvasH)
   192→        mCtx.drawImage(tempCanvas, 0, 0)
   193→        mCtx.filter = 'none'
   194→
   195→        // Upload to GPU texture
   196→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   197→        gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   198→        gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   199→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   200→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   201→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   202→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   203→        mask.ready = true
   204→      }
   205→
   206→      // --- Composite: inner shadow mask × shadowAlpha × shadowColor → scene ---
   207→      gl.enable(gl.BLEND)
   208→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   209→      gl.useProgram(this.innerShadowMaskCompositeProgram)
   210→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   211→      gl.enableVertexAttribArray(this.aPosLocIs)
   212→      gl.vertexAttribPointer(this.aPosLocIs, 2, gl.FLOAT, false, 0, 0)
   213→
   214→      gl.uniform2f(this.uIs['uCanvasSize'], this.canvas.width, this.canvas.height)
   215→      gl.uniform2f(this.uIs['uOffset'], sx * this.dpr, sy * this.dpr)
   216→      gl.uniform2f(this.uIs['uSize'], sw * this.dpr, sh * this.dpr)
   217→      gl.uniform4f(this.uIs['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   218→
   219→      gl.activeTexture(gl.TEXTURE0)
   220→      gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   221→      gl.uniform1i(this.uIs['uInnerShadowMask'], 0)
   222→      // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   223→      // physical canvas is SS× larger but the shader uses 1x coords for UV.
   224→      gl.uniform2f(this.uIs['uMaskOffset'], margin, margin)
   225→      gl.uniform2f(this.uIs['uMaskSize'], mask.w, mask.h)
   226→
   227→      // Shadow color (defaults to black [0,0,0] if not specified)
   228→      const color = shadowCfg.color ?? [0, 0, 0]
   229→      gl.uniform3f(this.uIs['uInnerShadowColor'], color[0], color[1], color[2])
   230→      gl.uniform1f(this.uIs['uInnerShadowAlpha'], shadowAlpha)
   231→
   232→      gl.uniform2f(this.uIs['uOriginalSize'], origSizeX, origSizeY)
   233→      gl.uniform1f(this.uIs['uOriginalCornerRadius'], origRadius)
   234→      gl.uniform2f(this.uIs['uLayerScale'], layerScaleX, layerScaleY)
   235→      gl.uniform1f(this.uIs['uElementRotation'], state.elementRotation)
   236→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   237→    }
   238→
   239→    // Inner shadow (faithful to the original — only ONE black inner shadow,
   240→    // no innerShadow2. LiquidToggle.kt: InnerShadow(radius=4dp*progress, alpha=progress))
   241→    if (el.innerShadow) {
   242→      drawInnerShadowPass(el.innerShadow, 0)
   243→    }
   244→
   245→    // --- Step 2c: Press glow (button + bottom-tab container) ---
   246→    // Faithful to InteractiveHighlight.kt: a flat white Plus-blend overlay
   247→    // + a radial white glow. For buttons, position = finger, intensity = st.pressProgress.
   248→    // For the bottom-tab container, position = indicator center (via the
   249→    // original's position lambda), intensity = toggleState.pressProgress.
   250→    const isContainer = !!el.isBottomTabContainer
   251→    const glowP = isButton ? p : (isContainer ? togglePressProgress : 0)
   252→    if ((isButton && el.isInteractive && st && p > 0.001) || (isContainer && togglePressProgress > 0.001)) {
   253→      // a. Flat white overlay
   254→      gl.useProgram(this.tintProgram)
   255→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   256→      gl.enableVertexAttribArray(this.aPosLocTn)
   257→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   258→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   259→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   260→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   261→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   262→      gl.uniform4f(
   263→        this.uTn['uCornerRadii'],
   264→        radii[0] * this.dpr,
   265→        radii[1] * this.dpr,
   266→        radii[2] * this.dpr,
   267→        radii[3] * this.dpr
   268→      )
   269→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   270→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   271→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   272→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   273→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   274→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.08 * glowP)
   275→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   276→
   277→      // b. Radial highlight at finger position (button) or indicator center (container).
   278→      gl.useProgram(this.highlightProgram)
   279→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   280→      gl.enableVertexAttribArray(this.aPosLocHl)
   281→      gl.vertexAttribPointer(this.aPosLocHl, 2, gl.FLOAT, false, 0, 0)
   282→      gl.blendFunc(gl.ONE, gl.ONE)
   283→      gl.uniform2f(this.uHl['uCanvasSize'], this.canvas.width, this.canvas.height)
   284→      gl.uniform2f(this.uHl['uOffset'], sx * this.dpr, sy * this.dpr)
   285→      gl.uniform2f(this.uHl['uSize'], sw * this.dpr, sh * this.dpr)
   286→      gl.uniform4f(
   287→        this.uHl['uCornerRadii'],
   288→        radii[0] * this.dpr,
   289→        radii[1] * this.dpr,
   290→        radii[2] * this.dpr,
   291→        radii[3] * this.dpr
   292→      )
   293→      gl.uniform2f(this.uHl['uOriginalSize'], origSizeX, origSizeY)
   294→      gl.uniform1f(this.uHl['uOriginalCornerRadius'], origRadius)
   295→      gl.uniform2f(this.uHl['uLayerScale'], layerScaleX, layerScaleY)
   296→      gl.uniform1f(this.uHl['uElementRotation'], state.elementRotation)
   297→      gl.uniform1f(this.uHl['uCornerStyle'], this.cornerStyle)
   298→      gl.uniform4f(this.uHl['uColor'], 1, 1, 1, 0.15 * glowP)
   299→      const minDim = Math.min(sw, sh) * this.dpr
   300→      gl.uniform1f(this.uHl['uRadius'], minDim * 1.5)
   301→      // Position: for buttons, the finger position. For the container, the
   302→      // indicator center — faithful to the original's position lambda which
   303→      // returns (dampedDragAnimation.value + 0.5) * tabWidth (indicator center
   304→      // relative to the container).
   305→      let px: number, py: number
   306→      if (isContainer) {
   307→        const tg = this.toggleStates.get(el.isBottomTabContainer!.groupId)
   308→        // tabW = container ORIGINAL width / tabsCount (not scaled — the position
   309→        // lambda runs in pre-scale local coords). indicator center =
   310→        // (fraction + 0.5) * tabW. Faithful to LiquidBottomTabs.kt:
   311→        //   position = (dampedDragAnimation.value + 0.5) * tabWidth
   312→        const tabsCount = el.isBottomTabContainer!.tabsCount ?? 4
   313→        const tabW = el.rect.w / tabsCount
   314→        const fraction = tg ? tg.fraction : 0
   315→        const indCenterX = (fraction + 0.5) * tabW
   316→        // Map original-local to scaled-local (the shader's uPosition is in
   317→        // scaled-local coords, 0..sw). scale = sw / el.rect.w.
   318→        const scaleToLocal = sw / el.rect.w
   319→        px = Math.max(0, Math.min(sw, indCenterX * scaleToLocal)) * this.dpr
   320→        py = (sh / 2) * this.dpr
   321→      } else {
   322→        px = Math.max(0, Math.min(sw, st!.dragX * state.layerScaleX)) * this.dpr
   323→        py = Math.max(0, Math.min(sh, st!.dragY * state.layerScaleY)) * this.dpr
   324→      }
   325→      gl.uniform2f(this.uHl['uPosition'], px, py)
   326→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   327→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   328→    }
   329→
   330→    // --- Step 2d: Toggle knob white overlay (faithful to LiquidToggle.kt
   331→    // / LiquidSlider.kt onDrawSurface) ---
   332→    if (el.isToggleKnob && togglePressProgress < 0.999) {
   333→      const whiteAlpha = 1.0 * (1 - togglePressProgress)
   334→      gl.useProgram(this.tintProgram)
   335→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   336→      gl.enableVertexAttribArray(this.aPosLocTn)
   337→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   338→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   339→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   340→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   341→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   342→      gl.uniform4f(
   343→        this.uTn['uCornerRadii'],
   344→        radii[0] * this.dpr,
   345→        radii[1] * this.dpr,
   346→        radii[2] * this.dpr,
   347→        radii[3] * this.dpr
   348→      )
   349→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   350→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   351→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   352→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   353→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   354→      // Faithful to LiquidToggle.kt onDrawSurface:
   355→      //   drawRect(Color.White.copy(alpha = 1f - progress))
   356→      // Solid white pebble at rest (alpha=1), fading to transparent when
   357→      // pressed (alpha=0) to reveal the glass refraction beneath.
   358→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, whiteAlpha)
   359→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   360→    }
   361→
   362→    // --- Step 2d2: Bottom tab indicator onDrawSurface (faithful to
   363→    // LiquidBottomTabs.kt indicator):
   364→    //   drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)
   365→    //   drawRect(Color.Black.copy(alpha = 0.03f * progress))
   366→    // First: theme-aware dim overlay fading OUT on press (SrcOver).
   367→    // Second: subtle black tint fading IN on press (SrcOver).
   368→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.dimColor) {
   369→      const dc = el.isBottomTabIndicator.dimColor
   370→      const p = togglePressProgress
   371→      gl.useProgram(this.tintProgram)
   372→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   373→      gl.enableVertexAttribArray(this.aPosLocTn)
   374→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   375→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   376→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   377→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   378→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   379→      gl.uniform4f(this.uTn['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   380→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   381→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   382→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   383→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   384→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   385→      // First overlay: dim color at 0.1 * (1 - progress) — fades out on press.
   386→      gl.uniform4f(this.uTn['uColor'], dc[0], dc[1], dc[2], 0.1 * (1 - p))
   387→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   388→      // Second overlay: black at 0.03 * progress — fades in on press.
   389→      gl.uniform4f(this.uTn['uColor'], 0, 0, 0, 0.03 * p)
   390→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   391→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   392→    }
   393→
   394→    // --- Step 2e: Foreground (label or icon) pass (button only) ---
   395→    if (isButton && (el.label || el.icon)) {
   396→      const fgTex = this.fgTextures.get(el.id)
   397→      if (fgTex) {
   398→        gl.useProgram(this.foregroundProgram)
   399→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   400→        gl.enableVertexAttribArray(this.aPosLocFg)
   401→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   402→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   403→
   404→        gl.activeTexture(gl.TEXTURE0)
   405→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   406→        gl.uniform1i(this.uFg['uTexture'], 0)
   407→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   408→        gl.uniform2f(this.uFg['uOffset'], sx * this.dpr, sy * this.dpr)
   409→        gl.uniform2f(this.uFg['uSize'], sw * this.dpr, sh * this.dpr)
   410→        gl.uniform4f(
   411→          this.uFg['uCornerRadii'],
   412→          radii[0] * this.dpr,
   413→          radii[1] * this.dpr,
   414→          radii[2] * this.dpr,
   415→          radii[3] * this.dpr
   416→        )
   417→        gl.uniform2f(this.uFg['uOriginalSize'], origSizeX, origSizeY)
   418→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], origRadius)
   419→      gl.uniform2f(this.uFg['uLayerScale'], layerScaleX, layerScaleY)
   420→      gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   421→      // Continuous-curvature mask for capsule foregrounds.
   422→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   423→        gl.activeTexture(gl.TEXTURE2)
   424→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   425→        gl.uniform1i(this.uFg['uContinuousSdf'], 2)
   426→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 1.0)
   427→        gl.uniform2f(this.uFg['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   428→        gl.uniform2f(this.uFg['uContinuousSdfElementSize'], state.origW * this.dpr, state.origH * this.dpr)
   429→      } else {
   430→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 0.0)
   431→      }
   432→        gl.uniform1f(this.uFg['uAlpha'], 1.0 - 0.15 * p)
   433→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   434→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   435→      }
   436→    }
   437→
   438→    // --- Step 2f: Rim highlight (Canvas2D stroke mask approach) ---
   439→    // Uses Canvas2D ctx.stroke() (browser-native Skia) to rasterize the G2
   440→    // Bezier path as a stroke mask. This is the SAME method the original
   441→    // uses (Skia drawOutline + paint.Stroke) — the browser's Canvas2D
   442→    // internally calls Skia's SkCanvas::drawPath, which tessellates the
   443→    // Bezier into triangles and rasterizes with hardware coverage AA.
   444→    //
   445→    // Advantages over the SDF approach:
   446→    //   - Exact G2 Bezier shape (not SDF approximation)
   447→    //   - Hardware coverage AA (sub-pixel accurate, no smoothstep needed)
   448→    //   - No SDF computation in shader (just one texture fetch)
   449→    //   - Adaptive tessellation (more samples at high-curvature corners)
   450→    //
   451→    // The stroke mask is cached by exact geometry. Highlight angle/alpha/press
   452→    // progress only affect the composite uniforms, not the mask, so caching is
   453→    // safe and avoids both per-frame rasterization and UV mismatch between a
   454→    // reused backing canvas and the logical mask size.
   455→    if (el.highlight && el.highlight.alpha > 0.001) {
   456→      const rimAlpha = (el.isToggleKnob || el.isBottomTabIndicator) ? elHighlightAlpha : el.highlight.alpha
   457→      const finalAlpha = rimAlpha * state.enterAlpha
   458→      if (finalAlpha > 0.001) {
   459→        // HighlightModifier.kt: ceil(width.toPx().coerceAtMost(minDimension / 2)) * 2
   460→        const widthPx = Math.min(
   461→          el.highlight.widthDp * this.dpr,
   462→          Math.min(origSizeX, origSizeY) * 0.5
   463→        )
   464→        const strokeWidthDevice = Math.max(1, Math.ceil(widthPx) * 2)
   465→        // Highlight data class: blurRadius defaults to width / 2. Honor it in the
   466→        // Canvas2D mask as well (for the default 0.25dp this is sub-pixel, but it
   467→        // keeps non-default highlights faithful too).
   468→        const blurPx = Math.max(0, (el.highlight.blurRadiusDp ?? el.highlight.widthDp / 2) * this.dpr)
   469→
   470→        // --- Generate stroke mask via Canvas2D (browser-native Skia) ---
   471→        // Element-local coordinates: (0,0) = element top-left in device px.
   472→        // Margin for stroke width + AA + blur spread.
   473→        const strokeMargin = Math.ceil(strokeWidthDevice) + 4  // logical margin
   474→        // Logical mask size (1x device px) — used for shader UV mapping
   475→        const maskW = Math.max(1, Math.ceil(origSizeX + 2 * strokeMargin))
   476→        const maskH = Math.max(1, Math.ceil(origSizeY + 2 * strokeMargin))
   477→        // Supersample for sharper stroke/blur rasterization.
   478→        // Cap SS so total mask DPR (this.dpr × SS) ≤ device's native DPR —
   479→        // no point rendering pixels beyond what the screen can display.
   480→        const deviceDpr = window.devicePixelRatio || 1
   481→        const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
   482→        const canvasW = maskW * SS
   483→        const canvasH = maskH * SS
   484→        const useG2 = !!el.useContinuousSdf
   485→        const maskKey = [
   486→          useG2 ? 'g2' : 'rr',
   487→          origSizeX.toFixed(3),
   488→          origSizeY.toFixed(3),
   489→          origRadius.toFixed(3),
   490→          strokeWidthDevice,
   491→          blurPx.toFixed(3),
   492→          strokeMargin,
   493→          maskW,
   494→          maskH,
   495→          `ss${SS}`,  // cache key includes supersample factor
   496→        ].join(':')
   497→
   498→        let mask = this.strokeMaskCache.get(maskKey)
   499→        if (!mask) {
   500→          const canvas = document.createElement('canvas')
   501→          canvas.width = canvasW  // 2x supersampled physical size
   502→          canvas.height = canvasH
   503→          const ctx = canvas.getContext('2d', { alpha: true })
   504→          if (!ctx) throw new Error('2D canvas not supported')
   505→          const tex = gl.createTexture()
   506→          if (!tex) throw new Error('WebGL texture allocation failed')
   507→          // w/h store the LOGICAL (1x) size for shader UV mapping;
   508→          // the physical canvas is SS times larger.
   509→          mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   510→          this.strokeMaskCache.set(maskKey, mask)
   511→
   512→          // Keep the cache bounded. 32 entries is far above the catalog's
   513→          // simultaneous highlight geometries and avoids unbounded growth on
   514→          // highly dynamic pages.
   515→          if (this.strokeMaskCache.size > 32) {
   516→            const oldestKey = this.strokeMaskCache.keys().next().value as string | undefined
   517→            if (oldestKey && oldestKey !== maskKey) {
   518→              const oldest = this.strokeMaskCache.get(oldestKey)
   519→              if (oldest) gl.deleteTexture(oldest.tex)
   520→              this.strokeMaskCache.delete(oldestKey)
   521→            }
   522→          }
   523→        }
   524→
   525→        if (!mask.ready) {
   526→          const smCtx = mask.ctx
   527→          smCtx.clearRect(0, 0, canvasW, canvasH)
   528→          smCtx.save()
   529→          // Scale up for 2x supersampling: draw in logical (1x) coordinates
   530→          // while the physical canvas is 2x. This gives sharper stroke + blur.
   531→          smCtx.scale(SS, SS)
   532→          smCtx.translate(strokeMargin, strokeMargin)
   533→
   534→          // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   535→          let path: Path2D
   536→          if (useG2) {
   537→            path = continuousCurvatureRoundedRectPath(smCtx, origSizeX, origSizeY, origRadius)
   538→          } else {
   539→            path = new Path2D()
   540→            const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   541→            path.moveTo(r, 0)
   542→            path.lineTo(origSizeX - r, 0)
   543→            path.arcTo(origSizeX, 0, origSizeX, r, r)
   544→            path.lineTo(origSizeX, origSizeY - r)
   545→            path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   546→            path.lineTo(r, origSizeY)
   547→            path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   548→            path.lineTo(0, r)
   549→            path.arcTo(0, 0, r, 0, r)
   550→            path.closePath()
   551→          }
   552→
   553→          // Stroke — browser-native Skia stroke rasterization.
   554→          // Faithful to HighlightModifier.kt: clipOutline(outline) BEFORE drawOutline,
   555→          // so only the INSIDE half of the centered stroke remains. Without this
   556→          // clip, Canvas2D keeps the outer half too and the rim highlight leaks
   557→          // outside the glass / looks twice as thick.
   558→          smCtx.clip(path)
   559→          smCtx.lineWidth = strokeWidthDevice
   560→          smCtx.strokeStyle = 'rgba(255,255,255,1)'
   561→          smCtx.lineJoin = 'round'
   562→          smCtx.lineCap = 'round'
   563→          // Approximate Skia BlurMaskFilter on the stroke paint. The clip above
   564→          // keeps the result inside the outline; the blur only softens that
   565→          // inside stroke band, like the original HighlightModifier layer.
   566→          smCtx.filter = blurPx > 0.01 ? `blur(${blurPx}px)` : 'none'
   567→          smCtx.stroke(path)
   568→          smCtx.filter = 'none'
   569→          smCtx.restore()
   570→
   571→          // Upload to GPU texture. Keep the same top-left UV convention as the
   572→          // foreground pass: with UNPACK_FLIP_Y_WEBGL=false, local y=0 samples
   573→          // the top row of the mask.
   574→          gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   575→          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   576→          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   577→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   578→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   579→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   580→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   581→          mask.ready = true
   582→        }
   583→
   584→        // --- Composite: stroke mask × intensity × color → scene FBO ---
   585→        gl.enable(gl.BLEND)
   586→        if (el.highlight.mode === 1) {
   587→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   588→        } else {
   589→          gl.blendFunc(gl.ONE, gl.ONE)
   590→        }
   591→        gl.useProgram(this.strokeMaskCompositeProgram)
   592→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   593→        gl.enableVertexAttribArray(this.aPosLocSm)
   594→        gl.vertexAttribPointer(this.aPosLocSm, 2, gl.FLOAT, false, 0, 0)
   595→        gl.uniform2f(this.uSm['uCanvasSize'], this.canvas.width, this.canvas.height)
   596→        gl.uniform2f(this.uSm['uOffset'], sx * this.dpr, sy * this.dpr)
   597→        gl.uniform2f(this.uSm['uSize'], sw * this.dpr, sh * this.dpr)
   598→        gl.uniform4f(this.uSm['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   599→        gl.activeTexture(gl.TEXTURE0)
   600→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   601→        gl.uniform1i(this.uSm['uStrokeMask'], 0)
   602→        // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   603→        // physical canvas is SS× larger but the shader uses 1x coords for UV:
   604→        //   UV = (localCoord + margin) / logicalMaskSize
   605→        // LINEAR filtering on the SS× texture gives supersampled quality.
   606→        gl.uniform2f(this.uSm['uMaskOffset'], strokeMargin, strokeMargin)
   607→        gl.uniform2f(this.uSm['uMaskSize'], mask.w, mask.h)
   608→        gl.uniform4f(this.uSm['uHighlightColor'], el.highlight.color[0], el.highlight.color[1], el.highlight.color[2], 1.0)
   609→        gl.uniform1f(this.uSm['uHighlightAngle'], el.useGravityAngle ? this.gravityAngle : el.highlight.angle)
   610→        gl.uniform1f(this.uSm['uHighlightFalloff'], el.highlight.falloff)
   611→        gl.uniform1f(this.uSm['uHighlightAlpha'], finalAlpha)
   612→        gl.uniform1f(this.uSm['uHighlightMode'], el.highlight.mode)
   613→        gl.uniform2f(this.uSm['uOriginalSize'], origSizeX, origSizeY)
   614→        gl.uniform1f(this.uSm['uOriginalCornerRadius'], origRadius)
   615→        gl.uniform2f(this.uSm['uLayerScale'], layerScaleX, layerScaleY)
   616→        gl.uniform1f(this.uSm['uElementRotation'], state.elementRotation)
   617→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   618→
   619→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   620→      }
   621→    }
   622→
   623→    // --- Step 2g: 内层背景板 rim highlight (指示器 only) ---
   624→    // The 内层背景板 (hidden Row's 56dp glass inside the 指示器) has its own
   625→    // Highlight.Default.copy(alpha=progress). This is drawn INSIDE the
   626→    // 指示器's element shader (sampleIndicatorBackdrop) so it's clipped by
   627→    // the 指示器's capsule SDF — no separate pass needed here.
   628→  },
   629→}

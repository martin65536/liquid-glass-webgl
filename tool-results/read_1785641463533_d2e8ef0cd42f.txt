    18→/* ------------------------------------------------------------------ *
    19→ * Shared LiquidSlider factory — used by both the Slider page and the
    20→ * Glass Playground. Creates track + fill + knob + interactions.
    21→ * ------------------------------------------------------------------ */
    22→// Unified per-group drag state (survives re-renders during liveUpdate /
    23→// gravityAngle changes). Used by ALL drag-based controls.
    24→const dragStates = new Map<string, { fraction: number; x: number; didDrag: boolean }>()
    25→
    26→/* ------------------------------------------------------------------ *
    27→ * makeDragInteractions — the ONE shared drag-interaction factory.
    28→ *
    29→ * Used by: sliders (continuous + stepped), toggles, bottom tabs.
    30→ * All share the same gesture pattern:
    31→ *   onDragStart → read current visual fraction → beginDrag
    32→ *   onDrag → relative drag (knob follows finger delta with spring lag)
    33→ *   onDragEnd → endDrag → snap → setTarget
    34→ *   onTap → jump to tapped fraction → snap → setTarget
    35→ *
    36→ * The control-specific behavior is injected via the `opts` object:
    37→ *   - getFraction: read current visual fraction (getToggleFraction / getTabFraction)
    38→ *   - beginDrag: start the drag (beginToggleDrag / beginTabDrag)
    39→ *   - drag: update during drag (dragToggle / dragTab)
    40→ *   - endDrag: release + return final fraction (endSliderDrag / endToggleDrag / endTabDrag)
    41→ *   - setTarget: programmatic target set (setToggleTarget / setTabSelected)
    42→ *   - snap: snap function (fraction → snapped fraction), or null = no snap
    43→ *   - liveUpdate: push fraction to onValueChange during drag
    44→ *   - onTapJump: if false, tap does nothing (e.g. toggle = toggle on tap, not jump)
    45→ * ------------------------------------------------------------------ */
    46→interface DragInteractionsOpts {
    47→  groupId: string
    48→  trackX: number
    49→  dragW: number
    50→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null
    51→  onValueChange: (fraction: number) => void
    52→  /** Optional lightweight callback fired on every drag move (regardless of
    53→   *  liveUpdate). Use this to update a display-only label WITHOUT triggering
    54→   *  a catalog rebuild (e.g. settings slider labels). onValueChange is still
    55→   *  fired on dragEnd (or every move if liveUpdate=true) for the real state. */
    56→  onLiveValue?: (fraction: number) => void
    57→  // Control-specific renderer calls:
    58→  getFraction: (r: LiquidGlassRenderer, groupId: string) => number
    59→  beginDrag: (r: LiquidGlassRenderer, groupId: string, fraction: number, count?: number) => void
    60→  drag: (r: LiquidGlassRenderer, groupId: string, startFraction: number, currentX: number, startX: number, dragW: number, count?: number) => void
    61→  endDrag: (r: LiquidGlassRenderer, groupId: string, count?: number) => number
    62→  setTarget: (r: LiquidGlassRenderer, groupId: string, fraction: number, count?: number) => void
    63→  count?: number // tabsCount for tabs (passed to begin/drag/end/setTarget)
    64→  snap?: (f: number) => number // snap function (null/undefined = no snap)
    65→  liveUpdate?: boolean
    66→  onTapJump?: boolean // default true; false = tap does nothing (toggle uses its own onTap)
    67→  didDragThreshold?: number // px of movement before didDrag=true (default 3)
    68→}
    69→
    70→export function makeDragInteractions(opts: DragInteractionsOpts): ElementInteraction {
    71→  const {
    72→    groupId, trackX, dragW, rendererRef, onValueChange, onLiveValue,
    73→    getFraction, beginDrag, drag, endDrag, setTarget,
    74→    count, snap, liveUpdate = false, onTapJump = true, didDragThreshold = 3,
    75→  } = opts
    76→
    77→  if (!dragStates.has(groupId)) dragStates.set(groupId, { fraction: 0, x: 0, didDrag: false })
    78→  const ds = dragStates.get(groupId)!
    79→
    80→  const fracFromPos = (px: number) => Math.max(0, Math.min(1, (px - trackX) / dragW))
    81→  const applySnap = (f: number) => (snap ? snap(f) : f)
    82→
    83→  return {
    84→    onTap: (pos) => {
    85→      if (!onTapJump) return
    86→      const f = applySnap(fracFromPos(pos.x))
    87→      const r = rendererRef?.current
    88→      if (r) setTarget(r, groupId, f, count)
    89→      onValueChange(f)
    90→    },
    91→    onDragStart: (pos) => {
    92→      const r = rendererRef?.current
    93→      if (!r) return
    94→      draggingGroups.add(groupId)
    95→      ds.fraction = getFraction(r, groupId)
    96→      ds.x = pos.x
    97→      ds.didDrag = false
    98→      beginDrag(r, groupId, ds.fraction, count)
    99→    },
   100→    onDrag: (pos) => {
   101→      const r = rendererRef?.current
   102→      if (!r) return
   103→      if (Math.abs(pos.x - ds.x) > didDragThreshold) ds.didDrag = true
   104→      drag(r, groupId, ds.fraction, pos.x, ds.x, dragW, count)
   105→      const f = getFraction(r, groupId)
   106→      if (onLiveValue) onLiveValue(f)
   107→      if (liveUpdate) {
   108→        onValueChange(f)
   109→      }
   110→    },
   111→    onDragEnd: () => {
   112→      const r = rendererRef?.current
   113→      if (!r) return
   114→      const rawF = endDrag(r, groupId, count)
   115→      // Keep draggingGroups flag set DURING onValueChange (which triggers
   116→      // setState → tabTargets/toggleTargets effect). The effect checks
   117→      // draggingGroups and skips setTabSelected/setToggleTarget, preventing
   118→      // it from zeroing velocity and fighting the spring. Delete AFTER
   119→      // onValueChange so the next render's effect can sync.
   120→      const snappedF = applySnap(rawF)
   121→      // Only call setTarget if the endDrag didn't already snap.
   122→      // For toggle/slider: endToggleDrag/endSliderDrag DON'T snap (they
   123→      // return the raw fraction), so setTarget is needed.
   124→      // For tabs: endTabDrag DOES snap (sets targetFraction to rounded
   125→      // index), so setTarget would be redundant and would zero velocity
   126→      // (fighting the spring). Skip setTarget for tabs (count != undefined
   127→      // && endDrag already snapped).
   128→      if (snap && count == null) setTarget(r, groupId, snappedF, count)
   129→      onValueChange(snappedF)
   130→      draggingGroups.delete(groupId)
   131→    },
   132→  }
   133→}

   136→ * Renderer method bindings — pre-wired adapters for each control type.
   137→ * These eliminate the boilerplate of passing lambdas every time.

   140→// Slider/Settings: getToggleFraction, beginToggleDrag, dragToggle,
   141→// endSliderDrag, setToggleTarget.
   142→export const sliderDragBindings = {
   143→  getFraction: (r: LiquidGlassRenderer, id: string) => r.getToggleFraction(id),
   144→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number) => r.beginToggleDrag(id, f),
   145→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number) =>
   146→    r.dragToggle(id, sf, cx, sx, dw),
   147→  endDrag: (r: LiquidGlassRenderer, id: string) => r.endSliderDrag(id),
   148→  setTarget: (r: LiquidGlassRenderer, id: string, f: number) => r.setToggleTarget(id, f),
   149→}
   150→
   151→// Toggle: getToggleTarget (not fraction!), beginToggleDrag, dragToggle,
   152→// endToggleDrag, setToggleTarget. Tap = toggle (no jump).
   153→export const toggleDragBindings = {
   154→  getFraction: (r: LiquidGlassRenderer, id: string) => r.getToggleTarget(id),
   155→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number) => r.beginToggleDrag(id, f),
   156→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number) =>
   157→    r.dragToggle(id, sf, cx, sx, dw),
   158→  endDrag: (r: LiquidGlassRenderer, id: string) => r.endToggleDrag(id),
   159→  setTarget: (r: LiquidGlassRenderer, id: string, f: number) => r.setToggleTarget(id, f),
   160→}
   161→
   162→// Bottom tabs: fraction (0..1) ↔ index (0..count-1) conversion at the
   163→// binding layer. The renderer works in INDEX space (0..count-1), but
   164→// makeDragInteractions works in FRACTION space (0..1). The bindings
   165→// convert: getFraction divides by (count-1), setTarget/beginDrag/drag
   166→// multiply by (count-1).
   167→export const tabDragBindings = {
   168→  getFraction: (r: LiquidGlassRenderer, id: string, count?: number) => {
   169→    const c = count ?? 3
   170→    // Use TARGET (not animated) fraction — faithful to the original which
   171→    // uses `targetValue` in onDrag, not the animated value. Starting a drag
   172→    // from the animated value (mid-spring) causes drift because the spring's
   173→    // residual motion adds to the finger delta.
   174→    return r.getTabTarget(id) / Math.max(1, c - 1)
   175→  },
   176→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number, count?: number) => {
   177→    const c = count ?? 3
   178→    r.beginTabDrag(id, f * (c - 1), c)
   179→  },
   180→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number, count?: number) => {
   181→    const c = count ?? 3
   182→    r.dragTab(id, sf * (c - 1), cx, sx, dw, c)
   183→  },
   184→  endDrag: (r: LiquidGlassRenderer, id: string, count?: number) => {
   185→    const c = count ?? 3
   186→    // endTabDrag returns an INDEX (0..c-1). Convert to fraction (0..1).
   187→    return r.endTabDrag(id, c) / Math.max(1, c - 1)
   188→  },
   189→  setTarget: (r: LiquidGlassRenderer, id: string, f: number, count?: number) => {
   190→    const c = count ?? 3
   191→    r.setTabSelected(id, Math.max(0, Math.min(c - 1, Math.round(f * (c - 1)))), c)
   192→  },
   193→}

   195→export function makeLiquidSlider(
   196→  idPrefix: string,
   197→  trackX: number,
   198→  trackY: number,
   199→  trackW: number,
   200→  groupId: string,
   201→  trackColor: [number, number, number, number],
   202→  accentColor: [number, number, number],
   203→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null,
   204→  onValueChange: (fraction: number) => void,
   205→  scroll = true,
   206→  liveUpdate = false,
   207→  initFraction = 0,
   208→  snap?: (f: number) => number,
   209→  onLiveValue?: (fraction: number) => void,
   210→): { elements: GlassElementConfig[]; interactions: Record<string, ElementInteraction> } {
   211→  const elements: GlassElementConfig[] = []
   212→  const interactions: Record<string, ElementInteraction> = {}
   213→  const dragW = trackW - SLIDER_KNOB_W / 2
   214→  const knobBaseX = trackX - SLIDER_KNOB_W / 4
   215→  const knobY = trackY + (SLIDER_TRACK_H - SLIDER_KNOB_H) / 2
   216→  // Knob rect.x is ALWAYS at fraction=0 (knobBaseX). The renderer's
   217→  // isToggleKnob.dragWidth drives the x offset via spring animation.
   218→  // Setting rect.x to initFraction would cause 2x displacement
   219→  // (rect.x + spring offset = initFraction*dragW + fraction*dragW).
   220→  const knobX = knobBaseX
   221→
   222→  // Track
   223→  const trackEl = makePlainRect(`${idPrefix}-track`, { x: trackX, y: trackY, w: trackW, h: SLIDER_TRACK_H }, trackColor, SLIDER_TRACK_H / 2)
   224→  trackEl.hitRect = { x: trackX, y: trackY + (SLIDER_TRACK_H - SLIDER_HIT_H) / 2, w: trackW, h: SLIDER_HIT_H }
   225→  trackEl.scroll = scroll
   226→  elements.push(trackEl)
   227→
   228→  // Fill — width driven by renderer via isSliderFill
   229→  const fillW = Math.max(SLIDER_TRACK_H, initFraction * trackW)
   230→  const fillEl = makePlainRect(`${idPrefix}-fill`, { x: trackX, y: trackY, w: fillW, h: SLIDER_TRACK_H }, [...accentColor, 1], SLIDER_TRACK_H / 2)
   231→  fillEl.isSliderFill = { groupId, trackX, trackW, knobW: SLIDER_KNOB_W, minW: 0 }
   232→  fillEl.scroll = scroll
   233→  elements.push(fillEl)
   234→
   235→  // Knob — frosted white at rest, glass when pressed (no highlight)
   236→  const knobEl = makeGlassShape(
   237→    `${idPrefix}-knob`,
   238→    { x: knobX, y: knobY, w: SLIDER_KNOB_W, h: SLIDER_KNOB_H },
   239→    {
   240→      cornerRadius: SLIDER_KNOB_H / 2,
   241→      refractionHeight: 10 * DP,
   242→      refractionAmount: -14 * DP,
   243→      blurRadius: 8 * DP,
   244→      saturation: 1.0,
   245→      surfaceColor: [0, 0, 0, 0],
   246→      highlight: { mode: 1, color: [1, 1, 1], angle: Math.PI / 4, falloff: 1.0, alpha: 1.0, widthDp: 0.5 / 1.5, blurRadiusDp: 0.25 / 1.5 },
   247→      outerShadow: { radius: 4 * DP, alpha: 0.05, offsetX: 0, offsetY: (4 / 6) * DP, color: [0, 0, 0] },
   248→      innerShadow: { radius: 4 * DP, alpha: 0.3, offsetX: 0, offsetY: 4 * DP }, // 0.15 was too faint; 0.3 matches visual intent
   249→      chromaticAberration: true,
   250→    },
   251→    scroll
   252→  )
   253→  knobEl.isToggleKnob = { groupId, dragWidth: dragW, velocityDivisor: 10 }
   254→  knobEl.hitRect = { x: knobX, y: knobY + (SLIDER_KNOB_H - SLIDER_HIT_H) / 2, w: SLIDER_KNOB_W, h: SLIDER_HIT_H }
   255→  elements.push(knobEl)
   256→
   257→  // Interactions — unified drag pattern via makeDragInteractions.
   258→  const interact = makeDragInteractions({
   259→    groupId, trackX, dragW, rendererRef, onValueChange, onLiveValue,
   260→    ...sliderDragBindings,
   261→    snap,
   262→    liveUpdate,
   263→  })
   264→  interactions[`${idPrefix}-track`] = interact
   265→  interactions[`${idPrefix}-knob`] = interact
   266→
   267→  return { elements, interactions }
   268→}

   401→/**
   402→ * Tab drag interactions — faithful to LiquidBottomTabs.kt's
   403→ * dampedDragAnimation gesture handling.
   404→ *
   405→ * The original uses DampedDragAnimation with:
   406→ *   valueRange = 0..(tabsCount-1)
   407→ *   pressedScale = 78/56
   408→ *   onDrag → updateValue(targetValue + dragAmount.x / tabWidth)
   409→ *   onDragStopped → snap to nearest tab, animateToValue
   410→ *
   411→ * We reuse the renderer's toggle-group state (which supports custom
   412→ * pressedScale via ensureToggleState) through the tab-specific API:
   413→ *   beginTabDrag / dragTab / endTabDrag.
   414→ */
   415→export function makeTabDragInteractions(
   416→  groupId: string,
   417→  tabWidth: number,
   418→  tabsCount: number,
   419→  onSelect: (i: number) => void,
   420→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null
   421→): ElementInteraction {
   422→  // Module-level drag state (survives catalog rebuilds).
   423→  if (!dragStates.has(groupId)) dragStates.set(groupId, { fraction: 0, x: 0, didDrag: false })
   424→  const ds = dragStates.get(groupId)!
   425→
   426→  return {
   427→    onTap: () => {
   428→      // Tab taps handled by tab-text interactions, not here.
   429→    },
   430→    onDragStart: (pos) => {
   431→      const r = rendererRef?.current
   432→      if (!r) return
   433→      draggingGroups.add(groupId)
   434→      // Use TARGET index (not animated) — faithful to original's targetValue.
   435→      ds.fraction = r.getTabTarget(groupId)
   436→      ds.x = pos.x
   437→      ds.didDrag = false
   438→      r.beginTabDrag(groupId, ds.fraction, tabsCount)
   439→    },
   440→    onDrag: (pos) => {
   441→      const r = rendererRef?.current
   442→      if (!r) return
   443→      if (Math.abs(pos.x - ds.x) > 3) ds.didDrag = true
   444→      // Relative drag: indicator follows finger delta from drag start.
   445→      // startTabIndex + (currentX - startX) / tabWidth, clamped to [0, count-1].
   446→      r.dragTab(groupId, ds.fraction, pos.x, ds.x, tabWidth, tabsCount)
   447→    },
   448→    onDragEnd: () => {
   449→      const r = rendererRef?.current
   450→      if (!r) return
   451→      // endTabDrag snaps to nearest integer, sets targetFraction, returns index.
   452→      const finalIndex = r.endTabDrag(groupId, tabsCount)
   453→      // Call onSelect AFTER endTabDrag (so targetFraction is already set).
   454→      // Keep draggingGroups set during onSelect to prevent tabTargets effect
   455→      // from calling setTabSelected (which would zero velocity).
   456→      if (ds.didDrag) {
   457→        onSelect(finalIndex)
   458→      }
   459→      draggingGroups.delete(groupId)
   460→    },
   461→  }
   462→}

   464→export function makeGlassShape(
   465→  id: string,
   466→  rect: { x: number; y: number; w: number; h: number },
   467→  opts: {
   468→    cornerRadius?: number
   469→    refractionHeight?: number
   470→    refractionAmount?: number
   471→    blurRadius?: number
   472→    saturation?: number
   473→    brightness?: number
   474→    contrast?: number
   475→    surfaceColor?: [number, number, number, number]
   476→    highlight?: GlassHighlight | null
   477→    outerShadow?: typeof DEFAULT_SHADOW | null
   478→    innerShadow?: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] } | null
   479→    depthEffect?: boolean
   480→    chromaticAberration?: boolean
   481→  } = {},
   482→  scroll = true
   483→): GlassElementConfig {
   484→  return {
   485→    id,
   486→    kind: 'glass-shape',
   487→    rect,
   488→    cornerRadius: opts.cornerRadius ?? rect.h / 2,
   489→    refractionHeight: opts.refractionHeight ?? 12 * DP,
   490→    refractionAmount: opts.refractionAmount ?? -24 * DP,
   491→    depthEffect: opts.depthEffect ?? false,
   492→    chromaticAberration: opts.chromaticAberration ?? false,
   493→    blurRadius: opts.blurRadius ?? 2 * DP,
   494→    saturation: opts.saturation ?? 1.5,
   495→    brightness: opts.brightness ?? 0,
   496→    contrast: opts.contrast ?? 1,
   497→    tintColor: [0, 0, 0, 0],

   515→/* ------------------------------------------------------------------ *
   516→ * Settings toggle — a compact toggle switch for the settings page.
   517→ * Creates a label text + toggle track + toggle knob, arranged in a row.
   518→ * Uses the same Liquid Toggle anatomy as build-toggle.ts but simplified
   519→ * for settings (no CombinedBackdrop, no drag, just tap to flip).
   520→ *
   521→ * Layout: label on the left, toggle on the right, within the given row.
   522→ * Returns elements and interactions to be merged into the caller's arrays.
   523→ * ------------------------------------------------------------------ */
   524→export function makeSettingsToggle(
   525→  id: string,
   526→  rowRect: { x: number; y: number; w: number; h: number },
   527→  label: string,
   528→  isOn: boolean,
   529→  onToggle: () => void,
   530→  palette: ThemePalette,
   531→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null,
   532→  scroll = true,
   533→): { elements: GlassElementConfig[]; interactions: Record<string, ElementInteraction> } {
   534→  const elements: GlassElementConfig[] = []
   535→  const interactions: Record<string, ElementInteraction> = {}
   536→
   537→  // Toggle dimensions (same as LiquidToggle.kt)
   538→  const TOGGLE_W = 64 * DP
   539→  const TOGGLE_H = 28 * DP
   540→  const TOGGLE_KNOB_W = 40 * DP
   541→  const TOGGLE_KNOB_H = 24 * DP
   542→  const TOGGLE_DRAG = 20 * DP
   543→  const TOGGLE_PADDING = 2 * DP
   544→
   545→  // Layout: label on the left, toggle on the right
   546→  const labelW = rowRect.w - TOGGLE_W - 8 * DP // 8dp gap between label and toggle
   547→  const trackX = rowRect.x + rowRect.w - TOGGLE_W
   548→  const trackY = rowRect.y + (rowRect.h - TOGGLE_H) / 2
   549→  const knobX = trackX + TOGGLE_PADDING
   550→  const knobY = trackY + (TOGGLE_H - TOGGLE_KNOB_H) / 2
   551→
   552→  // Label
   553→  const labelColor = palette.backIconColor
   554→  elements.push(
   555→    makeText(
   556→      `${id}-label`,
   557→      { x: rowRect.x, y: rowRect.y, w: labelW, h: rowRect.h },
   558→      label,
   559→      { color: labelColor, fontSizePx: 15, fontWeight: 400, align: 'left', paddingPx: 0, halo: palette.homeTextHalo }
   560→    )
   561→  )
   562→
   563→  // Track
   564→  const trackColorOff = palette.toggleTrackOff
   565→  const accentColor = palette.toggleAccent
   566→  const trackEl = makePlainRect(
   567→    `${id}-track`,
   568→    { x: trackX, y: trackY, w: TOGGLE_W, h: TOGGLE_H },
   569→    trackColorOff,
   570→    TOGGLE_H / 2,
   571→    scroll,
   572→  )
   573→  trackEl.isToggleTrack = {
   574→    groupId: id,
   575→    offColor: trackColorOff,
   576→    onColor: [...accentColor, 1] as [number, number, number, number],
   577→  }
   578→  elements.push(trackEl)
   579→
   580→  // Knob — same glass effects as the full LiquidToggle knob
   581→  const KNOB_HIGHLIGHT: GlassHighlight = {
   582→    mode: 1,
   583→    color: [1, 1, 1],
   584→    angle: Math.PI / 4,
   585→    falloff: 1.0,
   586→    alpha: 1.0,
   587→    widthDp: 0.5 / 1.5,
   588→    blurRadiusDp: 0.25 / 1.5,
   589→  }
   590→  const knobEl = makeGlassShape(
   591→    `${id}-knob`,
   592→    { x: knobX, y: knobY, w: TOGGLE_KNOB_W, h: TOGGLE_KNOB_H },
   593→    {
   594→      cornerRadius: TOGGLE_KNOB_H / 2,
   595→      refractionHeight: 5 * DP,
   596→      refractionAmount: -10 * DP,
   597→      blurRadius: 8 * DP,
   598→      saturation: 1.0,
   599→      surfaceColor: [0, 0, 0, 0],
   600→      highlight: KNOB_HIGHLIGHT,
   601→      outerShadow: { radius: 4 * DP, alpha: 0.05, offsetX: 0, offsetY: (4 / 6) * DP, color: [0, 0, 0] },
   602→      innerShadow: { radius: 4 * DP, alpha: 0.3, offsetX: 0, offsetY: 4 * DP },
   603→      chromaticAberration: true,
   604→    },
   605→    scroll,
   606→  )
   607→  knobEl.isToggleKnob = {
   608→    groupId: id,
   609→    dragWidth: TOGGLE_DRAG,
   610→    trackColorOff,
   611→    trackColorOn: [...accentColor, 1] as [number, number, number, number],
   612→    trackW: TOGGLE_W,
   613→    trackH: TOGGLE_H,
   614→    trackOriginalX: trackX,
   615→    trackOriginalY: trackY,
   616→    // Settings page has solid background — use the palette's solid backdrop color
   617→    solidBackdropColor: palette.toggleCardBg,
   618→  }
   619→  elements.push(knobEl)
   620→
   621→  // Interaction — tap to flip, drag to slide
   622→  const toggleInteract = makeDragInteractions({
   623→    groupId: id, trackX: 0, dragW: TOGGLE_DRAG, rendererRef,
   624→    onValueChange: (f) => {
   625→      const finalOn = f >= 0.5
   626→      onToggle()
   627→    },
   628→    ...toggleDragBindings,
   629→    snap: (f) => (f >= 0.5 ? 1 : 0),
   630→    onTapJump: false,
   631→  })
   632→  // Override onTap: toggle flip
   633→  toggleInteract.onTap = () => onToggle()
   634→
   635→  interactions[`${id}-track`] = toggleInteract
   636→  interactions[`${id}-knob`] = toggleInteract
   637→
   638→  return { elements, interactions }
   639→}

   751→/* ------------------------------------------------------------------ *
   752→ * applyVerticalCenter — offsets all element y positions (except the
   753→ * back button, which stays top-left) so the content is vertically
   754→ * centered within the viewport. Mirrors BackdropDemoScaffold's
   755→ * `Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center)`.
   756→ *
   757→ * Returns the new contentHeight (= H if centering applied, since the
   758→ * content now spans the full viewport visually).
   759→ * ------------------------------------------------------------------ */
   760→export function applyVerticalCenter(
   761→  elements: GlassElementConfig[],
   762→  contentTop: number,
   763→  contentHeight: number,
   764→  H: number
   765→): number {
   766→  const contentSize = contentHeight - contentTop
   767→  if (contentSize >= H) return contentHeight
   768→  const yOffset = Math.max(0, (H - contentSize) / 2 - contentTop)
   769→  if (yOffset <= 0) return contentHeight
   770→  for (const el of elements) {
   771→    // Back button, theme button, and full-screen overlays (scroll=false) stay
   772→    // at their fixed positions (not shifted by vertical centering).
   773→    if (el.id === '__back__' || el.id === '__theme__') continue
   774→    if (el.scroll === false && el.id !== '__pickimage__') continue
   775→    el.rect = { ...el.rect, y: el.rect.y + yOffset }
   776→    // Shift hitRect too (if set) so expanded touch targets follow the element.
   777→    if (el.hitRect) {
   778→      el.hitRect = { ...el.hitRect, y: el.hitRect.y + yOffset }
   779→    }
   780→    // Faithful fix: toggle knobs store the TRACK's original screen
   781→    // position separately in `isToggleKnob.trackOriginalY` (used by the
   782→    // renderer to compute the scaled track rect inside the knob's
   783→    // CombinedBackdrop). Since the track element's rect.y was just
   784→    // shifted by yOffset, we must shift trackOriginalY by the same
   785→    // amount — otherwise the scaled track rect would be at the wrong Y
   786→    // (off by yOffset * (1 - trackScaleY)), causing "no track visible
   787→    // inside the knob" after vertical centering.
   788→    if (el.isToggleKnob && el.isToggleKnob.trackOriginalY != null) {
   789→      el.isToggleKnob.trackOriginalY += yOffset
   790→    }
   791→    // Bottom tab indicator stores the CONTAINER rect separately for its
   792→    // CombinedBackdrop (the inset capsule SDF covers the container area).
   793→    // Shift it by the same yOffset so the SDF stays aligned.
   794→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.containerRect) {
   795→      el.isBottomTabIndicator.containerRect = {
   796→        ...el.isBottomTabIndicator.containerRect,
   797→        y: el.isBottomTabIndicator.containerRect.y + yOffset,
   798→      }
   799→    }
   800→    // Bottom tab content stores the CONTAINER center (scale origin) separately.
   801→    // Shift it by the same yOffset so the scale pivot stays aligned with the
   802→    // actual container position after vertical centering.
   803→    if (el.isBottomTabContent && el.isBottomTabContent.containerCenterY != null) {
   804→      el.isBottomTabContent.containerCenterY += yOffset
   805→    }
   806→    // Bottom tab indicator also scales around the container center — shift
   807→    // its pivot too.
   808→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.containerCenterY != null) {
   809→      el.isBottomTabIndicator.containerCenterY += yOffset
   810→    }
   811→    // Shift tab content rects (for blue tint mask) by yOffset too.
   812→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.tabContentRects) {
   813→      el.isBottomTabIndicator.tabContentRects = el.isBottomTabIndicator.tabContentRects.map(r => ({
   814→        ...r, y: r.y + yOffset,
   815→      }))
   816→    }
   817→  }
   818→  return contentHeight + yOffset
   819→}

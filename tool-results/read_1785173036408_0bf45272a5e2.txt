   427→   427→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   428→   428→  ensureBlurPrograms(tapCount: number): void {
   429→   429→    if (this.blurPrograms.has(tapCount)) return
   430→   430→    const gl = this.gl
   431→   431→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   432→   432→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   433→   433→    const mk = (fs: WebGLShader) => {
   434→   434→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   435→   435→      const p = gl.createProgram()!
   436→   436→      gl.attachShader(p, vs)
   437→   437→      gl.attachShader(p, fs)
   438→   438→      gl.bindAttribLocation(p, 0, 'aPos')
   439→   439→      gl.linkProgram(p)
   440→   440→      gl.deleteShader(vs)
   441→   441→      gl.deleteShader(fs)
   442→   442→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   443→   443→        const log = gl.getProgramInfoLog(p)
   444→   444→        gl.deleteProgram(p)
   445→   445→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   446→   446→      }
   447→   447→      return p
   448→   448→    }
   449→   449→    const hProg = mk(hFs)
   450→   450→    const vProg = mk(vFs)
   451→   451→    const uH: Record<string, WebGLUniformLocation | null> = {
   452→   452→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   453→   453→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   454→   454→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   455→   455→    }
   456→   456→    const uV: Record<string, WebGLUniformLocation | null> = {
   457→   457→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   458→   458→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   459→   459→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   460→   460→    }
   461→   461→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   462→   462→  }

   464→   464→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   465→   465→   *  blurred result into blurFboB, returns blurFboBTex.
   466→   466→   *  Saves/restores the currently-bound framebuffer.
   467→   467→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   468→   468→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   469→   469→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   470→   470→    const gl = this.gl
   471→   471→    const w = this.fboW
   472→   472→    const h = this.fboH
   473→   473→    // Compute tap count, capped by blurTapCap (performance knob).
   474→   474→    let taps = computeBlur1DTapCount(radius)
   475→   475→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   476→   476→    this.ensureBlurPrograms(taps)
   477→   477→    const entry = this.blurPrograms.get(taps)!
   478→   478→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   479→   479→    gl.disable(gl.BLEND)
   480→   480→
   481→   481→    // Pass 1: horizontal — srcTex → blurFboA
   482→   482→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   483→   483→    gl.viewport(0, 0, w, h)
   484→   484→    gl.useProgram(entry.hProg)
   485→   485→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   486→   486→    gl.enableVertexAttribArray(entry.aPosH)
   487→   487→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   488→   488→    gl.activeTexture(gl.TEXTURE0)
   489→   489→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   490→   490→    gl.uniform1i(entry.uH['uTexture'], 0)
   491→   491→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   492→   492→    gl.uniform1f(entry.uH['uRadius'], radius)
   493→   493→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   494→   494→
   495→   495→    // Pass 2: vertical — blurFboATex → blurFboB
   496→   496→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   497→   497→    gl.viewport(0, 0, w, h)
   498→   498→    gl.useProgram(entry.vProg)
   499→   499→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   500→   500→    gl.enableVertexAttribArray(entry.aPosV)
   501→   501→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   502→   502→    gl.activeTexture(gl.TEXTURE0)
   503→   503→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   504→   504→    gl.uniform1i(entry.uV['uTexture'], 0)
   505→   505→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   506→   506→    gl.uniform1f(entry.uV['uRadius'], radius)
   507→   507→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   508→   508→
   509→   509→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   510→   510→    gl.viewport(0, 0, w, h)
   511→   511→    return this.blurFboBTex!
   512→   512→  }

   514→   514→  /** Lazy-compile highlight blur programs (alpha-blurring, sigma semantics).
   515→   515→   *  Separate from ensureBlurPrograms because the shader is different

   517→   517→  ensureHighlightBlurPrograms(tapCount: number): void {

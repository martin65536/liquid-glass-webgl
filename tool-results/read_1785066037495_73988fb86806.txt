    86→/* ------------------------------------------------------------------ *
    87→ * Element kinds — extends the glass-button model to cover the catalog.
    88→ *
    89→ *   - 'button'          : existing glass button (default)
    90→ *   - 'glass-shape'     : glass rect with NO label and NO press effect
    91→ *                         (e.g. dialog card, tabbar background, toggle/slider knob)
    92→ *   - 'plain-rect'      : solid colored rounded rect (track, fill, card, scrim)
    93→ *   - 'progressive-blur': alpha-masked backdrop blur band
    94→ *   - 'text'            : unclipped text label (section titles, dialog body)
    95→ *
    96→ * The renderer treats each element uniformly through GlassElementConfig.
    97→ * ------------------------------------------------------------------ */
    98→
    99→export type ElementKind =
   100→  | 'button'
   101→  | 'glass-shape'
   102→  | 'plain-rect'
   103→  | 'progressive-blur'
   104→  | 'text'

   148→     *  purposes (matching ContentScale.Inside behavior). */
   149→    layoutSize?: number
   150→  }

   177→    offsetY: number
   178→  } | null
   179→  /**

   182→   * The renderer keeps a single scrollY for the whole canvas; elements
   183→   * without this flag are static (drawn at their rect.y as-is).
   184→   */

   200→   * Default 0 (no rotation).
   201→   */

   203→  /**
   204→   * Arbitrary per-frame scale transform (multiplied into the final scaleX/Y

   207→   * keeping rect.w/h fixed. Default 1 (identity — no extra scale).
   208→   */

   210→  elementScaleY?: number
   211→  /**
   212→   * If set, this element is a toggle knob. The renderer maintains an
   213→   * animated `fraction` (0..1) per groupId and applies:
   214→   *   - x-offset = fraction * dragWidth (knob slides between off/on)
   215→   *   - scale = lerp(1, 1.5, pressProgress) (knob grows on press)
   216→   *   - white overlay alpha = 1 - pressProgress (matches onDrawSurface in LiquidToggle.kt)
   217→   *
   218→   * Faithful to LiquidToggle.kt + DampedDragAnimation.kt:
   219→   *   - valueAnimation: spring(1f, 1000f) — critically damped, no overshoot
   220→   *   - pressProgress: spring(1f, 1000f) — critically damped
   221→   *   - scale: spring(0.6f/0.7f, 250f) — underdamped, slight overshoot
   222→   */

   224→    groupId: string
   225→    /** How far the knob moves from fraction=0 to fraction=1, in CSS px. */
   226→    dragWidth: number
   227→    /**
   228→     * Velocity divisor for the squash-and-stretch effect.
   229→     * Faithful to original:
   230→     *   - LiquidToggle.kt: velocity / 50
   231→     *   - LiquidSlider.kt: velocity / 10
   232→     * Defaults to 50 (toggle). Slider sets 10.
   233→     */
   234→    velocityDivisor?: number
   235→    /**

   237→     *   backdrop = rememberCombinedBackdrop(
   238→     *     backdrop,                                            // wallpaper
   239→     *     rememberBackdrop(trackBackdrop) { drawBackdrop ->   // track color
   240→     *       scale(scaleX, scaleY) { drawBackdrop() }
   241→     *     }
   242→     *   )
   243→     * When set, the knob samples the wallpaper (unscaled) + composited
   244→     * scaled track color, instead of the content-scaled scene. The track
   245→     * color is lerped between offColor and onColor by the toggle fraction.
   246→     * The scale is (lerp(2/3, 0.75, pressProgress), lerp(0, 0.75, pressProgress))
   247→     * around the knob's center.
   248→     */
   249→    trackColorOff?: [number, number, number, number]
   250→    trackColorOn?: [number, number, number, number]
   251→    /** Track's original width in CSS px (e.g. 64dp for toggle). */

   253→    /** Track's original height in CSS px (e.g. 28dp for toggle). */
   254→    trackH?: number
   255→    /**
   256→     * Track's original screen position (top-left, in CSS px). This is the
   257→     * FIXED position where the trackBackdrop is captured — NOT the knob's
   258→     * current position.
   259→     *
   260→     * Faithful to LiquidToggle.kt:
   261→     *   rememberBackdrop(trackBackdrop) { drawBackdrop ->
   262→     *     scale(scaleX, scaleY) { drawBackdrop() }  // pivot = knob.center
   263→     *   }
   264→     * The track content is at its original screen position, and the scale
   265→     * pivots around the KNOB's center. So the scaled track center is at:
   266→     *   knob_center + (track_center - knob_center) * scale
   267→     * The scaled track content moves PARTIALLY with the knob (at rate
   268→     * `1 - scale`), NOT fully with the knob.
   269→     */

   271→    trackOriginalY?: number
   272→    /**
   273→     * Solid backdrop color (RGBA 0..1) — used when the outer backdrop is a
   274→     * CanvasBackdrop (e.g. toggle inside a solid-color card). Faithful to
   275→     * ToggleContent.kt's t2 which uses:
   276→     *   rememberCanvasBackdrop { drawRect(backgroundColor) }
   277→     * When set, the knob samples this solid color instead of the wallpaper
   278→     * texture for the outer backdrop portion of the CombinedBackdrop.
   279→     * When unset, samples the wallpaper texture (LayerBackdrop case).
   280→     */
   281→    solidBackdropColor?: [number, number, number, number]
   282→  }
   283→  /**
   284→   * If set, this element is a toggle track. Its color is lerped between

   286→   * animated fraction.
   287→   */

   289→    groupId: string
   290→    offColor: [number, number, number, number]
   291→    onColor: [number, number, number, number]
   292→  }

   294→   * Slider fill — the accent-colored portion of a slider track. Its width is
   295→   * driven by the toggle group's animated `fraction` (spring) so it stays
   296→   * perfectly in sync with the knob's spring motion (no React-state lag).

   298→   * The fill extends from trackX to the knob's current center position:
   299→   *   fillEnd = trackX + knobW/4 + fraction * (trackW - knobW/2)
   300→   * This matches the knob center exactly (faithful to LiquidSlider.kt where

   305→    groupId: string
   306→    trackX: number
   307→    trackW: number
   308→    knobW: number
   309→    minW: number

   311→  /**

   316→   */

   318→  /**
   319→   * Bottom tabs content (text/icon) — scales up to 1.2 on press.

   325→    groupId: string

   331→    containerCenterY?: number
   332→    /** Container width (for computing the layerBlock scale). */
   333→    containerWidth?: number
   334→  }

   337→   * Faithful to LiquidBottomTabs.kt indicator layerBlock:
   338→   *   scaleX = dampedDragAnimation.scaleX  (spring 0.6, 250, 1→78/56)
   339→   *   scaleY = dampedDragAnimation.scaleY  (spring 0.7, 250, 1→78/56)
   340→   *   velocity = dampedDragAnimation.velocity / 10  (not 50!)
   341→   *   scaleX /= 1 - clamp(vel*0.75, -0.2, 0.2)
   342→   *   scaleY *= 1 - clamp(vel*0.25, -0.2, 0.2)
   343→   * Position: translationX = fraction * dragWidth + panelOffset

   346→    groupId: string
   347→    dragWidth: number

   350→     *    drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)

   414→   *   translationY = -48dp * (1 - derived)

   463→  backdropFbo?: boolean
   464→  /** GLOBAL backdrop blur radius (CSS px), applied to the wallpaper RIGHT

   515→/* ------------------------------------------------------------------ *
   516→ * ToggleGroupState — faithful port of DampedDragAnimation.kt.
   517→ *
   518→ * The renderer maintains one ToggleGroupState per groupId. The fraction
   519→ * (0..1) is animated with a critically damped spring (k=1000, ζ=1) so
   520→ * it tracks the target with no overshoot — matching the smooth knob
   521→ * glide of the original. The pressProgress (also critically damped)
   522→ * drives the knob scale (1→1.5) and the white overlay alpha (1→0).
   523→ *
   524→ * Faithful to DampedDragAnimation.kt:
   525→ *   - valueAnimation:        spring(1f, 1000f)  — critically damped
   526→ *   - velocityAnimation:     spring(0.5f, 300f) — underdamped (drag velocity)
   527→ *   - pressProgressAnimation: spring(1f, 1000f) — critically damped
   528→ *   - scaleXAnimation:       spring(0.6f, 250f) — underdamped, more bounce
   529→ *   - scaleYAnimation:       spring(0.7f, 250f) — underdamped, less bounce
   530→ *
   531→ * The velocity (tracked via a VelocityTracker on the value animation)
   532→ * drives the squash-and-stretch in the layerBlock:
   533→ *   scaleX /= 1 - clamp(vel/50 * 0.75, -0.2, 0.2)
   534→ *   scaleY *= 1 - clamp(vel/50 * 0.25, -0.2, 0.2)
   535→ * ------------------------------------------------------------------ */
   536→export interface ToggleGroupState {
   537→  // Knob position fraction (0..1). Animated with critically damped spring.
   538→  fraction: number
   539→  fractionVelocity: number
   540→  targetFraction: number
   541→  // Press progress (0 = released, 1 = pressed). Drives scale + white overlay.
   542→  pressProgress: number
   543→  pressVelocity: number
   544→  targetPress: number
   545→  // Knob scale X (1..1.5). Underdamped spring (ζ=0.6, k=250) — more bounce.
   546→  scaleX: number
   547→  scaleXVelocity: number
   548→  targetScaleX: number
   549→  // Knob scale Y (1..1.5). Underdamped spring (ζ=0.7, k=250) — less bounce.
   550→  scaleY: number
   551→  scaleYVelocity: number
   552→  targetScaleY: number
   553→  // Drag velocity (normalized 0..1 per second). Underdamped spring (ζ=0.5, k=300).
   554→  // Drives the squash-and-stretch in the layerBlock. Tracked via a simple
   555→  // velocity tracker on the fraction animation.
   556→  velocity: number
   557→  velocityVelocity: number
   558→  targetVelocity: number
   559→  // True while the user is dragging the knob. While true, drag deltas
   560→  // update targetFraction directly (no spring lag, matches onDrag in
   561→  // DampedDragAnimation which updates `fraction` instantly).
   562→  isDragging: boolean
   563→  // True after a drag release (endToggleDrag/endSliderDrag) — enables
   564→  // velocity tracking from fraction rate-of-change. Set false by
   565→  // setToggleTarget/setTabSelected (tap = no velocity animation).
   566→  // Faithful to DampedDragAnimation: velocity is only non-zero after a
   567→  // drag (VelocityTracker accumulates during drag); animateToValue (tap)
   568→  // checks `if (velocity != 0f)` — for taps velocity is 0 → no stretch.
   569→  trackVelocityAfterRelease: boolean
   570→  // VelocityTracker fed (time, fraction) every animation frame — faithful
   571→  // port of DampedDragAnimation.updateVelocity() which calls
   572→  // velocityTracker.addPosition(now, Offset(value, 0)) inside the
   573→  // valueAnimation.animateTo block. Replaces the old naive Δfraction/Δt
   574→  // difference (which was spike-prone on small dt) with a least-squares
   575→  // fit, matching Compose's VelocityTracker.calculateVelocity().
   576→  velocityTracker: VelocityTracker1D
   577→  // Retained for compatibility (no longer the primary velocity source).
   578→  lastFractionForVelocity: number
   579→  lastFractionTime: number
   580→
   581→  // pressedScale: target scale when pressed. 1.5 for toggle knob,
   582→  // 78/56 ≈ 1.393 for bottom tabs indicator.
   583→  // Set when the toggle group is first created (via ensureToggleState).
   584→  pressedScale: number
   585→  // valueRangeSpan = (valueRange.endInclusive - valueRange.start).
   586→  // Faithful to DampedDragAnimation.kt's updateVelocity():
   587→  //   targetVelocity = velocityTracker.calculateVelocity().x / valueRangeSpan
   588→  // Toggle/slider: valueRange = 0..1 → span 1 (no division).
   589→  // Bottom tabs:   valueRange = 0..(tabsCount-1) → span = tabsCount-1.
   590→  // The velocity is normalized to "progress/sec" so the squash-stretch
   591→  // magnitude is consistent regardless of how many tabs there are.
   592→  // Defaults to 1 (toggle/slider). Set to tabsCount-1 for tab groups.
   593→  valueRangeSpan: number
   594→
   595→  // Bottom tabs panelOffset (drag-driven horizontal shift of the whole bar).
   596→  // Faithful to LiquidBottomTabs.kt:
   597→  //   panelOffset = 4dp * sign(fraction) * EaseOut(|fraction|)
   598→  //   offsetAnimation snaps to value+dragAmount during drag,
   599→  //   animates to 0 on release with spring(1f, 300f) — critically damped.
   600→  panelOffset: number
   601→  panelOffsetVelocity: number
   602→  targetPanelOffset: number
   603→}

    38→/* ------------------------------------------------------------------ *
    39→ * LiquidGlassRenderer
    40→ *
    41→ * One opaque WebGL canvas. Render pipeline per frame:
    42→ *   1. Wallpaper background pass (cover-fit).
    43→ *   2. For each button (in order):
    44→ *      a. Outer drop shadow pass (if configured).
    45→ *      b. Element pass (refraction + vibrancy + tint + highlight),
    46→ *         with InteractiveHighlight-driven scale/translation/stretch.
    47→ *      c. White overlay pass (Plus blend, 8% * progress) on press.
    48→ *      d. Radial highlight pass (Plus blend, at finger position) on press.
    49→ *      e. Foreground pass (button label, with press-driven alpha fade).
    50→ *
    51→ * No DOM children — the canvas owns the entire visual surface.
    52→ *
    53→ * NOTE: The class methods are split across multiple files in this
    54→ * directory (methods-*.ts). Each methods file declares its method
    55→ * signatures via `declare module './index'` and exports a record of
    56→ * named functions. Those records are merged onto the prototype at the
    57→ * bottom of this file via Object.assign.
    58→ * ------------------------------------------------------------------ */

   468→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   469→  ensureBlurPrograms(tapCount: number): void {
   470→    if (this.blurPrograms.has(tapCount)) return
   471→    const gl = this.gl
   472→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   473→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   474→    const mk = (fs: WebGLShader) => {
   475→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   476→      const p = gl.createProgram()!
   477→      gl.attachShader(p, vs)
   478→      gl.attachShader(p, fs)
   479→      gl.bindAttribLocation(p, 0, 'aPos')
   480→      gl.linkProgram(p)
   481→      gl.deleteShader(vs)
   482→      gl.deleteShader(fs)
   483→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   484→        const log = gl.getProgramInfoLog(p)
   485→        gl.deleteProgram(p)
   486→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   487→      }
   488→      return p
   489→    }
   490→    const hProg = mk(hFs)
   491→    const vProg = mk(vFs)
   492→    const uH: Record<string, WebGLUniformLocation | null> = {
   493→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   494→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   495→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   496→    }
   497→    const uV: Record<string, WebGLUniformLocation | null> = {
   498→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   499→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   500→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   501→    }
   502→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   503→  }

   505→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   506→   *  blurred result into blurFboB, returns blurFboBTex.
   507→   *  Saves/restores the currently-bound framebuffer.
   508→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   509→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   510→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   511→    const gl = this.gl
   512→    const w = this.fboW
   513→    const h = this.fboH
   514→    // Compute tap count, capped by blurTapCap (performance knob).
   515→    let taps = computeBlur1DTapCount(radius)
   516→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   517→    this.ensureBlurPrograms(taps)
   518→    const entry = this.blurPrograms.get(taps)!
   519→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   520→    gl.disable(gl.BLEND)
   521→
   522→    // Pass 1: horizontal — srcTex → blurFboA
   523→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   524→    gl.viewport(0, 0, w, h)
   525→    gl.useProgram(entry.hProg)
   526→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   527→    gl.enableVertexAttribArray(entry.aPosH)
   528→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   529→    gl.activeTexture(gl.TEXTURE0)
   530→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   531→    gl.uniform1i(entry.uH['uTexture'], 0)
   532→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   533→    gl.uniform1f(entry.uH['uRadius'], radius)
   534→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   535→
   536→    // Pass 2: vertical — blurFboATex → blurFboB
   537→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   538→    gl.viewport(0, 0, w, h)
   539→    gl.useProgram(entry.vProg)
   540→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   541→    gl.enableVertexAttribArray(entry.aPosV)
   542→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   543→    gl.activeTexture(gl.TEXTURE0)
   544→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   545→    gl.uniform1i(entry.uV['uTexture'], 0)
   546→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   547→    gl.uniform1f(entry.uV['uRadius'], radius)
   548→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   549→
   550→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   551→    gl.viewport(0, 0, w, h)
   552→    return this.blurFboBTex!
   553→  }

   555→  /** Lazy-compile highlight blur programs (alpha-blurring, sigma semantics).

   594→  /** 2-pass Gaussian blur on a highlight stroke MASK (alpha only).
   595→   *  Faithful to Android BlurMaskFilter(NORMAL, sigma):
   596→   *    - sigma = blurRadiusPx (the Android radius param IS sigma)
   597→   *    - convolves the mask's ALPHA with a Gaussian kernel
   598→   *    - sub-pixel sigma (0.25px) still blurs (no 0.5 early-return)
   599→   *  Reads srcTex (alpha mask), writes blurFboB, returns blurFboBTex.
   600→   *  Saves/restores the currently-bound framebuffer. */
   601→  blurHighlightMask(srcTex: WebGLTexture, sigmaPx: number): WebGLTexture {
   602→    const gl = this.gl
   603→    const w = this.fboW
   604→    const h = this.fboH
   605→    let taps = computeHighlightBlurTapCount(sigmaPx)
   606→    taps = Math.min(taps, Math.max(3, this.blurTapCap | 0))
   607→    this.ensureHighlightBlurPrograms(taps)
   608→    const entry = this.highlightBlurPrograms.get(taps)!
   609→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   610→    gl.disable(gl.BLEND)
   611→
   612→    // Pass 1: horizontal — srcTex → blurFboA
   613→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   614→    gl.viewport(0, 0, w, h)
   615→    gl.useProgram(entry.hProg)
   616→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   617→    gl.enableVertexAttribArray(entry.aPosH)
   618→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   619→    gl.activeTexture(gl.TEXTURE0)
   620→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   621→    gl.uniform1i(entry.uH['uTexture'], 0)
   622→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   623→    gl.uniform1f(entry.uH['uRadius'], sigmaPx)
   624→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   625→
   626→    // Pass 2: vertical — blurFboATex → blurFboB
   627→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   628→    gl.viewport(0, 0, w, h)
   629→    gl.useProgram(entry.vProg)
   630→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   631→    gl.enableVertexAttribArray(entry.aPosV)
   632→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   633→    gl.activeTexture(gl.TEXTURE0)
   634→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   635→    gl.uniform1i(entry.uV['uTexture'], 0)
   636→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   637→    gl.uniform1f(entry.uV['uRadius'], sigmaPx)
   638→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   639→
   640→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   641→    gl.viewport(0, 0, w, h)
   642→    return this.blurFboBTex!
   643→  }

   645→  dispose() {
   646→    if (this.rafId !== null) cancelAnimationFrame(this.rafId)
   647→    this.rafId = null
   648→    if (this.animRafId !== null) cancelAnimationFrame(this.animRafId)
   649→    this.animRafId = null
   650→    const gl = this.gl
   651→    if (this.wallpaperTexture) gl.deleteTexture(this.wallpaperTexture)
   652→    for (const tex of this.fgTextures.values()) gl.deleteTexture(tex)
   653→    this.fgTextures.clear()
   654→    for (const entry of this.strokeMaskCache.values()) gl.deleteTexture(entry.tex)
   655→    this.strokeMaskCache.clear()
   656→    destroyCache(gl, this.innerShadowMaskCache)
   657→    if (this.fboA) gl.deleteFramebuffer(this.fboA)
   658→    if (this.fboATex) gl.deleteTexture(this.fboATex)
   659→    if (this.fboB) gl.deleteFramebuffer(this.fboB)
   660→    if (this.fboBTex) gl.deleteTexture(this.fboBTex)
   661→    this.fboA = this.fboB = null
   662→    this.fboATex = this.fboBTex = null
   663→    if (this.tabsBackdropFbo) gl.deleteFramebuffer(this.tabsBackdropFbo)
   664→    if (this.tabsBackdropTex) gl.deleteTexture(this.tabsBackdropTex)
   665→    this.tabsBackdropFbo = null
   666→    this.tabsBackdropTex = null
   667→    // GP element FBO + blur FBOs + programs
   668→    if (this.gpElementFbo) gl.deleteFramebuffer(this.gpElementFbo)
   669→    if (this.gpElementTex) gl.deleteTexture(this.gpElementTex)
   670→    if (this.blurFboA) gl.deleteFramebuffer(this.blurFboA)
   671→    if (this.blurFboATex) gl.deleteTexture(this.blurFboATex)
   672→    if (this.blurFboB) gl.deleteFramebuffer(this.blurFboB)
   673→    if (this.blurFboBTex) gl.deleteTexture(this.blurFboBTex)
   674→    this.gpElementFbo = this.blurFboA = this.blurFboB = null
   675→    this.gpElementTex = this.blurFboATex = this.blurFboBTex = null
   676→    if (this.highlightMaskFbo) gl.deleteFramebuffer(this.highlightMaskFbo)
   677→    if (this.highlightMaskTex) gl.deleteTexture(this.highlightMaskTex)
   678→    this.highlightMaskFbo = null
   679→    this.highlightMaskTex = null
   680→    if (this.dialogBackdropFbo) gl.deleteFramebuffer(this.dialogBackdropFbo)
   681→    if (this.dialogBackdropTex) gl.deleteTexture(this.dialogBackdropTex)
   682→    this.dialogBackdropFbo = null
   683→    this.dialogBackdropTex = null
   684→    this.dialogBackdropKey = null
   685→    // Per-element FBOs (elFbo + backdrop crop + el blur ping-pong)
   686→    if (this.elFbo) gl.deleteFramebuffer(this.elFbo)
   687→    if (this.elFboTex) gl.deleteTexture(this.elFboTex)
   688→    this.elFbo = null
   689→    this.elFboTex = null
   690→    this.elFboW = this.elFboH = 0
   691→    if (this.backdropCropFbo) gl.deleteFramebuffer(this.backdropCropFbo)
   692→    if (this.backdropCropTex) gl.deleteTexture(this.backdropCropTex)
   693→    this.backdropCropFbo = null
   694→    this.backdropCropTex = null
   695→    if (this.elBlurFboA) gl.deleteFramebuffer(this.elBlurFboA)
   696→    if (this.elBlurFboATex) gl.deleteTexture(this.elBlurFboATex)
   697→    if (this.elBlurFboB) gl.deleteFramebuffer(this.elBlurFboB)
   698→    if (this.elBlurFboBTex) gl.deleteTexture(this.elBlurFboBTex)
   699→    this.elBlurFboA = this.elBlurFboB = null
   700→    this.elBlurFboATex = this.elBlurFboBTex = null
   701→    for (const { hProg, vProg } of this.blurPrograms.values()) {
   702→      gl.deleteProgram(hProg)
   703→      gl.deleteProgram(vProg)
   704→    }
   705→    this.blurPrograms.clear()
   706→    for (const { hProg, vProg } of this.highlightBlurPrograms.values()) {
   707→      gl.deleteProgram(hProg)
   708→      gl.deleteProgram(vProg)
   709→    }
   710→    this.highlightBlurPrograms.clear()
   711→    if (this.sdfTexture) gl.deleteTexture(this.sdfTexture)
   712→    this.sdfTexture = null
   713→    for (const { tex } of this.continuousSdfPool.values()) gl.deleteTexture(tex)
   714→    this.continuousSdfPool.clear()
   715→    this.continuousSdfTexture = null
   716→    this.continuousSdfKey = null
   717→    gl.deleteProgram(this.elementProgram)
   718→    gl.deleteProgram(this.shadowProgram)
   719→    gl.deleteProgram(this.wallpaperProgram)
   720→    gl.deleteProgram(this.foregroundProgram)
   721→    gl.deleteProgram(this.highlightProgram)
   722→    gl.deleteProgram(this.tintProgram)
   723→    gl.deleteProgram(this.rimHighlightProgram)
   724→    gl.deleteProgram(this.highlightStrokeProgram)
   725→    gl.deleteProgram(this.highlightCompositeProgram)
   726→    gl.deleteProgram(this.strokeMaskCompositeProgram)
   727→    gl.deleteProgram(this.innerShadowMaskCompositeProgram)
   728→    gl.deleteProgram(this.plainRectProgram)
   729→    gl.deleteProgram(this.progressiveBlurProgram)
   730→    gl.deleteProgram(this.copyProgram)
   731→    gl.deleteProgram(this.solidFillProgram)
   732→    gl.deleteProgram(this.colorControlsProgram)
   733→    gl.deleteProgram(this.sceneTintProgram)
   734→    gl.deleteProgram(this.elFboCompositeProgram)
   735→    gl.deleteProgram(this.elFboCropProgram)
   736→    gl.deleteBuffer(this.quadBuffer)
   737→  }

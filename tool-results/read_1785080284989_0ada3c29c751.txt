    86→/* ------------------------------------------------------------------ *
    87→ * Element kinds — extends the glass-button model to cover the catalog.
    88→ *
    89→ *   - 'button'          : existing glass button (default)
    90→ *   - 'glass-shape'     : glass rect with NO label and NO press effect
    91→ *                         (e.g. dialog card, tabbar background, toggle/slider knob)
    92→ *   - 'plain-rect'      : solid colored rounded rect (track, fill, card, scrim)
    93→ *   - 'progressive-blur': alpha-masked backdrop blur band
    94→ *   - 'text'            : unclipped text label (section titles, dialog body)
    95→ *
    96→ * The renderer treats each element uniformly through GlassElementConfig.
    97→ * ------------------------------------------------------------------ */
    98→
    99→export type ElementKind =
   100→  | 'button'
   101→  | 'glass-shape'
   102→  | 'plain-rect'
   103→  | 'progressive-blur'
   104→  | 'text'

   173→   *  Faithful to InnerShadowModifier.kt. Default color = Black.copy(alpha=0.15f).

   175→   *  When color is set, the shader uses SrcOver blend with that color. */

   177→    radius: number
   178→    alpha: number
   179→    offsetX: number
   180→    offsetY: number

   182→     *  The shader uses: color = mix(color, shadowColor, ring * alpha) */

   186→   *  Typically a WHITE inner shadow offset upward (0, -radius), creating
   187→   *  a bright band at the bottom edge — paired with the black inner shadow
   188→   *  (offset downward) at the top edge. This makes toggle/slider knobs

   193→    alpha: number

   195→    offsetY: number
   196→    color: [number, number, number]
   197→  } | null
   198→  /**

   200→   * to the section top, and the renderer adds `scrollY` to its screen y.
   201→   * The renderer keeps a single scrollY for the whole canvas; elements
   202→   * without this flag are static (drawn at their rect.y as-is).
   203→   */

   205→  /**
   206→   * Touch hit-test rect (optional). When set, pointer hit-testing uses this

   211→   * the same way as `rect` (if `scroll` is true, y is section-relative).
   212→   */

   214→  /**

   216→   * Faithful to Compose graphicsLayer { rotationZ }. Used by Glass
   217→   * Playground's transformable glass square. The renderer rotates the
   218→   * SDF + refraction sampling so the glass shape rotates as a unit.
   219→   * Default 0 (no rotation).
   220→   */
   221→  elementRotation?: number
   222→  /**
   223→   * Arbitrary per-frame scale transform (multiplied into the final scaleX/Y
   224→   * alongside press, toggle, tab, and control-center transforms). Used by
   225→   * the perf benchmark's outer 12 glasses for scale-only animation while
   226→   * keeping rect.w/h fixed. Default 1 (identity — no extra scale).
   227→   */

   229→  elementScaleY?: number
   230→  /**
   231→   * If set, this element is a toggle knob. The renderer maintains an
   232→   * animated `fraction` (0..1) per groupId and applies:
   233→   *   - x-offset = fraction * dragWidth (knob slides between off/on)
   234→   *   - scale = lerp(1, 1.5, pressProgress) (knob grows on press)
   235→   *   - white overlay alpha = 1 - pressProgress (matches onDrawSurface in LiquidToggle.kt)
   236→   *
   237→   * Faithful to LiquidToggle.kt + DampedDragAnimation.kt:
   238→   *   - valueAnimation: spring(1f, 1000f) — critically damped, no overshoot
   239→   *   - pressProgress: spring(1f, 1000f) — critically damped
   240→   *   - scale: spring(0.6f/0.7f, 250f) — underdamped, slight overshoot
   241→   */
   242→  isToggleKnob?: {
   243→    groupId: string
   244→    /** How far the knob moves from fraction=0 to fraction=1, in CSS px. */
   245→    dragWidth: number
   246→    /**
   247→     * Velocity divisor for the squash-and-stretch effect.
   248→     * Faithful to original:
   249→     *   - LiquidToggle.kt: velocity / 50
   250→     *   - LiquidSlider.kt: velocity / 10
   251→     * Defaults to 50 (toggle). Slider sets 10.
   252→     */
   253→    velocityDivisor?: number
   254→    /**
   255→     * CombinedBackdrop track color info (faithful to LiquidToggle.kt):
   256→     *   backdrop = rememberCombinedBackdrop(
   257→     *     backdrop,                                            // wallpaper
   258→     *     rememberBackdrop(trackBackdrop) { drawBackdrop ->   // track color
   259→     *       scale(scaleX, scaleY) { drawBackdrop() }
   260→     *     }
   261→     *   )
   262→     * When set, the knob samples the wallpaper (unscaled) + composited
   263→     * scaled track color, instead of the content-scaled scene. The track
   264→     * color is lerped between offColor and onColor by the toggle fraction.
   265→     * The scale is (lerp(2/3, 0.75, pressProgress), lerp(0, 0.75, pressProgress))
   266→     * around the knob's center.
   267→     */

   269→    trackColorOn?: [number, number, number, number]
   270→    /** Track's original width in CSS px (e.g. 64dp for toggle). */

   272→    /** Track's original height in CSS px (e.g. 28dp for toggle). */
   273→    trackH?: number
   274→    /**
   275→     * Track's original screen position (top-left, in CSS px). This is the
   276→     * FIXED position where the trackBackdrop is captured — NOT the knob's
   277→     * current position.

   279→     * Faithful to LiquidToggle.kt:
   280→     *   rememberBackdrop(trackBackdrop) { drawBackdrop ->
   281→     *     scale(scaleX, scaleY) { drawBackdrop() }  // pivot = knob.center
   282→     *   }
   283→     * The track content is at its original screen position, and the scale
   284→     * pivots around the KNOB's center. So the scaled track center is at:
   285→     *   knob_center + (track_center - knob_center) * scale
   286→     * The scaled track content moves PARTIALLY with the knob (at rate
   287→     * `1 - scale`), NOT fully with the knob.
   288→     */
   289→    trackOriginalX?: number
   290→    trackOriginalY?: number
   291→    /**
   292→     * Solid backdrop color (RGBA 0..1) — used when the outer backdrop is a
   293→     * CanvasBackdrop (e.g. toggle inside a solid-color card). Faithful to
   294→     * ToggleContent.kt's t2 which uses:
   295→     *   rememberCanvasBackdrop { drawRect(backgroundColor) }
   296→     * When set, the knob samples this solid color instead of the wallpaper
   297→     * texture for the outer backdrop portion of the CombinedBackdrop.
   298→     * When unset, samples the wallpaper texture (LayerBackdrop case).
   299→     */
   300→    solidBackdropColor?: [number, number, number, number]
   301→  }
   302→  /**
   303→   * If set, this element is a toggle track. Its color is lerped between
   304→   * offColor and onColor based on the corresponding toggle group's
   305→   * animated fraction.
   306→   */

   308→    groupId: string
   309→    offColor: [number, number, number, number]
   310→    onColor: [number, number, number, number]
   311→  }
   312→  /**
   313→   * Slider fill — the accent-colored portion of a slider track. Its width is
   314→   * driven by the toggle group's animated `fraction` (spring) so it stays
   315→   * perfectly in sync with the knob's spring motion (no React-state lag).
   316→   *
   317→   * The fill extends from trackX to the knob's current center position:
   318→   *   fillEnd = trackX + knobW/4 + fraction * (trackW - knobW/2)
   319→   * This matches the knob center exactly (faithful to LiquidSlider.kt where
   320→   * both fill width and knob translationX use the same `progress`), so the
   321→   * fill never overshoots or undershoots the knob.
   322→   */

   324→    groupId: string
   325→    trackX: number
   326→    trackW: number
   327→    knobW: number
   328→    minW: number
   329→  }
   330→  /**
   331→   * Bottom tabs container — scales up on press (16dp/width).
   332→   * Faithful to LiquidBottomTabs.kt container layerBlock:
   333→   *   scale = lerp(1, 1 + 16dp/width, pressProgress)
   334→   * Also shifts by panelOffset during drag.
   335→   */

   337→  /**
   338→   * Bottom tabs content (text/icon) — scales up to 1.2 on press.
   339→   * Faithful to LiquidBottomTab.kt graphicsLayer:
   340→   *   scale = lerp(1, 1.2, pressProgress)
   341→   * Also shifts by panelOffset during drag.
   342→   */
   343→  isBottomTabContent?: {
   344→    groupId: string
   345→    /** Container center (the scale origin for the whole bar). Tab content

   347→     *  original where container is the parent and its transform applies
   348→     *  uniformly to all children. */

   350→    containerCenterY?: number
   351→    /** Container width (for computing the layerBlock scale). */
   352→    containerWidth?: number
   353→  }
   354→  /**
   355→   * Bottom tabs indicator — DampedDragAnimation with pressedScale=78/56.
   356→   * Faithful to LiquidBottomTabs.kt indicator layerBlock:
   357→   *   scaleX = dampedDragAnimation.scaleX  (spring 0.6, 250, 1→78/56)
   358→   *   scaleY = dampedDragAnimation.scaleY  (spring 0.7, 250, 1→78/56)
   359→   *   velocity = dampedDragAnimation.velocity / 10  (not 50!)
   360→   *   scaleX /= 1 - clamp(vel*0.75, -0.2, 0.2)
   361→   *   scaleY *= 1 - clamp(vel*0.25, -0.2, 0.2)
   362→   * Position: translationX = fraction * dragWidth + panelOffset

   365→    groupId: string
   366→    dragWidth: number
   367→    /** Theme-aware dim color for onDrawSurface (Black light / White dark).

   369→     *    drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)

   389→    containerCenterY?: number

   392→    /** Tab content element IDs (for looking up fgTextures to use as blue-tint

   433→   *   translationY = -48dp * (1 - derived)

   482→  backdropFbo?: boolean

   534→/* ------------------------------------------------------------------ *
   535→ * ToggleGroupState — faithful port of DampedDragAnimation.kt.
   536→ *
   537→ * The renderer maintains one ToggleGroupState per groupId. The fraction
   538→ * (0..1) is animated with a critically damped spring (k=1000, ζ=1) so
   539→ * it tracks the target with no overshoot — matching the smooth knob
   540→ * glide of the original. The pressProgress (also critically damped)
   541→ * drives the knob scale (1→1.5) and the white overlay alpha (1→0).
   542→ *
   543→ * Faithful to DampedDragAnimation.kt:
   544→ *   - valueAnimation:        spring(1f, 1000f)  — critically damped
   545→ *   - velocityAnimation:     spring(0.5f, 300f) — underdamped (drag velocity)
   546→ *   - pressProgressAnimation: spring(1f, 1000f) — critically damped
   547→ *   - scaleXAnimation:       spring(0.6f, 250f) — underdamped, more bounce
   548→ *   - scaleYAnimation:       spring(0.7f, 250f) — underdamped, less bounce
   549→ *
   550→ * The velocity (tracked via a VelocityTracker on the value animation)
   551→ * drives the squash-and-stretch in the layerBlock:
   552→ *   scaleX /= 1 - clamp(vel/50 * 0.75, -0.2, 0.2)
   553→ *   scaleY *= 1 - clamp(vel/50 * 0.25, -0.2, 0.2)
   554→ * ------------------------------------------------------------------ */
   555→export interface ToggleGroupState {
   556→  // Knob position fraction (0..1). Animated with critically damped spring.
   557→  fraction: number
   558→  fractionVelocity: number
   559→  targetFraction: number
   560→  // Press progress (0 = released, 1 = pressed). Drives scale + white overlay.
   561→  pressProgress: number
   562→  pressVelocity: number
   563→  targetPress: number
   564→  // Knob scale X (1..1.5). Underdamped spring (ζ=0.6, k=250) — more bounce.
   565→  scaleX: number
   566→  scaleXVelocity: number
   567→  targetScaleX: number
   568→  // Knob scale Y (1..1.5). Underdamped spring (ζ=0.7, k=250) — less bounce.
   569→  scaleY: number
   570→  scaleYVelocity: number
   571→  targetScaleY: number
   572→  // Drag velocity (normalized 0..1 per second). Underdamped spring (ζ=0.5, k=300).
   573→  // Drives the squash-and-stretch in the layerBlock. Tracked via a simple
   574→  // velocity tracker on the fraction animation.
   575→  velocity: number
   576→  velocityVelocity: number
   577→  targetVelocity: number
   578→  // True while the user is dragging the knob. While true, drag deltas
   579→  // update targetFraction directly (no spring lag, matches onDrag in
   580→  // DampedDragAnimation which updates `fraction` instantly).
   581→  isDragging: boolean
   582→  // True after a drag release (endToggleDrag/endSliderDrag) — enables
   583→  // velocity tracking from fraction rate-of-change. Set false by
   584→  // setToggleTarget/setTabSelected (tap = no velocity animation).
   585→  // Faithful to DampedDragAnimation: velocity is only non-zero after a
   586→  // drag (VelocityTracker accumulates during drag); animateToValue (tap)
   587→  // checks `if (velocity != 0f)` — for taps velocity is 0 → no stretch.
   588→  trackVelocityAfterRelease: boolean
   589→  // VelocityTracker fed (time, fraction) every animation frame — faithful
   590→  // port of DampedDragAnimation.updateVelocity() which calls
   591→  // velocityTracker.addPosition(now, Offset(value, 0)) inside the
   592→  // valueAnimation.animateTo block. Replaces the old naive Δfraction/Δt
   593→  // difference (which was spike-prone on small dt) with a least-squares
   594→  // fit, matching Compose's VelocityTracker.calculateVelocity().
   595→  velocityTracker: VelocityTracker1D
   596→  // Retained for compatibility (no longer the primary velocity source).
   597→  lastFractionForVelocity: number
   598→  lastFractionTime: number
   599→
   600→  // pressedScale: target scale when pressed. 1.5 for toggle knob,
   601→  // 78/56 ≈ 1.393 for bottom tabs indicator.
   602→  // Set when the toggle group is first created (via ensureToggleState).
   603→  pressedScale: number
   604→  // valueRangeSpan = (valueRange.endInclusive - valueRange.start).
   605→  // Faithful to DampedDragAnimation.kt's updateVelocity():
   606→  //   targetVelocity = velocityTracker.calculateVelocity().x / valueRangeSpan
   607→  // Toggle/slider: valueRange = 0..1 → span 1 (no division).
   608→  // Bottom tabs:   valueRange = 0..(tabsCount-1) → span = tabsCount-1.
   609→  // The velocity is normalized to "progress/sec" so the squash-stretch
   610→  // magnitude is consistent regardless of how many tabs there are.
   611→  // Defaults to 1 (toggle/slider). Set to tabsCount-1 for tab groups.
   612→  valueRangeSpan: number
   613→
   614→  // Bottom tabs panelOffset (drag-driven horizontal shift of the whole bar).
   615→  // Faithful to LiquidBottomTabs.kt:
   616→  //   panelOffset = 4dp * sign(fraction) * EaseOut(|fraction|)
   617→  //   offsetAnimation snaps to value+dragAmount during drag,
   618→  //   animates to 0 on release with spring(1f, 300f) — critically damped.
   619→  panelOffset: number
   620→  panelOffsetVelocity: number
   621→  targetPanelOffset: number
   622→}

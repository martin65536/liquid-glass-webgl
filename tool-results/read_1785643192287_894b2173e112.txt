    18→    18→/* ------------------------------------------------------------------ *
    19→    19→ * Shared LiquidSlider factory — used by both the Slider page and the
    20→    20→ * Glass Playground. Creates track + fill + knob + interactions.

    22→    22→// Unified per-group drag state (survives re-renders during liveUpdate /
    23→    23→// gravityAngle changes). Used by ALL drag-based controls.
    24→    24→const dragStates = new Map<string, { fraction: number; x: number; didDrag: boolean }>()
    25→    25→
    26→    26→/* ------------------------------------------------------------------ *
    27→    27→ * makeDragInteractions — the ONE shared drag-interaction factory.
    28→    28→ *
    29→    29→ * Used by: sliders (continuous + stepped), toggles, bottom tabs.
    30→    30→ * All share the same gesture pattern:
    31→    31→ *   onDragStart → read current visual fraction → beginDrag
    32→    32→ *   onDrag → relative drag (knob follows finger delta with spring lag)
    33→    33→ *   onDragEnd → endDrag → snap → setTarget
    34→    34→ *   onTap → jump to tapped fraction → snap → setTarget
    35→    35→ *
    36→    36→ * The control-specific behavior is injected via the `opts` object:
    37→    37→ *   - getFraction: read current visual fraction (getToggleFraction / getTabFraction)
    38→    38→ *   - beginDrag: start the drag (beginToggleDrag / beginTabDrag)
    39→    39→ *   - drag: update during drag (dragToggle / dragTab)
    40→    40→ *   - endDrag: release + return final fraction (endSliderDrag / endToggleDrag / endTabDrag)
    41→    41→ *   - setTarget: programmatic target set (setToggleTarget / setTabSelected)
    42→    42→ *   - snap: snap function (fraction → snapped fraction), or null = no snap
    43→    43→ *   - liveUpdate: push fraction to onValueChange during drag
    44→    44→ *   - onTapJump: if false, tap does nothing (e.g. toggle = toggle on tap, not jump)
    45→    45→ * ------------------------------------------------------------------ */
    46→    46→interface DragInteractionsOpts {
    47→    47→  groupId: string
    48→    48→  trackX: number
    49→    49→  dragW: number
    50→    50→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null
    51→    51→  onValueChange: (fraction: number) => void
    52→    52→  /** Optional lightweight callback fired on every drag move (regardless of
    53→    53→   *  liveUpdate). Use this to update a display-only label WITHOUT triggering
    54→    54→   *  a catalog rebuild (e.g. settings slider labels). onValueChange is still
    55→    55→   *  fired on dragEnd (or every move if liveUpdate=true) for the real state. */
    56→    56→  onLiveValue?: (fraction: number) => void
    57→    57→  // Control-specific renderer calls:
    58→    58→  getFraction: (r: LiquidGlassRenderer, groupId: string) => number
    59→    59→  beginDrag: (r: LiquidGlassRenderer, groupId: string, fraction: number, count?: number) => void
    60→    60→  drag: (r: LiquidGlassRenderer, groupId: string, startFraction: number, currentX: number, startX: number, dragW: number, count?: number) => void
    61→    61→  endDrag: (r: LiquidGlassRenderer, groupId: string, count?: number) => number
    62→    62→  setTarget: (r: LiquidGlassRenderer, groupId: string, fraction: number, count?: number) => void
    63→    63→  count?: number // tabsCount for tabs (passed to begin/drag/end/setTarget)
    64→    64→  snap?: (f: number) => number // snap function (null/undefined = no snap)
    65→    65→  liveUpdate?: boolean
    66→    66→  onTapJump?: boolean // default true; false = tap does nothing (toggle uses its own onTap)
    67→    67→  didDragThreshold?: number // px of movement before didDrag=true (default 3)
    68→    68→}
    69→    69→
    70→    70→export function makeDragInteractions(opts: DragInteractionsOpts): ElementInteraction {
    71→    71→  const {
    72→    72→    groupId, trackX, dragW, rendererRef, onValueChange, onLiveValue,
    73→    73→    getFraction, beginDrag, drag, endDrag, setTarget,
    74→    74→    count, snap, liveUpdate = false, onTapJump = true, didDragThreshold = 3,
    75→    75→  } = opts
    76→    76→
    77→    77→  if (!dragStates.has(groupId)) dragStates.set(groupId, { fraction: 0, x: 0, didDrag: false })
    78→    78→  const ds = dragStates.get(groupId)!
    79→    79→
    80→    80→  const fracFromPos = (px: number) => Math.max(0, Math.min(1, (px - trackX) / dragW))
    81→    81→  const applySnap = (f: number) => (snap ? snap(f) : f)
    82→    82→
    83→    83→  return {
    84→    84→    onTap: (pos) => {
    85→    85→      if (!onTapJump) return
    86→    86→      const f = applySnap(fracFromPos(pos.x))
    87→    87→      const r = rendererRef?.current
    88→    88→      if (r) setTarget(r, groupId, f, count)
    89→    89→      onValueChange(f)
    90→    90→    },
    91→    91→    onDragStart: (pos) => {
    92→    92→      const r = rendererRef?.current
    93→    93→      if (!r) return
    94→    94→      draggingGroups.add(groupId)
    95→    95→      ds.fraction = getFraction(r, groupId)
    96→    96→      ds.x = pos.x
    97→    97→      ds.didDrag = false
    98→    98→      beginDrag(r, groupId, ds.fraction, count)
    99→    99→    },
   100→   100→    onDrag: (pos) => {
   101→   101→      const r = rendererRef?.current
   102→   102→      if (!r) return
   103→   103→      if (Math.abs(pos.x - ds.x) > didDragThreshold) ds.didDrag = true
   104→   104→      drag(r, groupId, ds.fraction, pos.x, ds.x, dragW, count)
   105→   105→      const f = getFraction(r, groupId)
   106→   106→      if (onLiveValue) onLiveValue(f)
   107→   107→      if (liveUpdate) {
   108→   108→        onValueChange(f)
   109→   109→      }
   110→   110→    },
   111→   111→    onDragEnd: () => {
   112→   112→      const r = rendererRef?.current
   113→   113→      if (!r) return
   114→   114→      const rawF = endDrag(r, groupId, count)
   115→   115→      // Keep draggingGroups flag set DURING onValueChange (which triggers
   116→   116→      // setState → tabTargets/toggleTargets effect). The effect checks
   117→   117→      // draggingGroups and skips setTabSelected/setToggleTarget, preventing
   118→   118→      // it from zeroing velocity and fighting the spring. Delete AFTER
   119→   119→      // onValueChange so the next render's effect can sync.
   120→   120→      const snappedF = applySnap(rawF)
   121→   121→      // Only call setTarget if the endDrag didn't already snap.
   122→   122→      // For toggle/slider: endToggleDrag/endSliderDrag DON'T snap (they
   123→   123→      // return the raw fraction), so setTarget is needed.
   124→   124→      // For tabs: endTabDrag DOES snap (sets targetFraction to rounded
   125→   125→      // index), so setTarget would be redundant and would zero velocity
   126→   126→      // (fighting the spring). Skip setTarget for tabs (count != undefined
   127→   127→      // && endDrag already snapped).
   128→   128→      if (snap && count == null) setTarget(r, groupId, snappedF, count)
   129→   129→      onValueChange(snappedF)
   130→   130→      draggingGroups.delete(groupId)
   131→   131→    },
   132→   132→  }
   133→   133→}

   136→   136→ * Renderer method bindings — pre-wired adapters for each control type.

   140→   140→// Slider/Settings: getToggleFraction, beginToggleDrag, dragToggle,
   141→   141→// endSliderDrag, setToggleTarget.
   142→   142→export const sliderDragBindings = {
   143→   143→  getFraction: (r: LiquidGlassRenderer, id: string) => r.getToggleFraction(id),
   144→   144→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number) => r.beginToggleDrag(id, f),
   145→   145→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number) =>
   146→   146→    r.dragToggle(id, sf, cx, sx, dw),
   147→   147→  endDrag: (r: LiquidGlassRenderer, id: string) => r.endSliderDrag(id),
   148→   148→  setTarget: (r: LiquidGlassRenderer, id: string, f: number) => r.setToggleTarget(id, f),
   149→   149→}
   150→   150→
   151→   151→// Toggle: getToggleTarget (not fraction!), beginToggleDrag, dragToggle,
   152→   152→// endToggleDrag, setToggleTarget. Tap = toggle (no jump).
   153→   153→export const toggleDragBindings = {
   154→   154→  getFraction: (r: LiquidGlassRenderer, id: string) => r.getToggleTarget(id),
   155→   155→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number) => r.beginToggleDrag(id, f),
   156→   156→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number) =>
   157→   157→    r.dragToggle(id, sf, cx, sx, dw),
   158→   158→  endDrag: (r: LiquidGlassRenderer, id: string) => r.endToggleDrag(id),
   159→   159→  setTarget: (r: LiquidGlassRenderer, id: string, f: number) => r.setToggleTarget(id, f),
   160→   160→}
   161→   161→
   162→   162→// Bottom tabs: fraction (0..1) ↔ index (0..count-1) conversion at the
   163→   163→// binding layer. The renderer works in INDEX space (0..count-1), but
   164→   164→// makeDragInteractions works in FRACTION space (0..1). The bindings
   165→   165→// convert: getFraction divides by (count-1), setTarget/beginDrag/drag
   166→   166→// multiply by (count-1).
   167→   167→export const tabDragBindings = {
   168→   168→  getFraction: (r: LiquidGlassRenderer, id: string, count?: number) => {
   169→   169→    const c = count ?? 3
   170→   170→    // Use TARGET (not animated) fraction — faithful to the original which
   171→   171→    // uses `targetValue` in onDrag, not the animated value. Starting a drag
   172→   172→    // from the animated value (mid-spring) causes drift because the spring's
   173→   173→    // residual motion adds to the finger delta.
   174→   174→    return r.getTabTarget(id) / Math.max(1, c - 1)
   175→   175→  },
   176→   176→  beginDrag: (r: LiquidGlassRenderer, id: string, f: number, count?: number) => {
   177→   177→    const c = count ?? 3
   178→   178→    r.beginTabDrag(id, f * (c - 1), c)
   179→   179→  },
   180→   180→  drag: (r: LiquidGlassRenderer, id: string, sf: number, cx: number, sx: number, dw: number, count?: number) => {
   181→   181→    const c = count ?? 3
   182→   182→    r.dragTab(id, sf * (c - 1), cx, sx, dw, c)
   183→   183→  },
   184→   184→  endDrag: (r: LiquidGlassRenderer, id: string, count?: number) => {
   185→   185→    const c = count ?? 3
   186→   186→    // endTabDrag returns an INDEX (0..c-1). Convert to fraction (0..1).
   187→   187→    return r.endTabDrag(id, c) / Math.max(1, c - 1)
   188→   188→  },
   189→   189→  setTarget: (r: LiquidGlassRenderer, id: string, f: number, count?: number) => {
   190→   190→    const c = count ?? 3
   191→   191→    r.setTabSelected(id, Math.max(0, Math.min(c - 1, Math.round(f * (c - 1)))), c)
   192→   192→  },
   193→   193→}

   195→   195→export function makeLiquidSlider(
   196→   196→  idPrefix: string,
   197→   197→  trackX: number,
   198→   198→  trackY: number,
   199→   199→  trackW: number,
   200→   200→  groupId: string,
   201→   201→  trackColor: [number, number, number, number],
   202→   202→  accentColor: [number, number, number],
   203→   203→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null,
   204→   204→  onValueChange: (fraction: number) => void,
   205→   205→  scroll = true,
   206→   206→  liveUpdate = false,
   207→   207→  initFraction = 0,
   208→   208→  snap?: (f: number) => number,
   209→   209→  onLiveValue?: (fraction: number) => void,
   210→   210→): { elements: GlassElementConfig[]; interactions: Record<string, ElementInteraction> } {
   211→   211→  const elements: GlassElementConfig[] = []
   212→   212→  const interactions: Record<string, ElementInteraction> = {}
   213→   213→  const dragW = trackW - SLIDER_KNOB_W / 2
   214→   214→  const knobBaseX = trackX - SLIDER_KNOB_W / 4
   215→   215→  const knobY = trackY + (SLIDER_TRACK_H - SLIDER_KNOB_H) / 2
   216→   216→  // Knob rect.x is ALWAYS at fraction=0 (knobBaseX). The renderer's
   217→   217→  // isToggleKnob.dragWidth drives the x offset via spring animation.
   218→   218→  // Setting rect.x to initFraction would cause 2x displacement
   219→   219→  // (rect.x + spring offset = initFraction*dragW + fraction*dragW).
   220→   220→  const knobX = knobBaseX
   221→   221→
   222→   222→  // Track
   223→   223→  const trackEl = makePlainRect(`${idPrefix}-track`, { x: trackX, y: trackY, w: trackW, h: SLIDER_TRACK_H }, trackColor, SLIDER_TRACK_H / 2)
   224→   224→  trackEl.hitRect = { x: trackX, y: trackY + (SLIDER_TRACK_H - SLIDER_HIT_H) / 2, w: trackW, h: SLIDER_HIT_H }
   225→   225→  trackEl.scroll = scroll
   226→   226→  elements.push(trackEl)
   227→   227→
   228→   228→  // Fill — width driven by renderer via isSliderFill
   229→   229→  const fillW = Math.max(SLIDER_TRACK_H, initFraction * trackW)
   230→   230→  const fillEl = makePlainRect(`${idPrefix}-fill`, { x: trackX, y: trackY, w: fillW, h: SLIDER_TRACK_H }, [...accentColor, 1], SLIDER_TRACK_H / 2)
   231→   231→  fillEl.isSliderFill = { groupId, trackX, trackW, knobW: SLIDER_KNOB_W, minW: 0 }
   232→   232→  fillEl.scroll = scroll
   233→   233→  elements.push(fillEl)
   234→   234→
   235→   235→  // Knob — frosted white at rest, glass when pressed (no highlight)
   236→   236→  const knobEl = makeGlassShape(
   237→   237→    `${idPrefix}-knob`,
   238→   238→    { x: knobX, y: knobY, w: SLIDER_KNOB_W, h: SLIDER_KNOB_H },
   239→   239→    {
   240→   240→      cornerRadius: SLIDER_KNOB_H / 2,
   241→   241→      refractionHeight: 10 * DP,
   242→   242→      refractionAmount: -14 * DP,
   243→   243→      blurRadius: 8 * DP,
   244→   244→      saturation: 1.0,
   245→   245→      surfaceColor: [0, 0, 0, 0],
   246→   246→      highlight: { mode: 1, color: [1, 1, 1], angle: Math.PI / 4, falloff: 1.0, alpha: 1.0, widthDp: 0.5 / 1.5, blurRadiusDp: 0.25 / 1.5 },
   247→   247→      outerShadow: { radius: 4 * DP, alpha: 0.05, offsetX: 0, offsetY: (4 / 6) * DP, color: [0, 0, 0] },
   248→   248→      innerShadow: { radius: 4 * DP, alpha: 0.3, offsetX: 0, offsetY: 4 * DP }, // 0.15 was too faint; 0.3 matches visual intent
   249→   249→      chromaticAberration: true,
   250→   250→    },
   251→   251→    scroll
   252→   252→  )
   253→   253→  knobEl.isToggleKnob = { groupId, dragWidth: dragW, velocityDivisor: 10 }
   254→   254→  knobEl.hitRect = { x: knobX, y: knobY + (SLIDER_KNOB_H - SLIDER_HIT_H) / 2, w: SLIDER_KNOB_W, h: SLIDER_HIT_H }
   255→   255→  elements.push(knobEl)
   256→   256→
   257→   257→  // Interactions — unified drag pattern via makeDragInteractions.
   258→   258→  const interact = makeDragInteractions({
   259→   259→    groupId, trackX, dragW, rendererRef, onValueChange, onLiveValue,
   260→   260→    ...sliderDragBindings,
   261→   261→    snap,
   262→   262→    liveUpdate,
   263→   263→  })
   264→   264→  interactions[`${idPrefix}-track`] = interact
   265→   265→  interactions[`${idPrefix}-knob`] = interact
   266→   266→
   267→   267→  return { elements, interactions }
   268→   268→}

   401→   401→/**
   402→   402→ * Tab drag interactions — faithful to LiquidBottomTabs.kt's
   403→   403→ * dampedDragAnimation gesture handling.
   404→   404→ *
   405→   405→ * The original uses DampedDragAnimation with:
   406→   406→ *   valueRange = 0..(tabsCount-1)
   407→   407→ *   pressedScale = 78/56
   408→   408→ *   onDrag → updateValue(targetValue + dragAmount.x / tabWidth)

   410→   410→ *
   411→   411→ * We reuse the renderer's toggle-group state (which supports custom
   412→   412→ * pressedScale via ensureToggleState) through the tab-specific API:
   413→   413→ *   beginTabDrag / dragTab / endTabDrag.
   414→   414→ */
   415→   415→export function makeTabDragInteractions(
   416→   416→  groupId: string,
   417→   417→  tabWidth: number,
   418→   418→  tabsCount: number,
   419→   419→  onSelect: (i: number) => void,
   420→   420→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null> | null
   421→   421→): ElementInteraction {
   422→   422→  // Module-level drag state (survives catalog rebuilds).
   423→   423→  if (!dragStates.has(groupId)) dragStates.set(groupId, { fraction: 0, x: 0, didDrag: false })
   424→   424→  const ds = dragStates.get(groupId)!
   425→   425→
   426→   426→  return {
   427→   427→    onTap: () => {
   428→   428→      // Tab taps handled by tab-text interactions, not here.
   429→   429→    },
   430→   430→    onDragStart: (pos) => {
   431→   431→      const r = rendererRef?.current
   432→   432→      if (!r) return
   433→   433→      draggingGroups.add(groupId)
   434→   434→      // Use TARGET index (not animated) — faithful to original's targetValue.
   435→   435→      ds.fraction = r.getTabTarget(groupId)
   436→   436→      ds.x = pos.x
   437→   437→      ds.didDrag = false
   438→   438→      r.beginTabDrag(groupId, ds.fraction, tabsCount)
   439→   439→    },
   440→   440→    onDrag: (pos) => {
   441→   441→      const r = rendererRef?.current
   442→   442→      if (!r) return
   443→   443→      if (Math.abs(pos.x - ds.x) > 3) ds.didDrag = true
   444→   444→      // Relative drag: indicator follows finger delta from drag start.
   445→   445→      // startTabIndex + (currentX - startX) / tabWidth, clamped to [0, count-1].
   446→   446→      r.dragTab(groupId, ds.fraction, pos.x, ds.x, tabWidth, tabsCount)
   447→   447→    },
   448→   448→    onDragEnd: () => {
   449→   449→      const r = rendererRef?.current
   450→   450→      if (!r) return
   451→   451→      // endTabDrag snaps to nearest integer, sets targetFraction, returns index.
   452→   452→      const finalIndex = r.endTabDrag(groupId, tabsCount)
   453→   453→      // Call onSelect AFTER endTabDrag (so targetFraction is already set).
   454→   454→      // Keep draggingGroups set during onSelect to prevent tabTargets effect
   455→   455→      // from calling setTabSelected (which would zero velocity).
   456→   456→      if (ds.didDrag) {
   457→   457→        onSelect(finalIndex)
   458→   458→      }
   459→   459→      draggingGroups.delete(groupId)
   460→   460→    },
   461→   461→  }
   462→   462→}

   751→   751→/* ------------------------------------------------------------------ *
   752→   752→ * applyVerticalCenter — offsets all element y positions (except the
   753→   753→ * back button, which stays top-left) so the content is vertically
   754→   754→ * centered within the viewport. Mirrors BackdropDemoScaffold's
   755→   755→ * `Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center)`.
   756→   756→ *
   757→   757→ * Returns the new contentHeight (= H if centering applied, since the
   758→   758→ * content now spans the full viewport visually).
   759→   759→ * ------------------------------------------------------------------ */
   760→   760→export function applyVerticalCenter(
   761→   761→  elements: GlassElementConfig[],
   762→   762→  contentTop: number,
   763→   763→  contentHeight: number,
   764→   764→  H: number
   765→   765→): number {
   766→   766→  const contentSize = contentHeight - contentTop
   767→   767→  if (contentSize >= H) return contentHeight
   768→   768→  const yOffset = Math.max(0, (H - contentSize) / 2 - contentTop)
   769→   769→  if (yOffset <= 0) return contentHeight
   770→   770→  for (const el of elements) {
   771→   771→    // Back button, theme button, and full-screen overlays (scroll=false) stay
   772→   772→    // at their fixed positions (not shifted by vertical centering).
   773→   773→    if (el.id === '__back__' || el.id === '__theme__') continue
   774→   774→    if (el.scroll === false && el.id !== '__pickimage__') continue
   775→   775→    el.rect = { ...el.rect, y: el.rect.y + yOffset }
   776→   776→    // Shift hitRect too (if set) so expanded touch targets follow the element.
   777→   777→    if (el.hitRect) {
   778→   778→      el.hitRect = { ...el.hitRect, y: el.hitRect.y + yOffset }
   779→   779→    }
   780→   780→    // Faithful fix: toggle knobs store the TRACK's original screen
   781→   781→    // position separately in `isToggleKnob.trackOriginalY` (used by the
   782→   782→    // renderer to compute the scaled track rect inside the knob's
   783→   783→    // CombinedBackdrop). Since the track element's rect.y was just
   784→   784→    // shifted by yOffset, we must shift trackOriginalY by the same
   785→   785→    // amount — otherwise the scaled track rect would be at the wrong Y
   786→   786→    // (off by yOffset * (1 - trackScaleY)), causing "no track visible
   787→   787→    // inside the knob" after vertical centering.
   788→   788→    if (el.isToggleKnob && el.isToggleKnob.trackOriginalY != null) {
   789→   789→      el.isToggleKnob.trackOriginalY += yOffset
   790→   790→    }
   791→   791→    // Bottom tab indicator stores the CONTAINER rect separately for its
   792→   792→    // CombinedBackdrop (the inset capsule SDF covers the container area).
   793→   793→    // Shift it by the same yOffset so the SDF stays aligned.
   794→   794→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.containerRect) {
   795→   795→      el.isBottomTabIndicator.containerRect = {
   796→   796→        ...el.isBottomTabIndicator.containerRect,
   797→   797→        y: el.isBottomTabIndicator.containerRect.y + yOffset,
   798→   798→      }
   799→   799→    }
   800→   800→    // Bottom tab content stores the CONTAINER center (scale origin) separately.
   801→   801→    // Shift it by the same yOffset so the scale pivot stays aligned with the
   802→   802→    // actual container position after vertical centering.
   803→   803→    if (el.isBottomTabContent && el.isBottomTabContent.containerCenterY != null) {
   804→   804→      el.isBottomTabContent.containerCenterY += yOffset
   805→   805→    }
   806→   806→    // Bottom tab indicator also scales around the container center — shift
   807→   807→    // its pivot too.
   808→   808→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.containerCenterY != null) {
   809→   809→      el.isBottomTabIndicator.containerCenterY += yOffset
   810→   810→    }
   811→   811→    // Shift tab content rects (for blue tint mask) by yOffset too.
   812→   812→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.tabContentRects) {
   813→   813→      el.isBottomTabIndicator.tabContentRects = el.isBottomTabIndicator.tabContentRects.map(r => ({
   814→   814→        ...r, y: r.y + yOffset,
   815→   815→      }))
   816→   816→    }
   817→   817→  }
   818→   818→  return contentHeight + yOffset
   819→   819→}

   821→   821→

    63→/* ------------------------------------------------------------------ *
    64→ * ProgressConverter — faithful port of ProgressConverter.kt
    65→ *   convert(p) = (1 - exp(-|p|)) * sign(p)
    66→ *
    67→ * This dampens overscroll exponentially: as |p| grows, the converted
    68→ * value approaches ±1 asymptotically. Used by ControlCenterContent.kt's
    69→ * derivedStateOf to map raw enterProgress → visual progress:
    70→ *   p < 0  → convert(p)              (approaches -1)
    71→ *   0..1   → p                        (linear)
    72→ *   p > 1  → 1 + convert(p - 1)      (approaches 2)
    73→ * ------------------------------------------------------------------ */
    74→export function convertProgress(p: number): number {
    75→  return (1 - Math.exp(-Math.abs(p))) * Math.sign(p)
    76→}
    77→export function derivedProgress(p: number): number {
    78→  if (p < 0) return convertProgress(p)
    79→  if (p <= 1) return p
    80→  return 1 + convertProgress(p - 1)
    81→}

    83→/** Animate controlCenterEnter (raw) AND controlCenterSafeEnter (clamped)
    84→ *  to `target` (0 or 1) via springs.
    85→ *
    86→ *  Faithful to ControlCenterContent.kt onDragStopped:
    87→ *    enterProgressAnimation.animateTo(
    88→ *        target,
    89→ *        if (target > 0.5) spring(0.5, 300, 0.5/maxDragHeight)  // underdamped
    90→ *        else              spring(1.0, 300, 0.01),               // critical
    91→ *        velocity / maxDragHeight                                 // initial vel
    92→ *    )
    93→ *    safeEnterProgressAnimation.animateTo(
    94→ *        target, spring(1.0, 300, 0.01)   // always critical, NO initial vel
    95→ *    )
    96→ *
    97→ *  The raw spring BOUNCES on expand (underdamped ζ=0.5) giving the
    98→ *  characteristic overshoot when releasing an over-pulled control center.
    99→ *  The safe spring settles smoothly (critical ζ=1.0) so alpha/dim/blur
   100→ *  never overshoot.
   101→ *
   102→ *  Uses the ANALYTICAL spring solution (not numerical integration) for
   103→ *  exact Compose-spring behavior at any frame rate. */
   104→export function animateControlCenterEnter(
   105→  setState: (patch: Partial<CatalogState> | ((prev: CatalogState) => Partial<CatalogState>)) => void,
   106→  target: number,
   107→  maxDrag: number,
   108→  initialVelocity: number // progress/s — already converted from px/s
   109→) {
   110→  if (ccAnim.handle != null) cancelAnimationFrame(ccAnim.handle)
   111→
   112→  // Spring params (faithful to Compose spring(dampingRatio, stiffness, visibilityThreshold))
   113→  const stiffness = 300
   114→  const omega0 = Math.sqrt(stiffness) // natural frequency (m=1)
   115→  const dampingRatioRaw = target > 0.5 ? 0.5 : 1.0 // underdamped for expand, critical for collapse
   116→  const dampingRatioSafe = 1.0 // always critical
   117→  // Visibility thresholds — the spring is considered settled below these.
   118→  const visThresholdRaw = target > 0.5 ? 0.5 / maxDrag : 0.01
   119→  const visThresholdSafe = 0.01
   120→
   121→  let posRaw = -1
   122→  let posSafe = -1
   123→  let velRaw = initialVelocity
   124→  let velSafe = 0
   125→  let lastT = -1
   126→
   127→  /** Analytical spring step. Given current pos, vel, and dt, computes
   128→   *  the exact new pos and vel after dt. Uses the closed-form solution
   129→   *  of the damped harmonic oscillator. */
   130→  function springStep(
   131→    pos: number, vel: number, tgt: number,
   132→    ratio: number, dt: number
   133→  ): { p: number; v: number } {
   134→    const A = pos - tgt
   135→    if (ratio >= 1) {
   136→      // Critically damped: y(t) = (A + B*t) * e^(-ω₀t)
   137→      //   A = pos - target, B = vel + ω₀*A
   138→      const B = vel + omega0 * A
   139→      const ed = Math.exp(-omega0 * dt)
   140→      const newPos = tgt + (A + B * dt) * ed
   141→      const newVel = (vel - omega0 * B * dt) * ed
   142→      return { p: newPos, v: newVel }
   143→    }
   144→    // Underdamped: y(t) = e^(-ζω₀t) * (A*cos(ωd*t) + B*sin(ωd*t))
   145→    //   A = pos - target, B = (vel + ζω₀*A) / ωd
   146→    const omegaD = omega0 * Math.sqrt(1 - ratio * ratio)
   147→    const B = (vel + ratio * omega0 * A) / omegaD
   148→    const ed = Math.exp(-ratio * omega0 * dt)
   149→    const cd = Math.cos(omegaD * dt)
   150→    const sd = Math.sin(omegaD * dt)
   151→    const newPos = tgt + ed * (A * cd + B * sd)
   152→    const newVel = ed * ((B * omegaD - ratio * omega0 * A) * cd - (A * omegaD + ratio * omega0 * B) * sd)
   153→    return { p: newPos, v: newVel }
   154→  }
   155→
   156→  const step = (now: number) => {
   157→    setState((prev) => {
   158→      if (posRaw < 0) posRaw = prev.controlCenterEnter
   159→      if (posSafe < 0) posSafe = prev.controlCenterSafeEnter
   160→      let dt: number
   161→      if (lastT < 0) {
   162→        dt = 1 / 60
   163→      } else {
   164→        dt = Math.min(0.05, (now - lastT) / 1000)
   165→      }
   166→      lastT = now
   167→
   168→      // Analytical spring step — exact solution, no integration error.
   169→      const r = springStep(posRaw, velRaw, target, dampingRatioRaw, dt)
   170→      posRaw = r.p
   171→      velRaw = r.v
   172→      const s = springStep(posSafe, velSafe, target, dampingRatioSafe, dt)
   173→      posSafe = s.p
   174→      velSafe = s.v
   175→
   176→      const doneRaw = Math.abs(target - posRaw) < visThresholdRaw && Math.abs(velRaw) < visThresholdRaw * 10
   177→      const doneSafe = Math.abs(target - posSafe) < visThresholdSafe && Math.abs(velSafe) < visThresholdSafe * 10
   178→
   179→      if (doneRaw && doneSafe) {
   180→        ccAnim.handle = null
   181→        ccAnim.lastVelocity = 0
   182→        return { controlCenterEnter: target, controlCenterSafeEnter: target }
   183→      }
   184→      ccAnim.handle = requestAnimationFrame(step)
   185→      return { controlCenterEnter: posRaw, controlCenterSafeEnter: posSafe }
   186→    })
   187→  }
   188→  ccAnim.handle = requestAnimationFrame(step)
   189→}

   623→/* ------------------------------------------------------------------ *
   624→ * Text-measurement helper (hidden 2D canvas).
   625→ * ------------------------------------------------------------------ */
   626→let _measureCtx: CanvasRenderingContext2D | null = null
   627→export function measureTextWidth(text: string, fontPx: number, weight = 400): number {
   628→  if (typeof document !== 'undefined') {
   629→    if (!_measureCtx) {
   630→      const c = document.createElement('canvas')
   631→      _measureCtx = c.getContext('2d')
   632→    }
   633→    if (_measureCtx) {
   634→      _measureCtx.font = `${weight} ${fontPx}px ${FONT_FAMILY}`
   635→      return _measureCtx.measureText(text).width
   636→    }
   637→  }
   638→  return text.length * fontPx * 0.55
   639→}

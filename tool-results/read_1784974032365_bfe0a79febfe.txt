    31→type Theme = 'light' | 'dark'

    33→function useSystemTheme(): Theme {
    34→  const [theme, setTheme] = React.useState<Theme>('light')
    35→  React.useEffect(() => {
    36→    if (typeof window === 'undefined' || !window.matchMedia) return
    37→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    38→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    39→    update()
    40→    mq.addEventListener('change', update)
    41→    return () => mq.removeEventListener('change', update)
    42→  }, [])
    43→  return theme
    44→}

   268→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   269→    const MIN_FPS = 55
   270→    if (fps >= MIN_FPS) {
   271→      // This DPR works — try higher
   272→      perfLoRef.current = candidateDpr
   273→    } else {
   274→      // Too slow — try lower
   275→      perfHiRef.current = candidateDpr
   276→    }
   277→
   278→    if (iteration >= maxIterations || perfHiRef.current - perfLoRef.current <= 0.25) {
   279→      // Converged — use the highest DPR that worked (lo)
   280→      const bestDpr = Math.max(0.5, Math.min(deviceDpr, Math.round(perfLoRef.current * 4) / 4))
   281→      perfPhaseRef.current = 'done'
   282→      const isGood = bestDpr >= deviceDpr * 0.5
   283→      const statusText = isGood
   284→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   285→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   286→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   287→      setState({
   288→        customDpr: bestDpr,
   289→        perfDone: true,
   290→        perfResultDpr: bestDpr,
   291→        perfStatusText: statusText,
   292→        perfProgress: null,
   293→        // Don't reset perfGlassAngle — settle animation will handle it
   294→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   295→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   296→        perfProgressFrac: 1,
   297→      })
   298→    } else {
   299→      // More iterations needed — increment trigger to re-fire the effect
   300→      perfPhaseRef.current = 'idle'
   301→      setState((prev) => ({
   302→        perfProgress: 'running',
   303→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   304→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   305→      }))
   306→    }
   307→  }

   337→  React.useEffect(() => {

   339→    if (!rendererReady) return

   346→    let elapsed = 0
   347→    let lastT = 0
   348→    let raf = 0
   349→    // Exponential decay rate for deformMul: mul = e^(-3*t), reaches ~0.05 by 1s
   350→    const DECAY_RATE = 3
   351→    // Exit button: starts appearing after 0.3s, fully visible by 0.8s
   352→    const EXIT_DELAY = 0.3
   353→    const EXIT_DURATION = 0.5
   354→
   355→    const tick = (timestamp: number) => {
   356→      if (!lastT) lastT = timestamp

   358→      lastT = timestamp

   360→

   362→      angle += ORBIT_SPEED * dt
   363→
   364→      // Exponential decay: deformMul goes from 1 → ~0 over ~1 second
   365→      const mul = Math.exp(-DECAY_RATE * elapsed)
   366→
   367→      // Exit button: ease-out quadratic after delay
   368→      let exitProg = 0
   369→      if (elapsed > EXIT_DELAY) {
   370→        const t = Math.min(1, (elapsed - EXIT_DELAY) / EXIT_DURATION)
   371→        exitProg = 1 - (1 - t) * (1 - t) // ease-out
   372→      }
   373→
   374→      setState({ perfGlassAngle: angle, perfDeformMul: mul, perfExitProgress: exitProg })
   375→
   376→      if (mul <= 0.01 && exitProg >= 0.99) {
   377→        // Settle complete — glass angle irrelevant at mul=0
   378→        setState({ perfDeformMul: 0, perfExitProgress: 1 })
   379→        return
   380→      }
   381→      raf = requestAnimationFrame(tick)
   382→    }
   383→    raf = requestAnimationFrame(tick)
   384→    return () => cancelAnimationFrame(raf)

   396→  // setState supports both a partial patch and a functional updater.

   434→  // Transition animation uses 3 phases for directional slide:
   435→  //   fadeOut   — old content slides out + fades to 0
   436→  //   prepIn    — instantly place new content at opposite offset + opacity 0 (no transition)
   437→  //   fadeIn    — animate new content from offset → center + opacity 0 → 1
   438→  const [transPhase, setTransPhase] = React.useState<'idle' | 'fadeOut' | 'prepIn' | 'fadeIn'>('idle')
   439→  const transDirRef = React.useRef<'enter' | 'exit'>('enter')
   440→  const pendingDestRef = React.useRef<CatalogDestination | null>(null)
   441→  const TRANSITION_MS = 200 // duration for each phase (fade out / fade in)
   442→  const OFFSET_PX = 16 // slide distance in px

   444→  const onNavigate = React.useCallback((d: CatalogDestination) => {

   447→    } else {

   449→      pendingDestRef.current = d
   450→      transDirRef.current = 'enter'
   451→      setTransPhase('fadeOut')
   452→    }

   455→    }
   456→  }, [state.pageTransition])

   459→    const target = CatalogDestination.Home
   460→    if (!state.pageTransition) {
   461→      setDestination(target)
   462→    } else {
   463→      // Exit: old page slides RIGHT out, new page slides in from LEFT.
   464→      pendingDestRef.current = target
   465→      transDirRef.current = 'exit'

   467→    }
   468→    if (typeof window !== 'undefined' && window.history.state?.dest !== undefined) {
   469→      window.history.back()
   470→    }
   471→  }, [state.pageTransition])

   473→  // Phase progression: fadeOut → prepIn → fadeIn → idle
   474→  React.useEffect(() => {
   475→    if (transPhase === 'fadeOut') {
   476→      const timer = setTimeout(() => {
   477→        const dest = pendingDestRef.current ?? CatalogDestination.Home
   478→        setDestination(dest)
   479→        pendingDestRef.current = null
   480→        // prepIn: place new content at opposite offset with no transition
   481→        setTransPhase('prepIn')
   482→      }, TRANSITION_MS)
   483→      return () => clearTimeout(timer)
   484→    }
   485→    if (transPhase === 'prepIn') {
   486→      // After React renders the offset position (1 frame), start animated fadeIn
   487→      requestAnimationFrame(() => {
   488→        requestAnimationFrame(() => {
   489→          setTransPhase('fadeIn')
   490→        })
   491→      })
   492→      return
   493→    }
   494→    if (transPhase === 'fadeIn') {
   495→      const timer = setTimeout(() => {

   554→  // Device motion → gravity angle for glass highlight direction.
   555→  // Faithful to UISensor.kt:
   556→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   557→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   558→  //
   559→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   560→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   561→  // rebuilding the catalog (which would lose drag state and cause jank).
   562→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   563→  // renderer.gravityAngle live each frame.
   564→  //
   565→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   566→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   567→  // device's acceleration INCLUDING gravity in device-space axes:
   568→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   569→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   570→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   571→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   572→  // vectors — atan2(beta, gamma) does NOT match the original.)
   573→  //
   574→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   575→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   576→  // Default 45° (matches UISensor.kt's initial value).
   577→  React.useEffect(() => {
   578→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   579→    let smoothed = 45
   580→    const alpha = 0.5
   581→    const handler = (e: DeviceMotionEvent) => {
   582→      const acc = e.accelerationIncludingGravity
   583→      if (!acc || acc.x == null || acc.y == null) return
   584→      const x = acc.x
   585→      const y = acc.y
   586→      let target = Math.atan2(y, x) * 180 / Math.PI
   587→      // Shortest-path interpolation: wrap the delta to [-180, 180] so
   588→      // rotating through ±180° doesn't cause a 358° jump.
   589→      let delta = target - smoothed
   590→      while (delta > 180) delta -= 360
   591→      while (delta < -180) delta += 360
   592→      smoothed += delta * alpha
   593→      // Wrap smoothed to [-180, 180] to keep the value bounded.
   594→      while (smoothed > 180) smoothed -= 360
   595→      while (smoothed < -180) smoothed += 360
   596→      const rad = smoothed * Math.PI / 180
   597→      rendererRef.current?.setGravityAngle(rad)
   598→    }
   599→    window.addEventListener('devicemotion', handler)
   600→    return () => window.removeEventListener('devicemotion', handler)
   601→  }, [rendererRef])

     1→     1→import type { LiquidGlassRenderer } from './index'
     2→     2→import type { GlassRenderState } from './methods-render-glass'
     3→     3→import { DP } from './spring'
     4→     4→import { continuousCurvatureRoundedRectPath } from './continuous-curve'
     5→     5→
     6→     6→declare module './index' {
     7→     7→  interface LiquidGlassRenderer {
     8→     8→    renderGlassPostPasses(state: GlassRenderState): void
     9→     9→  }
    10→    10→}
    11→    11→
    12→    12→export const glassPostPassMethods = {
    13→    13→  /** Steps 2b–2f: Inner shadow, press glow, white overlay, foreground, rim highlight.
    14→    14→   *  These all composite on top of the glass body (already drawn to
    15→    15→   *  otherFbo by renderGlassElementPass). */
    16→    16→  renderGlassPostPasses(this: LiquidGlassRenderer, state: GlassRenderState) {
    17→    17→    const gl = this.gl
    18→    18→    const { el, st, isButton, p, sx, sy, sw, sh, radii, togglePressProgress, elHighlightAlpha } = state
    19→    19→
    20→    20→    // Original-space SDF uniforms — shared by all post-pass shaders so their
    21→    21→    // capsule clip is computed in ORIGINAL space (faithful to graphicsLayer
    22→    22→    // { scaleX, scaleY } post-scaling). See element.ts / highlight.ts.
    23→    23→    const origSizeX = state.origW * this.dpr
    24→    24→    const origSizeY = state.origH * this.dpr
    25→    25→    const origRadius = state.origCornerRadius * this.dpr
    26→    26→    const layerScaleX = state.layerScaleX
    27→    27→    const layerScaleY = state.layerScaleY
    28→    28→
    29→    29→    // --- Step 2b: Inner shadow post-passes (Canvas2D ring mask approach) ---
    30→    30→    // Inner shadows are drawn on the element surface, underneath the press glow.
    31→    31→    // Each shadow uses a Canvas2D-generated blurred ring mask (fill shape →
    32→    32→    // destination-out offset shape → blur), composited via the
    33→    33→    // INNER_SHADOW_MASK_COMPOSITE_FRAGMENT_SHADER with SrcOver blend.
    34→    34→    // SDF clip in the shader ensures the shadow stays inside the shape boundary.
    35→    35→
    36→    36→    /** Helper: draw one inner shadow post-pass (shadow1 or shadow2). */
    37→    37→    const drawInnerShadowPass = (
    38→    38→      shadowCfg: { radius: number; alpha: number; offsetX: number; offsetY: number; color?: [number, number, number] },
    39→    39→      shadowIndex: number // 0 = innerShadow, 1 = innerShadow2
    40→    40→    ) => {
    41→    41→      // Progress modulation — faithful to the original inline shader:
    42→    42→      //   - Toggle knobs: radius, alpha, offset all modulated by togglePressProgress
    43→    43→      //   - Bottom-tab indicator: radius, alpha, offset modulated by togglePressProgress
    44→    44→      //   - Other elements (magnifier): static params, no modulation
    45→    45→      const progress = (el.isToggleKnob || el.isBottomTabIndicator) ? togglePressProgress : 1
    46→    46→      let shadowAlpha = shadowCfg.alpha * progress * state.enterAlpha
    47→    47→      let shadowRadius = shadowCfg.radius * progress
    48→    48→      let shadowOffsetX = shadowCfg.offsetX * progress
    49→    49→      let shadowOffsetY = shadowCfg.offsetY * progress
    50→    50→
    51→    51→      if (shadowAlpha <= 0.001 || shadowRadius <= 0.5) return
    52→    52→
    53→    53→      // Blur sigma = radius / 3 (BlurMaskFilter semantics: sigma = radius / 3)
    54→    54→      const blurSigma = (shadowRadius / 3) * this.dpr  // device px
    55→    55→      // Margin for blur spread + AA
    56→    56→      const margin = Math.ceil(blurSigma * 3) + 2
    57→    57→      // Mask dimensions in device px (origSize + 2*margin)
    58→    58→      const maskW = Math.max(1, Math.ceil(origSizeX + 2 * margin))
    59→    59→      const maskH = Math.max(1, Math.ceil(origSizeY + 2 * margin))
    60→    60→      // Supersampling for sharper mask rasterization
    61→    61→      const deviceDpr = window.devicePixelRatio || 1
    62→    62→      const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
    63→    63→      const canvasW = maskW * SS
    64→    64→      const canvasH = maskH * SS
    65→    65→      const useG2 = !!el.useContinuousSdf
    66→    66→
    67→    67→      // Offset in device px (already × progress)
    68→    68→      const offsetXDp = shadowOffsetX * this.dpr
    69→    69→      const offsetYDp = shadowOffsetY * this.dpr
    70→    70→
    71→    71→      const maskKey = [
    72→    72→        'is',
    73→    73→        shadowIndex,
    74→    74→        useG2 ? 'g2' : 'rr',
    75→    75→        origSizeX.toFixed(3),
    76→    76→        origSizeY.toFixed(3),
    77→    77→        origRadius.toFixed(3),
    78→    78→        offsetXDp.toFixed(3),
    79→    79→        offsetYDp.toFixed(3),
    80→    80→        blurSigma.toFixed(3),
    81→    81→        margin,
    82→    82→        maskW,
    83→    83→        maskH,
    84→    84→        `ss${SS}`,
    85→    85→      ].join(':')
    86→    86→
    87→    87→      let mask = this.innerShadowMaskCache.get(maskKey)
    88→    88→      if (!mask) {
    89→    89→        const canvas = document.createElement('canvas')
    90→    90→        canvas.width = canvasW
    91→    91→        canvas.height = canvasH
    92→    92→        const ctx = canvas.getContext('2d', { alpha: true })
    93→    93→        if (!ctx) throw new Error('2D canvas not supported')
    94→    94→        const tex = gl.createTexture()
    95→    95→        if (!tex) throw new Error('WebGL texture allocation failed')
    96→    96→        mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
    97→    97→        this.innerShadowMaskCache.set(maskKey, mask)
    98→    98→
    99→    99→        // Keep the cache bounded (32 entries)
   100→   100→        if (this.innerShadowMaskCache.size > 32) {
   101→   101→          const oldestKey = this.innerShadowMaskCache.keys().next().value as string | undefined
   102→   102→          if (oldestKey && oldestKey !== maskKey) {
   103→   103→            const oldest = this.innerShadowMaskCache.get(oldestKey)
   104→   104→            if (oldest) gl.deleteTexture(oldest.tex)
   105→   105→            this.innerShadowMaskCache.delete(oldestKey)
   106→   106→          }
   107→   107→        }
   108→   108→      }
   109→   109→
   110→   110→      if (!mask.ready) {
   111→   111→        // --- Generate the ring mask via Canvas2D (two-canvas approach) ---
   112→   112→        // Step 1: draw hard-edge ring on a temp canvas (with SS× supersampling)
   113→   113→        // Step 2: blur the ring onto the output canvas at 1:1 physical pixels
   114→   114→        //         (no ctx.scale — drawImage and blur must be in physical-px space)
   115→   115→        //   - blur radius = blurSigma * SS physical px (= blurSigma logical px)
   116→   116→        //   - drawImage at 1:1 physical (canvasW × canvasH → canvasW × canvasH)
   117→   117→
   118→   118→        // --- Temp canvas for the hard-edge ring ---
   119→   119→        const tempCanvas = document.createElement('canvas')
   120→   120→        tempCanvas.width = canvasW
   121→   121→        tempCanvas.height = canvasH
   122→   122→        const tCtx = tempCanvas.getContext('2d', { alpha: true })
   123→   123→        if (!tCtx) throw new Error('2D canvas not supported')
   124→   124→
   125→   125→        // Draw ring on temp canvas: fill shape → destination-out offset shape
   126→   126→        // Scale(SS, SS) for supersampling: draw in logical (device-px) coordinates
   127→   127→        // while the physical canvas is SS× larger.
   128→   128→        tCtx.save()
   129→   129→        tCtx.scale(SS, SS)
   130→   130→        tCtx.translate(margin, margin)
   131→   131→
   132→   132→        // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   133→   133→        let path: Path2D
   134→   134→        if (useG2) {
   135→   135→          path = continuousCurvatureRoundedRectPath(tCtx, origSizeX, origSizeY, origRadius)
   136→   136→        } else {
   137→   137→          path = new Path2D()
   138→   138→          const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   139→   139→          path.moveTo(r, 0)
   140→   140→          path.lineTo(origSizeX - r, 0)
   141→   141→          path.arcTo(origSizeX, 0, origSizeX, r, r)
   142→   142→          path.lineTo(origSizeX, origSizeY - r)
   143→   143→          path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   144→   144→          path.lineTo(r, origSizeY)
   145→   145→          path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   146→   146→          path.lineTo(0, r)
   147→   147→          path.arcTo(0, 0, r, 0, r)
   148→   148→          path.closePath()
   149→   149→        }
   150→   150→
   151→   151→        // Fill the shape with white (creates full interior)
   152→   152→        tCtx.globalCompositeOperation = 'source-over'
   153→   153→        tCtx.fillStyle = 'white'
   154→   154→        tCtx.fill(path)
   155→   155→
   156→   156→        // Draw the OFFSET shape with destination-out (removes offset interior)
   157→   157→        tCtx.globalCompositeOperation = 'destination-out'
   158→   158→        tCtx.save()
   159→   159→        tCtx.translate(offsetXDp, offsetYDp)
   160→   160→        tCtx.fill(path) // same path, shifted by offset
   161→   161→        tCtx.restore()
   162→   162→
   163→   163→        tCtx.globalCompositeOperation = 'source-over'
   164→   164→        tCtx.restore()
   165→   165→
   166→   166→        // --- Blur the ring onto the output canvas (1:1 physical pixels) ---
   167→   167→        // NO ctx.scale here — drawImage and ctx.filter blur must operate in
   168→   168→        // physical-pixel space to avoid SS× size distortion.
   169→   169→        //   blurSigma is in device px (logical); the SS× canvas has SS physical px
   170→   170→        //   per logical px, so the physical blur radius = blurSigma * SS.
   171→   171→        const mCtx = mask.ctx
   172→   172→        mCtx.clearRect(0, 0, canvasW, canvasH)
   173→   173→        if (blurSigma > 0.01) {
   174→   174→          mCtx.filter = `blur(${blurSigma * SS}px)`
   175→   175→        } else {
   176→   176→          mCtx.filter = 'none'
   177→   177→        }
   178→   178→        // 1:1 physical-pixel drawImage: temp(canvasW×canvasH) → mask(canvasW×canvasH)
   179→   179→        mCtx.drawImage(tempCanvas, 0, 0)
   180→   180→        mCtx.filter = 'none'
   181→   181→
   182→   182→        // Upload to GPU texture
   183→   183→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   184→   184→        gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   185→   185→        gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   186→   186→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   187→   187→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   188→   188→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   189→   189→        gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   190→   190→        mask.ready = true
   191→   191→      }
   192→   192→
   193→   193→      // --- Composite: inner shadow mask × shadowAlpha × shadowColor → scene ---
   194→   194→      gl.enable(gl.BLEND)
   195→   195→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   196→   196→      gl.useProgram(this.innerShadowMaskCompositeProgram)
   197→   197→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   198→   198→      gl.enableVertexAttribArray(this.aPosLocIs)
   199→   199→      gl.vertexAttribPointer(this.aPosLocIs, 2, gl.FLOAT, false, 0, 0)
   200→   200→
   201→   201→      gl.uniform2f(this.uIs['uCanvasSize'], this.canvas.width, this.canvas.height)
   202→   202→      gl.uniform2f(this.uIs['uOffset'], sx * this.dpr, sy * this.dpr)
   203→   203→      gl.uniform2f(this.uIs['uSize'], sw * this.dpr, sh * this.dpr)
   204→   204→      gl.uniform4f(this.uIs['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   205→   205→
   206→   206→      gl.activeTexture(gl.TEXTURE0)
   207→   207→      gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   208→   208→      gl.uniform1i(this.uIs['uInnerShadowMask'], 0)
   209→   209→      // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   210→   210→      // physical canvas is SS× larger but the shader uses 1x coords for UV.
   211→   211→      gl.uniform2f(this.uIs['uMaskOffset'], margin, margin)
   212→   212→      gl.uniform2f(this.uIs['uMaskSize'], mask.w, mask.h)
   213→   213→
   214→   214→      // Shadow color (defaults to black [0,0,0] if not specified)
   215→   215→      const color = shadowCfg.color ?? [0, 0, 0]
   216→   216→      gl.uniform3f(this.uIs['uInnerShadowColor'], color[0], color[1], color[2])
   217→   217→      gl.uniform1f(this.uIs['uInnerShadowAlpha'], shadowAlpha)
   218→   218→
   219→   219→      gl.uniform2f(this.uIs['uOriginalSize'], origSizeX, origSizeY)
   220→   220→      gl.uniform1f(this.uIs['uOriginalCornerRadius'], origRadius)
   221→   221→      gl.uniform2f(this.uIs['uLayerScale'], layerScaleX, layerScaleY)
   222→   222→      gl.uniform1f(this.uIs['uElementRotation'], state.elementRotation)
   223→   223→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   224→   224→    }
   225→   225→
   226→   226→    // Inner shadow 1 (typically dark/black, offset downward)
   227→   227→    if (el.innerShadow) {
   228→   228→      drawInnerShadowPass(el.innerShadow, 0)
   229→   229→    }
   230→   230→    // Inner shadow 2 (typically white/bright, offset upward — 3D bevel)
   231→   231→    if (el.innerShadow2) {
   232→   232→      drawInnerShadowPass(el.innerShadow2, 1)
   233→   233→    }
   234→   234→
   235→   235→    // --- Step 2c: Press glow (button + bottom-tab container) ---
   236→   236→    // Faithful to InteractiveHighlight.kt: a flat white Plus-blend overlay
   237→   237→    // + a radial white glow. For buttons, position = finger, intensity = st.pressProgress.
   238→   238→    // For the bottom-tab container, position = indicator center (via the
   239→   239→    // original's position lambda), intensity = toggleState.pressProgress.
   240→   240→    const isContainer = !!el.isBottomTabContainer
   241→   241→    const glowP = isButton ? p : (isContainer ? togglePressProgress : 0)
   242→   242→    if ((isButton && el.isInteractive && st && p > 0.001) || (isContainer && togglePressProgress > 0.001)) {
   243→   243→      // a. Flat white overlay
   244→   244→      gl.useProgram(this.tintProgram)
   245→   245→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   246→   246→      gl.enableVertexAttribArray(this.aPosLocTn)
   247→   247→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   248→   248→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE)
   249→   249→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   250→   250→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   251→   251→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   252→   252→      gl.uniform4f(
   253→   253→        this.uTn['uCornerRadii'],
   254→   254→        radii[0] * this.dpr,
   255→   255→        radii[1] * this.dpr,
   256→   256→        radii[2] * this.dpr,
   257→   257→        radii[3] * this.dpr
   258→   258→      )
   259→   259→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   260→   260→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   261→   261→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   262→   262→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   263→   263→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   264→   264→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, 0.08 * glowP)
   265→   265→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   266→   266→
   267→   267→      // b. Radial highlight at finger position (button) or indicator center (container).
   268→   268→      gl.useProgram(this.highlightProgram)
   269→   269→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   270→   270→      gl.enableVertexAttribArray(this.aPosLocHl)
   271→   271→      gl.vertexAttribPointer(this.aPosLocHl, 2, gl.FLOAT, false, 0, 0)
   272→   272→      gl.blendFunc(gl.ONE, gl.ONE)
   273→   273→      gl.uniform2f(this.uHl['uCanvasSize'], this.canvas.width, this.canvas.height)
   274→   274→      gl.uniform2f(this.uHl['uOffset'], sx * this.dpr, sy * this.dpr)
   275→   275→      gl.uniform2f(this.uHl['uSize'], sw * this.dpr, sh * this.dpr)
   276→   276→      gl.uniform4f(
   277→   277→        this.uHl['uCornerRadii'],
   278→   278→        radii[0] * this.dpr,
   279→   279→        radii[1] * this.dpr,
   280→   280→        radii[2] * this.dpr,
   281→   281→        radii[3] * this.dpr
   282→   282→      )
   283→   283→      gl.uniform2f(this.uHl['uOriginalSize'], origSizeX, origSizeY)
   284→   284→      gl.uniform1f(this.uHl['uOriginalCornerRadius'], origRadius)
   285→   285→      gl.uniform2f(this.uHl['uLayerScale'], layerScaleX, layerScaleY)
   286→   286→      gl.uniform1f(this.uHl['uElementRotation'], state.elementRotation)
   287→   287→      gl.uniform1f(this.uHl['uCornerStyle'], this.cornerStyle)
   288→   288→      gl.uniform4f(this.uHl['uColor'], 1, 1, 1, 0.15 * glowP)
   289→   289→      const minDim = Math.min(sw, sh) * this.dpr
   290→   290→      gl.uniform1f(this.uHl['uRadius'], minDim * 1.5)
   291→   291→      // Position: for buttons, the finger position. For the container, the
   292→   292→      // indicator center — faithful to the original's position lambda which
   293→   293→      // returns (dampedDragAnimation.value + 0.5) * tabWidth (indicator center
   294→   294→      // relative to the container).
   295→   295→      let px: number, py: number
   296→   296→      if (isContainer) {
   297→   297→        const tg = this.toggleStates.get(el.isBottomTabContainer!.groupId)
   298→   298→        // tabW = container ORIGINAL width / tabsCount (not scaled — the position
   299→   299→        // lambda runs in pre-scale local coords). indicator center =
   300→   300→        // (fraction + 0.5) * tabW. Faithful to LiquidBottomTabs.kt:
   301→   301→        //   position = (dampedDragAnimation.value + 0.5) * tabWidth
   302→   302→        const tabsCount = el.isBottomTabContainer!.tabsCount ?? 4
   303→   303→        const tabW = el.rect.w / tabsCount
   304→   304→        const fraction = tg ? tg.fraction : 0
   305→   305→        const indCenterX = (fraction + 0.5) * tabW
   306→   306→        // Map original-local to scaled-local (the shader's uPosition is in
   307→   307→        // scaled-local coords, 0..sw). scale = sw / el.rect.w.
   308→   308→        const scaleToLocal = sw / el.rect.w
   309→   309→        px = Math.max(0, Math.min(sw, indCenterX * scaleToLocal)) * this.dpr
   310→   310→        py = (sh / 2) * this.dpr
   311→   311→      } else {
   312→   312→        px = Math.max(0, Math.min(sw, st!.dragX * state.layerScaleX)) * this.dpr
   313→   313→        py = Math.max(0, Math.min(sh, st!.dragY * state.layerScaleY)) * this.dpr
   314→   314→      }
   315→   315→      gl.uniform2f(this.uHl['uPosition'], px, py)
   316→   316→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   317→   317→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   318→   318→    }
   319→   319→
   320→   320→    // --- Step 2d: Toggle knob white overlay (faithful to LiquidToggle.kt
   321→   321→    // / LiquidSlider.kt onDrawSurface) ---
   322→   322→    if (el.isToggleKnob && togglePressProgress < 0.999) {
   323→   323→      const whiteAlpha = 1.0 * (1 - togglePressProgress)
   324→   324→      gl.useProgram(this.tintProgram)
   325→   325→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   326→   326→      gl.enableVertexAttribArray(this.aPosLocTn)
   327→   327→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   328→   328→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   329→   329→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   330→   330→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   331→   331→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   332→   332→      gl.uniform4f(
   333→   333→        this.uTn['uCornerRadii'],
   334→   334→        radii[0] * this.dpr,
   335→   335→        radii[1] * this.dpr,
   336→   336→        radii[2] * this.dpr,
   337→   337→        radii[3] * this.dpr
   338→   338→      )
   339→   339→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   340→   340→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   341→   341→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   342→   342→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   343→   343→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   344→   344→      // Faithful to LiquidToggle.kt onDrawSurface:
   345→   345→      //   drawRect(Color.White.copy(alpha = 1f - progress))
   346→   346→      // Solid white pebble at rest (alpha=1), fading to transparent when
   347→   347→      // pressed (alpha=0) to reveal the glass refraction beneath.
   348→   348→      gl.uniform4f(this.uTn['uColor'], 1, 1, 1, whiteAlpha)
   349→   349→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   350→   350→    }
   351→   351→
   352→   352→    // --- Step 2d2: Bottom tab indicator onDrawSurface (faithful to
   353→   353→    // LiquidBottomTabs.kt indicator):
   354→   354→    //   drawRect(if (isLightTheme) Color.Black.copy(0.1f) else Color.White.copy(0.1f), alpha = 1f - progress)
   355→   355→    //   drawRect(Color.Black.copy(alpha = 0.03f * progress))
   356→   356→    // First: theme-aware dim overlay fading OUT on press (SrcOver).
   357→   357→    // Second: subtle black tint fading IN on press (SrcOver).
   358→   358→    if (el.isBottomTabIndicator && el.isBottomTabIndicator.dimColor) {
   359→   359→      const dc = el.isBottomTabIndicator.dimColor
   360→   360→      const p = togglePressProgress
   361→   361→      gl.useProgram(this.tintProgram)
   362→   362→      gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   363→   363→      gl.enableVertexAttribArray(this.aPosLocTn)
   364→   364→      gl.vertexAttribPointer(this.aPosLocTn, 2, gl.FLOAT, false, 0, 0)
   365→   365→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   366→   366→      gl.uniform2f(this.uTn['uCanvasSize'], this.canvas.width, this.canvas.height)
   367→   367→      gl.uniform2f(this.uTn['uOffset'], sx * this.dpr, sy * this.dpr)
   368→   368→      gl.uniform2f(this.uTn['uSize'], sw * this.dpr, sh * this.dpr)
   369→   369→      gl.uniform4f(this.uTn['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   370→   370→      gl.uniform2f(this.uTn['uOriginalSize'], origSizeX, origSizeY)
   371→   371→      gl.uniform1f(this.uTn['uOriginalCornerRadius'], origRadius)
   372→   372→      gl.uniform2f(this.uTn['uLayerScale'], layerScaleX, layerScaleY)
   373→   373→      gl.uniform1f(this.uTn['uElementRotation'], state.elementRotation)
   374→   374→      gl.uniform1f(this.uTn['uCornerStyle'], this.cornerStyle)
   375→   375→      // First overlay: dim color at 0.1 * (1 - progress) — fades out on press.
   376→   376→      gl.uniform4f(this.uTn['uColor'], dc[0], dc[1], dc[2], 0.1 * (1 - p))
   377→   377→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   378→   378→      // Second overlay: black at 0.03 * progress — fades in on press.
   379→   379→      gl.uniform4f(this.uTn['uColor'], 0, 0, 0, 0.03 * p)
   380→   380→      gl.drawArrays(gl.TRIANGLES, 0, 6)
   381→   381→      gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   382→   382→    }
   383→   383→
   384→   384→    // --- Step 2e: Foreground (label or icon) pass (button only) ---
   385→   385→    if (isButton && (el.label || el.icon)) {
   386→   386→      const fgTex = this.fgTextures.get(el.id)
   387→   387→      if (fgTex) {
   388→   388→        gl.useProgram(this.foregroundProgram)
   389→   389→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   390→   390→        gl.enableVertexAttribArray(this.aPosLocFg)
   391→   391→        gl.vertexAttribPointer(this.aPosLocFg, 2, gl.FLOAT, false, 0, 0)
   392→   392→        gl.blendFunc(gl.ONE, gl.ONE_MINUS_SRC_ALPHA)
   393→   393→
   394→   394→        gl.activeTexture(gl.TEXTURE0)
   395→   395→        gl.bindTexture(gl.TEXTURE_2D, fgTex)
   396→   396→        gl.uniform1i(this.uFg['uTexture'], 0)
   397→   397→        gl.uniform2f(this.uFg['uCanvasSize'], this.canvas.width, this.canvas.height)
   398→   398→        gl.uniform2f(this.uFg['uOffset'], sx * this.dpr, sy * this.dpr)
   399→   399→        gl.uniform2f(this.uFg['uSize'], sw * this.dpr, sh * this.dpr)
   400→   400→        gl.uniform4f(
   401→   401→          this.uFg['uCornerRadii'],
   402→   402→          radii[0] * this.dpr,
   403→   403→          radii[1] * this.dpr,
   404→   404→          radii[2] * this.dpr,
   405→   405→          radii[3] * this.dpr
   406→   406→        )
   407→   407→        gl.uniform2f(this.uFg['uOriginalSize'], origSizeX, origSizeY)
   408→   408→        gl.uniform1f(this.uFg['uOriginalCornerRadius'], origRadius)
   409→   409→      gl.uniform2f(this.uFg['uLayerScale'], layerScaleX, layerScaleY)
   410→   410→      gl.uniform1f(this.uFg['uCornerStyle'], this.cornerStyle)
   411→   411→      // Continuous-curvature mask for capsule foregrounds.
   412→   412→      if (el.useContinuousSdf && this.continuousSdfTexture) {
   413→   413→        gl.activeTexture(gl.TEXTURE2)
   414→   414→        gl.bindTexture(gl.TEXTURE_2D, this.continuousSdfTexture)
   415→   415→        gl.uniform1i(this.uFg['uContinuousSdf'], 2)
   416→   416→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 1.0)
   417→   417→        gl.uniform2f(this.uFg['uContinuousSdfTexSize'], this.continuousSdfTexSize[0], this.continuousSdfTexSize[1])
   418→   418→        gl.uniform2f(this.uFg['uContinuousSdfElementSize'], state.origW * this.dpr, state.origH * this.dpr)
   419→   419→      } else {
   420→   420→        gl.uniform1f(this.uFg['uUseContinuousSdf'], 0.0)
   421→   421→      }
   422→   422→        gl.uniform1f(this.uFg['uAlpha'], 1.0 - 0.15 * p)
   423→   423→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   424→   424→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   425→   425→      }
   426→   426→    }
   427→   427→
   428→   428→    // --- Step 2f: Rim highlight (Canvas2D stroke mask approach) ---
   429→   429→    // Uses Canvas2D ctx.stroke() (browser-native Skia) to rasterize the G2
   430→   430→    // Bezier path as a stroke mask. This is the SAME method the original
   431→   431→    // uses (Skia drawOutline + paint.Stroke) — the browser's Canvas2D
   432→   432→    // internally calls Skia's SkCanvas::drawPath, which tessellates the
   433→   433→    // Bezier into triangles and rasterizes with hardware coverage AA.
   434→   434→    //
   435→   435→    // Advantages over the SDF approach:
   436→   436→    //   - Exact G2 Bezier shape (not SDF approximation)
   437→   437→    //   - Hardware coverage AA (sub-pixel accurate, no smoothstep needed)
   438→   438→    //   - No SDF computation in shader (just one texture fetch)
   439→   439→    //   - Adaptive tessellation (more samples at high-curvature corners)
   440→   440→    //
   441→   441→    // The stroke mask is cached by exact geometry. Highlight angle/alpha/press
   442→   442→    // progress only affect the composite uniforms, not the mask, so caching is
   443→   443→    // safe and avoids both per-frame rasterization and UV mismatch between a
   444→   444→    // reused backing canvas and the logical mask size.
   445→   445→    if (el.highlight && el.highlight.alpha > 0.001) {
   446→   446→      const rimAlpha = (el.isToggleKnob || el.isBottomTabIndicator) ? elHighlightAlpha : el.highlight.alpha
   447→   447→      const finalAlpha = rimAlpha * state.enterAlpha
   448→   448→      if (finalAlpha > 0.001) {
   449→   449→        // HighlightModifier.kt: ceil(width.toPx().coerceAtMost(minDimension / 2)) * 2
   450→   450→        const widthPx = Math.min(
   451→   451→          el.highlight.widthDp * this.dpr,
   452→   452→          Math.min(origSizeX, origSizeY) * 0.5
   453→   453→        )
   454→   454→        const strokeWidthDevice = Math.max(1, Math.ceil(widthPx) * 2)
   455→   455→        // Highlight data class: blurRadius defaults to width / 2. Honor it in the
   456→   456→        // Canvas2D mask as well (for the default 0.25dp this is sub-pixel, but it
   457→   457→        // keeps non-default highlights faithful too).
   458→   458→        const blurPx = Math.max(0, (el.highlight.blurRadiusDp ?? el.highlight.widthDp / 2) * this.dpr)
   459→   459→
   460→   460→        // --- Generate stroke mask via Canvas2D (browser-native Skia) ---
   461→   461→        // Element-local coordinates: (0,0) = element top-left in device px.
   462→   462→        // Margin for stroke width + AA + blur spread.
   463→   463→        const strokeMargin = Math.ceil(strokeWidthDevice) + 4  // logical margin
   464→   464→        // Logical mask size (1x device px) — used for shader UV mapping
   465→   465→        const maskW = Math.max(1, Math.ceil(origSizeX + 2 * strokeMargin))
   466→   466→        const maskH = Math.max(1, Math.ceil(origSizeY + 2 * strokeMargin))
   467→   467→        // Supersample for sharper stroke/blur rasterization.
   468→   468→        // Cap SS so total mask DPR (this.dpr × SS) ≤ device's native DPR —
   469→   469→        // no point rendering pixels beyond what the screen can display.
   470→   470→        const deviceDpr = window.devicePixelRatio || 1
   471→   471→        const SS = Math.min(2, Math.max(1, Math.floor(deviceDpr / this.dpr)))
   472→   472→        const canvasW = maskW * SS
   473→   473→        const canvasH = maskH * SS
   474→   474→        const useG2 = !!el.useContinuousSdf
   475→   475→        const maskKey = [
   476→   476→          useG2 ? 'g2' : 'rr',
   477→   477→          origSizeX.toFixed(3),
   478→   478→          origSizeY.toFixed(3),
   479→   479→          origRadius.toFixed(3),
   480→   480→          strokeWidthDevice,
   481→   481→          blurPx.toFixed(3),
   482→   482→          strokeMargin,
   483→   483→          maskW,
   484→   484→          maskH,
   485→   485→          `ss${SS}`,  // cache key includes supersample factor
   486→   486→        ].join(':')
   487→   487→
   488→   488→        let mask = this.strokeMaskCache.get(maskKey)
   489→   489→        if (!mask) {
   490→   490→          const canvas = document.createElement('canvas')
   491→   491→          canvas.width = canvasW  // 2x supersampled physical size
   492→   492→          canvas.height = canvasH
   493→   493→          const ctx = canvas.getContext('2d', { alpha: true })
   494→   494→          if (!ctx) throw new Error('2D canvas not supported')
   495→   495→          const tex = gl.createTexture()
   496→   496→          if (!tex) throw new Error('WebGL texture allocation failed')
   497→   497→          // w/h store the LOGICAL (1x) size for shader UV mapping;
   498→   498→          // the physical canvas is SS times larger.
   499→   499→          mask = { tex, canvas, ctx, w: maskW, h: maskH, ready: false }
   500→   500→          this.strokeMaskCache.set(maskKey, mask)
   501→   501→
   502→   502→          // Keep the cache bounded. 32 entries is far above the catalog's
   503→   503→          // simultaneous highlight geometries and avoids unbounded growth on
   504→   504→          // highly dynamic pages.
   505→   505→          if (this.strokeMaskCache.size > 32) {
   506→   506→            const oldestKey = this.strokeMaskCache.keys().next().value as string | undefined
   507→   507→            if (oldestKey && oldestKey !== maskKey) {
   508→   508→              const oldest = this.strokeMaskCache.get(oldestKey)
   509→   509→              if (oldest) gl.deleteTexture(oldest.tex)
   510→   510→              this.strokeMaskCache.delete(oldestKey)
   511→   511→            }
   512→   512→          }
   513→   513→        }
   514→   514→
   515→   515→        if (!mask.ready) {
   516→   516→          const smCtx = mask.ctx
   517→   517→          smCtx.clearRect(0, 0, canvasW, canvasH)
   518→   518→          smCtx.save()
   519→   519→          // Scale up for 2x supersampling: draw in logical (1x) coordinates
   520→   520→          // while the physical canvas is 2x. This gives sharper stroke + blur.
   521→   521→          smCtx.scale(SS, SS)
   522→   522→          smCtx.translate(strokeMargin, strokeMargin)
   523→   523→
   524→   524→          // Build the path (element-local, 0..origSizeX × 0..origSizeY)
   525→   525→          let path: Path2D
   526→   526→          if (useG2) {
   527→   527→            path = continuousCurvatureRoundedRectPath(smCtx, origSizeX, origSizeY, origRadius)
   528→   528→          } else {
   529→   529→            path = new Path2D()
   530→   530→            const r = Math.min(origRadius, origSizeX / 2, origSizeY / 2)
   531→   531→            path.moveTo(r, 0)
   532→   532→            path.lineTo(origSizeX - r, 0)
   533→   533→            path.arcTo(origSizeX, 0, origSizeX, r, r)
   534→   534→            path.lineTo(origSizeX, origSizeY - r)
   535→   535→            path.arcTo(origSizeX, origSizeY, origSizeX - r, origSizeY, r)
   536→   536→            path.lineTo(r, origSizeY)
   537→   537→            path.arcTo(0, origSizeY, 0, origSizeY - r, r)
   538→   538→            path.lineTo(0, r)
   539→   539→            path.arcTo(0, 0, r, 0, r)
   540→   540→            path.closePath()
   541→   541→          }
   542→   542→
   543→   543→          // Stroke — browser-native Skia stroke rasterization.
   544→   544→          // Faithful to HighlightModifier.kt: clipOutline(outline) BEFORE drawOutline,
   545→   545→          // so only the INSIDE half of the centered stroke remains. Without this
   546→   546→          // clip, Canvas2D keeps the outer half too and the rim highlight leaks
   547→   547→          // outside the glass / looks twice as thick.
   548→   548→          smCtx.clip(path)
   549→   549→          smCtx.lineWidth = strokeWidthDevice
   550→   550→          smCtx.strokeStyle = 'rgba(255,255,255,1)'
   551→   551→          smCtx.lineJoin = 'round'
   552→   552→          smCtx.lineCap = 'round'
   553→   553→          // Approximate Skia BlurMaskFilter on the stroke paint. The clip above
   554→   554→          // keeps the result inside the outline; the blur only softens that
   555→   555→          // inside stroke band, like the original HighlightModifier layer.
   556→   556→          smCtx.filter = blurPx > 0.01 ? `blur(${blurPx}px)` : 'none'
   557→   557→          smCtx.stroke(path)
   558→   558→          smCtx.filter = 'none'
   559→   559→          smCtx.restore()
   560→   560→
   561→   561→          // Upload to GPU texture. Keep the same top-left UV convention as the
   562→   562→          // foreground pass: with UNPACK_FLIP_Y_WEBGL=false, local y=0 samples
   563→   563→          // the top row of the mask.
   564→   564→          gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   565→   565→          gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false)
   566→   566→          gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, mask.canvas)
   567→   567→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR)
   568→   568→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR)
   569→   569→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE)
   570→   570→          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE)
   571→   571→          mask.ready = true
   572→   572→        }
   573→   573→
   574→   574→        // --- Composite: stroke mask × intensity × color → scene FBO ---
   575→   575→        gl.enable(gl.BLEND)
   576→   576→        if (el.highlight.mode === 1) {
   577→   577→          gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   578→   578→        } else {
   579→   579→          gl.blendFunc(gl.ONE, gl.ONE)
   580→   580→        }
   581→   581→        gl.useProgram(this.strokeMaskCompositeProgram)
   582→   582→        gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   583→   583→        gl.enableVertexAttribArray(this.aPosLocSm)
   584→   584→        gl.vertexAttribPointer(this.aPosLocSm, 2, gl.FLOAT, false, 0, 0)
   585→   585→        gl.uniform2f(this.uSm['uCanvasSize'], this.canvas.width, this.canvas.height)
   586→   586→        gl.uniform2f(this.uSm['uOffset'], sx * this.dpr, sy * this.dpr)
   587→   587→        gl.uniform2f(this.uSm['uSize'], sw * this.dpr, sh * this.dpr)
   588→   588→        gl.uniform4f(this.uSm['uCornerRadii'], radii[0] * this.dpr, radii[1] * this.dpr, radii[2] * this.dpr, radii[3] * this.dpr)
   589→   589→        gl.activeTexture(gl.TEXTURE0)
   590→   590→        gl.bindTexture(gl.TEXTURE_2D, mask.tex)
   591→   591→        gl.uniform1i(this.uSm['uStrokeMask'], 0)
   592→   592→        // uMaskOffset/uMaskSize are in LOGICAL (1x device px) space — the
   593→   593→        // physical canvas is SS× larger but the shader uses 1x coords for UV:
   594→   594→        //   UV = (localCoord + margin) / logicalMaskSize
   595→   595→        // LINEAR filtering on the SS× texture gives supersampled quality.
   596→   596→        gl.uniform2f(this.uSm['uMaskOffset'], strokeMargin, strokeMargin)
   597→   597→        gl.uniform2f(this.uSm['uMaskSize'], mask.w, mask.h)
   598→   598→        gl.uniform4f(this.uSm['uHighlightColor'], el.highlight.color[0], el.highlight.color[1], el.highlight.color[2], 1.0)
   599→   599→        gl.uniform1f(this.uSm['uHighlightAngle'], el.useGravityAngle ? this.gravityAngle : el.highlight.angle)
   600→   600→        gl.uniform1f(this.uSm['uHighlightFalloff'], el.highlight.falloff)
   601→   601→        gl.uniform1f(this.uSm['uHighlightAlpha'], finalAlpha)
   602→   602→        gl.uniform1f(this.uSm['uHighlightMode'], el.highlight.mode)
   603→   603→        gl.uniform2f(this.uSm['uOriginalSize'], origSizeX, origSizeY)
   604→   604→        gl.uniform1f(this.uSm['uOriginalCornerRadius'], origRadius)
   605→   605→        gl.uniform2f(this.uSm['uLayerScale'], layerScaleX, layerScaleY)
   606→   606→        gl.uniform1f(this.uSm['uElementRotation'], state.elementRotation)
   607→   607→        gl.drawArrays(gl.TRIANGLES, 0, 6)
   608→   608→
   609→   609→        gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA)
   610→   610→      }
   611→   611→    }
   612→   612→
   613→   613→    // --- Step 2g: 内层背景板 rim highlight (指示器 only) ---
   614→   614→    // The 内层背景板 (hidden Row's 56dp glass inside the 指示器) has its own
   615→   615→    // Highlight.Default.copy(alpha=progress). This is drawn INSIDE the
   616→   616→    // 指示器's element shader (sampleIndicatorBackdrop) so it's clipped by
   617→   617→    // the 指示器's capsule SDF — no separate pass needed here.
   618→   618→  },
   619→   619→}

   233→  // Device motion → gravity angle for glass highlight direction.
   234→  // Faithful to UISensor.kt:
   235→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   236→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   237→  //
   238→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   239→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   240→  // rebuilding the catalog (which would lose drag state and cause jank).
   241→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   242→  // renderer.gravityAngle live each frame.
   243→  //
   244→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   245→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   246→  // device's acceleration INCLUDING gravity in device-space axes:
   247→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   248→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   249→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   250→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   251→  // vectors — atan2(beta, gamma) does NOT match the original.)
   252→  //
   253→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   254→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   255→  // Default 45° (matches UISensor.kt's initial value).
   256→  React.useEffect(() => {
   257→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return

   259→    // that uses it (el.useGravityAngle=true). Listening on all pages causes
   260→    // ~60 setGravityAngle calls/sec → 60 full WebGL renders/sec → high GPU.
   261→    if (destination !== CatalogDestination.ControlCenter) return
   262→    let smoothed = 45
   263→    const alpha = 0.5
   264→    const handler = (e: DeviceMotionEvent) => {
   265→      const acc = e.accelerationIncludingGravity
   266→      if (!acc || acc.x == null || acc.y == null) return
   267→      const x = acc.x
   268→      const y = acc.y
   269→      let target = Math.atan2(y, x) * 180 / Math.PI
   270→      let delta = target - smoothed
   271→      while (delta > 180) delta -= 360
   272→      while (delta < -180) delta += 360
   273→      smoothed += delta * alpha
   274→      while (smoothed > 180) smoothed -= 360
   275→      while (smoothed < -180) smoothed += 360
   276→      const rad = smoothed * Math.PI / 180
   277→      rendererRef.current?.setGravityAngle(rad)
   278→    }
   279→    window.addEventListener('devicemotion', handler)
   280→    return () => window.removeEventListener('devicemotion', handler)

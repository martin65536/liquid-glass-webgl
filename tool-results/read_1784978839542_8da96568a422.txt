    33→function useSystemTheme(): Theme {
    34→  const [theme, setTheme] = React.useState<Theme>('light')
    35→  React.useEffect(() => {
    36→    if (typeof window === 'undefined' || !window.matchMedia) return
    37→    const mq = window.matchMedia('(prefers-color-scheme: dark)')
    38→    const update = () => setTheme(mq.matches ? 'dark' : 'light')
    39→    update()
    40→    mq.addEventListener('change', update)
    41→    return () => mq.removeEventListener('change', update)
    42→  }, [])
    43→  return theme
    44→}

   287→  function finishIteration(fps: number, candidateDpr: number, deviceDpr: number, iteration: number, maxIterations: number) {
   288→    const MIN_FPS = 55
   289→    if (fps >= MIN_FPS) {
   290→      // This DPR works — set lo to it (we know it's viable)
   291→      perfLoRef.current = candidateDpr
   292→    } else {
   293→      // Too slow — set hi to it (search lower)
   294→      perfHiRef.current = candidateDpr
   295→    }
   296→
   297→    // Convergence: range too tight to differentiate, or max iterations reached.
   298→    // Use a tight threshold (< 0.125) so we don't stop early and miss deviceDpr.
   299→    if (iteration >= maxIterations || (perfHiRef.current - perfLoRef.current) < 0.125) {
   300→      // Converged — use the highest DPR that worked (lo)
   301→      const bestDpr = Math.max(deviceDpr / 2, Math.min(deviceDpr, Math.round(perfLoRef.current * 4) / 4))
   302→      perfPhaseRef.current = 'done'
   303→      const isGood = bestDpr >= deviceDpr * 0.5
   304→      const statusText = isGood
   305→        ? `检测完成！推荐 DPR：${bestDpr} · 设备可流畅运行液态玻璃`
   306→        : `检测完成 · 推荐 DPR：${bestDpr} · 性能有限，已自动降低画质`
   307→      try { window.localStorage.setItem(PERF_KEY, String(bestDpr)) } catch {}
   308→      setState({
   309→        customDpr: bestDpr,
   310→        perfDone: true,
   311→        perfResultDpr: bestDpr,
   312→        perfStatusText: statusText,
   313→        perfProgress: null,
   314→        // Don't reset perfGlassAngle — settle animation will handle it
   315→        // perfDeformMul stays at 1 — settle animation will decay it to 0
   316→        // perfExitProgress stays at 0 — settle animation will animate it to 1
   317→        perfProgressFrac: 1,
   318→      })
   319→    } else {
   320→      // More iterations needed — increment trigger to re-fire the effect
   321→      perfPhaseRef.current = 'idle'
   322→      setState((prev) => ({
   323→        perfProgress: 'running',
   324→        perfRoundTrigger: prev.perfRoundTrigger + 1,
   325→        perfStatusText: `第${iteration}/${maxIterations}轮 · DPR: ${candidateDpr} · FPS: ${fps}fps`,
   326→      }))
   327→    }
   328→  }

   385→  // from deformed to square (perfDeformMul decays 1→0) and animate the

   394→
   395→    const ORBIT_SPEED = 2 * Math.PI // keep rotation going during settle
   396→    const PROG_LERP_SPEED = 4 // progress bar lerp speed
   397→    let angle = state.perfGlassAngle || 0
   398→    let progAnimated = state.perfProgressFracAnimated ?? 0
   399→    let elapsed = 0
   400→    let lastT = 0
   401→    let raf = 0
   402→    // Exponential decay rate for deformMul: mul = e^(-3*t), reaches ~0.05 by 1s
   403→    const DECAY_RATE = 3

   405→    const EXIT_DELAY = 0.3

   409→      if (!lastT) lastT = timestamp
   410→      const dt = Math.min(0.05, (timestamp - lastT) / 1000)
   411→      lastT = timestamp
   412→      elapsed += dt
   413→

   418→      const mul = Math.exp(-DECAY_RATE * elapsed)
   419→
   420→      // Exit button: ease-out quadratic after delay
   421→      let exitProg = 0
   422→      if (elapsed > EXIT_DELAY) {
   423→        const t = Math.min(1, (elapsed - EXIT_DELAY) / EXIT_DURATION)
   424→        exitProg = 1 - (1 - t) * (1 - t) // ease-out
   425→      }
   426→
   427→      // Smoothly animate progress to 1.0 (use ref for latest value)
   428→      const progTarget = perfProgressFracRef.current
   429→      progAnimated += (progTarget - progAnimated) * Math.min(1, PROG_LERP_SPEED * dt)
   430→
   431→      setState({ perfGlassAngle: angle, perfDeformMul: mul, perfExitProgress: exitProg, perfProgressFracAnimated: progAnimated })
   432→
   433→      if (mul <= 0.01 && exitProg >= 0.99 && progAnimated >= 0.99) {
   434→        // Settle complete — glass angle irrelevant at mul=0
   435→        setState({ perfDeformMul: 0, perfExitProgress: 1, perfProgressFracAnimated: 1 })
   436→        return
   437→      }
   438→      raf = requestAnimationFrame(tick)
   439→    }
   440→    raf = requestAnimationFrame(tick)
   441→    return () => cancelAnimationFrame(raf)

   453→  // setState supports both a partial patch and a functional updater.
   454→  // The functional form is critical for drag callbacks (slider, magnifier,
   455→  // lock screen, toggle) so they always read the latest state — avoiding
   456→  // stale closures when multiple pointermove events fire between React renders.
   457→  // Also persists Settings fields (customDpr, globalSeparableBlur, blurTapCap,
   458→  // blurDownsample) to localStorage so they survive page reloads.
   459→  const setState = React.useCallback(
   460→    (patch: Partial<CatalogState> | ((prev: CatalogState) => Partial<CatalogState>)) => {
   461→      setStateRaw((prev) => {
   462→        const p = typeof patch === 'function' ? patch(prev) : patch

   611→  // Device motion → gravity angle for glass highlight direction.
   612→  // Faithful to UISensor.kt:
   613→  //   gravityAngle = gravityAngle * (1-alpha) + atan2(y, x) * 180/PI * alpha
   614→  // with alpha = 0.5, updated on every sensor event (~60Hz).
   615→  //
   616→  // CRITICAL: we push the angle DIRECTLY to the renderer via
   617→  // rendererRef.current.setGravityAngle() — NOT via React state. This avoids
   618→  // rebuilding the catalog (which would lose drag state and cause jank).
   619→  // CC tiles opt in via el.useGravityAngle=true; the rim highlight pass reads
   620→  // renderer.gravityAngle live each frame.
   621→  //
   622→  // SOURCE: DeviceMotionEvent.accelerationIncludingGravity is the exact Web
   623→  // equivalent of Android's Sensor.TYPE_ACCELEROMETER — both report the
   624→  // device's acceleration INCLUDING gravity in device-space axes:
   625→  //   x: right is positive  (Android values[0] / Web acceleration.x)
   626→  //   y: top is positive    (Android values[1] / Web acceleration.y)
   627→  // atan2(y, x) therefore has the SAME semantics on both platforms.
   628→  // (DeviceOrientationEvent's beta/gamma are EULER ANGLES, not acceleration
   629→  // vectors — atan2(beta, gamma) does NOT match the original.)
   630→  //
   631→  // EMA smoothing + shortest-path angle interpolation handles the atan2
   632→  // discontinuity at ±180° (rotating through 180° doesn't jump to -180°).
   633→  // Default 45° (matches UISensor.kt's initial value).
   634→  React.useEffect(() => {
   635→    if (typeof window === 'undefined' || !('DeviceMotionEvent' in window)) return
   636→    let smoothed = 45
   637→    const alpha = 0.5
   638→    const handler = (e: DeviceMotionEvent) => {
   639→      const acc = e.accelerationIncludingGravity
   640→      if (!acc || acc.x == null || acc.y == null) return
   641→      const x = acc.x
   642→      const y = acc.y
   643→      let target = Math.atan2(y, x) * 180 / Math.PI
   644→      // Shortest-path interpolation: wrap the delta to [-180, 180] so
   645→      // rotating through ±180° doesn't cause a 358° jump.
   646→      let delta = target - smoothed
   647→      while (delta > 180) delta -= 360
   648→      while (delta < -180) delta += 360
   649→      smoothed += delta * alpha
   650→      // Wrap smoothed to [-180, 180] to keep the value bounded.
   651→      while (smoothed > 180) smoothed -= 360
   652→      while (smoothed < -180) smoothed += 360
   653→      const rad = smoothed * Math.PI / 180
   654→      rendererRef.current?.setGravityAngle(rad)
   655→    }
   656→    window.addEventListener('devicemotion', handler)
   657→    return () => window.removeEventListener('devicemotion', handler)
   658→  }, [rendererRef])

    34→    34→/* ------------------------------------------------------------------ *
    35→    35→ * LiquidGlassRenderer
    36→    36→ *
    37→    37→ * One opaque WebGL canvas. Render pipeline per frame:
    38→    38→ *   1. Wallpaper background pass (cover-fit).
    39→    39→ *   2. For each button (in order):
    40→    40→ *      a. Outer drop shadow pass (if configured).
    41→    41→ *      b. Element pass (refraction + vibrancy + tint + highlight),
    42→    42→ *         with InteractiveHighlight-driven scale/translation/stretch.
    43→    43→ *      c. White overlay pass (Plus blend, 8% * progress) on press.
    44→    44→ *      d. Radial highlight pass (Plus blend, at finger position) on press.
    45→    45→ *      e. Foreground pass (button label, with press-driven alpha fade).
    46→    46→ *
    47→    47→ * No DOM children — the canvas owns the entire visual surface.
    48→    48→ *
    49→    49→ * NOTE: The class methods are split across multiple files in this
    50→    50→ * directory (methods-*.ts). Each methods file declares its method
    51→    51→ * signatures via `declare module './index'` and exports a record of
    52→    52→ * named functions. Those records are merged onto the prototype at the
    53→    53→ * bottom of this file via Object.assign.
    54→    54→ * ------------------------------------------------------------------ */

   432→   432→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   433→   433→  ensureBlurPrograms(tapCount: number): void {
   434→   434→    if (this.blurPrograms.has(tapCount)) return
   435→   435→    const gl = this.gl
   436→   436→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   437→   437→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   438→   438→    const mk = (fs: WebGLShader) => {
   439→   439→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   440→   440→      const p = gl.createProgram()!
   441→   441→      gl.attachShader(p, vs)
   442→   442→      gl.attachShader(p, fs)
   443→   443→      gl.bindAttribLocation(p, 0, 'aPos')
   444→   444→      gl.linkProgram(p)
   445→   445→      gl.deleteShader(vs)
   446→   446→      gl.deleteShader(fs)
   447→   447→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   448→   448→        const log = gl.getProgramInfoLog(p)
   449→   449→        gl.deleteProgram(p)
   450→   450→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   451→   451→      }
   452→   452→      return p
   453→   453→    }
   454→   454→    const hProg = mk(hFs)
   455→   455→    const vProg = mk(vFs)
   456→   456→    const uH: Record<string, WebGLUniformLocation | null> = {
   457→   457→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   458→   458→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   459→   459→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   460→   460→    }
   461→   461→    const uV: Record<string, WebGLUniformLocation | null> = {
   462→   462→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   463→   463→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   464→   464→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   465→   465→    }
   466→   466→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   467→   467→  }

   469→   469→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   470→   470→   *  blurred result into blurFboB, returns blurFboBTex.
   471→   471→   *  Saves/restores the currently-bound framebuffer.
   472→   472→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   473→   473→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   474→   474→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   475→   475→    const gl = this.gl
   476→   476→    const w = this.fboW
   477→   477→    const h = this.fboH
   478→   478→    // Compute tap count, capped by blurTapCap (performance knob).
   479→   479→    let taps = computeBlur1DTapCount(radius)
   480→   480→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   481→   481→    this.ensureBlurPrograms(taps)
   482→   482→    const entry = this.blurPrograms.get(taps)!
   483→   483→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   484→   484→    gl.disable(gl.BLEND)
   485→   485→
   486→   486→    // Pass 1: horizontal — srcTex → blurFboA
   487→   487→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   488→   488→    gl.viewport(0, 0, w, h)
   489→   489→    gl.useProgram(entry.hProg)
   490→   490→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   491→   491→    gl.enableVertexAttribArray(entry.aPosH)
   492→   492→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   493→   493→    gl.activeTexture(gl.TEXTURE0)
   494→   494→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   495→   495→    gl.uniform1i(entry.uH['uTexture'], 0)
   496→   496→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   497→   497→    gl.uniform1f(entry.uH['uRadius'], radius)
   498→   498→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   499→   499→
   500→   500→    // Pass 2: vertical — blurFboATex → blurFboB
   501→   501→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   502→   502→    gl.viewport(0, 0, w, h)
   503→   503→    gl.useProgram(entry.vProg)
   504→   504→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   505→   505→    gl.enableVertexAttribArray(entry.aPosV)
   506→   506→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   507→   507→    gl.activeTexture(gl.TEXTURE0)
   508→   508→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   509→   509→    gl.uniform1i(entry.uV['uTexture'], 0)
   510→   510→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   511→   511→    gl.uniform1f(entry.uV['uRadius'], radius)
   512→   512→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   513→   513→
   514→   514→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   515→   515→    gl.viewport(0, 0, w, h)
   516→   516→    return this.blurFboBTex!
   517→   517→  }

   558→   558→  /** 2-pass Gaussian blur on a highlight stroke MASK (alpha only).
   559→   559→   *  Faithful to Android BlurMaskFilter(NORMAL, sigma):
   560→   560→   *    - sigma = blurRadiusPx (the Android radius param IS sigma)
   561→   561→   *    - convolves the mask's ALPHA with a Gaussian kernel
   562→   562→   *    - sub-pixel sigma (0.25px) still blurs (no 0.5 early-return)

   567→   567→    const w = this.fboW
   568→   568→    const h = this.fboH
   569→   569→    let taps = computeHighlightBlurTapCount(sigmaPx)
   570→   570→    taps = Math.min(taps, Math.max(3, this.blurTapCap | 0))
   571→   571→    this.ensureHighlightBlurPrograms(taps)
   572→   572→    const entry = this.highlightBlurPrograms.get(taps)!
   573→   573→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   574→   574→    gl.disable(gl.BLEND)
   575→   575→
   576→   576→    // Pass 1: horizontal — srcTex → blurFboA
   577→   577→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   578→   578→    gl.viewport(0, 0, w, h)
   579→   579→    gl.useProgram(entry.hProg)
   580→   580→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   581→   581→    gl.enableVertexAttribArray(entry.aPosH)
   582→   582→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   583→   583→    gl.activeTexture(gl.TEXTURE0)

   586→   586→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   587→   587→    gl.uniform1f(entry.uH['uRadius'], sigmaPx)
   588→   588→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   589→   589→
   590→   590→    // Pass 2: vertical — blurFboATex → blurFboB
   591→   591→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   592→   592→    gl.viewport(0, 0, w, h)
   593→   593→    gl.useProgram(entry.vProg)
   594→   594→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   595→   595→    gl.enableVertexAttribArray(entry.aPosV)
   596→   596→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)

   605→   605→    gl.viewport(0, 0, w, h)

   614→   614→    const gl = this.gl
   615→   615→    if (this.wallpaperTexture) gl.deleteTexture(this.wallpaperTexture)

   618→   618→    for (const entry of this.strokeMaskCache.values()) gl.deleteTexture(entry.tex)

   621→   621→    this.innerShadowMaskCache.clear()
   622→   622→    if (this.fboA) gl.deleteFramebuffer(this.fboA)
   623→   623→    if (this.fboATex) gl.deleteTexture(this.fboATex)
   624→   624→    if (this.fboB) gl.deleteFramebuffer(this.fboB)
   625→   625→    if (this.fboBTex) gl.deleteTexture(this.fboBTex)
   626→   626→    this.fboA = this.fboB = null
   627→   627→    this.fboATex = this.fboBTex = null
   628→   628→    if (this.tabsBackdropFbo) gl.deleteFramebuffer(this.tabsBackdropFbo)
   629→   629→    if (this.tabsBackdropTex) gl.deleteTexture(this.tabsBackdropTex)
   630→   630→    this.tabsBackdropFbo = null
   631→   631→    this.tabsBackdropTex = null
   632→   632→    // GP element FBO + blur FBOs + programs
   633→   633→    if (this.gpElementFbo) gl.deleteFramebuffer(this.gpElementFbo)
   634→   634→    if (this.gpElementTex) gl.deleteTexture(this.gpElementTex)

   636→   636→    if (this.blurFboATex) gl.deleteTexture(this.blurFboATex)
   637→   637→    if (this.blurFboB) gl.deleteFramebuffer(this.blurFboB)
   638→   638→    if (this.blurFboBTex) gl.deleteTexture(this.blurFboBTex)
   639→   639→    this.gpElementFbo = this.blurFboA = this.blurFboB = null
   640→   640→    this.gpElementTex = this.blurFboATex = this.blurFboBTex = null
   641→   641→    if (this.highlightMaskFbo) gl.deleteFramebuffer(this.highlightMaskFbo)
   642→   642→    if (this.highlightMaskTex) gl.deleteTexture(this.highlightMaskTex)
   643→   643→    this.highlightMaskFbo = null
   644→   644→    this.highlightMaskTex = null
   645→   645→    if (this.dialogBackdropFbo) gl.deleteFramebuffer(this.dialogBackdropFbo)
   646→   646→    if (this.dialogBackdropTex) gl.deleteTexture(this.dialogBackdropTex)
   647→   647→    this.dialogBackdropFbo = null
   648→   648→    this.dialogBackdropTex = null
   649→   649→    this.dialogBackdropKey = null
   650→   650→    for (const { hProg, vProg } of this.blurPrograms.values()) {
   651→   651→      gl.deleteProgram(hProg)
   652→   652→      gl.deleteProgram(vProg)
   653→   653→    }
   654→   654→    this.blurPrograms.clear()
   655→   655→    for (const { hProg, vProg } of this.highlightBlurPrograms.values()) {
   656→   656→      gl.deleteProgram(hProg)
   657→   657→      gl.deleteProgram(vProg)
   658→   658→    }
   659→   659→    this.highlightBlurPrograms.clear()
   660→   660→    if (this.sdfTexture) gl.deleteTexture(this.sdfTexture)
   661→   661→    this.sdfTexture = null
   662→   662→    for (const { tex } of this.continuousSdfPool.values()) gl.deleteTexture(tex)
   663→   663→    this.continuousSdfPool.clear()
   664→   664→    this.continuousSdfTexture = null
   665→   665→    this.continuousSdfKey = null
   666→   666→    gl.deleteProgram(this.elementProgram)
   667→   667→    gl.deleteProgram(this.shadowProgram)
   668→   668→    gl.deleteProgram(this.wallpaperProgram)
   669→   669→    gl.deleteProgram(this.foregroundProgram)
   670→   670→    gl.deleteProgram(this.highlightProgram)
   671→   671→    gl.deleteProgram(this.tintProgram)
   672→   672→    gl.deleteProgram(this.rimHighlightProgram)
   673→   673→    gl.deleteProgram(this.highlightStrokeProgram)
   674→   674→    gl.deleteProgram(this.highlightCompositeProgram)
   675→   675→    gl.deleteProgram(this.strokeMaskCompositeProgram)

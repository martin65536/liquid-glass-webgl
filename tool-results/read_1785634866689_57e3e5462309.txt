     1→     1→/* ------------------------------------------------------------------ *
     2→     2→ * Element shader utilities — GLSL helper functions used by the element
     3→     3→ * fragment shader: backdrop sampling (Gaussian disc), color
     4→     4→ * controls (saturation/brightness/contrast), HSV conversion + Hue blend.
     5→     5→ *
     6→     6→ * The blur sampling functions are dynamically generated with a tap count
     7→     7→ * that scales with quality needs. WebGL1 requires constant loop bounds,
     8→     8→ * so we unroll the Gaussian kernel in JS and emit a fixed sequence of
     9→     9→ * texture2D calls. This gives high-quality Gaussian blur at any radius.
    10→    10→ * ------------------------------------------------------------------ */
    11→    11→
    12→    12→/** Generate a Gaussian disc sampling pattern in JS.
    13→    13→ *  Returns an array of {x, y, w} for a Gaussian blur with sigma=1.0
    14→    14→ *  (the shader scales offsets by the actual radius at runtime).
    15→    15→ *  Uses a golden-angle spiral (Vogel's method) for even disc coverage.
    16→    16→ *  `tapCount` controls quality (more taps = smoother).
    17→    17→ */
    18→    18→function generateGaussianDisc(tapCount: number): Array<{ x: number; y: number; w: number }> {
    19→    19→  const taps: Array<{ x: number; y: number; w: number }> = []
    20→    20→  if (tapCount <= 1) {
    21→    21→    taps.push({ x: 0, y: 0, w: 1.0 })
    22→    22→    return taps
    23→    23→  }
    24→    24→  const goldenAngle = Math.PI * (3.0 - Math.sqrt(5.0))
    25→    25→  const maxRadius = 3.0 // 3-sigma cutoff
    26→    26→  let totalW = 0
    27→    27→  for (let i = 0; i < tapCount; i++) {
    28→    28→    const t = (i + 0.5) / tapCount
    29→    29→    const r = maxRadius * Math.sqrt(t)
    30→    30→    const angle = i * goldenAngle
    31→    31→    const x = r * Math.cos(angle)
    32→    32→    const y = r * Math.sin(angle)
    33→    33→    const dist2 = x * x + y * y
    34→    34→    const w = Math.exp(-0.5 * dist2)
    35→    35→    taps.push({ x, y, w })
    36→    36→    totalW += w
    37→    37→  }
    38→    38→  if (totalW > 0) {
    39→    39→    for (const t of taps) t.w /= totalW
    40→    40→  }
    41→    41→  return taps
    42→    42→}
    43→    43→
    44→    44→/** Generate GLSL texture2D calls for a Gaussian blur.
    45→    45→ *  Offsets are in units of radius (sigma = radius). The shader multiplies
    46→    46→ *  by pxToUv to convert to UV-space offset.
    47→    47→ */
    48→    48→function generateBlurGLSL(taps: Array<{ x: number; y: number; w: number }>, sampler: string, uvVar: string, pxToUvExpr: string): string {
    49→    49→  if (taps.length === 1) {
    50→    50→    return `    return texture2D(${sampler}, ${uvVar});\n`
    51→    51→  }
    52→    52→  let code = ''
    53→    53→  for (const t of taps) {
    54→    54→    const ox = t.x.toFixed(6)
    55→    55→    const oy = t.y.toFixed(6)
    56→    56→    const w = t.w.toFixed(8)
    57→    57→    code += `    sum += texture2D(${sampler}, ${uvVar} + vec2(${ox}, ${oy}) * ${pxToUvExpr}) * ${w};\n`
    58→    58→  }
    59→    59→  return code
    60→    60→}

    71→    71→  const wallpaperBlurCode = generateBlurGLSL(taps, 'uWallpaperSampler', 'uv', 'pxToUv')
    72→    72→

    77→    77→vec3 rgb2hsv(vec3 c);
    78→    78→vec3 hsv2rgb(vec3 c);

    81→    81→float circleMap(float x) {
    82→    82→    return 1.0 - sqrt(1.0 - x * x);

    84→    84→
    85→    85→// SDF-texture glass sampling (faithful to SdfShader.kt).
    86→    86→// Samples the clock_sdf texture at element-local coords.
    87→    87→// Returns vec4(intensity, maskAlpha, normalX, normalY); zeroes if outside.
    88→    88→vec4 sampleSdfTexture(vec2 localPx) {
    89→    89→    vec2 uv = vec2(localPx.x / uOriginalSize.x,
    90→    90→                   localPx.y / uOriginalSize.y);
    91→    91→    if (uv.x < 0.0 || uv.y < 0.0 || uv.x > 1.0 || uv.y > 1.0) {
    92→    92→        return vec4(0.0);
    93→    93→    }
    94→    94→    vec4 v = texture2D(uSdfTexSampler, uv);
    95→    95→    float sd = v.r * 2.0 - 1.0;
    96→    96→    float mask = smoothstep(0.5, 1.0, v.a);
    97→    97→    if (mask <= 0.0) return vec4(0.0);
    98→    98→    if (mask < 1.0) sd = 0.0;
    99→    99→    vec2 normal = normalize(v.gb * 2.0 - 1.0);
   100→   100→    float intensity = circleMap(1.0 - min(1.0, -sd * 1.5));
   101→   101→    return vec4(intensity, mask, normal.x, normal.y);
   102→   102→}

   104→   104→// Convert a canvas-pixel coordinate (top-left origin) to scene-texture UV.
   105→   105→// The scene texture is the same size as the canvas, and is rendered with
   106→   106→// gl_FragCoord (bottom-left origin). So UV = (canvasPx.x / canvasW, 1 -
   107→   107→// canvasPx.y / canvasH). The Y flip happens here so the rest of the shader
   108→   108→// can work in top-left-origin canvas px.
   109→   109→vec2 sceneUv(vec2 canvasPx) {
   110→   110→    return vec2(canvasPx.x / uCanvasSize.x, 1.0 - canvasPx.y / uCanvasSize.y);
   111→   111→}

   113→   113→// Gaussian disc blur — ${tapCount} taps, dynamically generated in JS.
   114→   114→// Offsets are in units of radius (sigma = radius), scaled at runtime.
   115→   115→// radius < 0.5 falls back to single tap (no visible blur).
   116→   116→//
   117→   117→// When uSampleWallpaper > 0.5, samples the CLEAN wallpaper (uWallpaperSampler
   118→   118→// via coverUv) instead of the scene FBO (uBackdrop via sceneUv), AND applies
   119→   119→// the scrim (uScrimColor) to replicate the original's wallpaper+scrim composited
   120→   120→// LayerBackdrop. The scrim is applied INSIDE sampleBackdrop so EVERY sampling
   121→   121→// site — the initial backdrop sample, the refraction re-sample, and each
   122→   122→// chromatic-aberration channel — gets the same wallpaper+scrim composite.
   123→   123→// This fixes the "scrim not applied at edges" bug where the refraction band
   124→   124→// re-sampled the clean wallpaper (without scrim), making the edge brighter
   125→   125→// than the interior.
   126→   126→vec4 sampleBackdrop(vec2 canvasPx, float radius) {
   127→   127→    if (uSampleWallpaper > 0.5) {
   128→   128→        vec2 uv = coverUv(canvasPx);
   129→   129→        vec4 c;
   130→   130→        if (radius < 0.5) {
   131→   131→            c = texture2D(uWallpaperSampler, uv);
   132→   132→        } else {
   133→   133→            vec2 pxToUv = radius * canvasPxToUvScale();
   134→   134→            vec4 sum = vec4(0.0);
   135→   135→${wallpaperBlurCode}            c = sum;
   136→   136→        }
   137→   137→        // Apply scrim (SrcOver) so the backdrop = wallpaper+scrim, opaque.
   138→   138→        if (uScrimColor.a > 0.001) {
   139→   139→            c.rgb = uScrimColor.rgb * uScrimColor.a + c.rgb * (1.0 - uScrimColor.a);
   140→   140→            c.a = 1.0;
   141→   141→        }
   142→   142→        return c;
   143→   143→    }
   144→   144→    vec2 uv = sceneUv(canvasPx);
   145→   145→    if (radius < 0.5) {
   146→   146→        return texture2D(uBackdrop, uv);
   147→   147→    }
   148→   148→    vec2 pxToUv = radius / uCanvasSize;
   149→   149→    vec4 sum = vec4(0.0);
   150→   150→${backdropBlurCode}    return sum;
   151→   151→}
   152→   152→
   153→   153→// Gaussian disc blur of the WALLPAPER (uWallpaperSampler via coverUv).
   154→   154→// Used by the SDF-texture glass path (LockScreen) — faithful to the original's
   155→   155→// blur(2dp) effect applied before the SDF shader.
   156→   156→vec4 sampleWallpaperBlurred(vec2 canvasPx, float radius) {
   157→   157→    vec2 uv = coverUv(canvasPx);
   158→   158→    if (radius < 0.5) {
   159→   159→        return texture2D(uWallpaperSampler, uv);
   160→   160→    }
   161→   161→    vec2 pxToUv = radius * canvasPxToUvScale();
   162→   162→    vec4 sum = vec4(0.0);
   163→   163→${wallpaperBlurCode}    return sum;
   164→   164→}

   166→   166→// --- Toggle knob CombinedBackdrop sampling (faithful to LiquidToggle.kt) ---
   167→   167→// The knob's backdrop is a CombinedBackdrop of:
   168→   168→//   1. Outer backdrop:
   169→   169→//      - LayerBackdrop (wallpaper) for t1 → sample uWallpaperSampler
   170→   170→//      - CanvasBackdrop (solid color) for t2 → use uSolidBackdropColor
   171→   171→//   2. Scaled trackBackdrop (track color rect, clipped to Capsule, scaled
   172→   172→//      by lerp(2/3, 0.75, pressProgress) x lerp(0, 0.75, pressProgress)
   173→   173→//      around the knob's center)
   174→   174→//
   175→   175→// This function samples the outer backdrop (wallpaper OR solid color) with blur,
   176→   176→// then composites the scaled track color on top using a rounded-rect SDF
   177→   177→// at the uTrackRect position (center + half-size + corner radius).
   178→   178→//
   179→   179→// The track color SDF is also blurred by approximating the blur as a
   180→   180→// smoothstep over uBlurRadius — this matches the original where the blur
   181→   181→// effect is applied to the CombinedBackdrop (outer + track color).
   182→   182→vec4 sampleToggleBackdrop(vec2 canvasPx, float radius) {
   183→   183→    // 1. Sample outer backdrop with blur.
   184→   184→    vec4 wp;
   185→   185→    if (uUseSolidBackdrop > 0.5) {
   186→   186→        // CanvasBackdrop case (t2): solid color fills the entire knob area.
   187→   187→        // Faithful to: rememberCanvasBackdrop { drawRect(backgroundColor) }
   188→   188→        // The drawRect fills the DrawScope (knob's bounds) with the color,
   189→   189→        // so every pixel of the knob's backdrop is the solid color.
   190→   190→        wp = uSolidBackdropColor;
   191→   191→    } else if (radius < 0.5) {
   192→   192→        // LayerBackdrop case (t1): sample wallpaper texture unscaled.
   193→   193→        // IMPORTANT: use coverUv (cover-fit) to match the wallpaper background
   194→   194→        // pass (WALLPAPER_FRAGMENT_SHADER). Using sceneUv (raw normalization)
   195→   195→        // here would sample the wrong texel when the wallpaper aspect ratio
   196→   196→        // differs from the canvas — causing the knob to see a shifted/misaligned
   197→   197→        // wallpaper that doesn't match what's displayed behind it.
   198→   198→        vec2 uv = coverUv(canvasPx);
   199→   199→        wp = texture2D(uWallpaperSampler, uv);
   200→   200→    } else {
   201→   201→        // LayerBackdrop case (t1) with blur: 9-tap poisson disc on wallpaper.
   202→   202→        // Use coverUv for the center sample, and convert the blur radius from
   203→   203→        // canvas px to UV-space using canvasPxToUvScale() (which accounts for
   204→   204→        // the cover-fit aspect ratio cropping).
   205→   205→        vec2 uv = coverUv(canvasPx);
   206→   206→        vec2 pxToUv = radius * canvasPxToUvScale();
   207→   207→        vec4 sum = vec4(0.0);
   208→   208→        float total = 0.0;
   209→   209→        sum += texture2D(uWallpaperSampler, uv) * 0.25; total += 0.25;
   210→   210→        sum += texture2D(uWallpaperSampler, uv + vec2( 1.000,  0.000) * pxToUv) * 0.12; total += 0.12;
   211→   211→        sum += texture2D(uWallpaperSampler, uv + vec2(-1.000,  0.000) * pxToUv) * 0.12; total += 0.12;
   212→   212→        sum += texture2D(uWallpaperSampler, uv + vec2( 0.000,  1.000) * pxToUv) * 0.12; total += 0.12;
   213→   213→        sum += texture2D(uWallpaperSampler, uv + vec2( 0.000, -1.000) * pxToUv) * 0.12; total += 0.12;
   214→   214→        sum += texture2D(uWallpaperSampler, uv + vec2( 0.707,  0.707) * pxToUv) * 0.0675; total += 0.0675;
   215→   215→        sum += texture2D(uWallpaperSampler, uv + vec2( 0.707, -0.707) * pxToUv) * 0.0675; total += 0.0675;
   216→   216→        sum += texture2D(uWallpaperSampler, uv + vec2(-0.707,  0.707) * pxToUv) * 0.0675; total += 0.0675;
   217→   217→        sum += texture2D(uWallpaperSampler, uv + vec2(-0.707, -0.707) * pxToUv) * 0.0675; total += 0.0675;
   218→   218→        wp = sum / total;
   219→   219→    }
   220→   220→
   221→   221→    // 2. Composite scaled track color on top.
   222→   222→    // The track rect is centered at uTrackRect.xy with half-size uTrackRect.zw,
   223→   223→    // and corner radius uTrackCornerRadius. We compute the SDF of this
   224→   224→    // rounded rect at canvasPx, then apply a smoothstep for edge AA + blur.
   225→   225→    // If uTrackColor.a == 0.0 OR the track rect is degenerate (halfW or
   226→   226→    // halfH < 0.5px, which happens at rest when scaleY=0), skip compositing.
   227→   227→    // Faithful to original: scale(scaleX, 0) { drawRect() } draws nothing.
   228→   228→    if (uTrackColor.a > 0.001 && uTrackRect.z > 0.5 && uTrackRect.w > 0.5) {
   229→   229→        vec2 trackCenter = uTrackRect.xy;
   230→   230→        vec2 trackHalf = uTrackRect.zw;
   231→   231→        vec2 trackLocal = canvasPx - trackCenter;
   232→   232→        // sdRoundedRect expects centered coord (relative to center).
   233→   233→        // Use uniform corner radius = uTrackCornerRadius.
   234→   234→        float tr = uTrackCornerRadius;
   235→   235→        // Approximate the rounded-rect SDF (matches sdRoundedRect from SDF_GLSL).
   236→   236→        vec2 q = abs(trackLocal) - trackHalf + vec2(tr);
   237→   237→        float trackSd = length(max(q, vec2(0.0))) + min(max(q.x, q.y), 0.0) - tr;
   238→   238→        // Blur the edge by uBlurRadius (approximate Gaussian edge feather).
   239→   239→        // Inside (trackSd < -radius) → mask=1; outside (trackSd > radius) → mask=0.
   240→   240→        // Use max(radius, 1.0) to guarantee at least 1px smoothstep for AA
   241→   241→        // — when fully pressed, blurRadius=0, but edges must still be smooth.
   242→   242→        float aaRadius = max(radius, 1.0);
   243→   243→        float mask = 1.0 - smoothstep(-aaRadius, aaRadius, trackSd);
   244→   244→        // Composite: srcOver (track color over outer backdrop).
   245→   245→        float a = mask * uTrackColor.a;
   246→   246→        wp.rgb = mix(wp.rgb, uTrackColor.rgb, a);
   247→   247→        wp.a = mix(wp.a, 1.0, a);
   248→   248→    }
   249→   249→    return wp;
   250→   250→}

   294→   294→    // 2. 内层背景板 (Inner backdrop) SDF — the hidden Row's 56dp glass capsule.
   295→   295→    //    Faithful to LiquidBottomTabs.kt: the hidden Row has NO layerBlock,
   296→   296→    //    so its glass does NOT scale with the container. Only panelOffset
   297→   297→    //    shifts it (translationX = panelOffset).
   298→   298→    vec2 capsuleHalf = max(uContainerRect.zw, vec2(0.0));
   299→   299→    float cr = max(uContainerCornerRadius, 0.0);
   300→   300→    // Center = rectCenter + panelOffset (NO container scale).
   301→   301→    vec2 scaledCenter = uContainerRect.xy + vec2(uIndicatorPanelOffset, 0.0);
   302→   302→    vec2 capsuleLocal = canvasPx - scaledCenter;
   303→   303→    vec2 cq = abs(capsuleLocal) - capsuleHalf + vec2(cr);
   304→   304→    float capsuleSd = length(max(cq, vec2(0.0))) + min(max(cq.x, cq.y), 0.0) - cr;
   305→   305→    // Mask: interpolate between 1.0 (at rest) and smoothstep (when pressed).
   306→   306→    // At rest (progress=0): mask=1.0 — no separate smoothstep transition at
   307→   307→    // the containerRect boundary, because it overlaps with the indicator's own
   308→   308→    // edge (both 56dp capsules). A second smoothstep here would reveal raw
   309→   309→    // wallpaper at the indicator edge, causing jagged aliasing. With mask=1.0,
   310→   310→    // the indicator always shows the glass scene inside its shape, and edgeAlpha
   311→   311→    // smoothly fades to transparent — matching the container glass behind it.
   312→   312→    // When pressed (progress=1): restore the original smoothstep mask for the
   313→   313→    // CombinedBackdrop clipping. Refraction displaces samples away from the
   314→   314→    // shared edge, so the smoothstep no longer causes jaggies; and the inner
   315→   315→    // backdrop capsule clip preserves the correct CombinedBackdrop visual
   316→   316→    // (scene inside capsule, wallpaper outside).
   317→   317→    float indicatorAaRadius = max(radius, 1.0);
   318→   318→    float smoothstepMask = 1.0 - smoothstep(-indicatorAaRadius, indicatorAaRadius, capsuleSd);
   319→   319→    float mask = mix(1.0, smoothstepMask, uIndicatorPressProgress);

   321→   321→    // 2b. 内层背景板 shadow (Shadow.Default) — faithful to LiquidBottomTabs.kt
   322→   322→    //     hidden Row's drawBackdrop: shadow defaults to Shadow.Default when not specified.
   323→   323→    //     Shadow.Default: radius=24dp, offset=DpOffset(0, radius/6=4dp), color=Black@0.1, alpha=1.
   324→   324→    //     In the CombinedBackdrop, the shadow is composited between wallpaper (outer)
   325→   325→    //     and glass body (inner). Through the semi-transparent glass body, this shadow
   326→   326→    //     bleeds through near the capsule edges — most visible near the top edge where
   327→   327→    //     the shadow offset (0, +4dp) makes those pixels "outside" the shadow capsule
   328→   328→    //     (shadow capsule top = original top + 4dp, so original top is outside it).
   329→   329→    //     Implementation mirrors ShadowModifier.kt:
   330→   330→    //       1. Shift capsule by shadow offset → shadow shape SDF
   331→   331→    //       2. Gaussian falloff (MaskFilter.makeBlur sigma = radius directly)
   332→   332→    //       3. Mask inside original capsule (ShadowMaskPaint BlendMode.Clear)
   333→   333→    //       4. Darken wallpaper by Black@0.1 × shadowIntensity
   334→   334→    float shadowOffsetYpx = (24.0 / 6.0) * uDpr; // DpOffset(0, radius/6) in device px
   335→   335→    vec2 shadowLocal = capsuleLocal - vec2(0.0, shadowOffsetYpx);
   336→   336→    vec2 shadowCq2 = abs(shadowLocal) - capsuleHalf + vec2(cr);
   337→   337→    float shadowSd = length(max(shadowCq2, vec2(0.0))) + min(max(shadowCq2.x, shadowCq2.y), 0.0) - cr;
   338→   338→    // Shadow intensity: Gaussian falloff from shadow shape edge.
   339→   339→    // MaskFilter.makeBlur(FilterBlurMode.NORMAL, radius) takes sigma = radius directly.
   340→   340→    float shadowSigma = max(24.0 * uDpr, 1.0); // sigma = 24dp in device px
   341→   341→    float shadowIntensity = 0.5 * exp(-shadowSd * shadowSd / (2.0 * shadowSigma * shadowSigma));
   342→   342→    // Mask shadow inside the original capsule (ShadowMaskPaint BlendMode.Clear
   343→   343→    // removes shadow where the shape itself is drawn, so shadow only appears outside).
   344→   344→    shadowIntensity *= smoothstep(-1.0, 1.0, capsuleSd);
   345→   345→    // Darken wallpaper by Black@0.1 × shadowIntensity (SrcOver compositing).
   346→   346→    wp.rgb *= (1.0 - shadowIntensity * 0.1);
   347→   347→
   348→   348→    // 3. Sample the GLASS LAYER FBO (wallpaper + container glass, NO tab text).
   349→   349→    //    This is a snapshot taken after the container glass is rendered but
   350→   350→    //    before tab-content is drawn — so it has no white/black text to bleed
   351→   351→    //    through. The blue tab text is drawn on top via fgTexture (step 4).
   352→   352→    vec2 sceneUv2 = sceneUv(canvasPx - vec2(uIndicatorPanelOffset, 0.0));
   353→   353→    vec4 scene = texture2D(uTabsGlassLayer, sceneUv2);
   354→   354→
   355→   355→    // 4. Draw blue 标签内容 (tab content: icons/labels) on top of the glass layer.
   356→   356→    //    Use each tab's fgTexture alpha as a hard mask (step) — pixels inside
   357→   357→    //    the icon/label shape become blue, everything else stays the glass
   358→   358→    //    layer's natural color. No white edges (hard replace, no mix).
   359→   359→    //    Faithful to LiquidBottomTabs.kt: the hidden Row's tab content gets
   360→   360→    //    LocalLiquidBottomTabScale = lerp(1, 1.2, pressProgress) + panelOffset
   361→   361→    //    (NOT the container scale — the hidden Row is a sibling of the
   362→   362→    //    container, not a child, so the container layerBlock doesn't apply).
   363→   363→    float contentScale = 1.0 + 0.2 * uIndicatorPressProgress;
   364→   364→    float tabMask = 0.0;
   365→   365→    for (int i = 0; i < 8; i++) {
   366→   366→        if (float(i) >= uTabContentCount) break;
   367→   367→        vec4 r = uTabContentRects[i];
   368→   368→        if (r.z > 0.5 && r.w > 0.5) {
   369→   369→            // Tab content scales around its OWN center (not container center)
   370→   370→            // by contentScale, then shifts by panelOffset.
   371→   371→            vec2 tabCenter = r.xy + vec2(uIndicatorPanelOffset, 0.0);
   372→   372→            vec2 scaledHalf = r.zw * contentScale;
   373→   373→            vec2 localPx = canvasPx - (tabCenter - scaledHalf);
   374→   374→            vec2 uv = localPx / (scaledHalf * 2.0);
   375→   375→            if (all(greaterThanEqual(uv, vec2(0.0))) && all(lessThanEqual(uv, vec2(1.0)))) {
   376→   376→                float a = 0.0;
   377→   377→                if (i == 0) a = texture2D(uTabContentTex0, uv).a;
   378→   378→                else if (i == 1) a = texture2D(uTabContentTex1, uv).a;
   379→   379→                else if (i == 2) a = texture2D(uTabContentTex2, uv).a;
   380→   380→                else if (i == 3) a = texture2D(uTabContentTex3, uv).a;
   381→   381→                else if (i == 4) a = texture2D(uTabContentTex4, uv).a;
   382→   382→                else if (i == 5) a = texture2D(uTabContentTex5, uv).a;
   383→   383→                else if (i == 6) a = texture2D(uTabContentTex6, uv).a;
   384→   384→                else if (i == 7) a = texture2D(uTabContentTex7, uv).a;
   385→   385→                tabMask = max(tabMask, a);
   386→   386→            }
   387→   387→        }
   388→   388→    }
   389→   389→    // Use fgTexture alpha directly as the blue compositing factor. fgTexture
   390→   390→    // is LINEAR-filtered so its alpha has smooth AA edges — no smoothstep
   391→   391→    // threshold needed (which caused jaggies by hard-clipping the AA gradient).
   392→   392→    vec3 sceneColor = mix(scene.rgb, uIndicatorAccent.rgb, tabMask);
   393→   393→
   394→   394→    // 5. Composite scene over wallpaper (SrcOver).
   395→   395→    //    At rest (mask≈1.0): a ≈ scene.a — glass scene composited at natural opacity.
   396→   396→    //    When pressed (mask=smoothstep): a = scene.a * mask — CombinedBackdrop clip.
   397→   397→    float a = scene.a * mask;
   398→   398→    vec3 resultRgb = mix(wp.rgb, sceneColor, a);
   399→   399→
   400→   400→    // 6. 内层背景板 rim highlight — faithful to LiquidBottomTabs.kt hidden Row:
   401→   401→    //    highlight = { Highlight.Default.copy(alpha = progress) }
   402→   402→    //    The HighlightModifier draws a STROKE (width=0.5dp, strokeWidth=2px)
   403→   403→    //    blurred by 0.25dp, clipped inside the capsule, colored by the
   404→   404→    //    DefaultHighlightShaderString AGSL shader:
   405→   405→    //      float2 grad = gradSdRoundedRect(centeredCoord, halfSize, gradRadius);
   406→   406→    //      float2 normal = float2(cos(angle), sin(angle));
   407→   407→    //      float d = dot(grad, normal);
   408→   408→    //      float intensity = pow(abs(d), falloff);
   409→   409→    //      return color * intensity;   // color = White(1.0), alpha=1*progress
   410→   410→    //    with angle=45°, falloff=1, gradRadius = min(radius*1.5, min(halfW, halfH)).
   411→   411→    //    The stroke's outward half (capsuleSd > 0) is clipped, leaving the inner
   412→   412→    //    half. Final contribution = White(1.0) * intensity * strokeMask * progress,
   413→   413→    //    added with Plus blend (additive).
   414→   414→    //    NOTE: this is the SAME as the 指示器's own rim highlight (step 2f in
   415→   415→    //    post-passes) — both use Highlight.Default. The only difference is the
   416→   416→    //    SDF: here it's the 内层背景板 capsule (inset 4dp), there it's the
   417→   417→    //    指示器's own capsule. The shader math is identical.
   418→   418→    //
   419→   419→    //    The stroke mask is now sampled from a pre-rasterized Canvas2D texture
   420→   420→    //    (uInnerStrokeMask) instead of computed analytically (65-tap Gaussian
   421→   421→    //    convolution of a hard-edge stroke band). This gives browser-native Skia
   422→   422→    //    hardware coverage AA — identical quality to the outer indicator rim
   423→   423→    //    highlight. The Canvas2D pipeline does ctx.clip(path) → ctx.stroke(path)
   424→   424→    //    → ctx.filter=blur, which naturally removes the outer half and provides
   425→   425→    //    sub-pixel AA. No per-pixel SDF loops, no smoothstep clipAA needed.
   426→   426→    float highlightAlpha = uIndicatorPressProgress;
   427→   427→    if (highlightAlpha > 0.001) {
   428→   428→        // SDF gradient + Default highlight intensity (angle=45°, falloff=1).
   429→   429→        // This part is identical to the AGSL DefaultHighlightShaderString.
   430→   430→        float indRadius = max(cr, 0.0);
   431→   431→        float indHalfMin = min(capsuleHalf.x, capsuleHalf.y);
   432→   432→        float gradRadius = min(indRadius * 1.5, indHalfMin);
   433→   433→        vec2 grad = gradSdRoundedRect(capsuleLocal, capsuleHalf, gradRadius);
   434→   434→        vec2 normal = vec2(0.70710678, 0.70710678); // cos(45°), sin(45°)
   435→   435→        float d = dot(grad, normal);
   436→   436→        float intensity = pow(abs(d), 1.0);
   437→   437→
   438→   438→        // Sample the pre-rasterized Canvas2D stroke mask texture.
   439→   439→        // UV mapping: capsuleLocal (centered, -halfW..+halfW) → element-local
   440→   440→        // (0..2*halfW) by adding capsuleHalf → add margin offset → divide
   441→   441→        // by maskSize. This is the same convention as the outer indicator
   442→   442→        // stroke mask (STROKE_MASK_COMPOSITE_FRAGMENT_SHADER).
   443→   443→        vec2 innerLocal = capsuleLocal + capsuleHalf;
   444→   444→        vec2 innerMaskUv = (innerLocal + uInnerStrokeMaskOffset) / uInnerStrokeMaskSize;
   445→   445→        // Bounds check — discard samples outside the mask texture.
   446→   446→        float innerMask = 0.0;
   447→   447→        if (innerMaskUv.x >= 0.0 && innerMaskUv.x <= 1.0 &&
   448→   448→            innerMaskUv.y >= 0.0 && innerMaskUv.y <= 1.0) {
   449→   449→            innerMask = texture2D(uInnerStrokeMask, innerMaskUv).a;
   450→   450→        }
   451→   451→
   452→   452→        // White(0.5) * intensity * innerMask * progress, Plus blend (additive).
   453→   453→        // Faithful to HighlightStyle.Default: color = White.copy(alpha=0.5f).
   454→   454→        // The AGSL shader uses this 0.5 alpha, NOT color.copy(alpha=1f).
   455→   455→        // Same fix as DEFAULT_HIGHLIGHT.alpha = 0.5 (was previously 1.0).
   456→   456→        // No clipAA needed — the Canvas2D clip(path) before stroke already removes
   457→   457→        // the outer half, and Skia hardware coverage provides AA.
   458→   458→        resultRgb += vec3(0.5) * intensity * innerMask * highlightAlpha;
   459→   459→    }
   460→   460→
   461→   461→    return vec4(resultRgb, 1.0);
   462→   462→}

   464→   464→// Magnifier backdrop sampling — faithful to MagnifierContent.kt's
   465→   465→// onDrawBackdrop: withTransform({ scale(1.5); translate(top=-80dp) }, drawBackdrop).
   466→   466→// Zoom around the magnifier center, then offset Y toward cursor.
   467→   467→vec4 sampleMagnifier(vec2 canvasPx, float radius) {
   468→   468→    vec2 magCenter = uElementOffset + uElementSize * 0.5;
   469→   469→    vec2 zoomedCoord = magCenter + (canvasPx - magCenter) / uMagnifierZoom;
   470→   470→    vec2 cursorCoord = vec2(zoomedCoord.x, zoomedCoord.y + uMagnifierOffsetY);

   474→   474→// colorControls — exact port of ColorFilter.kt colorControlsColorFilter.
   475→   475→// saturation 1.5, brightness 0, contrast 1 -> pure saturation boost.
   476→   476→vec3 applyColorControls(vec3 c, float brightness, float contrast, float saturation) {
   477→   477→    float invSat = 1.0 - saturation;
   478→   478→    float r = 0.213 * invSat;
   479→   479→    float g = 0.715 * invSat;
   480→   480→    float b = 0.072 * invSat;
   481→   481→    float t = (0.5 - contrast * 0.5 + brightness) * 255.0;
   482→   482→    float cs = contrast * saturation;
   483→   483→    float cr = contrast * r;
   484→   484→    float cg = contrast * g;
   485→   485→    float cb = contrast * b;
   486→   486→    vec3 outc;
   487→   487→    outc.r = (cr + cs) * c.r + cg * c.g + cb * c.b + t / 255.0;
   488→   488→    outc.g = cr * c.r + (cg + cs) * c.g + cb * c.b + t / 255.0;
   489→   489→    outc.b = cr * c.r + cg * c.g + (cb + cs) * c.b + t / 255.0;
   490→   490→    return outc;
   491→   491→}

   493→   493→// --- HSV conversion + BlendMode.Hue ---------------------------
   494→   494→// Faithful port of Skia's BlendMode.Hue (non-separable blend).
   495→   495→// Hue blend: result takes hue from src, saturation+value from dst.
   496→   496→// Used by drawRect(tint, BlendMode.Hue) in onDrawSurface.
   497→   497→vec3 rgb2hsv(vec3 c) {
   498→   498→    float maxC = max(c.r, max(c.g, c.b));
   499→   499→    float minC = min(c.r, min(c.g, c.b));
   500→   500→    float delta = maxC - minC;
   501→   501→    float v = maxC;
   502→   502→    float s = maxC < 1e-6 ? 0.0 : delta / maxC;
   503→   503→    float h = 0.0;
   504→   504→    if (delta > 1e-6) {
   505→   505→        if (maxC == c.r) {
   506→   506→            h = mod((c.g - c.b) / delta, 6.0);
   507→   507→        } else if (maxC == c.g) {
   508→   508→            h = (c.b - c.r) / delta + 2.0;
   509→   509→        } else {
   510→   510→            h = (c.r - c.g) / delta + 4.0;
   511→   511→        }
   512→   512→        h *= 60.0;
   513→   513→        if (h < 0.0) h += 360.0;
   514→   514→    }
   515→   515→    return vec3(h / 360.0, s, v);
   516→   516→}
   517→   517→
   518→   518→vec3 hsv2rgb(vec3 c) {
   519→   519→    float h = c.x * 6.0;
   520→   520→    float s = c.y;
   521→   521→    float v = c.z;
   522→   522→    float i = floor(h);
   523→   523→    float f = h - i;
   524→   524→    float p = v * (1.0 - s);
   525→   525→    float q = v * (1.0 - s * f);
   526→   526→    float t = v * (1.0 - s * (1.0 - f));
   527→   527→    i = mod(i, 6.0);
   528→   528→    if (i < 1.0) return vec3(v, t, p);
   529→   529→    if (i < 2.0) return vec3(q, v, p);
   530→   530→    if (i < 3.0) return vec3(p, v, t);
   531→   531→    if (i < 4.0) return vec3(p, q, v);
   532→   532→    if (i < 5.0) return vec3(t, p, v);
   533→   533→    return vec3(v, p, q);
   534→   534→}
   535→   535→
   536→   536→// BlendMode.Hue: take hue from src, sat+val from dst.
   537→   537→vec3 blendHue(vec3 dst, vec3 src) {
   538→   538→    vec3 dh = rgb2hsv(dst);
   539→   539→    vec3 sh = rgb2hsv(src);
   540→   540→    return hsv2rgb(vec3(sh.x, dh.y, dh.z));
   541→   541→}

    23→/* ------------------------------------------------------------------ *
    24→ * CapsuleSdfDebugOverlay
    25→ *
    26→ * Debug overlay for capsule SDF texture generation profiling. Shows
    27→ * per-step timings for each generateContinuousCurvatureMask call +
    28→ * GPU upload time, plus an optional "pack images" view that renders
    29→ * the CPU-side RGBA cache (R=coverage, G=SDF) so you can visually
    30→ * inspect what each cache entry looks like.
    31→ *
    32→ * Toggled from the Performance Monitor panel's "DEBUG OVERLAYS" section.
    33→ *
    34→ * Dragging: pointer events with setPointerCapture + touch-action:none
    35→ * so it works on both mouse and touch. The collapsed badge and the
    36→ * expanded header are both draggable.
    37→ *
    38→ * Polls every 200ms (decoupled from render loop).
    39→ * ------------------------------------------------------------------ */
    40→
    41→interface Props {
    42→  rendererRef: React.MutableRefObject<LiquidGlassRenderer | null>
    43→}
    44→
    45→const POLL_MS = 200
    46→const MAX_ROWS = 8
    47→
    48→export function CapsuleSdfDebugOverlay({ rendererRef }: Props) {
    49→  const [timings, setTimings] = React.useState<CapsuleSdfTiming[]>([])
    50→  const [poolSize, setPoolSize] = React.useState(0)
    51→  // Current texSize (the last-loaded SDF texture's size, one of 128/256/512/1024) +
    52→  // breakdown of how many pool entries use each texSize. Dynamic since
    53→  // Task 66: texSize is 2× oversampling rounded up to POT, clamped [128,1024].
    54→  const [curTexSize, setCurTexSize] = React.useState(0)
    55→  const [pool128, setPool128] = React.useState(0)
    56→  const [pool256, setPool256] = React.useState(0)
    57→  const [pool512, setPool512] = React.useState(0)
    58→  const [pool1024, setPool1024] = React.useState(0)
    59→  const [lastGenMs, setLastGenMs] = React.useState(0)
    60→  const [lastUploadMs, setLastUploadMs] = React.useState(0)
    61→  const [lastKey, setLastKey] = React.useState('')
    62→  const [collapsed, setCollapsed] = React.useState(false)
    63→  const [showPackImages, setShowPackImages] = React.useState(false)
    64→  const [showHighlightMasks, setShowHighlightMasks] = React.useState(false)
    65→  // All entries in the CPU maskCache — we show EVERY entry (active + orphan)
    66→  // so the user can inspect the full cache state, including stale slider-drag
    67→  // or previous-page entries that will age out via the 32MB LRU budget.
    68→  // Each entry is tagged `active` (matches a currently-on-screen
    69→  // useContinuousSdf element) vs `orphan` (no longer on screen) so orphans
    70→  // can be visually dimmed rather than hidden.
    71→  const [maskEntries, setMaskEntries] = React.useState<Array<MaskCacheEntry & { active: boolean }>>([])
    72→  const [maskActiveCount, setMaskActiveCount] = React.useState(0)
    73→  // Total entries in the CPU maskCache (always equals maskEntries.length when
    74→  // "img" is on; also polled when "img" is off for the always-visible summary).
    75→  // The cache is bounded by a 32MB byte-budget LRU (continuous-mask.ts
    76→  // MAX_MASK_CACHE_BYTES) so orphans auto-evict once the budget is exceeded;
    77→  // 'clr' does a full manual purge (for cold-start timing measurement).
    78→  // maskBytes/maskMaxBytes show the live fill ratio.
    79→  const [maskTotalCount, setMaskTotalCount] = React.useState(0)
    80→  const [maskBytes, setMaskBytes] = React.useState(0)
    81→  const [maskMaxBytes] = React.useState(getMaskCacheMaxBytes())
    82→  const [highlightMaskEntries, setHighlightMaskEntries] = React.useState<Array<{
    83→    key: string
    84→    canvas: HTMLCanvasElement
    85→    w: number
    86→    h: number
    87→    ready: boolean
    88→  }>>([])
    89→  const [holeR, setHoleR] = React.useState(false)
    90→  const [holeG, setHoleG] = React.useState(false)
    91→  // Edge scan result — populated asynchronously. The scan button sets a
    92→  // pending flag + requestRender(); the render loop does gl.readPixels at
    93→  // the end of the frame (while the drawing buffer is valid) and stores
    94→  // the result in renderer._edgeScanResult. We poll for it.
    95→  const [edgeScan, setEdgeScan] = React.useState<EdgeScanResult | null>(null)
    96→  // The scanId we last consumed — used to detect new results in the poll.
    97→  const lastConsumedScanId = React.useRef(0)
    98→  const [pos, setPos] = React.useState({ x: -1, y: 120 })
    99→
   100→  // Track the CURRENT visible viewport height (excludes the mobile dynamic
   101→  // navigation bar — URL bar / bottom toolbar — which shrinks/grows as the
   102→  // user scrolls). CSS `100vh` is locked to the LARGEST possible viewport
   103→  // (it does NOT shrink when the navbar appears), so an overlay sized to
   104→  // `calc(100vh - y)` overflows behind the dynamic navbar and becomes
   105→  // unreachable on mobile. `dvh` would fix this but has poor browser
   106→  // compatibility (user rejected it), so we track `window.innerHeight`
   107→  // ourselves and re-render on resize. We listen to BOTH `window.resize`
   108→  // (fires on orientation change + most navbar show/hide) and
   109→  // `visualViewport.resize` (fires on iOS Safari dynamic-navbar transitions
   110→  // where innerHeight updates slightly later) so the overlay snaps to the
   111→  // new height without a visible lag.
   112→  const [vh, setVh] = React.useState(() =>
   113→    typeof window !== 'undefined' ? window.innerHeight : 800)
   114→  React.useEffect(() => {
   115→    const update = () => setVh(window.innerHeight)
   116→    update()
   117→    window.addEventListener('resize', update, { passive: true })
   118→    const vv = typeof visualViewport !== 'undefined' ? visualViewport : null
   119→    if (vv) {
   120→      vv.addEventListener('resize', update)
   121→      vv.addEventListener('scroll', update)
   122→    }
   123→    return () => {
   124→      window.removeEventListener('resize', update)
   125→      if (vv) {
   126→        vv.removeEventListener('resize', update)
   127→        vv.removeEventListener('scroll', update)
   128→      }
   129→    }
   130→  }, [])
   131→
   132→  // Read the renderer's actual probe flags on mount (they may have been
   133→  // toggled by a previous overlay instance). The flags are independent —
   134→  // both R and G can be ON at the same time.
   135→  React.useEffect(() => {
   136→    const r = rendererRef.current
   137→    if (r) {
   138→      setHoleR(r.debugSdfHoleTopLeftR)
   139→      setHoleG(r.debugSdfHoleTopLeftG)
   140→    }
   141→  }, [rendererRef])
   142→
   143→  const flipHole = (channel: 'R' | 'G') => {
   144→    const r = rendererRef.current
   145→    if (!r) return
   146→    if (channel === 'R') {
   147→      const next = !holeR
   148→      r.debugSdfHoleTopLeftR = next
   149→      setHoleR(next)
   150→    } else {
   151→      const next = !holeG
   152→      r.debugSdfHoleTopLeftG = next
   153→      setHoleG(next)
   154→    }
   155→    // Clear the ENTIRE GPU capsule SDF pool (not just the snapshot map).
   156→    // The pool key includes both probe flags (r${holeR},g${holeG}), so each
   157→    // probe combination gets its own texture. Without clearing, a previously-
   158→    // visited probe state's texture stays in the pool — on re-entry it's a
   159→    // pool HIT, so loadContinuousSdf skips texImage2D entirely, and the
   160→    // _debugUploadedSdfTexMap snapshot is never re-written (snapshots only
   161→    // fill on pool MISS). Result: the "GPU upload (probed)" view showed
   162→    // NOTHING for any probe state visited more than once — notably R+G
   163→    // after both had been toggled individually first.
   164→    //
   165→    // Clearing the pool forces every element to MISS on the next render →
   166→    // re-upload → re-snapshot. clearCapsuleSdfPool also clears the snapshot
   167→    // map (so we don't need a separate _debugUploadedSdfTexMap.clear()).
   168→    // The CPU maskCache is NOT touched (clearCapsuleSdfPool only clears the
   169→    // GPU side) — the clean shape stays cached, so re-generation skips the
   170→    // Canvas2D raster + distance transform and only pays the texImage2D +
   171→    // 挖0 cost (~1ms per element). markAllDirty forces every capsule
   172→    // element's elFbo to re-raster so the new texture is sampled immediately.
   173→    r.clearCapsuleSdfPool()
   174→    r.markAllDirty()
   175→    r.requestRender()
   176→  }
   177→
   178→  React.useEffect(() => {
   179→    const id = setInterval(() => {
   180→      const r = rendererRef.current
   181→      if (!r) return
   182→      const all = getCapsuleSdfTimings()
   183→      setTimings(all.slice(-MAX_ROWS).reverse())
   184→      setPoolSize(r.continuousSdfPool.size)
   185→      // Current texSize = the texSize of the last-loaded SDF texture
   186→      // (renderer.continuousSdfTexSize is set by loadContinuousSdf each
   187→      // frame). 0 until the first capsule element renders.
   188→      setCurTexSize(r.continuousSdfTexSize ? r.continuousSdfTexSize[0] : 0)
   189→      // Pool breakdown by texSize — shows how many GPU textures are at each
   190→      // tier (128² knobs/tracks, 256² cards/buttons, 512² GP/dialog,
   191→      // 1024² very large). Empty until capsule elements render.
   192→      let n128 = 0, n256 = 0, n512 = 0, n1024 = 0
   193→      for (const v of r.continuousSdfPool.values()) {
   194→        if (v.texSize >= 1024) n1024++
   195→        else if (v.texSize >= 512) n512++
   196→        else if (v.texSize >= 256) n256++
   197→        else n128++
   198→      }
   199→      setPool128(n128)
   200→      setPool256(n256)
   201→      setPool512(n512)
   202→      setPool1024(n1024)
   203→      setLastGenMs(r._lastCapsuleGenMs)
   204→      setLastUploadMs(r._lastCapsuleUploadMs)
   205→      setLastKey(r._lastCapsuleKey || '')
   206→      // CPU maskCache size + bytes are O(1) reads (getMaskCacheSize /
   207→      // getMaskCacheBytes) so we update them every poll — the always-visible
   208→      // summary line shows "CPU cache: N entries, Z.Z MB / 32 MB" so the
   209→      // user can watch the LRU budget fill up + evict even with "img" off.
   210→      setMaskTotalCount(getMaskCacheSize())
   211→      setMaskBytes(getMaskCacheBytes())
   212→      // Only materialize the full entry array when the user wants to see
   213→      // pack images — Array.from over the whole map every poll is wasteful
   214→      // otherwise.
   215→      if (showPackImages) {
   216→        // Show EVERY entry in the maskCache — both active (matching a
   217→        // currently-on-screen useContinuousSdf element) and orphan (left
   218→        // over from old slider positions / previous pages). The cache is
   219→        // bounded by a 32MB byte-budget LRU, so orphans age out
   220→        // automatically once the budget is exceeded; showing them lets the
   221→        // user inspect the full cache state and watch eviction happen.
   222→        // Each entry is tagged `active` so orphans can be visually dimmed
   223→        // (desaturated + lower opacity) rather than hidden.
   224→        //
   225→        // Key format: "w,h,radius,texSize,sX" (continuous-mask.ts). The
   226→        // first 3 segments (w,h,radius) are the element-identity part —
   227→        // texSize + skipSdf are derived from dpr/quality/noContinuousSdf
   228→        // and are the same for all elements on a given settings config, so
   229→        // matching on the first 3 segments is sufficient to decide active.
   230→        const allEntries = getMaskCacheEntries()
   231→        const activePrefixes = new Set<string>()
   232→        for (const e of r.buttonConfigs) {
   233→          if (e.useContinuousSdf && e.rect.w > 0 && e.rect.h > 0) {
   234→            // Key uses the raw (un-rounded) w/h/radius values, joined by ','.
   235→            // Match continuous-mask.ts's `${w},${h},${radius},` prefix.
   236→            activePrefixes.add(`${e.rect.w},${e.rect.h},${e.cornerRadius},`)
   237→          }
   238→        }
   239→        let activeCount = 0
   240→        const tagged = allEntries.map(e => {
   241→          // Extract the first 3 comma-separated segments + the trailing ','.
   242→          const c1 = e.key.indexOf(',')
   243→          const c2 = e.key.indexOf(',', c1 + 1)
   244→          const c3 = e.key.indexOf(',', c2 + 1)
   245→          const isActive = c1 >= 0 && c2 >= 0 && c3 >= 0
   246→            && activePrefixes.has(e.key.slice(0, c3 + 1))
   247→          if (isActive) activeCount++
   248→          return { ...e, active: isActive }
   249→        })
   250→        setMaskEntries(tagged)
   251→        setMaskActiveCount(activeCount)
   252→      }
   253→      // Same for highlight stroke masks.
   254→      if (showHighlightMasks) {
   255→        setHighlightMaskEntries(Array.from(r.strokeMaskCache.entries()).map(([k, v]) => ({
   256→          key: k, canvas: v.canvas, w: v.w, h: v.h, ready: v.ready,
   257→        })))
   258→      }
   259→      // Edge scan: poll for a completed result. The scan button sets a
   260→      // pending flag + requestRender(); the render loop does gl.readPixels
   261→      // at the end of the frame and stores the result. We detect new
   262→      // results by comparing scanId to the last one we consumed.
   263→      const es = r._edgeScanResult
   264→      if (es && es.scanId !== lastConsumedScanId.current) {
   265→        lastConsumedScanId.current = es.scanId
   266→        setEdgeScan(es)
   267→      }
   268→    }, POLL_MS)
   269→    return () => clearInterval(id)
   270→  }, [rendererRef, showPackImages, showHighlightMasks])
   271→
   272→  // --- Dragging (mouse + touch) ---
   273→  // touch-action:none on the drag handles prevents the browser from
   274→  // interpreting a touch drag as a page scroll / pull-to-refresh, so the
   275→  // pointer events reach our handlers. setPointerCapture ensures we keep
   276→  // receiving pointermove even if the finger leaves the handle element.
   277→  const dragRef = React.useRef<{ sx: number; sy: number; px: number; py: number } | null>(null)
   278→  const onPointerDown = (e: React.PointerEvent) => {
   279→    // Don't start a drag when the user clicks a button inside the header
   280→    // (the collapse "-" button). Those stop propagation on their own.
   281→    if ((e.target as HTMLElement).closest('button')) return
   282→    const el = e.currentTarget as HTMLElement
   283→    try { el.setPointerCapture(e.pointerId) } catch { /* ignore */ }
   284→    dragRef.current = { sx: e.clientX, sy: e.clientY, px: pos.x, py: pos.y }
   285→  }
   286→  const onPointerMove = (e: React.PointerEvent) => {
   287→    if (!dragRef.current) return
   288→    const dx = e.clientX - dragRef.current.sx
   289→    const dy = e.clientY - dragRef.current.sy
   290→    const newX = dragRef.current.px === -1
   291→      ? window.innerWidth - 360 + dx  // approximate; will snap on release
   292→      : dragRef.current.px + dx
   293→    const newY = Math.max(0, Math.min(vh - 40, dragRef.current.py + dy))
   294→    setPos({ x: Math.max(0, newX), y: newY })
   295→  }
   296→  const onPointerUp = () => { dragRef.current = null }
   297→  const onPointerCancel = () => { dragRef.current = null }
   298→
   299→  const left = pos.x === -1 ? undefined : pos.x
   300→  const right = pos.x === -1 ? 8 : undefined
   301→
   302→  if (collapsed) {
   303→    return (
   304→      <div
   305→        onPointerDown={onPointerDown}
   306→        onPointerMove={onPointerMove}
   307→        onPointerUp={onPointerUp}
   308→        onPointerCancel={onPointerCancel}
   309→        onClick={() => setCollapsed(false)}
   310→        style={{
   311→          position: 'absolute', top: pos.y, left, right,
   312→          background: 'rgba(0,0,0,0.85)', color: '#0f0',
   313→          font: 'bold 12px monospace', padding: '6px 10px',
   314→          borderRadius: 6, zIndex: 60, cursor: 'grab',
   315→          border: '1px solid #0f0',
   316→          touchAction: 'none',  // critical for touch drag
   317→          userSelect: 'none',
   318→        }}
   319→      >
   320→        Capsule SDF [{poolSize}] {(lastGenMs + lastUploadMs).toFixed(1)}ms {curTexSize ? `${curTexSize}²` : ''}
   321→      </div>
   322→    )
   323→  }
   324→
   325→  // Find the bottleneck step in the last MISS generation
   326→  const lastMiss = timings.find(t => !t.cacheHit)
   327→  const bottleneck = lastMiss ? (() => {
   328→    const steps: [string, number][] = [
   329→      ['canvas', lastMiss.stepCanvasSetup],
   330→      ['pathDraw', lastMiss.stepPathDraw],
   331→      ['getImage', lastMiss.stepGetImageData],
   332→      ['alpha', lastMiss.stepAlphaExtract],
   333→      ['init', lastMiss.stepInitArrays],
   334→      ['fwdPass', lastMiss.stepForwardPass],
   335→      ['bwdPass', lastMiss.stepBackwardPass],
   336→      ['pack', lastMiss.stepPack],
   337→    ]
   338→    steps.sort((a, b) => b[1] - a[1])
   339→    return steps[0]
   340→  })() : null
   341→
   342→  // CPU maskCache fill ratio (0..1) — drives the green→red color on the
   343→  // "CPU cache: N entries, Z.Z/32 MB LRU" summary line so the user can see
   344→  // at a glance how close the byte budget is to triggering eviction.
   345→  const fillPct = maskMaxBytes > 0 ? maskBytes / maskMaxBytes : 0
   346→
   347→  return (
   348→    <div
   349→      style={{
   350→        position: 'absolute', top: pos.y, left, right,
   351→        width: 340,
   352→        // Cap height to viewport so the overlay never overflows the screen.
   353→        // The header + summary stay pinned; everything below scrolls.
   354→        // Floor at 220px so the overlay stays usable when dragged near the
   355→        // bottom of the viewport (pos.y can be up to vh-40). Uses the JS-
   356→        // tracked `vh` (window.innerHeight) instead of CSS `100vh` because
   357→        // `100vh` on mobile is locked to the LARGEST possible viewport and
   358→        // does NOT shrink when the dynamic navigation bar appears — an
   359→        // overlay sized to `100vh - y` would overflow behind the navbar.
   360→        maxHeight: Math.max(220, vh - pos.y - 8),
   361→        display: 'flex', flexDirection: 'column',
   362→        background: 'rgba(0,0,0,0.92)', color: '#0f0',
   363→        font: '11px monospace', borderRadius: 8, zIndex: 60,
   364→        border: '1px solid #0f0', overflow: 'hidden',
   365→        boxShadow: '0 4px 20px rgba(0,0,0,0.5)',
   366→      }}
   367→    >
   368→      {/* WebKit scrollbar styling (Firefox uses scrollbarColor inline). */}
   369→      <style>{`
   370→        .capsule-debug-scroll::-webkit-scrollbar { width: 8px; }
   371→        .capsule-debug-scroll::-webkit-scrollbar-track { background: transparent; }
   372→        .capsule-debug-scroll::-webkit-scrollbar-thumb {
   373→          background: rgba(0,255,0,0.35); border-radius: 4px;
   374→        }
   375→        .capsule-debug-scroll::-webkit-scrollbar-thumb:hover {
   376→          background: rgba(0,255,0,0.55);
   377→        }
   378→      `}</style>
   379→      {/* Header — drag handle (pinned, non-scrolling) */}
   380→      <div
   381→        onPointerDown={onPointerDown}
   382→        onPointerMove={onPointerMove}
   383→        onPointerUp={onPointerUp}
   384→        onPointerCancel={onPointerCancel}
   385→        style={{
   386→          display: 'flex', justifyContent: 'space-between', alignItems: 'center',
   387→          padding: '6px 10px', background: 'rgba(0,255,0,0.15)',
   388→          cursor: 'grab', borderBottom: '1px solid rgba(0,255,0,0.3)',
   389→          fontWeight: 'bold', fontSize: 12,
   390→          touchAction: 'none',  // critical for touch drag
   391→          userSelect: 'none',
   392→        }}
   393→      >
   394→        <span>Capsule SDF Debug</span>
   395→        <span style={{ display: 'flex', gap: 6 }}>
   396→          <button
   397→            onPointerDown={(e) => e.stopPropagation()}
   398→            onClick={() => flipHole('R')}
   399→            title="PROBE clip source: zero R (coverage) in the TOP-LEFT 1/4 of the capsule SDF texture (image row<128 && col<128). Done on a COPY at GPU upload — CPU maskCache stays clean; GPU pool key includes the flag so toggling is instant. Due to UNPACK_FLIP_Y + Y-down centeredOrigRot, this image-top-left region maps to the element's BOTTOM-LEFT corner on screen. If that corner of the glass body then VANISHES (sampleClipMask → 0 → discard), the clip edge really comes from sampling R. If nothing changes, the edge is from elsewhere (analytic sdRoundedRect / scissor / elFbo composite bounds). Independent of G — both can be ON."
   400→            style={{
   401→              background: holeR ? 'rgba(255,0,255,0.35)' : 'none',
   402→              border: `1px solid ${holeR ? '#f0f' : '#0f0'}`,
   403→              color: holeR ? '#f0f' : '#0f0',
   404→              cursor: 'pointer', fontSize: 10, padding: '0 5px', borderRadius: 3, fontWeight: 'bold',
   405→            }}
   406→          >R</button>
   407→          <button
   408→            onPointerDown={(e) => e.stopPropagation()}
   409→            onClick={() => flipHole('G')}
   410→            title="PROBE stroke/SDF source: zero G (SDF) in the TOP-LEFT 1/4 of the capsule SDF texture. Same copy-at-upload mechanism as R — CPU maskCache untouched, GPU pool key includes the flag. If the highlight / rim-stroke shape in the corresponding corner then changes or disappears, it really does come from sampling G (sampleClipSdf). Independent of R — both can be ON."
   411→            style={{
   412→              background: holeG ? 'rgba(255,0,255,0.35)' : 'none',
   413→              border: `1px solid ${holeG ? '#f0f' : '#0f0'}`,
   414→              color: holeG ? '#f0f' : '#0f0',
   415→              cursor: 'pointer', fontSize: 10, padding: '0 5px', borderRadius: 3, fontWeight: 'bold',
   416→            }}
   417→          >G</button>
   418→          <button
   419→            onPointerDown={(e) => e.stopPropagation()}
   420→            onClick={() => {
   421→              // Clear BOTH the CPU mask cache (Uint8Array entries) and the
   422→              // GPU texture pool (WebGLTextures). Next render re-generates
   423→              // everything on demand — useful to measure cold-start timings
   424→              // or to verify the SDF texture is actually being used.
   425→              clearMaskCache()
   426→              rendererRef.current?.clearCapsuleSdfPool()
   427→              setMaskEntries([])
   428→              setMaskTotalCount(0)
   429→              setMaskActiveCount(0)
   430→              setMaskBytes(0)
   431→              rendererRef.current?.requestRender?.()
   432→            }}
   433→            title="Clear both CPU mask cache + GPU texture pool. Forces re-generation on next render."
   434→            style={{
   435→              background: 'rgba(255,68,68,0.2)', border: '1px solid #f44',
   436→              color: '#f88', cursor: 'pointer', fontSize: 10, padding: '0 4px', borderRadius: 3,
   437→            }}
   438→          >clr</button>
   439→          <button
   440→            onPointerDown={(e) => e.stopPropagation()}
   441→            onClick={() => setShowPackImages(v => !v)}
   442→            title="Toggle visualization of cached RGBA pack textures (R=coverage, G=SDF)"

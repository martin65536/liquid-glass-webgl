    39→/* ------------------------------------------------------------------ *
    40→ * LiquidGlassRenderer
    41→ *
    42→ * One opaque WebGL canvas. Render pipeline per frame:
    43→ *   1. Wallpaper background pass (cover-fit).
    44→ *   2. For each button (in order):
    45→ *      a. Outer drop shadow pass (if configured).
    46→ *      b. Element pass (refraction + vibrancy + tint + highlight),
    47→ *         with InteractiveHighlight-driven scale/translation/stretch.
    48→ *      c. White overlay pass (Plus blend, 8% * progress) on press.
    49→ *      d. Radial highlight pass (Plus blend, at finger position) on press.
    50→ *      e. Foreground pass (button label, with press-driven alpha fade).
    51→ *
    52→ * No DOM children — the canvas owns the entire visual surface.
    53→ *
    54→ * NOTE: The class methods are split across multiple files in this
    55→ * directory (methods-*.ts). Each methods file declares its method
    56→ * signatures via `declare module './index'` and exports a record of
    57→ * named functions. Those records are merged onto the prototype at the
    58→ * bottom of this file via Object.assign.
    59→ * ------------------------------------------------------------------ */

   523→  /** Lazy-compile horizontal + vertical blur programs for a 1D tap count. */
   524→  ensureBlurPrograms(tapCount: number): void {
   525→    if (this.blurPrograms.has(tapCount)) return
   526→    const gl = this.gl
   527→    const hFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'horizontal'))
   528→    const vFs = compileShader(gl, gl.FRAGMENT_SHADER, generateSeparableBlurShader(tapCount, 'vertical'))
   529→    const mk = (fs: WebGLShader) => {
   530→      const vs = compileShader(gl, gl.VERTEX_SHADER, VERTEX_SHADER)
   531→      const p = gl.createProgram()!
   532→      gl.attachShader(p, vs)
   533→      gl.attachShader(p, fs)
   534→      gl.bindAttribLocation(p, 0, 'aPos')
   535→      gl.linkProgram(p)
   536→      gl.deleteShader(vs)
   537→      gl.deleteShader(fs)
   538→      if (!gl.getProgramParameter(p, gl.LINK_STATUS)) {
   539→        const log = gl.getProgramInfoLog(p)
   540→        gl.deleteProgram(p)
   541→        throw new Error('Blur program link error (taps=' + tapCount + '): ' + log)
   542→      }
   543→      return p
   544→    }
   545→    const hProg = mk(hFs)
   546→    const vProg = mk(vFs)
   547→    const uH: Record<string, WebGLUniformLocation | null> = {
   548→      uTexture: gl.getUniformLocation(hProg, 'uTexture'),
   549→      uTexSize: gl.getUniformLocation(hProg, 'uTexSize'),
   550→      uRadius: gl.getUniformLocation(hProg, 'uRadius'),
   551→    }
   552→    const uV: Record<string, WebGLUniformLocation | null> = {
   553→      uTexture: gl.getUniformLocation(vProg, 'uTexture'),
   554→      uTexSize: gl.getUniformLocation(vProg, 'uTexSize'),
   555→      uRadius: gl.getUniformLocation(vProg, 'uRadius'),
   556→    }
   557→    this.blurPrograms.set(tapCount, { hProg, vProg, uH, uV, aPosH: 0, aPosV: 0 })
   558→  }

   560→  /** 2-pass blur a source texture by `radius` px. Reads srcTex, writes the
   561→   *  blurred result into blurFboB, returns blurFboBTex.
   562→   *  Saves/restores the currently-bound framebuffer.
   563→   *  Uses this.blurTapCap to cap 1D tap count (performance knob).
   564→   *  (blurDownsample is reserved for future use — currently always full-res.) */
   565→  blurTexture(srcTex: WebGLTexture, radius: number): WebGLTexture {
   566→    const gl = this.gl
   567→    const w = this.fboW
   568→    const h = this.fboH
   569→    // Compute tap count, capped by blurTapCap (performance knob).
   570→    let taps = computeBlur1DTapCount(radius)
   571→    taps = Math.min(taps, Math.max(1, this.blurTapCap | 0))
   572→    this.ensureBlurPrograms(taps)
   573→    const entry = this.blurPrograms.get(taps)!
   574→    const savedFb = gl.getParameter(gl.FRAMEBUFFER_BINDING)
   575→    gl.disable(gl.BLEND)
   576→
   577→    // Pass 1: horizontal — srcTex → blurFboA
   578→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboA)
   579→    gl.viewport(0, 0, w, h)
   580→    gl.useProgram(entry.hProg)
   581→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   582→    gl.enableVertexAttribArray(entry.aPosH)
   583→    gl.vertexAttribPointer(entry.aPosH, 2, gl.FLOAT, false, 0, 0)
   584→    gl.activeTexture(gl.TEXTURE0)
   585→    gl.bindTexture(gl.TEXTURE_2D, srcTex)
   586→    gl.uniform1i(entry.uH['uTexture'], 0)
   587→    gl.uniform2f(entry.uH['uTexSize'], w, h)
   588→    gl.uniform1f(entry.uH['uRadius'], radius)
   589→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   590→
   591→    // Pass 2: vertical — blurFboATex → blurFboB
   592→    gl.bindFramebuffer(gl.FRAMEBUFFER, this.blurFboB)
   593→    gl.viewport(0, 0, w, h)
   594→    gl.useProgram(entry.vProg)
   595→    gl.bindBuffer(gl.ARRAY_BUFFER, this.quadBuffer)
   596→    gl.enableVertexAttribArray(entry.aPosV)
   597→    gl.vertexAttribPointer(entry.aPosV, 2, gl.FLOAT, false, 0, 0)
   598→    gl.activeTexture(gl.TEXTURE0)
   599→    gl.bindTexture(gl.TEXTURE_2D, this.blurFboATex!)
   600→    gl.uniform1i(entry.uV['uTexture'], 0)
   601→    gl.uniform2f(entry.uV['uTexSize'], w, h)
   602→    gl.uniform1f(entry.uV['uRadius'], radius)
   603→    gl.drawArrays(gl.TRIANGLES, 0, 6)
   604→
   605→    gl.bindFramebuffer(gl.FRAMEBUFFER, savedFb)
   606→    gl.viewport(0, 0, w, h)
   607→    return this.blurFboBTex!
   608→  }
